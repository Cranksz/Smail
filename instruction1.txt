  function buildBrushGeometry(pts, speeds, brushType, baseR) {
    const curve = new THREE.CatmullRomCurve3(pts);
    const segs = Math.min(240, Math.max(12, pts.length * 3));

    if (brushType === '2d' || brushType === 'ribbon') {
      const samplePts = curve.getPoints(segs);
      const pos = new Float32Array((segs + 1) * 2 * 3);
      const uvs = new Float32Array((segs + 1) * 2 * 2);
      const idx = [];
      const tang = new THREE.Vector3(), side = new THREE.Vector3();
      const refNormal = brushType === '2d' ? sketchPlaneNormal2D : new THREE.Vector3(0, 1, 0);
      const maxHalfW = baseR * (brushType === '2d' ? 1.35 : 2.35);

      for (let i = 0; i <= segs; i++) {
        const t = i / segs;
        curve.getTangent(t, tang).normalize();
        side.crossVectors(tang, refNormal);
        if (side.lengthSq() < 0.001) side.set(1, 0, 0);
        // Smooth rounded taper at both stroke tips
        const tipTaper = Math.min(1, Math.sin(Math.min(0.5, t) * Math.PI) * 2.2, Math.sin(Math.min(0.5, 1 - t) * Math.PI) * 2.2);
        let speedScale = 1.0;
        if (brushType === '2d') {
          const spIdx = Math.min(speeds.length - 1, Math.floor(t * speeds.length));
          speedScale = THREE.MathUtils.clamp(1.35 - (speeds[spIdx] || 0.25) * 0.55, 0.55, 1.45);
        }
        side.normalize().multiplyScalar(Math.max(0.0005, maxHalfW * tipTaper * speedScale));
        const p = samplePts[i];
        pos[i * 6] = p.x - side.x; pos[i * 6 + 1] = p.y - side.y; pos[i * 6 + 2] = p.z - side.z;
        pos[i * 6 + 3] = p.x + side.x; pos[i * 6 + 4] = p.y + side.y; pos[i * 6 + 5] = p.z + side.z;
        uvs[i * 4] = t; uvs[i * 4 + 1] = 0;
        uvs[i * 4 + 2] = t; uvs[i * 4 + 3] = 1;
        if (i < segs) {
          const a = i * 2, b = i * 2 + 1, c = (i + 1) * 2, d = (i + 1) * 2 + 1;
          idx.push(a, b, c, b, d, c);
        }
      }
      const geo = new THREE.BufferGeometry();
      geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
      geo.setAttribute('uv', new THREE.BufferAttribute(uvs, 2));
      geo.setIndex(idx);
      geo.computeVertexNormals();
      return geo;
    }

    // Smoothly Capped & Tapered 3D Sculpted Tube (No hollow open tube ends!)
    const radSegs = brushType === 'rope' ? 12 : 16;
    const geo = new THREE.TubeGeometry(curve, segs, baseR, radSegs, false);
    const posAttr = geo.attributes.position;
    const cPt = new THREE.Vector3(), radial = new THREE.Vector3();

    for (let i = 0; i < posAttr.count; i++) {
      const ring = Math.floor(i / (radSegs + 1));
      const seg = i % (radSegs + 1);
      const t = Math.min(1, ring / segs);
      curve.getPoint(t, cPt);
      radial.fromBufferAttribute(posAttr, i).sub(cPt);

      // Smooth rounded endcaps at t=0 and t=1
      let cap = 1.0;
      if (t < 0.06) cap = Math.sin((t / 0.06) * (Math.PI * 0.5));
      else if (t > 0.94) cap = Math.sin(((1.0 - t) / 0.06) * (Math.PI * 0.5));
      cap = Math.max(0.05, cap);

      let scale = cap;
      if (brushType === 'calligraphy') {
        const spIdx = Math.min(speeds.length - 1, Math.floor(t * speeds.length));
        const spd = speeds[spIdx] || 0.25;
        scale *= THREE.MathUtils.clamp(1.65 - spd * 0.85, 0.42, 1.75);
      } else if (brushType === 'rope') {
        const ang = (seg / radSegs) * Math.PI * 2 + t * 44.0;
        scale *= 0.86 + 0.24 * Math.sin(ang * 3.0);
      }
      radial.multiplyScalar(scale).add(cPt);
      posAttr.setXYZ(i, radial.x, radial.y, radial.z);
    }
    geo.computeVertexNormals();
    return geo;
  }

    if (p === 'float') return { bounce: 0.55, friction: 0.99, grav: false };
    return { bounce: 0.44, friction: 0.92, grav: !!APP.drawingGravity };
  }

  function finalizeStroke() {
    if (previewMesh) {
      APP.scene.remove(previewMesh);
      if (previewMesh.geometry) previewMesh.geometry.dispose();
      if (previewMesh.material) previewMesh.material.dispose();
      previewMesh = null;
    }
    sketchPlane2D = null;
    if (currentPoints.length < 2) {
      currentPoints = [];
      currentSpeeds = [];
      return;
    }

    const colHex = PALETTE[drawCfg.colorIndex % PALETTE.length];
    const baseR = SIZE_MAP[drawCfg.sizeKey] || 0.0095;
    const rawMesh = buildStrokeMesh(currentPoints, currentSpeeds, drawCfg.brush, colHex, baseR);
    rawMesh.geometry.computeBoundingBox();
    const box = rawMesh.geometry.boundingBox;
    const center = new THREE.Vector3();
    box.getCenter(center);
    rawMesh.geometry.translate(-center.x, -center.y, -center.z);

    const localPts = downsamplePoints(currentPoints, 0.022).map((p) => p.sub(center));

    const strokeGroup = new THREE.Group();
    strokeGroup.name = `${drawCfg.brush}_Stroke`;
    strokeGroup.position.copy(center);
    strokeGroup.add(rawMesh);
    APP.scene.add(strokeGroup);

    const phys = getPhysicsValues();
    const isDynamic = phys.grav;

    const ent = APP.registerEntity(strokeGroup, {
      name: `${isDynamic ? 'Dynamic' : 'Static'} ${drawCfg.brush} Stroke`,
      category: 'stroke',
      isStroke: true,
      hasGravity: isDynamic,
      restitution: phys.bounce,
      velocity: new THREE.Vector3(0, 0, 0),
      meta: {
        isDynamicStroke: isDynamic,
        isStaticStroke: !isDynamic,
        localPoints: localPts,
        tubeRadius: baseR
      }
    });
    if (ent.physics) {
      ent.physics.bounce = phys.bounce;
      ent.physics.friction = phys.friction;
      ent.physics.gravity = isDynamic;
    }

    strokeHistory.push(ent);
    currentPoints = [];
    currentSpeeds = [];
    return ent;
  }

  function stepSculpting(dt) {
    const ptr = getActivePointerPoint(false);
    if (!ptr || drawCfg.mode !== 'sculpt') {
      sculptRing.visible = false;
      prevSculptPos = null;
      return;
    }

    sculptRing.visible = true;
    sculptRing.position.copy(ptr.pos);
    const s = drawCfg.sculptRadius / 0.032;
    sculptRing.scale.setScalar(s * 0.35);
    if (APP.camera) sculptRing.lookAt(APP.camera.position);

    if (!ptr.down) {
      prevSculptPos = null;
      return;
    }

    const handDelta = prevSculptPos ? tmpSculptDelta.subVectors(ptr.pos, prevSculptPos) : tmpSculptDelta.set(0, 0, 0);
    prevSculptPos = ptr.pos.clone();

    const brushR = drawCfg.sculptRadius;
    const str = drawCfg.sculptStrength * Math.min(0.05, dt || 0.016);
    const mode = drawCfg.sculptBrush;

    for (const ent of APP.entities || []) {
      if (!ent || !ent.group?.parent || ent.isPen || ent.isStroke || ent.category === 'world_fx' || ent.category === 'portal' || ent.category === 'board' || ent.category === 'cloth') continue;
      ent.group.traverse((mesh) => {
        if (!mesh.isMesh || !mesh.geometry?.attributes?.position || mesh.userData?.nxrDecal) return;
        const geo = mesh.geometry;
        const posA = geo.attributes.position;
        if (!geo.attributes.normal) geo.computeVertexNormals();
        const normA = geo.attributes.normal;

        mesh.updateMatrixWorld(true);
        tmpBoxStat.setFromObject(mesh).expandByScalar(brushR);
        if (!tmpBoxStat.containsPoint(ptr.pos)) return;

        tmpSculptLocal.copy(ptr.pos);
        mesh.worldToLocal(tmpSculptLocal);
        const localScale = mesh.getWorldScale(tmpV).x || 1;
        const localR = brushR / Math.max(0.05, localScale);
        const localRSq = localR * localR;

        let touched = false;
        if (mode === 'smooth') {
          tmpCentroid.set(0, 0, 0);
          let cnt = 0;
          for (let i = 0; i < posA.count; i++) {
            tmpV.fromBufferAttribute(posA, i);
            if (tmpV.distanceToSquared(tmpSculptLocal) < localRSq) { tmpCentroid.add(tmpV); cnt++; }
          }
          if (cnt > 2) {
            tmpCentroid.divideScalar(cnt);
            for (let i = 0; i < posA.count; i++) {
              tmpV.fromBufferAttribute(posA, i);
              const dSq = tmpV.distanceToSquared(tmpSculptLocal);
              if (dSq < localRSq) {
                const w = 0.5 * (1 + Math.cos(Math.PI * Math.sqrt(dSq) / localR));
                tmpV.lerp(tmpCentroid, Math.min(0.35, w * str * 8.0));
                posA.setXYZ(i, tmpV.x, tmpV.y, tmpV.z);
                touched = true;
              }
            }
          }
        } else {
          const invRot = new THREE.Matrix3().getNormalMatrix(mesh.matrixWorld).invert();
          const localDrag = handDelta.clone().applyMatrix3(invRot);
          for (let i = 0; i < posA.count; i++) {
            tmpV.fromBufferAttribute(posA, i);
            const dSq = tmpV.distanceToSquared(tmpSculptLocal);
            if (dSq >= localRSq) continue;
            const d = Math.sqrt(dSq);
            const w = 0.5 * (1 + Math.cos(Math.PI * d / localR));
            if (mode === 'grab') {
              tmpV.addScaledVector(localDrag, w);
            } else if (mode === 'pinch') {
              tmpV.lerp(tmpSculptLocal, w * str * 1.6);
            } else {
              tmpN.fromBufferAttribute(normA, i);
              const sign = mode === 'carve' ? -1 : 1;
              tmpV.addScaledVector(tmpN, sign * w * str * 0.25);
            }
            posA.setXYZ(i, tmpV.x, tmpV.y, tmpV.z);
            touched = true;
          }
        }

        if (touched) {
          posA.needsUpdate = true;
          geo.computeVertexNormals();
          geo.computeBoundingBox();
          geo.computeBoundingSphere();
        }
      });
    }
  }

  function closestPointOnSegment(p, a, b, out) {
    const abX = b.x - a.x, abY = b.y - a.y, abZ = b.z - a.z;
    const lenSq = abX * abX + abY * abY + abZ * abZ;
    if (lenSq < 1e-8) return out.copy(a);
    let t = ((p.x - a.x) * abX + (p.y - a.y) * abY + (p.z - a.z) * abZ) / lenSq;
    if (t < 0) t = 0;
    else if (t > 1) t = 1;
    return out.set(a.x + abX * t, a.y + abY * t, a.z + abZ * t);
  }

  function stepStrokeCollisions() {
    if (!drawCfg.collisions) return;
    const entities = APP.entities || [];
    const dynStrokes = [], statStrokes = [], solidBoxes = [];

    for (let i = 0; i < entities.length; i++) {
      const e = entities[i];
      if (!e || !e.group?.parent) continue;
      if (e.isStroke) {
        if (e.physics?.gravity && !e.isHeld && !e.heldByBot) dynStrokes.push(e);
        else if (e.physics && !e.physics.gravity) statStrokes.push(e);
      } else if (!e.isPen && e.category !== 'world_fx' && e.category !== 'portal' && e.category !== 'board') {
        solidBoxes.push(e);
      }
    }

    if (!dynStrokes.length) return;

    for (let d = 0; d < dynStrokes.length; d++) {
      const dyn = dynStrokes[d];
      const rad = dyn.meta?.tubeRadius || 0.0095;
      const colDist = rad * 2.25;
      dyn.group.updateMatrixWorld(true);
      tmpBoxDyn.setFromObject(dyn.group);

      if (!tmpBoxDyn.isEmpty() && tmpBoxDyn.min.y < 0) {
        dyn.group.position.y += -tmpBoxDyn.min.y;
        if (dyn.physics.vel.y < 0) {
          dyn.physics.vel.y = Math.abs(dyn.physics.vel.y) > 0.25 ? -dyn.physics.vel.y * (dyn.physics.bounce ?? 0.44) : 0;
        }
        dyn.physics.vel.x *= dyn.physics.friction ?? 0.92;
        dyn.physics.vel.z *= dyn.physics.friction ?? 0.92;
        dyn.group.updateMatrixWorld(true);
        tmpBoxDyn.setFromObject(dyn.group);
      }

      tmpBoxDyn.expandByScalar(colDist);
      const dynPts = dyn.meta?.localPoints;
      if (!dynPts || !dynPts.length) continue;

      for (let s = 0; s < statStrokes.length; s++) {
        const stat = statStrokes[s];
        const statPts = stat.meta?.localPoints;
        if (!statPts || statPts.length < 2) continue;

        stat.group.updateMatrixWorld(true);
        tmpBoxStat.setFromObject(stat.group).expandByScalar(colDist);
        if (!tmpBoxDyn.intersectsBox(tmpBoxStat)) continue;

        const dynMat = dyn.group.matrixWorld, statMat = stat.group.matrixWorld;
        let deepestPen = 0, hitNX = 0, hitNY = 0, hitNZ = 0;

        for (let pi = 0; pi < dynPts.length; pi++) {
          tmpPtW.copy(dynPts[pi]).applyMatrix4(dynMat);
          for (let si = 0; si < statPts.length - 1; si++) {
            tmpSegA.copy(statPts[si]).applyMatrix4(statMat);
            tmpSegB.copy(statPts[si + 1]).applyMatrix4(statMat);
            closestPointOnSegment(tmpPtW, tmpSegA, tmpSegB, tmpClosest);
            tmpDiff.subVectors(tmpPtW, tmpClosest);
            const distSq = tmpDiff.lengthSq();
            if (distSq < colDist * colDist) {
              const dist = Math.sqrt(distSq);
              const pen = colDist - dist;
              if (pen > deepestPen) {
                deepestPen = pen;
                if (dist > 1e-5) tmpNormal.copy(tmpDiff).multiplyScalar(1 / dist);
                else tmpNormal.set(0, 1, 0);
                hitNX = tmpNormal.x; hitNY = tmpNormal.y; hitNZ = tmpNormal.z;
              }
            }
          }
        }

        if (deepestPen > 0) {
          dyn.group.position.x += hitNX * deepestPen;
          dyn.group.position.y += hitNY * deepestPen;
          dyn.group.position.z += hitNZ * deepestPen;
          const v = dyn.physics.vel;
          const vn = v.x * hitNX + v.y * hitNY + v.z * hitNZ;
          if (vn < 0) {
            const bounce = Math.abs(vn) > 0.28 ? (dyn.physics.bounce ?? 0.44) : 0.0;
            const imp = -(1 + bounce) * vn;
            v.x += hitNX * imp; v.y += hitNY * imp; v.z += hitNZ * imp;
            v.x *= 0.94; v.z *= 0.94;
            if (hitNY > 0.65 && Math.abs(v.y) < 0.06) v.y = 0;
          }
          dyn.group.updateMatrixWorld(true);
          tmpBoxDyn.setFromObject(dyn.group).expandByScalar(colDist);
        }
      }

      for (let oi = 0; oi < solidBoxes.length; oi++) {
        const obj = solidBoxes[oi];
        obj.group.updateMatrixWorld(true);
        tmpBoxStat.setFromObject(obj.group);
        if (!tmpBoxDyn.intersectsBox(tmpBoxStat)) continue;
        const dynMat = dyn.group.matrixWorld;
        let maxPen = 0, nx = 0, ny = 1, nz = 0;
        for (let pi = 0; pi < dynPts.length; pi++) {
          tmpPtW.copy(dynPts[pi]).applyMatrix4(dynMat);
          tmpBoxStat.clampPoint(tmpPtW, tmpClosest);
          tmpDiff.subVectors(tmpPtW, tmpClosest);
          const distSq = tmpDiff.lengthSq();
          if (distSq < rad * rad) {
            const dist = Math.sqrt(distSq);
            const pen = rad - dist;
            if (pen > maxPen) {
              maxPen = pen;
              if (dist > 1e-5) tmpNormal.copy(tmpDiff).multiplyScalar(1 / dist);
              else tmpNormal.set(0, 1, 0);
              nx = tmpNormal.x; ny = tmpNormal.y; nz = tmpNormal.z;
            }
          }
        }
        if (maxPen > 0) {
          dyn.group.position.x += nx * maxPen; dyn.group.position.y += ny * maxPen; dyn.group.position.z += nz * maxPen;
          const v = dyn.physics.vel;
          const vn = v.x * nx + v.y * ny + v.z * nz;
          if (vn < 0) {
            const imp = -(1 + (dyn.physics.bounce ?? 0.4)) * vn;
            v.x += nx * imp; v.y += ny * imp; v.z += nz * imp;
          }
        }
      }
    }
  }

  function createVolumetricFireMaterial(opts = {}) {
    const uniforms = {
      uTime: { value: 0.0 },
      uBend: { value: new THREE.Vector3(0, 0, 0) },
      uIntensity: { value: Number(opts.intensity) || 1.0 },
      uFlameH: { value: Number(opts.height) || 0.24 }
    };
    const mat = new THREE.ShaderMaterial({
      uniforms,
      transparent: true,
      depthWrite: false,
      side: THREE.DoubleSide,
      blending: THREE.AdditiveBlending,
      vertexShader: [
        'uniform float uTime; uniform vec3 uBend; uniform float uIntensity; uniform float uFlameH;',
        'varying vec2 vUv; varying float vHeight;',
        'void main() {',
        '  vUv = uv; vHeight = clamp(uv.y, 0.0, 1.0);',
        '  vec3 p = position; float h2 = vHeight * vHeight;',
        '  p.xz *= (1.0 - vHeight * 0.45) * mix(0.88, 1.32, clamp(uIntensity - 0.5, 0.0, 1.2));',
        '  p.y *= mix(0.92, 1.55, clamp((uIntensity - 0.8) * 0.55, 0.0, 1.2));',
        '  float sway = sin(uTime * 10.5 + position.y * 20.0) * 0.014 * h2;',
        '  p.x += uBend.x * h2 + sway;',
        '  p.z += uBend.z * h2 + cos(uTime * 8.2 + position.y * 17.0) * 0.014 * h2;',
        '  gl_Position = projectionMatrix * modelViewMatrix * vec4(p, 1.0);',
        '}'
      ].join('\n'),
      fragmentShader: [
        'uniform float uTime; uniform float uIntensity; varying vec2 vUv; varying float vHeight;',
        'float fHash(vec2 p) { return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453123); }',
        'float fNoise(vec2 p) { vec2 i = floor(p); vec2 f = fract(p); f = f * f * (3.0 - 2.0 * f); return mix(mix(fHash(i), fHash(i + vec2(1.0, 0.0)), f.x), mix(fHash(i + vec2(0.0, 1.0)), fHash(i + vec2(1.0, 1.0)), f.x), f.y); }',
        'float fbm(vec2 p) { float v = 0.0; float a = 0.52; for (int i = 0; i < 4; i++) { v += a * fNoise(p); p *= 2.07; a *= 0.5; } return v; }',
        'void main() {',
        '  vec2 uv = vUv;',
        '  float n = fbm(vec2(uv.x * 4.5, uv.y * 3.8 - uTime * 3.8));',
        '  float n2 = fbm(vec2(uv.x * 8.2 + 3.1, uv.y * 6.8 - uTime * 5.6));',

        '  if (alpha < 0.01) discard;',
        '  vec3 blueCore = vec3(0.32, 0.65, 1.0); vec3 whiteHot = vec3(1.0, 0.96, 0.82); vec3 gold = vec3(1.0, 0.62, 0.10); vec3 emberRed = vec3(0.90, 0.16, 0.02);',

      if (!hp) continue;
      const last = _prevHandMap.get(k);
      if (last) _tmpHVel.subVectors(hp, last).divideScalar(Math.max(0.005, dt));
      else _tmpHVel.set(0, 0, 0);
      _prevHandMap.set(k, hp.clone());

      const dist = hp.distanceTo(flameWorldPos);
      if (dist < 0.75) {
        const prox = 1.0 - dist / 0.75;
        boost += prox * 0.95;
        _tmpDir.subVectors(hp, flameWorldPos).setY(0);
        if (_tmpDir.lengthSq() > 0.0001) _tmpDir.normalize();
        wind.x += _tmpHVel.x * prox * 0.085 + _tmpDir.x * prox * 0.032;
        wind.z += _tmpHVel.z * prox * 0.085 + _tmpDir.z * prox * 0.032;
      }
    }
    return { wind, boost };
  }

        flame.uniforms.uBend.value.copy(localBend);
        flame.uniforms.uIntensity.value += (targetIntensity - flame.uniforms.uIntensity.value) * Math.min(1, delta * 10);

        fireLight.intensity = 1.9 * flame.uniforms.uIntensity.value;
        coalMat.emissiveIntensity = 1.8 + flame.uniforms.uIntensity.value * 0.95;

        const arr = sparkGeo.attributes.position.array;
        for (let i = 0; i < sparkCount; i++) {
          const sv = sparkVel[i];
          arr[i * 3] += (sv.x + localBend.x * 2.6 + Math.sin(t * 7 + i) * 0.03) * delta;
          arr[i * 3 + 1] += sv.y * flame.uniforms.uIntensity.value * delta;
          arr[i * 3 + 2] += (sv.z + localBend.z * 2.6 + Math.cos(t * 6 + i) * 0.03) * delta;

    APP.bot?.onEntityCreated?.(entity);

  function spawnHandFire(opts = {}) {
    if (APP._handFireEnt && APP.entities.includes(APP._handFireEnt)) {
      if (opts.toggle !== false && opts.active === false) {

    const sLife = new Float32Array(sparkCount);

    }));
    sPts.frustumCulled = false;
    handFireGroup.add(sPts);

    APP.scene.add(handFireGroup);
    const handPref = String(opts.hand || '').toLowerCase();
    const prevPos = new THREE.Vector3(), hVel = new THREE.Vector3(), curBend = new THREE.Vector3(), tmpBox = new THREE.Box3();
    let hasPrev = false, touchTimer = 0, igniteCooldown = 0;

    const ent = APP.registerEntity(handFireGroup, {
      name: 'Virtual Hand Fire',
      category: 'world_fx',
      grabbable: false,
      physics: { enabled: false },
      onUpdate: function (dt) {
        const delta = Math.max(0.004, Math.min(0.04, dt || 0.016));
        const t = performance.now() * 0.001;
        flame.uniforms.uTime.value = t;
        igniteCooldown = Math.max(0, igniteCooldown - delta);

        const hands = APP.hands || {};
        const pick = (handPref && hands[handPref]?.active) ? hands[handPref]
          : (hands.right?.active ? hands.right : (hands.left?.active ? hands.left : hands.dhand));
        const hp = getHandWorldPoint(pick);

        if (!hp) { handFireGroup.visible = false; return; }
        handFireGroup.visible = true;
        handFireGroup.position.copy(hp);

        if (hasPrev) hVel.subVectors(hp, prevPos).divideScalar(delta);
        else { hVel.set(0, 0, 0); hasPrev = true; }
        prevPos.copy(hp);

        const spd = hVel.length();
        const targetBend = new THREE.Vector3(-hVel.x * 0.075, 0, -hVel.z * 0.075).clampLength(0, 0.22);
        curBend.lerp(targetBend, Math.min(1, delta * 14));
        flame.uniforms.uBend.value.copy(curBend);
        const targetInt = 1.05 + Math.min(1.4, spd * 0.45) + Math.sin(t * 16) * 0.1;
        flame.uniforms.uIntensity.value += (targetInt - flame.uniforms.uIntensity.value) * Math.min(1, delta * 12);
        hLight.intensity = 1.6 * flame.uniforms.uIntensity.value;

        const spawnCnt = spd > 0.4 ? 3 : 1;
        for (let k = 0; k < spawnCnt; k++) {
          const idx = nextS; nextS = (nextS + 1) % sparkCount;
          sPos[idx * 3] = (Math.random() - 0.5) * 0.05;

  function burnObjectPart(opts = {}) {
    const q = String(opts.target || 'latest').trim();
    let target = q ? APP.findEntity?.(q) : null;
    if (!target || target.category === 'world_fx' || target.category === 'board' || target.category === 'portal') {
      const cands = (APP.entities || []).filter((e) => e && e.group?.parent && e.category !== 'world_fx' && e.category !== 'board' && e.category !== 'portal' && !e.isPen);
      target = cands[cands.length - 1] || null;
    }
    if (!target) return { ok: false, error: 'No target object found to burn.' };

    if (target.category === 'cloth' && APP.sim?.burnCloth) {
      return APP.sim.burnCloth({ origin: opts.part || 'bottom' });
    }

    target.group.updateMatrixWorld(true);
    const box = new THREE.Box3().setFromObject(target.group);
    if (box.isEmpty()) return { ok: false, error: 'Target bounds empty.' };
    const size = new THREE.Vector3(), center = new THREE.Vector3();
    box.getSize(size); box.getCenter(center);

    const part = String(opts.part || 'top').toLowerCase();
    const burnCenter = center.clone();
    if (Array.isArray(opts.point) && opts.point.length === 3) {
      burnCenter.set(opts.point[0], opts.point[1], opts.point[2]);
    } else if (part === 'top') {
      burnCenter.y = box.max.y - size.y * 0.14;
    } else if (part === 'bottom' || part === 'base') {
      burnCenter.y = box.min.y + size.y * 0.14;
    } else if (part === 'left') {
      burnCenter.x = box.min.x + size.x * 0.18;
    } else if (part === 'right') {
      burnCenter.x = box.max.x - size.x * 0.18;
    } else if (part === 'front') {
      burnCenter.z = box.max.z - size.z * 0.18;
    }

    const burnRadius = part === 'all' ? Math.max(size.x, size.y, size.z) * 0.85 : Math.max(0.14, Math.min(size.x, size.y, size.z) * 0.55);

    const localFlames = [];
    const anchorPts = [];
    const vTmp = new THREE.Vector3();
    target.group.traverse((c) => {
      if (!c.isMesh || !c.geometry?.attributes?.position || c.userData?.nxrDecal) return;
      const posA = c.geometry.attributes.position;
      const step = Math.max(1, Math.floor(posA.count / 90));
      for (let i = 0; i < posA.count; i += step) {
        vTmp.fromBufferAttribute(posA, i).applyMatrix4(c.matrixWorld);
        if (vTmp.distanceTo(burnCenter) <= burnRadius) {
          anchorPts.push(target.group.worldToLocal(vTmp.clone()));
        }
      }
      if (c.material && !c.material.isShaderMaterial) {
        const mats = Array.isArray(c.material) ? c.material : [c.material];
        c.material = mats.map((m) => {
          const cl = m.clone();
          const uTime = { value: 0 };
          const localBurnC = target.group.worldToLocal(burnCenter.clone());
          cl.userData.nxrBurnTime = uTime;
          cl.customProgramCacheKey = () => 'nxr_part_burn_v1';
          cl.onBeforeCompile = (shader) => {
            shader.uniforms.uPBTime = uTime;
            shader.uniforms.uPBCenter = { value: localBurnC };
            shader.uniforms.uPBRad = { value: burnRadius };
            shader.vertexShader = shader.vertexShader
              .replace('#include <common>', '#include <common>\nvarying vec3 vPBPos;')
              .replace('#include <begin_vertex>', '#include <begin_vertex>\nvPBPos = position;');
            shader.fragmentShader = shader.fragmentShader
              .replace('#include <common>', '#include <common>\nuniform float uPBTime; uniform vec3 uPBCenter; uniform float uPBRad;\nvarying vec3 vPBPos;\nfloat pbHash(vec3 p){p=fract(p*0.3183+0.1);p*=17.0;return fract(p.x*p.y*p.z*(p.x+p.y+p.z));}')
              .replace('#include <map_fragment>', '#include <map_fragment>\nfloat pbD=1.0-smoothstep(uPBRad*0.35,uPBRad*1.05,distance(vPBPos,uPBCenter));\ndiffuseColor.rgb=mix(diffuseColor.rgb,vec3(0.03,0.02,0.02),pbD*0.92);')
              .replace('#include <emissivemap_fragment>', '#include <emissivemap_fragment>\nfloat pbCrk=step(0.62,pbHash(floor(vPBPos*42.0)));\nfloat pbFlick=0.65+0.35*sin(uPBTime*18.0+vPBPos.y*25.0);\ntotalEmissiveRadiance+=vec3(1.0,0.28,0.02)*pbD*pbCrk*pbFlick*3.2;');
          };
          cl.needsUpdate = true;
          return cl;
        });
        if (!Array.isArray(c.material)) c.material = c.material[0];
      }
    });

    if (!anchorPts.length) anchorPts.push(target.group.worldToLocal(burnCenter.clone()));

    const fireOverlay = new THREE.Group();
    fireOverlay.name = `Burn_${part}_${target.name}`;
    const clusterCount = Math.min(6, Math.max(2, anchorPts.length));
    for (let i = 0; i < clusterCount; i++) {
      const p = anchorPts[Math.floor((i / clusterCount) * anchorPts.length)];
      const fSize = Math.max(0.09, Math.min(0.24, burnRadius * 0.72));
      const cl = createFlameCluster(fSize, fSize * 1.65, 0);
      cl.group.position.copy(p);
      fireOverlay.add(cl.group);
      localFlames.push(cl);
    }

    const burnLight = new THREE.PointLight(0xff6b1a, 2.0, 4.2);
    burnLight.position.copy(target.group.worldToLocal(burnCenter.clone())).add(new THREE.Vector3(0, 0.12, 0));
    fireOverlay.add(burnLight);

    target.group.add(fireOverlay);
    const curBend = new THREE.Vector3(), fWorld = new THREE.Vector3();

    const fxEnt = APP.registerEntity(new THREE.Group(), {
      name: `Fire on ${target.name} (${part})`,
      category: 'world_fx',
      grabbable: false,
      physics: { enabled: false },
      onUpdate: function (dt) {
        if (!target.group?.parent || !fireOverlay.parent) {
          APP.removeEntity?.(fxEnt, false);
          return;
        }
        const delta = Math.max(0.004, Math.min(0.04, dt || 0.016));
        const t = performance.now() * 0.001;
        fireOverlay.getWorldPosition(fWorld);
        const hw = sampleHandWindForce(fWorld, delta);
        const targetBend = hw.wind.clampLength(0, 0.18);
        curBend.lerp(targetBend, Math.min(1, delta * 10));
        const localBend = curBend.clone().applyQuaternion(target.group.quaternion.clone().invert());

        for (let i = 0; i < localFlames.length; i++) {
          localFlames[i].uniforms.uTime.value = t + i * 0.7;
          localFlames[i].uniforms.uBend.value.copy(localBend);
          localFlames[i].uniforms.uIntensity.value = 1.0 + hw.boost + Math.sin(t * 14 + i) * 0.1;
        }
        burnLight.intensity = 1.7 + hw.boost * 1.4 + Math.sin(t * 17) * 0.25;
        target.group.traverse((c) => {
          if (c.isMesh && c.material?.userData?.nxrBurnTime) c.material.userData.nxrBurnTime.value = t;
        });
      }
    });

    return { ok: true, id: fxEnt.id, message: `Ignited ${part} part of ${target.name}.` };
  }

  function extinguishAll() {
    let cnt = 0;
    for (const e of [...(APP.entities || [])]) {
      if (e && (e.name?.startsWith('Fire on ') || e === APP._handFireEnt)) {
        APP.removeEntity?.(e, false);
        cnt++;
      }
    }
    APP._handFireEnt = null;
    return { ok: true, message: cnt ? `Extinguished ${cnt} fire effect(s).` : 'No active fire effects to extinguish.' };
  }

      if (f === 'object' || f === 'part' || f === 'object_part') return burnObjectPart(opts);
      if (f === 'campfire' || f === 'bonfire') return spawnCampfire();
      return spawnTorch();
    }
  };

    if (low.includes('campfire') || low.includes('bonfire') || low.includes('مخيم')) return spawnCampfire();

      const rimIndices = [];
      for (let i = 0; i < SEGMENTS; i++) {
        const a = i * 2, b = i * 2 + 1, c = (i + 1) * 2, d = (i + 1) * 2 + 1;
        rimIndices.push(a, b, c, b, d, c);
      }
      for (let i = 0; i <= SEGMENTS; i++) {
        const u = i / SEGMENTS;
        rimUvs[i * 4] = u; rimUvs[i * 4 + 1] = 0; rimUvs[i * 4 + 2] = u; rimUvs[i * 4 + 3] = 1;
      }
      rimGeo.setAttribute('position', new THREE.BufferAttribute(rimVerts, 3));
      rimGeo.setAttribute('uv', new THREE.BufferAttribute(rimUvs, 2));
      rimGeo.setIndex(rimIndices);

      const rimUniforms = { uTime: { value: 0 } };
      const rimMesh = new THREE.Mesh(rimGeo, new THREE.ShaderMaterial({
        uniforms: rimUniforms,
        transparent: true, depthWrite: false, side: THREE.DoubleSide, blending: THREE.AdditiveBlending,
        vertexShader: 'varying vec2 vUv; void main() { vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0); }',
        fragmentShader: [
          'uniform float uTime; varying vec2 vUv;',
          'void main() {',
          '  float core = exp(-pow((vUv.y - 0.5) * 5.0, 2.0));',
          '  float wave = 0.5 + 0.5 * sin(vUv.x * 25.1327 - uTime * 4.5);',
          '  float wave2 = 0.5 + 0.5 * cos(vUv.x * 43.9822 + uTime * 3.1);',
          '  vec3 col = mix(vec3(0.0, 0.96, 0.72), vec3(0.22, 0.74, 0.98), wave);',
          '  col = mix(col, vec3(0.75, 0.45, 1.0), wave2 * 0.45);',
          '  col = mix(col, vec3(0.92, 1.0, 0.98), pow(core, 2.4));',
          '  gl_FragColor = vec4(col * 1.35, core * (0.75 + 0.25 * wave));',
          '}'
        ].join('\n')
      }));
      portalGroup.add(rimMesh);

      updateDancyAperture = function (t) {
        const mPos = maskGeo.attributes.position.array;
        const rPos = rimGeo.attributes.position.array;
        mPos[0] = 0; mPos[1] = 0; mPos[2] = 0;
        for (let i = 0; i <= SEGMENTS; i++) {
          const theta = (i / SEGMENTS) * Math.PI * 2;
          const w = 1.0 + 0.085 * Math.sin(3.0 * theta + t * 2.6) + 0.055 * Math.cos(5.0 * theta - t * 3.4) + 0.030 * Math.sin(7.0 * theta + t * 4.8);
          const x = Math.cos(theta) * BASE_RX * w, y = Math.sin(theta) * BASE_RY * w;
          const zWave = Math.sin(4.0 * theta + t * 2.2) * 0.025;
          const mIdx = (i + 1) * 3;
          mPos[mIdx] = x; mPos[mIdx + 1] = y; mPos[mIdx + 2] = 0;
          const rIdx = i * 6;
          rPos[rIdx] = x * 0.94; rPos[rIdx + 1] = y * 0.94; rPos[rIdx + 2] = zWave;
          rPos[rIdx + 3] = x * 1.12; rPos[rIdx + 4] = y * 1.12; rPos[rIdx + 5] = zWave;
        }
        maskGeo.attributes.position.needsUpdate = true;
        rimGeo.attributes.position.needsUpdate = true;
        rimUniforms.uTime.value = t;
      };
    }

    // Per-Eye Texture Swap in onBeforeRender (Zero RenderTarget switching mid-pass!)
    portalSurfaceMesh.onBeforeRender = function (renderer, sc, activeCam) {
      const xrCam = (renderer.xr?.isPresenting && typeof renderer.xr.getCamera === 'function') ? renderer.xr.getCamera() : null;
      const isRight = !!(xrCam && xrCam.cameras && xrCam.cameras.length > 1 && activeCam === xrCam.cameras[1]);
      portalUniforms.uPortalTex.value = isRight ? portalRT_R.texture : portalRT_L.texture;
    };

    const portalLight = new THREE.PointLight(isDoor ? 0xffe4b5 : 0x00f0b4, 1.4, 3.8);
    portalLight.position.set(0, isDoor ? 1.1 : 0, 0.25);
    portalGroup.add(portalLight);

    const portalCam = new THREE.PerspectiveCamera(65, 1, 0.05, 40);
    portalCam.matrixAutoUpdate = false;

    function renderEyeCamera(r, eyeCam, targetRT) {
      eyeCam.updateMatrixWorld(true);
      portalCam.matrixWorld.copy(eyeCam.matrixWorld);
      portalCam.matrixWorldInverse.copy(eyeCam.matrixWorldInverse);
      portalCam.projectionMatrix.copy(eyeCam.projectionMatrix);
      portalCam.projectionMatrixInverse.copy(eyeCam.projectionMatrixInverse);
      r.setRenderTarget(targetRT);
      r.clear();
      r.render(portalScene, portalCam);
    }

    const ent = APP.registerEntity(portalGroup, {
      name: (isDoor ? 'Door Portal to ' : 'Portal to ') + (asset.name || 'World'),
      category: 'portal',
      holdable: !isDoor,
      physics: { gravity: false, enabled: false },
      onUpdate: function (dt) {
        if (!portalGroup.parent) {
          portalRT_L.dispose();
          portalRT_R.dispose();
          if (woodTex) woodTex.dispose();
          return;
        }
        const t = performance.now() * 0.001;
        portalUniforms.uTime.value = t;
        if (updateDancyAperture) updateDancyAperture(t);
        if (doorAnimFn) doorAnimFn(Math.min(0.04, dt || 0.016));

        worldGroup.children.forEach((c) => c.userData?.updateAnim?.(t));

        portalGroup.updateMatrixWorld(true);
        worldAnchor.position.copy(portalGroup.position);
        worldAnchor.quaternion.copy(portalGroup.quaternion);
        worldAnchor.scale.copy(portalGroup.scale);

        const r = APP.renderer;
        const mainCam = APP.camera;
        if (!r || !mainCam) return;

        const xrCam = (r.xr?.isPresenting && typeof r.xr.getCamera === 'function') ? r.xr.getCamera() : null;
        const leftCam = (xrCam && xrCam.cameras && xrCam.cameras[0]) ? xrCam.cameras[0] : mainCam;
        const rightCam = (xrCam && xrCam.cameras && xrCam.cameras[1]) ? xrCam.cameras[1] : null;

        const prevRT = r.getRenderTarget();
        const prevXR = r.xr ? r.xr.enabled : false;
        const prevAutoClear = r.autoClear;
        try {
          if (r.xr) r.xr.enabled = false;
          r.autoClear = true;
          renderEyeCamera(r, leftCam, portalRT_L);
          if (rightCam) renderEyeCamera(r, rightCam, portalRT_R);
        } finally {
          r.autoClear = prevAutoClear;
          r.setRenderTarget(prevRT);
          if (r.xr) r.xr.enabled = prevXR;
        }
      }
    });

    if (updateDancyAperture) updateDancyAperture(performance.now() * 0.001);
    APP.emit?.('entity:created', ent);
    APP.bot?.onEntityCreated?.(ent);

    return { ok: true, id: ent.id, style, message: `Opened stereo-accurate ${isDoor ? '3D Door' : 'Dancy Vortex'} portal to "${asset.name}".` };
  }

  const PORTAL_TOOL = {
    name: 'open_portal',
    description: 'Opens a stereo-accurate 3D parallax portal window into any world (space, forest, ocean, cyber/city, lava/volcano, desert, snow, or loaded GLB/image). Randomly picks between a dancing plasma vortex ("dancy") or a realistic swinging oak doorway ("door"), or specify `style`.',
    parameters: {
      type: 'object',
      properties: {
        world: { type: 'string', description: 'World name or theme e.g. "space", "forest", "ocean", "cyber", "lava", "desert", "snow", or model/image name.' },
        style: { type: 'string', description: 'Optional: "dancy" | "door" (omit to pick randomly).' }
      }
    }
  };
  APP.registerBotTool?.(PORTAL_TOOL, (a) => buildPortal(a?.world || '', a?.style));

  APP.openPortal = buildPortal;
  return await buildPortal(typeof command === 'string' ? command : '');
})();
