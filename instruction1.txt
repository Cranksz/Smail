@autorun
@pin: Drawing Gravity
@desc: Draw metallic reflective 3D tube strokes. Toggling Drawing Gravity applies ONLY to next drawings so dynamic drawings collide with static ones.
@match: drawing & gravity | draw & gravity | toggle & gravity | gravity & on | gravity & off | start drawing | stop drawing | جاذبية & الرسم
@code:
(function () {
  const THREE = APP.THREE;
  if (!THREE || !APP.scene) return;

  if (typeof APP._inst1DrawCleanup === 'function') {
    try { APP._inst1DrawCleanup(); } catch (_) {}
  }

  // Override so toggling Drawing Gravity ONLY affects FUTURE drawings
  APP.setDrawingGravity = function setDrawingGravityOnlyNext(val) {
    APP.drawingGravity = !!val;
    APP.toolbar?.syncState?.();
    const coreGravBtn = document.getElementById('core-grav-btn');
    if (coreGravBtn) {
      coreGravBtn.textContent = 'Draw Gravity: ' + (APP.drawingGravity ? 'ON' : 'OFF');
      coreGravBtn.classList.toggle('active', !!APP.drawingGravity);
    }
    APP.emit?.('drawing:gravity_changed', APP.drawingGravity);
    return APP.drawingGravity;
  };

  const STROKE_RADIUS = 0.0095;
  const COLLISION_DIST = STROKE_RADIUS * 2.25;
  // Sleek anodized metallic chrome tints
  const METAL_PALETTE = [0x6ef3d6, 0x7dd3fc, 0xf9a8d4, 0xfde047, 0xc4b5fd, 0xe2e8f0];

  let isDrawing = false;
  let currentPoints = [];
  let previewMesh = null;
  let drawColorIndex = 0;

  const tmpNibPos = new THREE.Vector3();
  const tmpClosest = new THREE.Vector3();
  const tmpSegA = new THREE.Vector3();
  const tmpSegB = new THREE.Vector3();
  const tmpPtW = new THREE.Vector3();
  const tmpDiff = new THREE.Vector3();
  const tmpNormal = new THREE.Vector3();
  const tmpBoxDyn = new THREE.Box3();
  const tmpBoxStat = new THREE.Box3();

  function cancelActiveStroke() {
    isDrawing = false;
    currentPoints = [];
    if (previewMesh) {
      APP.scene.remove(previewMesh);
      if (previewMesh.geometry) previewMesh.geometry.dispose();
      if (previewMesh.material) previewMesh.material.dispose();
      previewMesh = null;
    }
  }

  function isPenAliveAndHeld() {
    const pen = APP._activePen;
    if (!pen) return false;
    const stillExists = Array.isArray(APP.entities) && APP.entities.includes(pen);
    const inScene = !!(pen.group && pen.group.parent);
    if (!stillExists || !inScene) {
      APP._activePen = null;
      APP.drawingEnabled = false;
      APP.desktopDrawActive = false;
      cancelActiveStroke();
      return false;
    }
    return !!pen.isHeld;
  }

  function getStrokeDrawPoint() {
    const hands = APP.hands || {};
    if (APP._activePen && !isPenAliveAndHeld()) return null;

    if (APP._activePen && APP._activePen.nibMesh && APP._activePen.isHeld) {
      const pen = APP._activePen;
      const h = pen.attachedHandKey ? hands[pen.attachedHandKey] : pen.heldBy;
      const activeDown = h
        ? !!(h.trigger || h.pinching)
        : Object.values(hands).some((x) => x && x.active && (x.trigger || x.pinching));
      if (activeDown) {
        pen.nibMesh.getWorldPosition(tmpNibPos);
        return tmpNibPos.clone();
      }
      return null;
    }

    if (!APP.drawingEnabled) return null;
    for (const k of Object.keys(hands)) {
      const h = hands[k];
      if (!h || !h.active) continue;
      if (k !== 'dhand' && (h.pinching || h.trigger)) {
        if (h.isHand && h.fingerTips?.index) return h.fingerTips.index.clone();
        if (h.rayOrigin && h.rayDir) return h.rayOrigin.clone().addScaledVector(h.rayDir, 0.45);
      }
      if (k === 'dhand' && h.trigger && h.rayOrigin && h.rayDir) {
        return h.rayOrigin.clone().addScaledVector(h.rayDir, 0.72);
      }
    }
    return null;
  }

  function createMetallicReflectiveMaterial(colorHex) {
    const MatCls = THREE.MeshPhysicalMaterial || THREE.MeshStandardMaterial;
    const mat = new MatCls({
      color: colorHex,
      metalness: 0.96,
      roughness: 0.10,
      envMap: APP.cubeRenderTarget?.texture || APP.roomEnvTexture || null,
      envMapIntensity: 1.85,
      emissive: colorHex,
      emissiveIntensity: 0.10
    });
    if (THREE.MeshPhysicalMaterial && mat instanceof THREE.MeshPhysicalMaterial) {
      mat.clearcoat = 1.0;
      mat.clearcoatRoughness = 0.06;
      mat.reflectivity = 1.0;
    }
    return mat;
  }

  function buildTubeMesh(pts, colorHex) {
    const curve = new THREE.CatmullRomCurve3(pts);
    const tubularSegs = Math.min(220, Math.max(8, pts.length * 3));
    const geo = new THREE.TubeGeometry(curve, tubularSegs, STROKE_RADIUS, 12, false);
    const mat = createMetallicReflectiveMaterial(colorHex);
    const mesh = new THREE.Mesh(geo, mat);
    mesh.castShadow = true;
    mesh.receiveShadow = true;
    return mesh;
  }

  function rebuildPreview(colorHex) {
    if (previewMesh) {
      APP.scene.remove(previewMesh);
      if (previewMesh.geometry) previewMesh.geometry.dispose();
      if (previewMesh.material) previewMesh.material.dispose();
      previewMesh = null;
    }
    if (currentPoints.length < 2) return;
    previewMesh = buildTubeMesh(currentPoints, colorHex);
    APP.scene.add(previewMesh);
  }

  function downsamplePoints(pts, minStep) {
    if (pts.length <= 2) return pts.map((p) => p.clone());
    const out = [pts[0].clone()];
    for (let i = 1; i < pts.length - 1; i++) {
      if (pts[i].distanceTo(out[out.length - 1]) >= minStep) out.push(pts[i].clone());
    }
    out.push(pts[pts.length - 1].clone());
    return out;
  }

  function finalizeStroke(colorHex) {
    if (previewMesh) {
      APP.scene.remove(previewMesh);
      if (previewMesh.geometry) previewMesh.geometry.dispose();
      if (previewMesh.material) previewMesh.material.dispose();
      previewMesh = null;
    }
    if (currentPoints.length < 2) {
      currentPoints = [];
      return;
    }

    const rawMesh = buildTubeMesh(currentPoints, colorHex);
    rawMesh.geometry.computeBoundingBox();
    const box = rawMesh.geometry.boundingBox;
    const center = new THREE.Vector3();
    box.getCenter(center);
    rawMesh.geometry.translate(-center.x, -center.y, -center.z);

    const localPts = downsamplePoints(currentPoints, 0.022).map((p) => p.sub(center));

    const strokeGroup = new THREE.Group();
    strokeGroup.name = 'Metallic_3D_Stroke';
    strokeGroup.position.copy(center);
    strokeGroup.add(rawMesh);
    APP.scene.add(strokeGroup);

    const isDynamic = !!APP.drawingGravity;

    const ent = APP.registerEntity(strokeGroup, {
      name: isDynamic ? 'Dynamic Metallic Stroke' : 'Static Metallic Stroke',
      category: 'stroke',
      isStroke: true,
      hasGravity: isDynamic,
      restitution: 0.44,
      velocity: new THREE.Vector3(0, 0, 0),
      meta: {
        isDynamicStroke: isDynamic,
        isStaticStroke: !isDynamic,
        localPoints: localPts,
        tubeRadius: STROKE_RADIUS
      }
    });

    currentPoints = [];
    drawColorIndex = (drawColorIndex + 1) % METAL_PALETTE.length;
    return ent;
  }

  function closestPointOnSegment(p, a, b, out) {
    const abX = b.x - a.x, abY = b.y - a.y, abZ = b.z - a.z;
    const lenSq = abX * abX + abY * abY + abZ * abZ;
    if (lenSq < 1e-8) return out.copy(a);
    let t = ((p.x - a.x) * abX + (p.y - a.y) * abY + (p.z - a.z) * abZ) / lenSq;
    if (t < 0) t = 0;
    else if (t > 1) t = 1;
    return out.set(a.x + abX * t, a.y + abY * t, a.z + abZ * t);
  }

  function stepStrokeCollisions() {
    const entities = APP.entities || [];
    const dynStrokes = [];
    const statStrokes = [];

    for (let i = 0; i < entities.length; i++) {
      const e = entities[i];
      if (!e || !e.isStroke || !e.group || !e.group.parent) continue;
      if (e.physics?.gravity && !e.isHeld && !e.heldByBot) dynStrokes.push(e);
      else if (e.physics && !e.physics.gravity) statStrokes.push(e);
    }

    if (!dynStrokes.length) return;

    for (let d = 0; d < dynStrokes.length; d++) {
      const dyn = dynStrokes[d];
      dyn.group.updateMatrixWorld(true);
      tmpBoxDyn.setFromObject(dyn.group);

      if (!tmpBoxDyn.isEmpty() && tmpBoxDyn.min.y < 0) {
        dyn.group.position.y += -tmpBoxDyn.min.y;
        if (dyn.physics.vel.y < 0) {
          dyn.physics.vel.y = Math.abs(dyn.physics.vel.y) > 0.25 ? -dyn.physics.vel.y * dyn.physics.bounce : 0;
        }
        dyn.physics.vel.x *= dyn.physics.friction;
        dyn.physics.vel.z *= dyn.physics.friction;
        dyn.group.updateMatrixWorld(true);
        tmpBoxDyn.setFromObject(dyn.group);
      }

      if (!statStrokes.length) continue;
      tmpBoxDyn.expandByScalar(COLLISION_DIST);
      const dynPts = dyn.meta?.localPoints;
      if (!dynPts || !dynPts.length) continue;

      for (let s = 0; s < statStrokes.length; s++) {
        const stat = statStrokes[s];
        const statPts = stat.meta?.localPoints;
        if (!statPts || statPts.length < 2) continue;

        stat.group.updateMatrixWorld(true);
        tmpBoxStat.setFromObject(stat.group).expandByScalar(COLLISION_DIST);
        if (!tmpBoxDyn.intersectsBox(tmpBoxStat)) continue;

        const dynMat = dyn.group.matrixWorld;
        const statMat = stat.group.matrixWorld;
        let deepestPen = 0;
        let hitNX = 0, hitNY = 0, hitNZ = 0;

        for (let pi = 0; pi < dynPts.length; pi++) {
          tmpPtW.copy(dynPts[pi]).applyMatrix4(dynMat);
          for (let si = 0; si < statPts.length - 1; si++) {
            tmpSegA.copy(statPts[si]).applyMatrix4(statMat);
            tmpSegB.copy(statPts[si + 1]).applyMatrix4(statMat);
            closestPointOnSegment(tmpPtW, tmpSegA, tmpSegB, tmpClosest);
            tmpDiff.subVectors(tmpPtW, tmpClosest);
            const distSq = tmpDiff.lengthSq();
            if (distSq < COLLISION_DIST * COLLISION_DIST) {
              const dist = Math.sqrt(distSq);
              const pen = COLLISION_DIST - dist;
              if (pen > deepestPen) {
                deepestPen = pen;
                if (dist > 1e-5) tmpNormal.copy(tmpDiff).multiplyScalar(1 / dist);
                else tmpNormal.set(0, 1, 0);
                hitNX = tmpNormal.x; hitNY = tmpNormal.y; hitNZ = tmpNormal.z;
              }
            }
          }
        }

        if (deepestPen > 0) {
          dyn.group.position.x += hitNX * deepestPen;
          dyn.group.position.y += hitNY * deepestPen;
          dyn.group.position.z += hitNZ * deepestPen;

          const v = dyn.physics.vel;
          const vn = v.x * hitNX + v.y * hitNY + v.z * hitNZ;
          if (vn < 0) {
            const bounce = Math.abs(vn) > 0.30 ? dyn.physics.bounce : 0.0;
            const imp = -(1 + bounce) * vn;
            v.x += hitNX * imp;
            v.y += hitNY * imp;
            v.z += hitNZ * imp;
            v.x *= 0.94;
            v.z *= 0.94;
            if (hitNY > 0.65 && Math.abs(v.y) < 0.06) v.y = 0;
          }
          dyn.group.updateMatrixWorld(true);
          tmpBoxDyn.setFromObject(dyn.group).expandByScalar(COLLISION_DIST);
        }
      }
    }
  }

  function update() {
    if ((isDrawing || APP._activePen) && APP.entities.length === 0) {
      APP._activePen = null;
      APP.drawingEnabled = false;
      APP.desktopDrawActive = false;
      cancelActiveStroke();
    }

    const pt = getStrokeDrawPoint();
    if (pt) {
      if (!isDrawing) {
        isDrawing = true;
        currentPoints = [pt];
      } else {
        const last = currentPoints[currentPoints.length - 1];
        if (last.distanceTo(pt) >= 0.015) {
          currentPoints.push(pt);
          rebuildPreview(METAL_PALETTE[drawColorIndex]);
        }
      }
    } else if (isDrawing) {
      isDrawing = false;
      finalizeStroke(METAL_PALETTE[drawColorIndex]);
    }

    stepStrokeCollisions();
  }

  const unsubFrame = APP.onUpdate ? APP.onUpdate(update) : null;
  if (!unsubFrame && APP.frameHooks) APP.frameHooks.add(update);

  APP._inst1DrawCleanup = function () {
    if (typeof unsubFrame === 'function') unsubFrame();
    else APP.frameHooks?.delete(update);
    cancelActiveStroke();
  };

  APP.registerAbilityHandler?.('spatial_drawing_gravity', function (text) {
    const low = String(text || '').toLowerCase();
    if (low.includes('start drawing')) { APP.drawingEnabled = true; return { ok: true, message: '3D drawing enabled.' }; }
    if (low.includes('stop drawing')) { APP.drawingEnabled = false; cancelActiveStroke(); return { ok: true, message: '3D drawing paused.' }; }
    let next = !APP.drawingGravity;
    if (low.includes('on') || low.includes('تشغيل')) next = true;
    else if (low.includes('off') || low.includes('ايقاف') || low.includes('إيقاف')) next = false;
    APP.setDrawingGravity(next);
    return { ok: true, message: 'Drawing Gravity is now ' + (next ? 'ON (applies to next drawings)' : 'OFF (next drawings stay static)') + '.' };
  });
})();
@end

(function () {
  const THREE = APP.THREE;
  if (!THREE || !APP.scene) return { ok: false, error: 'Scene not ready.' };

  const torchGroup = new THREE.Group();
  torchGroup.name = 'Realistic_Fire_Torch';

  const woodMat = new THREE.MeshStandardMaterial({ color: 0x3b2314, roughness: 0.88, metalness: 0.05 });
  const handle = new THREE.Mesh(new THREE.CylinderGeometry(0.022, 0.015, 0.44, 18), woodMat);
  handle.position.y = 0.22;
  torchGroup.add(handle);

  const ironMat = new THREE.MeshStandardMaterial({ color: 0x1f242d, roughness: 0.36, metalness: 0.88 });
  const collar = new THREE.Mesh(new THREE.CylinderGeometry(0.044, 0.025, 0.075, 18), ironMat);
  collar.position.y = 0.45;
  torchGroup.add(collar);

  // Forged iron cage prongs around coal bed
  for (let i = 0; i < 6; i++) {
    const a = (i / 6) * Math.PI * 2;
    const rib = new THREE.Mesh(new THREE.BoxGeometry(0.006, 0.07, 0.006), ironMat);
    rib.position.set(Math.cos(a) * 0.042, 0.49, Math.sin(a) * 0.042);
    rib.rotation.z = -Math.cos(a) * 0.18;
    rib.rotation.x = Math.sin(a) * 0.18;
    torchGroup.add(rib);
  }

  const coalMat = new THREE.MeshStandardMaterial({
    color: 0x1a0802,
    emissive: 0xff3b00,
    emissiveIntensity: 2.4,
    roughness: 0.75
  });
  const coals = new THREE.Mesh(new THREE.SphereGeometry(0.036, 16, 14), coalMat);
  coals.scale.set(1, 0.6, 1);
  coals.position.y = 0.485;
  torchGroup.add(coals);

  // Realistic GLSL FBM Turbulent Volumetric Fire Shader
  const fireUniforms = {
    uTime: { value: 0.0 },
    uBend: { value: new THREE.Vector3(0, 0, 0) },
    uIntensity: { value: 1.0 }
  };

  const fireMat = new THREE.ShaderMaterial({
    uniforms: fireUniforms,
    transparent: true,
    depthWrite: false,
    side: THREE.DoubleSide,
    blending: THREE.AdditiveBlending,
    vertexShader: [
      'uniform float uTime;',
      'uniform vec3 uBend;',
      'uniform float uIntensity;',
      'varying vec2 vUv;',
      'varying float vHeight;',
      'void main() {',
      '  vUv = uv;',
      '  vHeight = clamp((position.y + 0.12) / 0.24, 0.0, 1.0);',
      '  vec3 p = position;',
      '  float h2 = vHeight * vHeight;',
      '  p.xz *= (1.0 - vHeight * 0.48) * mix(0.85, 1.25, clamp(uIntensity - 0.5, 0.0, 1.0));',
      '  p.y *= mix(0.9, 1.45, clamp((uIntensity - 0.8) * 0.6, 0.0, 1.0));',
      '  float sway = sin(uTime * 9.0 + position.y * 18.0) * 0.012 * h2;',
      '  p.x += uBend.x * h2 + sway;',
      '  p.z += uBend.z * h2 + cos(uTime * 7.5 + position.y * 15.0) * 0.012 * h2;',
      '  gl_Position = projectionMatrix * modelViewMatrix * vec4(p, 1.0);',
      '}'
    ].join('\n'),
    fragmentShader: [
      'uniform float uTime;',
      'uniform float uIntensity;',
      'varying vec2 vUv;',
      'varying float vHeight;',
      'float fHash(vec2 p) { return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453123); }',
      'float fNoise(vec2 p) {',
      '  vec2 i = floor(p); vec2 f = fract(p); f = f * f * (3.0 - 2.0 * f);',
      '  return mix(mix(fHash(i), fHash(i + vec2(1.0, 0.0)), f.x), mix(fHash(i + vec2(0.0, 1.0)), fHash(i + vec2(1.0, 1.0)), f.x), f.y);',
      '}',
      'float fbm(vec2 p) {',
      '  float v = 0.0; float a = 0.52;',
      '  for (int i = 0; i < 4; i++) { v += a * fNoise(p); p *= 2.07; a *= 0.5; }',
      '  return v;',
      '}',
      'void main() {',
      '  vec2 uv = vUv;',

    portalGroup.add(portalLight);

    const ent = APP.registerEntity(portalGroup, {
      name: 'Portal to ' + (asset.name || 'World'),
      category: 'portal',
      holdable: false,
      physics: { gravity: false, enabled: false },
      onUpdate: function () {
        const t = performance.now() * 0.001;
        updateDancyAperture(t);

        // Keep clipping plane synced with portalGroup's exact world position & facing normal (-Z)
        portalGroup.updateMatrixWorld(true);
        const normal = new THREE.Vector3(0, 0, -1).applyQuaternion(portalGroup.quaternion).normalize();
        const origin = new THREE.Vector3();
        portalGroup.getWorldPosition(origin);
        worldClipPlane.setFromNormalAndCoplanarPoint(normal, origin);
      }
    });

    updateDancyAperture(performance.now() * 0.001);
    APP.emit?.('entity:created', ent);
    APP.bot?.onEntityCreated?.(ent);

    return { ok: true, message: `Opened dancing 3D portal to "${asset.name}" (${asset.kind}).` };
  }

  APP.openPortal = buildDancyPortal;
  return await buildDancyPortal(typeof command === 'string' ? command : '');
})();
