  function createSoftGlowTexture() {
    const c = document.createElement('canvas');
    c.width = 64;
    c.height = 64;
    const ctx = c.getContext('2d');
    const g = ctx.createRadialGradient(32, 32, 2, 32, 32, 30);
    g.addColorStop(0, 'rgba(255,255,255,1)');
    g.addColorStop(0.35, 'rgba(255,255,255,0.7)');
    g.addColorStop(1, 'rgba(255,255,255,0)');
    ctx.fillStyle = g;
    ctx.fillRect(0, 0, 64, 64);
    const tex = new THREE.CanvasTexture(c);
    tex.needsUpdate = true;
    return tex;
  }

  function createMatrixAtlasTexture() {
    const c = document.createElement('canvas');
    c.width = 256;
    c.height = 512;
    const ctx = c.getContext('2d');
    ctx.clearRect(0, 0, 256, 512);
    const glyphs = 'アイウエオカキクケコサシスセソタチツテト0123456789ABCDEFZ*+<>';
    const cols = 8;
    const rows = 16;
    const cw = 256 / cols;
    const ch = 512 / rows;
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.font = 'bold 22px monospace';

    for (let col = 0; col < cols; col++) {
      for (let row = 0; row < rows; row++) {
        const chIdx = Math.floor(Math.random() * glyphs.length);
        const alpha = (row + 1) / rows;
        const isHead = row === rows - 1;
        ctx.fillStyle = isHead
          ? 'rgba(210, 255, 225, 0.98)'
          : 'rgba(20, 255, 110, ' + (0.18 + alpha * 0.72).toFixed(2) + ')';
        if (isHead) {
          ctx.shadowColor = '#6effa8';
          ctx.shadowBlur = 10;
        } else {
          ctx.shadowBlur = 0;
        }
        ctx.fillText(glyphs[chIdx], col * cw + cw * 0.5, row * ch + ch * 0.5);
      }
    }
    const tex = new THREE.CanvasTexture(c);
    tex.wrapS = THREE.RepeatWrapping;
    tex.wrapT = THREE.RepeatWrapping;
    tex.needsUpdate = true;
    return tex;
  }

    ctx.lineWidth = 2;
    ctx.strokeRect(0, 0, 256, 256);
    ctx.strokeStyle = 'rgba(150, 170, 195, 0.14)';
    ctx.lineWidth = 1;
    for (let i = 64; i < 256; i += 64) {
      ctx.beginPath();
      ctx.moveTo(i, 0);
      ctx.lineTo(i, 256);
      ctx.stroke();
      ctx.beginPath();
      ctx.moveTo(0, i);
      ctx.lineTo(256, i);
      ctx.stroke();

    tex.wrapS = THREE.RepeatWrapping;
    tex.wrapT = THREE.RepeatWrapping;
    tex.repeat.set(36, 36);
    tex.anisotropy = 4;
    tex.needsUpdate = true;
    return tex;

  function attachXRLightEstimation(session) {
    if (!renderer || !renderer.xr || !THREE.XREstimatedLight) return;
    if (xrLight) return;
    try {
      xrLight = new THREE.XREstimatedLight(renderer);
      xrLight.addEventListener('estimationstart', function () {
        xrLightProbeActive = true;
        if (currentEnv === 'default' && APP.isAR) {
          scene.add(xrLight);
          if (xrLight.environment) {
            scene.environment = xrLight.environment;
          }
          if (APP.ambientLight) APP.ambientLight.intensity = 0.25;
          if (APP.dirLight) APP.dirLight.intensity = 0.35;
        }
      });
      xrLight.addEventListener('estimationend', function () {
        xrLightProbeActive = false;
        scene.remove(xrLight);
        if (APP.defaultEnvironmentTex) {
          scene.environment = APP.defaultEnvironmentTex;
        }
        if (APP.ambientLight) APP.ambientLight.intensity = 0.65;
        if (APP.dirLight) APP.dirLight.intensity = 1.25;
      });
    } catch (e) {
      // Device/browser may not support light-estimation
    }
  }

        scene.fog = null;

          scene.add(xrLight);

      } else {
        scene.background = new THREE.Color(0x070b13);
        scene.fog = new THREE.FogExp2(0x070b13, 0.045);
        if (renderer) renderer.setClearAlpha(1);
        studioGroup.visible = true;

      if (APP.dirLight) APP.dirLight.intensity = inAR && xrLightProbeActive ? 0.35 : 1.25;
    } else if (currentEnv === 'matrix_black') {
      if (xrLight && xrLight.parent === scene) scene.remove(xrLight);
      scene.background = new THREE.Color(0x000201);
      scene.fog = new THREE.FogExp2(0x000a03, 0.055);

      matrixGroup.visible = true;

  function setEnvironment(mode) {
    const clean = String(mode || 'default').toLowerCase().trim();
    if (ENV_MODES.includes(clean)) {
      currentEnv = clean;
    } else if (clean.includes('matrix')) {
      currentEnv = 'matrix_black';
    } else if (clean.includes('construct') || clean.includes('gray') || clean.includes('white')) {
      currentEnv = 'construct_gray';
    } else {
      currentEnv = 'default';
    }
    applyEnvironmentVisuals();
    if (typeof APP.emit === 'function') {
      try { APP.emit('visual:env_changed', currentEnv); } catch (_) {}
    }
    return currentEnv;
  }

  function cycleEnvironment() {
    const idx = ENV_MODES.indexOf(currentEnv);
    return setEnvironment(ENV_MODES[(idx + 1) % ENV_MODES.length]);
  }

  // Dissolve / Materialization Shader via onBeforeCompile
  // ====================================================================
  function patchMaterialWithDissolve(mat) {
    if (!mat || typeof mat.onBeforeCompile !== 'function') return null;
    if (mat.userData && mat.userData.dissolveUniforms) {
      return mat.userData.dissolveUniforms;
    }

    const uniforms = {
      uDissolve: { value: 0.0 },
      uDissolveMode: { value: 0.0 },
      uMinY: { value: 0.0 },
      uMaxY: { value: 1.0 },
      uTime: { value: 0.0 },
      uEdgeColor: { value: new THREE.Color(0x00ff88) }
    };

    const prevOnBeforeCompile = mat.onBeforeCompile;
    const prevCacheKey = typeof mat.customProgramCacheKey === 'function' ? mat.customProgramCacheKey() : '';

    mat.customProgramCacheKey = function () {
      return (prevCacheKey ? prevCacheKey + '|' : '') + 'wxrmr_dissolve_v1';
    };

    mat.onBeforeCompile = function (shader, r) {
      if (prevOnBeforeCompile && prevOnBeforeCompile !== mat.onBeforeCompile) {
        try { prevOnBeforeCompile.call(mat, shader, r); } catch (_) {}
      }

      shader.uniforms.uDissolve = uniforms.uDissolve;
      shader.uniforms.uDissolveMode = uniforms.uDissolveMode;
      shader.uniforms.uMinY = uniforms.uMinY;
      shader.uniforms.uMaxY = uniforms.uMaxY;
      shader.uniforms.uTime = uniforms.uTime;
      shader.uniforms.uEdgeColor = uniforms.uEdgeColor;

      // Vertex injection: pass world position safely across Standard/Physical/Basic
      shader.vertexShader = shader.vertexShader
        .replace(
          'void main() {',
          'varying vec3 vDissolveWorldPos;\nvoid main() {'
        )
        .replace(
          '#include <project_vertex>',
          '#include <project_vertex>\nvDissolveWorldPos = (modelMatrix * vec4(transformed, 1.0)).xyz;'
        );

      // Fragment injection: noise + scanline discard + glowing edge
      const fragHeader = [
        'uniform float uDissolve;',
        'uniform float uDissolveMode;',
        'uniform float uMinY;',
        'uniform float uMaxY;',
        'uniform float uTime;',
        'uniform vec3 uEdgeColor;',
        'varying vec3 vDissolveWorldPos;',
        'float wxrHash(vec3 p) {',
        '  p = fract(p * 0.3183099 + 0.1);',
        '  p *= 17.0;',
        '  return fract(p.x * p.y * p.z * (p.x + p.y + p.z));',
        '}',
        'float wxrNoise(vec3 x) {',
        '  vec3 i = floor(x);',
        '  vec3 f = fract(x);',
        '  f = f * f * (3.0 - 2.0 * f);',
        '  return mix(',
        '    mix(mix(wxrHash(i + vec3(0.0,0.0,0.0)), wxrHash(i + vec3(1.0,0.0,0.0)), f.x),',
        '        mix(wxrHash(i + vec3(0.0,1.0,0.0)), wxrHash(i + vec3(1.0,1.0,0.0)), f.x), f.y),',
        '    mix(mix(wxrHash(i + vec3(0.0,0.0,1.0)), wxrHash(i + vec3(1.0,0.0,1.0)), f.x),',
        '        mix(wxrHash(i + vec3(0.0,1.0,1.0)), wxrHash(i + vec3(1.0,1.0,1.0)), f.x), f.y), f.z);',
        '}',
        'void main() {'
      ].join('\n');

      const fragDiscardBlock = [
        '#include <clipping_planes_fragment>',
        'float wxrEdgeGlow = 0.0;',
        'if (uDissolve > 0.002) {',
        '  if (uDissolveMode < 0.5) {',
        '    float h = clamp((vDissolveWorldPos.y - uMinY) / max(0.02, uMaxY - uMinY), 0.0, 1.0);',
        '    float n = (wxrNoise(vDissolveWorldPos * 12.0 + vec3(0.0, uTime * 2.5, 0.0)) - 0.5) * 0.14;',
        '    float scan = sin(vDissolveWorldPos.y * 85.0 - uTime * 14.0) * 0.03;',
        '    float cutoff = (1.0 - uDissolve) * 1.16 - 0.08;',
        '    float v = h + n + scan;',
        '    if (v > cutoff + 0.09) discard;',
        '    wxrEdgeGlow = smoothstep(cutoff - 0.04, cutoff + 0.09, v);',
        '  } else {',
        '    float n = wxrNoise(vDissolveWorldPos * 9.5 + vec3(0.0, -uTime * 1.8, 0.0));',
        '    float scan = sin(vDissolveWorldPos.y * 90.0 + uTime * 16.0) * 0.5 + 0.5;',
        '    float field = n * 0.75 + scan * 0.25;',
        '    float thresh = uDissolve * 1.15 - 0.05;',
        '    if (field < thresh) discard;',
        '    wxrEdgeGlow = 1.0 - smoothstep(thresh, thresh + 0.13, field);',
        '  }',
        '}'
      ].join('\n');

      const fragGlowBlock = [
        'if (wxrEdgeGlow > 0.001) {',
        '  gl_FragColor.rgb += uEdgeColor * (wxrEdgeGlow * 2.6);',
        '}',
        '#include <dithering_fragment>'
      ].join('\n');

      shader.fragmentShader = shader.fragmentShader
        .replace('void main() {', fragHeader)
        .replace('#include <clipping_planes_fragment>', fragDiscardBlock)
        .replace('#include <dithering_fragment>', fragGlowBlock);
    };

    mat.userData = mat.userData || {};
    mat.userData.dissolveUniforms = uniforms;
    mat.needsUpdate = true;
    return uniforms;
  }

  function applyDissolveShader(rootObj, opts) {
    if (!rootObj) return [];
    const options = opts || {};
    const list = [];
    const clonedMap = new Map();

    rootObj.updateMatrixWorld(true);
    const box = new THREE.Box3().setFromObject(rootObj);
    const minY = isFinite(box.min.y) ? box.min.y : 0;
    const maxY = isFinite(box.max.y) && box.max.y > minY ? box.max.y : minY + 1;

    rootObj.traverse((node) => {
      if (!node.isMesh || !node.material) return;
      const mats = Array.isArray(node.material) ? node.material : [node.material];
      const outMats = [];

      for (let i = 0; i < mats.length; i++) {
        let m = mats[i];
        if (!m) continue;
        // Clone shared material per object once so independent dissolves do not interfere
        if (!m.userData || m.userData.dissolveOwner !== rootObj.uuid) {
          if (clonedMap.has(m)) {
            m = clonedMap.get(m);
          } else {
            const cloned = m.clone();
            cloned.userData = Object.assign({}, m.userData || {}, { dissolveOwner: rootObj.uuid });
            clonedMap.set(m, cloned);
            m = cloned;
          }
        }
        const u = patchMaterialWithDissolve(m);
        if (u) {
          u.uMinY.value = minY;
          u.uMaxY.value = maxY;
          if (options.color) u.uEdgeColor.value.set(options.color);
          if (typeof options.mode === 'number') u.uDissolveMode.value = options.mode;
          if (typeof options.progress === 'number') u.uDissolve.value = options.progress;
          list.push(u);
        }
        outMats.push(m);
      }

      node.material = Array.isArray(node.material) ? outMats : outMats[0];
    });

    return list;
  }

  const activeAnims = [];
  const activeParticles = [];

  function spawnPulseRing(pos, colorHex, maxRadius) {
    const geo = new THREE.RingGeometry(0.08, 0.14, 40);
    geo.rotateX(-Math.PI / 2);
    const mat = new THREE.MeshBasicMaterial({
      color: colorHex || 0x00ff88,
      transparent: true,
      opacity: 0.9,
      blending: THREE.AdditiveBlending,
      depthWrite: false,
      side: THREE.DoubleSide
    });
    const ring = new THREE.Mesh(geo, mat);
    ring.position.set(pos.x, Math.max(0.004, pos.y + 0.008), pos.z);
    scene.add(ring);

    activeRings.push({
      mesh: ring,
      age: 0,
      duration: 0.75,
      maxScale: (maxRadius || 0.9) / 0.14
    });
  }

  function spawnParticles(centerPos, sizeVec, opts) {
    const options = opts || {};
    const count = options.count || 56;
    const style = options.style || 'matrix'; // 'matrix' | 'fire' | 'smoke'
    const spreadX = Math.max(0.18, (sizeVec && sizeVec.x ? sizeVec.x : 0.4) * 0.65);
    const spreadY = Math.max(0.18, (sizeVec && sizeVec.y ? sizeVec.y : 0.5));
    const spreadZ = Math.max(0.18, (sizeVec && sizeVec.z ? sizeVec.z : 0.4) * 0.65);

    const positions = new Float32Array(count * 3);
    const colors = new Float32Array(count * 3);
    const velocities = new Float32Array(count * 3);

    const cPrimary = new THREE.Color(
      style === 'fire' ? 0xff6a1a : style === 'smoke' ? 0x8b98a8 : 0x00ff88
    );
    const cSecondary = new THREE.Color(
      style === 'fire' ? 0xffd24d : style === 'smoke' ? 0x3b4450 : 0x99ffcc
    );

    for (let i = 0; i < count; i++) {
      const i3 = i * 3;
      positions[i3] = centerPos.x + (Math.random() - 0.5) * spreadX;
      positions[i3 + 1] = centerPos.y + (Math.random() - 0.5) * spreadY;
      positions[i3 + 2] = centerPos.z + (Math.random() - 0.5) * spreadZ;

      const mixCol = cPrimary.clone().lerp(cSecondary, Math.random());
      colors[i3] = mixCol.r;
      colors[i3 + 1] = mixCol.g;
      colors[i3 + 2] = mixCol.b;

      velocities[i3] = (Math.random() - 0.5) * (style === 'smoke' ? 0.45 : 0.65);
      velocities[i3 + 1] = (style === 'fire' ? 0.9 : 0.55) + Math.random() * 0.95;
      velocities[i3 + 2] = (Math.random() - 0.5) * (style === 'smoke' ? 0.45 : 0.65);
    }

    const geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.BufferAttribute(positions, 3));
    geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));

    const mat = new THREE.PointsMaterial({
      size: style === 'smoke' ? 0.14 : 0.085,
      map: glowTexture,
      vertexColors: true,
      transparent: true,
      opacity: style === 'smoke' ? 0.65 : 0.95,
      blending: style === 'smoke' ? THREE.NormalBlending : THREE.AdditiveBlending,
      depthWrite: false
    });

    const points = new THREE.Points(geo, mat);
    points.frustumCulled = false;
    scene.add(points);

    activeParticles.push({
      points,
      velocities,
      age: 0,
      duration: options.duration || 0.95,
      baseOpacity: mat.opacity
    });
  }

  function materializeObject(obj, opts) {
    if (!obj) return;
    const options = opts || {};
    const color = options.color || 0x00ff88;
    const duration = options.duration || 0.85;

    obj.updateMatrixWorld(true);
    const box = new THREE.Box3().setFromObject(obj);
    const center = new THREE.Vector3();
    const size = new THREE.Vector3();
    box.getCenter(center);
    box.getSize(size);

    const uniformsList = applyDissolveShader(obj, {
      mode: 0.0,
      progress: 1.0,
      color
    });

    const minY = isFinite(box.min.y) ? box.min.y : 0;
    const radius = Math.max(0.35, Math.max(size.x, size.z) * 0.9);
    spawnPulseRing(new THREE.Vector3(center.x, minY, center.z), color, radius);
    spawnParticles(center, size, { style: 'matrix', count: 52, duration: duration + 0.15 });

    activeAnims.push({
      obj,
      uniformsList,
      mode: 0,
      age: 0,
      duration,
      onComplete: options.onComplete
    });
  }

  function disposeObjectResources(obj) {
    if (!obj) return;
    obj.traverse((node) => {
      if (node.geometry) {
        try { node.geometry.dispose(); } catch (_) {}
      }
      if (node.material) {
        const mats = Array.isArray(node.material) ? node.material : [node.material];
        for (const m of mats) {
          if (!m) continue;
          for (const key of Object.keys(m)) {
            const val = m[key];
            if (val && val.isTexture && val !== glowTexture && val !== matrixAtlasTex && val !== constructGridTex) {
              try { val.dispose(); } catch (_) {}
            }
          }
          try { m.dispose(); } catch (_) {}
        }
      }
    });
  }

  function dissolveObject(obj, opts) {
    if (!obj) return;
    const options = opts || {};
    const style = options.style || 'fire'; // 'fire' | 'matrix' | 'smoke'
    const color = options.color || (style === 'matrix' ? 0x00ff88 : style === 'smoke' ? 0x99aabb : 0xff6b1a);
    const duration = options.duration || 0.78;
    const removeOnDone = options.removeOnDone !== false;

    obj.updateMatrixWorld(true);
    const box = new THREE.Box3().setFromObject(obj);
    const center = new THREE.Vector3();
    const size = new THREE.Vector3();
    box.getCenter(center);
    box.getSize(size);

    const uniformsList = applyDissolveShader(obj, {
      mode: 1.0,
      progress: 0.0,
      color
    });

    const minY = isFinite(box.min.y) ? box.min.y : 0;
    const radius = Math.max(0.4, Math.max(size.x, size.z) * 1.05);
    spawnPulseRing(new THREE.Vector3(center.x, minY, center.z), color, radius);
    spawnParticles(center, size, { style, count: 68, duration: duration + 0.25 });

    activeAnims.push({
      obj,
      uniformsList,
      mode: 1,
      age: 0,
      duration,
      removeOnDone,
      onComplete: function () {
        if (removeOnDone) {
          if (obj.parent) obj.parent.remove(obj);
          if (Array.isArray(APP.entities)) {
            const idx = APP.entities.findIndex((e) => e === obj || (e && e.mesh === obj));
            if (idx !== -1) APP.entities.splice(idx, 1);
          }
          disposeObjectResources(obj);
        }
        if (typeof options.onComplete === 'function') options.onComplete();
      }
    });
  }

  // ====================================================================
  // Per-Frame Visual Animation Update
  // ====================================================================
  function update(dt, elapsed) {
    const delta = Math.min(0.05, dt || 0.016);
    const now = elapsed || (performance.now() * 0.001);

    // Animate Matrix falling glyph curtains when visible
    if (matrixGroup.visible) {
      for (let i = 0; i < curtainDrops.length; i++) {
        const d = curtainDrops[i];
        d.tex.offset.y = (d.tex.offset.y + delta * d.speed) % 1;
        d.mesh.position.y = d.baseY + Math.sin(now * 0.8 + d.phase) * 0.12;
      }
    }

    // Update materialize / dissolve object animations
    for (let i = activeAnims.length - 1; i >= 0; i--) {
      const a = activeAnims[i];
      a.age += delta;
      const t = Math.min(1, a.age / a.duration);
      const val = a.mode === 0 ? 1.0 - t : t;

      for (let j = 0; j < a.uniformsList.length; j++) {
        const u = a.uniformsList[j];
        u.uDissolve.value = val;
        u.uTime.value = now;
      }

      if (t >= 1) {
        activeAnims.splice(i, 1);
        if (typeof a.onComplete === 'function') {
          try { a.onComplete(); } catch (e) { console.warn('[visual] onComplete error:', e); }
        }
      }
    }

    // Update particle bursts
    for (let i = activeParticles.length - 1; i >= 0; i--) {
      const p = activeParticles[i];
      p.age += delta;
      const t = Math.min(1, p.age / p.duration);
      const posAttr = p.points.geometry.getAttribute('position');
      const arr = posAttr.array;
      const vel = p.velocities;

      for (let k = 0; k < arr.length; k += 3) {
        arr[k] += vel[k] * delta;
        arr[k + 1] += vel[k + 1] * delta;
        arr[k + 2] += vel[k + 2] * delta;
      }
      posAttr.needsUpdate = true;
      p.points.material.opacity = p.baseOpacity * (1 - t);

      if (t >= 1) {
        scene.remove(p.points);
        p.points.geometry.dispose();
        p.points.material.dispose();
        activeParticles.splice(i, 1);
      }
    }

    // Update expanding ground pulse rings
    for (let i = activeRings.length - 1; i >= 0; i--) {
      const r = activeRings[i];
      r.age += delta;
      const t = Math.min(1, r.age / r.duration);
      const easeOut = 1 - Math.pow(1 - t, 3);
      const s = 1 + easeOut * r.maxScale;
      r.mesh.scale.set(s, 1, s);
      r.mesh.material.opacity = 0.85 * (1 - t);

      if (t >= 1) {
        scene.remove(r.mesh);
        r.mesh.geometry.dispose();
        r.mesh.material.dispose();
        activeRings.splice(i, 1);
      }
    }
  }

  // Subscribe to lifecycle hooks exposed by index.html
  if (typeof APP.onUpdate === 'function') {
    APP.onUpdate(update);
  }
  if (typeof APP.on === 'function') {
    APP.on('xr:start', function (info) {
      attachXRLightEstimation(info && info.session);
      applyEnvironmentVisuals();
    });
    APP.on('xr:end', function () {
      if (xrLight && xrLight.parent === scene) scene.remove(xrLight);
      xrLightProbeActive = false;
      applyEnvironmentVisuals();
    });
  }

  // Expose public visual APIs
  APP.applyDissolveShader = applyDissolveShader;
  APP.materializeObject = materializeObject;
  APP.dissolveObject = dissolveObject;
  APP.visual = {
    ENV_MODES,
    setEnvironment,
    cycleEnvironment,
    getEnvironment: function () { return currentEnv; },
    applyDissolveShader,
    materializeObject,
    dissolveObject,
    spawnPulseRing,
    spawnParticles,
    disposeObjectResources,
    update
  };

  applyEnvironmentVisuals();

  // Voice / text command handler for switching environments
  if (typeof APP.registerAbilityHandler === 'function') {
    APP.registerAbilityHandler('visual_world_fx', function (text) {
      const low = String(text || '').toLowerCase();
      if (low.includes('matrix') || low.includes('ماتريكس')) {
        const mode = setEnvironment('matrix_black');
        return { ok: true, message: 'Switched environment to ' + mode };
      }
      if (low.includes('construct') || low.includes('gray') || low.includes('grey') || low.includes('البيضاء')) {
        const mode = setEnvironment('construct_gray');
        return { ok: true, message: 'Switched environment to ' + mode };
      }
      if (low.includes('passthrough') || low.includes('default') || low.includes('الواقع')) {
        const mode = setEnvironment('default');
        return { ok: true, message: 'Switched environment to ' + mode };
      }
      const mode = cycleEnvironment();
      return { ok: true, message: 'Cycled environment to ' + mode };
    });
  }
})();
