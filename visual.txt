  function createMatrixAtlasTexture() {
    const c = document.createElement('canvas');
    c.width = 256;
    c.height = 512;
    const ctx = c.getContext('2d');
    ctx.clearRect(0, 0, 256, 512);
    const glyphs = 'アイウエオカキクケコサシスセソタチツテトナニヌネノ0123456789ABCDEF';
    const cols = 8;
    const rows = 16;
    const cw = 256 / cols;
    const ch = 512 / rows;
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.font = 'bold 21px monospace';

    for (let col = 0; col < cols; col++) {
      for (let row = 0; row < rows; row++) {
        const chIdx = Math.floor(Math.random() * glyphs.length);
        const alpha = (row + 1) / rows;
        const isHead = row === rows - 1;
        ctx.fillStyle = isHead
          ? 'rgba(225, 255, 240, 0.98)'
          : 'rgba(0, 255, 136, ' + (0.14 + alpha * 0.78).toFixed(2) + ')';
        ctx.shadowColor = isHead ? '#b4ffdc' : '#00ff88';
        ctx.shadowBlur = isHead ? 10 : 4;
        ctx.fillText(glyphs[chIdx], col * cw + cw * 0.5, row * ch + ch * 0.5);
      }
    }
    const tex = new THREE.CanvasTexture(c);
    tex.wrapS = THREE.RepeatWrapping;
    tex.wrapT = THREE.RepeatWrapping;
    tex.needsUpdate = true;
    return tex;
  }

  function patchMaterialWithMatrixShader(mat) {
    if (!mat || typeof mat.onBeforeCompile !== 'function') return null;
    if (mat.userData && mat.userData.matrixUniforms) {
      return mat.userData.matrixUniforms;
    }

    const uniforms = {
      uDissolve: { value: 0.0 },
      uDissolveMode: { value: 0.0 },
      uMinY: { value: 0.0 },
      uMaxY: { value: 1.0 },
      uTime: { value: 0.0 },
      uMatrixGreen: { value: new THREE.Color(0x00ff88) },
      uMatrixCyan: { value: new THREE.Color(0x38bdf8) }
    };

    const prevCompile = mat.onBeforeCompile;
    const prevKey = typeof mat.customProgramCacheKey === 'function' ? mat.customProgramCacheKey() : '';

    mat.customProgramCacheKey = function () {
      return (prevKey ? prevKey + '|' : '') + 'wxr_matrix_sim_v2';
    };

    mat.onBeforeCompile = function (shader, r) {
      if (prevCompile && prevCompile !== mat.onBeforeCompile) {
        try { prevCompile.call(mat, shader, r); } catch (_) {}
      }

      shader.uniforms.uDissolve = uniforms.uDissolve;
      shader.uniforms.uDissolveMode = uniforms.uDissolveMode;
      shader.uniforms.uMinY = uniforms.uMinY;
      shader.uniforms.uMaxY = uniforms.uMaxY;
      shader.uniforms.uTime = uniforms.uTime;
      shader.uniforms.uMatrixGreen = uniforms.uMatrixGreen;
      shader.uniforms.uMatrixCyan = uniforms.uMatrixCyan;

      shader.vertexShader = shader.vertexShader
        .replace('void main() {', 'varying vec3 vMtxWorldPos;\nvoid main() {')
        .replace(
          '#include <project_vertex>',
          '#include <project_vertex>\nvMtxWorldPos = (modelMatrix * vec4(transformed, 1.0)).xyz;'
        );

      const fragHeader = [
        'uniform float uDissolve;',
        'uniform float uDissolveMode;',
        'uniform float uMinY;',
        'uniform float uMaxY;',
        'uniform float uTime;',
        'uniform vec3 uMatrixGreen;',
        'uniform vec3 uMatrixCyan;',
        'varying vec3 vMtxWorldPos;',
        'float mtxHash(vec3 p) {',
        '  p = fract(p * vec3(443.897, 441.423, 437.195));',
        '  p += dot(p, p.yzx + 19.19);',
        '  return fract((p.x + p.y) * p.z);',
        '}',
        'float mtxGrid(vec3 p, float scale, float thickness) {',
        '  vec3 f = abs(fract(p * scale) - 0.5);',
        '  float d = min(min(f.x, f.y), f.z);',
        '  return 1.0 - smoothstep(0.5 - thickness, 0.5, d);',
        '}',
        'float mtxRain(vec3 p, float t) {',
        '  vec2 col = floor(p.xz * 32.0);',
        '  float seed = mtxHash(vec3(col, 1.0));',
        '  float speed = 1.8 + seed * 3.2;',
        '  float stream = fract(p.y * 6.5 + t * speed + seed * 10.0);',
        '  return smoothstep(0.55, 0.98, stream);',
        '}',
        'void main() {'
      ].join('\n');

      const fragDiscard = [
        '#include <clipping_planes_fragment>',
        'float mtxHoloWeight = 0.0;',
        'float mtxEdgeMask = 0.0;',
        'float mtxRainPulse = 0.0;',
        'if (uDissolve > 0.002) {',
        '  float hSpan = max(0.04, uMaxY - uMinY);',
        '  float hNorm = clamp((vMtxWorldPos.y - uMinY) / hSpan, 0.0, 1.0);',
        '  vec3 voxelCell = floor(vMtxWorldPos * 28.0);',
        '  float cellRnd = mtxHash(voxelCell);',
        '  float lattice = mtxGrid(vMtxWorldPos, 28.0, 0.075);',
        '  mtxRainPulse = mtxRain(vMtxWorldPos, uTime);',
        '  if (uDissolveMode < 0.5) {',
        '    float scanY = mix(-0.14, 1.14, 1.0 - uDissolve);',
        '    float delta = hNorm - scanY;',
        '    float jitter = (cellRnd - 0.5) * 0.10;',
        '    if (delta + jitter > 0.14) discard;',
        '    if (delta + jitter > 0.02 && lattice < 0.25 && mtxRainPulse < 0.45) discard;',
        '    mtxHoloWeight = smoothstep(-0.08, 0.14, delta + jitter);',
        '    mtxEdgeMask = max(lattice * 0.95, 1.0 - smoothstep(0.0, 0.028, abs(delta)));',
        '  } else {',
        '    float sweep = mix(1.16, -0.16, uDissolve);',
        '    float delta = hNorm - sweep;',
        '    float thresh = uDissolve * 1.18 - 0.08;',
        '    float field = cellRnd * 0.65 + (1.0 - hNorm) * 0.35;',
        '    if (field < thresh && (delta > 0.06 || lattice < 0.3)) discard;',
        '    mtxHoloWeight = 1.0 - smoothstep(thresh, thresh + 0.20, field);',
        '    mtxEdgeMask = max(lattice, 1.0 - smoothstep(0.0, 0.035, abs(delta)));',
        '  }',
        '}'
      ].join('\n');

      const fragColor = [
        'if (uDissolve > 0.002 && (mtxHoloWeight > 0.01 || mtxEdgeMask > 0.01)) {',
        '  vec3 holoCol = mix(uMatrixGreen, uMatrixCyan, mtxRainPulse * 0.55);',
        '  vec3 hotCore = vec3(0.85, 1.0, 0.94);',
        '  vec3 synth = mix(holoCol, hotCore, clamp(mtxEdgeMask * 0.8, 0.0, 1.0));',
        '  gl_FragColor.rgb = mix(gl_FragColor.rgb, synth, clamp(mtxHoloWeight * 0.85, 0.0, 0.92));',
        '  gl_FragColor.rgb += synth * (mtxEdgeMask * 1.45 + mtxRainPulse * 0.65 * mtxHoloWeight);',
        '}',
        '#include <dithering_fragment>'
      ].join('\n');

      shader.fragmentShader = shader.fragmentShader
        .replace('void main() {', fragHeader)
        .replace('#include <clipping_planes_fragment>', fragDiscard)
        .replace('#include <dithering_fragment>', fragColor);
    };

    mat.userData = mat.userData || {};
    mat.userData.matrixUniforms = uniforms;
    mat.needsUpdate = true;
    return uniforms;
  }

  function collectAndPatchUniforms(rootObj, opts) {
    if (!rootObj) return [];
    const list = [];
    const clonedMap = new Map();

    rootObj.updateMatrixWorld(true);
    const box = new THREE.Box3().setFromObject(rootObj);
    const minY = isFinite(box.min.y) ? box.min.y : 0;
    const maxY = isFinite(box.max.y) && box.max.y > minY ? box.max.y : minY + 1;

    rootObj.traverse((node) => {
      if (!node.isMesh || !node.material) return;
      const mats = Array.isArray(node.material) ? node.material : [node.material];
      const out = [];

      for (let i = 0; i < mats.length; i++) {
        let m = mats[i];
        if (!m) continue;
        if (!m.userData || m.userData.matrixOwner !== rootObj.uuid) {
          if (clonedMap.has(m)) {
            m = clonedMap.get(m);
          } else {
            const c = m.clone();
            c.userData = Object.assign({}, m.userData || {}, { matrixOwner: rootObj.uuid });
            clonedMap.set(m, c);
            m = c;
          }
        }
        const u = patchMaterialWithMatrixShader(m);
        if (u) {
          u.uMinY.value = minY;
          u.uMaxY.value = maxY;
          if (typeof opts.mode === 'number') u.uDissolveMode.value = opts.mode;
          if (typeof opts.progress === 'number') u.uDissolve.value = opts.progress;
          list.push(u);
        }
        out.push(m);
      }
      node.material = Array.isArray(node.material) ? out : out[0];
    });
    return list;
  }

      reverse: !!reverse,

  function spawnMatrixGlyphStream(centerPos, sizeVec, isDissolve) {
    const count = 44;
    const spreadX = Math.max(0.20, sizeVec.x * 0.85);
    const spreadY = Math.max(0.24, sizeVec.y * 0.95);
    const spreadZ = Math.max(0.20, sizeVec.z * 0.85);

    const positions = new Float32Array(count * 3);
    const colors = new Float32Array(count * 3);
    const velocities = new Float32Array(count * 3);

    const cGreen = new THREE.Color(0x00ff88);
    const cCyan = new THREE.Color(0x38bdf8);

    for (let i = 0; i < count; i++) {
      const i3 = i * 3;
      positions[i3] = centerPos.x + (Math.random() - 0.5) * spreadX;
      positions[i3 + 1] = centerPos.y + (Math.random() - 0.5) * spreadY;
      positions[i3 + 2] = centerPos.z + (Math.random() - 0.5) * spreadZ;

      const mix = cGreen.clone().lerp(cCyan, Math.random() * 0.55);
      colors[i3] = mix.r;
      colors[i3 + 1] = mix.g;
      colors[i3 + 2] = mix.b;

      velocities[i3] = (Math.random() - 0.5) * 0.14;
      velocities[i3 + 1] = isDissolve ? -(0.35 + Math.random() * 0.75) : (0.35 + Math.random() * 0.7);
      velocities[i3 + 2] = (Math.random() - 0.5) * 0.14;
    }

    const geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.BufferAttribute(positions, 3));
    geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));

    const mat = new THREE.PointsMaterial({
      size: 0.055,
      map: glyphSparkTex,
      vertexColors: true,
      transparent: true,
      opacity: 0.92,
      blending: THREE.AdditiveBlending,
      depthWrite: false
    });

    const pts = new THREE.Points(geo, mat);
    pts.frustumCulled = false;
    scene.add(pts);
    activeGlyphStreams.push({ points: pts, velocities, age: 0, duration: 1.05 });
  }

  function materializeObject(obj, opts) {
    if (!obj) return;
    const options = opts || {};
    const duration = options.duration || 0.95;

    obj.updateMatrixWorld(true);
    const box = new THREE.Box3().setFromObject(obj);
    const center = new THREE.Vector3();
    const size = new THREE.Vector3();
    box.getCenter(center);
    box.getSize(size);
    const minY = isFinite(box.min.y) ? box.min.y : 0;
    const maxY = isFinite(box.max.y) && box.max.y > minY ? box.max.y : minY + 0.5;
    const radius = Math.max(0.25, Math.max(size.x, size.z) * 0.65);

    const uniformsList = collectAndPatchUniforms(obj, { mode: 0.0, progress: 1.0 });
    spawnFloorGridPulse(new THREE.Vector3(center.x, minY, center.z), radius);
    spawnLaserScanRing(center, radius, minY, maxY, false);
    spawnMatrixGlyphStream(center, size, false);

    activeAnims.push({
      obj,
      uniformsList,
      mode: 0,
      age: 0,
      duration,
      onComplete: options.onComplete
    });
  }

  function dissolveObject(obj, opts) {
    if (!obj) return;
    const options = opts || {};
    const duration = options.duration || 0.88;
    const removeOnDone = options.removeOnDone !== false;

    obj.updateMatrixWorld(true);
    const box = new THREE.Box3().setFromObject(obj);
    const center = new THREE.Vector3();
    const size = new THREE.Vector3();
    box.getCenter(center);
    box.getSize(size);
    const minY = isFinite(box.min.y) ? box.min.y : 0;
    const maxY = isFinite(box.max.y) && box.max.y > minY ? box.max.y : minY + 0.5;
    const radius = Math.max(0.25, Math.max(size.x, size.z) * 0.65);

    const uniformsList = collectAndPatchUniforms(obj, { mode: 1.0, progress: 0.0 });
    spawnFloorGridPulse(new THREE.Vector3(center.x, minY, center.z), radius);
    spawnLaserScanRing(center, radius, minY, maxY, true);
    spawnMatrixGlyphStream(center, size, true);

    activeAnims.push({
      obj,
      uniformsList,
      mode: 1,
      age: 0,
      duration,
      onComplete: function () {
        if (removeOnDone) {
          if (obj.parent) obj.parent.remove(obj);
          if (typeof APP.disposeObject3D === 'function') APP.disposeObject3D(obj);
        }
        if (typeof options.onComplete === 'function') options.onComplete();
      }
    });
  }

  // Unified entrypoint matching index.html (`applyDissolveShader(group, 'materialize')`)
  function applyDissolveShader(rootObj, modeOrOpts, extraOpts) {
    if (typeof modeOrOpts === 'string') {
      const m = modeOrOpts.toLowerCase();
      if (m === 'dissolve') {
        dissolveObject(rootObj, Object.assign({ removeOnDone: false }, extraOpts || {}));
      } else {
        materializeObject(rootObj, extraOpts || {});
      }
      return;
    }
    return collectAndPatchUniforms(rootObj, modeOrOpts || {});
  }

  function dissolveAndRemove(rootObj, onComplete) {
    dissolveObject(rootObj, { removeOnDone: false, onComplete });
  }

  // ====================================================================
  // 5. PER-FRAME ANIMATION LOOP
  // ====================================================================
  function update(dt, elapsed) {
    const delta = Math.min(0.05, dt || 0.016);
    const now = elapsed || (performance.now() * 0.001);

    if (matrixGroup.visible) {
      for (let i = 0; i < curtainDrops.length; i++) {
        const d = curtainDrops[i];
        d.tex.offset.y = (d.tex.offset.y + delta * d.speed) % 1;
        d.mesh.position.y = d.baseY + Math.sin(now * 0.8 + d.phase) * 0.1;
      }
    }

    for (let i = activeAnims.length - 1; i >= 0; i--) {
      const a = activeAnims[i];
      a.age += delta;
      const t = Math.min(1, a.age / a.duration);
      const val = a.mode === 0 ? 1.0 - t : t;
      for (let j = 0; j < a.uniformsList.length; j++) {
        a.uniformsList[j].uDissolve.value = val;
        a.uniformsList[j].uTime.value = now;
      }
      if (t >= 1) {
        activeAnims.splice(i, 1);
        if (typeof a.onComplete === 'function') {
          try { a.onComplete(); } catch (_) {}
        }
      }
    }

    for (let i = activeGlyphStreams.length - 1; i >= 0; i--) {
      const s = activeGlyphStreams[i];
      s.age += delta;
      const t = Math.min(1, s.age / s.duration);
      const posAttr = s.points.geometry.getAttribute('position');
      const arr = posAttr.array;
      for (let k = 0; k < arr.length; k += 3) {
        arr[k] += s.velocities[k] * delta;
        arr[k + 1] += s.velocities[k + 1] * delta;
        arr[k + 2] += s.velocities[k + 2] * delta;
      }
      posAttr.needsUpdate = true;
      s.points.material.opacity = 0.92 * (1 - t);
      if (t >= 1) {
        scene.remove(s.points);
        s.points.geometry.dispose();
        s.points.material.dispose();
        activeGlyphStreams.splice(i, 1);
      }
    }

    for (let i = activeScanRings.length - 1; i >= 0; i--) {
      const r = activeScanRings[i];
      r.age += delta;
      const t = Math.min(1, r.age / r.duration);
      if (r.isLaserSweep) {
        const y = r.reverse
          ? THREE.MathUtils.lerp(r.maxY, r.minY, t)
          : THREE.MathUtils.lerp(r.minY, r.maxY, t);
        r.mesh.position.y = y;
        r.mesh.material.opacity = Math.sin(t * Math.PI) * 0.95;
      } else {
        const ease = 1 - Math.pow(1 - t, 3);
        const sc = 1 + ease * r.maxScale;
        r.mesh.scale.set(sc, 1, sc);
        r.mesh.material.opacity = 0.85 * (1 - t);
      }
      if (t >= 1) {
        scene.remove(r.mesh);
        r.mesh.geometry.dispose();
        r.mesh.material.dispose();
        activeScanRings.splice(i, 1);
      }
    }
  }

  const unsubFrame = APP.onUpdate ? APP.onUpdate(update) : null;
  if (!unsubFrame && APP.frameHooks) APP.frameHooks.add(update);

  const unsubXRStart = APP.on ? APP.on('xr:start', applyEnvironmentVisuals) : null;
  const unsubXREnd = APP.on ? APP.on('xr:end', applyEnvironmentVisuals) : null;

  APP._visualCleanup = function () {
    if (typeof unsubFrame === 'function') unsubFrame();
    else if (APP.frameHooks) APP.frameHooks.delete(update);
    if (typeof unsubXRStart === 'function') unsubXRStart();
    if (typeof unsubXREnd === 'function') unsubXREnd();
    if (envGroup.parent) envGroup.parent.remove(envGroup);
  };

  APP.applyDissolveShader = applyDissolveShader;
  APP.materializeObject = materializeObject;
  APP.dissolveObject = dissolveObject;
  APP.visual = {
    ENV_MODES,
    setEnvironment,
    cycleEnvironment,
    getEnvironment: () => currentEnv,
    onXRModeChange: () => applyEnvironmentVisuals(),
    applyDissolveShader,
    dissolveAndRemove,
    materializeObject,
    dissolveObject,
    update
  };

  applyEnvironmentVisuals();

  APP.registerAbilityHandler?.('visual_world_fx', function (text) {
    const low = String(text || '').toLowerCase();
    if (low.includes('matrix') || low.includes('ماتريكس')) return { ok: true, message: 'Env: ' + setEnvironment('matrix_black') };
    if (low.includes('construct') || low.includes('gray') || low.includes('البيضاء')) return { ok: true, message: 'Env: ' + setEnvironment('construct_gray') };
    if (low.includes('passthrough') || low.includes('default') || low.includes('الواقع')) return { ok: true, message: 'Env: ' + setEnvironment('default') };
    return { ok: true, message: 'Env: ' + cycleEnvironment() };
  });
})();
