  function stepCapePhysics(dt, elapsed) {
    capeMesh.visible = capeEnabled && botEnabled;
    if (!capeMesh.visible) return;

    const safeDt = Math.min(0.025, Math.max(0.004, dt || 0.016));
    bodyGroup.updateMatrixWorld(true);

    // Check if Geeny jumped far or uninitialized -> reset cleanly without vertex stretch
    tmpCapeW.copy(capePinnedLocal[Math.floor(CAPE_COLS / 2)]);
    bodyGroup.localToWorld(tmpCapeW);
    if (!capeInitialized || !isFinite(capeCur[0].x) || capeCur[0].distanceTo(tmpCapeW) > 1.2) {
      snapCapeToRest();
    }

    // 1. Update pinned shoulder row (row 0)
    for (let c = 0; c < CAPE_COLS; c++) {
      tmpCapeW.copy(capePinnedLocal[c]);
      bodyGroup.localToWorld(tmpCapeW);
      capeCur[c].copy(tmpCapeW);
      capePrev[c].copy(tmpCapeW);
    }

    // Rear world direction for aerodynamic flutter
    tmpCapeBack.set(0, 0, -1).applyQuaternion(bodyGroup.getWorldQuaternion(new THREE.Quaternion())).normalize();
    const gravStep = -4.8 * safeDt * safeDt;
    const maxStepSpeed = 0.042; // Strict per-step clamp -> zero vertex explosion

    // 2. Verlet integration for rows 1..CAPE_ROWS-1
    for (let r = 1; r < CAPE_ROWS; r++) {
      for (let c = 0; c < CAPE_COLS; c++) {
        const idx = r * CAPE_COLS + c;
        const cur = capeCur[idx];
        const prev = capePrev[idx];

        let vx = (cur.x - prev.x) * 0.965;
        let vy = (cur.y - prev.y) * 0.965;
        let vz = (cur.z - prev.z) * 0.965;
        const spdSq = vx * vx + vy * vy + vz * vz;
        if (spdSq > maxStepSpeed * maxStepSpeed) {
          const s = maxStepSpeed / Math.sqrt(spdSq);
          vx *= s; vy *= s; vz *= s;
        }

        prev.copy(cur);
        const flutter = (Math.sin(elapsed * 4.4 + c * 0.65 + r * 0.42) * 0.5 + 0.5) * 0.00048;
        cur.x += vx + tmpCapeBack.x * flutter;
        cur.y += vy + gravStep;
        cur.z += vz + tmpCapeBack.z * flutter;
      }
    }

    // 3. Multi-pass Constraint Solver + Body Sphere & 18-Segment Tentacle Collisions
    const bodyCenter = botRoot.position;
    const R_BODY = 0.174;
    const R_BODY_SQ = R_BODY * R_BODY;

    for (let iter = 0; iter < 5; iter++) {
      // A. Structural / Shear / Bend distance constraints
      for (let k = 0; k < capeConstraints.length; k++) {
        const con = capeConstraints[k];
        const ia = con[0], ib = con[1], restLen = con[2], stiff = con[3];
        const pa = capeCur[ia], pb = capeCur[ib];

        tmpCapeDelta.subVectors(pb, pa);
        const d = tmpCapeDelta.length();
        if (d < 1e-6) continue;
        const diff = ((d - restLen) / d) * stiff;

        const pinA = ia < CAPE_COLS;
        const pinB = ib < CAPE_COLS;
        if (pinA && pinB) continue;
        if (pinA) {
          pb.addScaledVector(tmpCapeDelta, -diff);
        } else if (pinB) {
          pa.addScaledVector(tmpCapeDelta, diff);
        } else {
          pa.addScaledVector(tmpCapeDelta, diff * 0.5);
          pb.addScaledVector(tmpCapeDelta, -diff * 0.5);
        }
      }

      // B. Smooth collision with Geeny Body Sphere, 18 Tentacle Segments, Claw, and Floor
      for (let idx = CAPE_COLS; idx < CAPE_COUNT; idx++) {
        const p = capeCur[idx];

        // Body sphere collision
        tmpCapeDelta.subVectors(p, bodyCenter);
        let dSq = tmpCapeDelta.lengthSq();
        if (dSq < R_BODY_SQ) {
          const d = Math.sqrt(dSq);
          if (d > 1e-5) p.copy(bodyCenter).addScaledVector(tmpCapeDelta, R_BODY / d);
          else p.set(bodyCenter.x, bodyCenter.y, bodyCenter.z - R_BODY);
        }

        // 18 Tentacle segments collision
        for (let s = 0; s < TENTACLE_SEGMENTS; s++) {
          const segPos = segments[s].position;
          const rTent = THREE.MathUtils.lerp(0.050, 0.032, s / (TENTACLE_SEGMENTS - 1));
          tmpCapeDelta.subVectors(p, segPos);
          dSq = tmpCapeDelta.lengthSq();
          if (dSq < rTent * rTent) {
            const d = Math.sqrt(dSq);
            if (d > 1e-5) p.copy(segPos).addScaledVector(tmpCapeDelta, rTent / d);
            else p.copy(segPos).add(new THREE.Vector3(0, 0, -rTent));
          }
        }

        // Claw tip collision
        const clawPos = clawGroup.position;
        const rClaw = 0.045;
        tmpCapeDelta.subVectors(p, clawPos);
        dSq = tmpCapeDelta.lengthSq();
        if (dSq < rClaw * rClaw) {
          const d = Math.sqrt(dSq);
          if (d > 1e-5) p.copy(clawPos).addScaledVector(tmpCapeDelta, rClaw / d);
        }

        // Floor clamp
        if (p.y < 0.02) p.y = 0.02;
      }
    }

    // 4. Write validated positions + recompute smooth normals
    const arr = capeGeo.attributes.position.array;
    for (let i = 0; i < CAPE_COUNT; i++) {
      const p = capeCur[i];
      if (!isFinite(p.x) || !isFinite(p.y) || !isFinite(p.z)) {
        snapCapeToRest();
        break;
      }
      arr[i * 3] = p.x;
      arr[i * 3 + 1] = p.y;
      arr[i * 3 + 2] = p.z;
    }
    capeGeo.attributes.position.needsUpdate = true;
    capeGeo.computeVertexNormals();
  }

  async function setCapeTexture(nameOrUrl) {
    const q = String(nameOrUrl || '').trim();
    const low = q.toLowerCase();
    if (!q || low === 'algeria' || low === 'algerian' || low === 'flag' || low === 'default' || low.includes('الجزائر')) {
      capeMat.map = defaultCapeTex;
      capeMat.needsUpdate = true;
      setCapeEnabled(true);
      return { ok: true, texture: 'Algerian Flag', message: 'Geeny cape set to Algerian Flag.' };
    }

    const clean = q.replace(/\.[^.]+$/, '').toLowerCase();
    let url = q.startsWith('http') || q.startsWith('blob:') || q.startsWith('./') || q.startsWith('data:') ? q : null;

    if (!url && APP.importedImages) {
      const rec = APP.importedImages.get(clean) || Array.from(APP.importedImages.values()).find((r) => r.cleanName.includes(clean));
      if (rec) url = rec.url;
    }
    if (!url) {
      for (const p of [`./assets/${clean}.png`, `./assets/${clean}.jpg`, `./assets/${clean}.jpeg`, `./assets/${clean}.webp`, `./${clean}.png`, `./${clean}.jpg`]) {
        try { const h = await fetch(p, { method: 'HEAD' }); if (h.ok) { url = p; break; } } catch (_) {}
      }
    }
    if (!url) {
      const avail = APP.importedImages ? Array.from(APP.importedImages.keys()).join(', ') : '';
      return { ok: false, error: `Picture "${q}" not found in /assets/ or imported images.` + (avail ? ` Available: ${avail}` : '') };
    }

    try {
      const tex = await new THREE.TextureLoader().loadAsync(url);
      tex.colorSpace = THREE.SRGBColorSpace;
      tex.anisotropy = 4;
      if (customCapeTex && customCapeTex !== defaultCapeTex) {
        try { customCapeTex.dispose(); } catch (_) {}
      }
      customCapeTex = tex;
      capeMat.map = tex;
      capeMat.needsUpdate = true;
      setCapeEnabled(true);
      return { ok: true, texture: q, message: `Updated Geeny cape texture to "${q}".` };
    } catch (e) {
      return { ok: false, error: 'Failed to load cape texture: ' + e.message };
    }
  }

  function drawBoardRoundedRect(x, y, w, h, r, fill, stroke) {
    bCtx.beginPath();
    bCtx.moveTo(x + r, y);
    bCtx.arcTo(x + w, y, x + w, y + h, r);
    bCtx.arcTo(x + w, y + h, x, y + h, r);
    bCtx.arcTo(x, y + h, x, y, r);
    bCtx.arcTo(x, y, x + w, y, r);
    bCtx.closePath();
    if (fill) { bCtx.fillStyle = fill; bCtx.fill(); }
    if (stroke) { bCtx.strokeStyle = stroke; bCtx.lineWidth = 3; bCtx.stroke(); }
  }

  function wrapCanvasText(ctx, text, x, y, maxWidth, lineHeight) {
    const words = String(text || '').split(/\s+/);
    let line = '';
    let cy = y;
    for (let i = 0; i < words.length; i++) {
      const test = line ? line + ' ' + words[i] : words[i];
      if (ctx.measureText(test).width > maxWidth && line) {
        ctx.fillText(line, x, cy);
        line = words[i];
        cy += lineHeight;
      } else {
        line = test;
      }
    }
    if (line) ctx.fillText(line, x, cy);
    return cy + lineHeight;
  }

  function executeDrawCommands(cmds, area) {
    if (!Array.isArray(cmds)) return;
    const ox = area.x, oy = area.y, aw = area.w, ah = area.h;
    const px = (nx) => ox + Number(nx || 0) * aw;
    const py = (ny) => oy + Number(ny || 0) * ah;

    for (const c of cmds) {
      if (!c || !c.type) continue;
      bCtx.save();
      bCtx.strokeStyle = c.color || '#00f0b4';
      bCtx.fillStyle = c.fill || c.color || 'rgba(0,240,180,0.18)';
      bCtx.lineWidth = c.width || 5;
      bCtx.lineCap = 'round';
      bCtx.lineJoin = 'round';

      if (c.type === 'circle') {
        bCtx.beginPath();
        bCtx.arc(px(c.cx ?? 0.5), py(c.cy ?? 0.5), (c.r ?? 0.15) * Math.min(aw, ah), 0, Math.PI * 2);
        if (c.filled) bCtx.fill();
        bCtx.stroke();
      } else if (c.type === 'ellipse') {
        bCtx.beginPath();
        bCtx.ellipse(px(c.cx ?? 0.5), py(c.cy ?? 0.5), (c.rx ?? 0.2) * aw, (c.ry ?? 0.15) * ah, c.rot || 0, 0, Math.PI * 2);
        if (c.filled) bCtx.fill();
        bCtx.stroke();
      } else if (c.type === 'line') {
        bCtx.beginPath();
        bCtx.moveTo(px(c.x1), py(c.y1));
        bCtx.lineTo(px(c.x2), py(c.y2));
        bCtx.stroke();
      } else if (c.type === 'arc') {
        bCtx.beginPath();
        bCtx.arc(px(c.cx ?? 0.5), py(c.cy ?? 0.5), (c.r ?? 0.1) * Math.min(aw, ah), c.start ?? 0, c.end ?? Math.PI);
        bCtx.stroke();
      } else if (c.type === 'poly' && Array.isArray(c.points) && c.points.length >= 2) {
        bCtx.beginPath();
        bCtx.moveTo(px(c.points[0][0]), py(c.points[0][1]));
        for (let i = 1; i < c.points.length; i++) bCtx.lineTo(px(c.points[i][0]), py(c.points[i][1]));
        if (c.closed) bCtx.closePath();
        if (c.filled) bCtx.fill();
        bCtx.stroke();
      } else if (c.type === 'curve') {
        bCtx.beginPath();
        bCtx.moveTo(px(c.x1), py(c.y1));
        bCtx.quadraticCurveTo(px(c.cx), py(c.cy), px(c.x2), py(c.y2));
        bCtx.stroke();
      } else if (c.type === 'text') {
        bCtx.fillStyle = c.color || '#e2f8ff';
        bCtx.font = 'bold ' + (c.size || 24) + 'px system-ui, sans-serif';
        bCtx.textAlign = 'center';
        bCtx.fillText(String(c.text || ''), px(c.x ?? 0.5), py(c.y ?? 0.5));
      }
      bCtx.restore();
    }
  }

  function computeObjCenterAndRadius(obj) {
    obj.updateMatrixWorld(true);
    tmpBox.setFromObject(obj);
    if (tmpBox.isEmpty()) {
      obj.getWorldPosition(vObjCenter);
      return { center: vObjCenter.clone(), radius: 0.12 };

    tmpBox.getCenter(vObjCenter);
    tmpBox.getSize(vObjSize);
    const radius = Math.max(0.06, Math.min(0.32, Math.max(vObjSize.x, vObjSize.z) * 0.45));
    return { center: vObjCenter.clone(), radius };
  }

  function evalCubicBezier(p0, c1, c2, p3, t, out) {
    const u = 1 - t;
    out.set(0, 0, 0)
      .addScaledVector(p0, u * u * u)
      .addScaledVector(c1, 3 * u * u * t)
      .addScaledVector(c2, 3 * u * t * t)
      .addScaledVector(p3, t * t * t);
    return out;
  }

  function updateTentacleIK(dt, elapsed) {
    vP0.set(0, -0.14, 0);
    botRoot.localToWorld(vP0);

    const tipLerp = 1 - Math.exp(-9.5 * dt);
    state.currentTipPos.lerp(state.tipTargetPos, tipLerp);
    vP3.copy(state.currentTipPos);

    const dist = vP0.distanceTo(vP3);
    vC1.copy(vP0)
      .add(new THREE.Vector3(0, -Math.max(0.24, dist * 0.48), 0))
      .addScaledVector(state.velocity, -0.18);

    tmpVecA.subVectors(vP3, vP0).normalize();
    vC2.copy(vP3)
      .addScaledVector(tmpVecA, -Math.max(0.18, dist * 0.36))
      .addScaledVector(state.velocity, -0.12);
    vC2.y += Math.sin(elapsed * 2.2) * 0.045 + 0.08;

    for (let i = 0; i < TENTACLE_SEGMENTS; i++) {
      const t = i / TENTACLE_SEGMENTS;
      const tNext = (i + 1) / TENTACLE_SEGMENTS;
      evalCubicBezier(vP0, vC1, vC2, vP3, t, vPt);
      evalCubicBezier(vP0, vC1, vC2, vP3, tNext, vNext);
      segments[i].position.copy(vPt);
      segments[i].lookAt(vNext);
    }

    clawGroup.position.copy(vP3);
    if (state.mode === 'showing' && state.targetObj) {
      const info = computeObjCenterAndRadius(state.targetObj);
      clawGroup.lookAt(info.center);
      const beamDist = Math.max(0.15, Math.min(0.95, vP3.distanceTo(info.center)));
      showBeam.scale.set(1, 1, beamDist / 0.60);
      showBeam.position.z = beamDist * 0.52;
    } else {
      evalCubicBezier(vP0, vC1, vC2, vP3, 0.9, tmpVecB);
      tmpVecA.subVectors(vP3, tmpVecB).normalize();
      if (tmpVecA.lengthSq() > 0.001) clawGroup.lookAt(tmpVecB.copy(vP3).add(tmpVecA));
    }

    clawClosure += (clawTargetClosure - clawClosure) * Math.min(1, dt * 10);
    const openAngle = THREE.MathUtils.lerp(0.62, -0.14, clawClosure);
    for (let p = 0; p < prongs.length; p++) prongs[p].rotation.y = openAngle;
  }

  function updateRobot(dt, elapsed) {
    if (!botEnabled) return;
    const delta = Math.min(0.05, dt || 0.016);
    const now = elapsed || (performance.now() * 0.001);

    const hp = APP.headPos || new THREE.Vector3(0, 1.5, 0);
    const hf = APP.headFwd || new THREE.Vector3(0, 0, -1);
    const flatFwd = new THREE.Vector3(hf.x, 0, hf.z).normalize();
    if (flatFwd.lengthSq() < 0.01) flatFwd.set(0, 0, -1);
    const right = new THREE.Vector3().crossVectors(flatFwd, new THREE.Vector3(0, 1, 0)).normalize();

    if (state.mode === 'idle') {
      const distToUser = state.worldAnchor.distanceTo(hp);
      if (distToUser > 3.0) {
        state.worldAnchor.copy(hp).addScaledVector(flatFwd, 1.15).addScaledVector(right, 0.45);
        state.worldAnchor.y = Math.max(1.05, hp.y + 0.04);
      } else if (distToUser < 0.45) {
        tmpVecA.subVectors(state.worldAnchor, hp).setY(0).normalize();
        if (tmpVecA.lengthSq() < 0.01) tmpVecA.copy(flatFwd);
        state.worldAnchor.copy(hp).addScaledVector(tmpVecA, 0.72);
        state.worldAnchor.y = Math.max(1.05, hp.y + 0.04);
      }

      if (now > state.nextThoughtTime && !boardState.visible) {
        state.nextThoughtTime = now + 7 + Math.random() * 5;
        const ents = Array.isArray(APP.entities) ? APP.entities : [];
        if (ents.length > 0 && Math.random() < 0.58) {
          const pick = ents[Math.floor(Math.random() * ents.length)];
          const obj = pick?.group || pick?.mesh;
          if (obj && !pick.isHeld) {
            const info = computeObjCenterAndRadius(obj);
            if (info.center.distanceTo(hp) < 2.6) {
              state.worldAnchor.copy(info.center).add(new THREE.Vector3((Math.random() - 0.5) * 0.45, 0.38, (Math.random() - 0.5) * 0.45));
              state.lookTargetPos.copy(info.center);
            }
          }
        } else {
          state.lookTargetPos.copy(hp);
        }
      } else if (APP.botSpeaking || APP.micLevel > 0.12 || boardState.visible) {
        state.lookTargetPos.copy(hp);
      }

      state.hoverTargetPos.copy(state.worldAnchor).add(new THREE.Vector3(
        Math.sin(now * 0.7) * 0.055,
        Math.sin(now * 1.7) * 0.035,
        Math.cos(now * 0.6) * 0.045
      ));

      state.tipTargetPos.copy(botRoot.position).add(new THREE.Vector3(
        Math.sin(now * 1.25) * 0.14,
        -0.46 + Math.cos(now * 1.55) * 0.05,
        -0.08 + Math.cos(now * 0.9) * 0.10
      ));
    } else if (state.mode === 'grabbing') {
      if (!state.targetObj || !state.targetObj.parent) {
        tentacle_release();
      } else {
        const info = computeObjCenterAndRadius(state.targetObj);
        state.hoverTargetPos.copy(info.center).add(new THREE.Vector3(0.28, 0.46, 0.22));
        state.tipTargetPos.copy(info.center);
        state.lookTargetPos.copy(info.center);

        if (state.currentTipPos.distanceTo(info.center) < 0.16) {
          state.grabProgress += delta;
          clawTargetClosure = 0.88;
          if (state.grabProgress > 0.22) {
            state.heldObj = state.targetObj;
            state.heldEnt = getEntityRecord(state.heldObj);
            state.heldRadius = info.radius;
            if (state.heldEnt) state.heldEnt.heldByBot = true;
            state.heldObj.userData = state.heldObj.userData || {};
            state.heldObj.userData.heldByBot = true;
            state.mode = 'holding';
          }
        }
      }
    } else if (state.mode === 'holding') {
      if (!state.heldObj || !state.heldObj.parent) {
        tentacle_release();
      } else {
        state.hoverTargetPos.copy(state.worldAnchor).add(new THREE.Vector3(0, Math.sin(now * 1.5) * 0.025, 0));
        state.tipTargetPos.copy(botRoot.position).add(new THREE.Vector3(-0.10, -0.44, 0.22));
        state.lookTargetPos.copy(hp);
      }
    } else if (state.mode === 'handing') {
      if (!state.heldObj || !state.heldObj.parent) {
        tentacle_release();
      } else {
        state.hoverTargetPos.copy(hp).addScaledVector(flatFwd, 0.72);
        state.hoverTargetPos.y = Math.max(0.92, hp.y + 0.02);

        let handAnchor = null;
        let handGrabbing = false;
        const hands = APP.hands || {};
        for (const k of Object.keys(hands)) {
          const h = hands[k];
          if (!h || !h.active) continue;
          const pos = h.isHand ? h.fingerTips?.index : h.pos;
          if (pos) {
            if (!handAnchor || pos.distanceTo(state.currentTipPos) < handAnchor.distanceTo(state.currentTipPos)) {
              handAnchor = pos;
            }
            if ((h.pinching || h.trigger || h.gripping) && pos.distanceTo(state.currentTipPos) < 0.22) {
              handGrabbing = true;
            }
          }
        }

        if (handAnchor && handAnchor.distanceTo(hp) < 1.1) {
          state.tipTargetPos.copy(handAnchor).addScaledVector(flatFwd, -0.05);
        } else {
          state.tipTargetPos.copy(hp).addScaledVector(flatFwd, 0.42);
          state.tipTargetPos.y = Math.max(0.75, hp.y - 0.20);
        }
        state.lookTargetPos.copy(hp);

        if (handGrabbing || state.heldEnt?.grabbedByHand) tentacle_release();
      }
    } else if (state.mode === 'showing') {
      if (!state.targetObj || !state.targetObj.parent) {
        tentacle_release();
      } else {
        const info = computeObjCenterAndRadius(state.targetObj);
        state.hoverTargetPos.copy(info.center).add(new THREE.Vector3(0.36, 0.40 + Math.sin(now * 2.0) * 0.03, 0.22));
        state.tipTargetPos.copy(info.center).add(new THREE.Vector3(0.18, 0.12, 0.12));
        state.worldAnchor.copy(state.hoverTargetPos);
        state.lookTargetPos.copy(info.center);

        state.showTimer -= delta;
        if (state.showTimer <= 0) tentacle_release();
      }
    }

    const stiffness = 22.0;
    const damping = 8.4;
    tmpVecA.subVectors(state.hoverTargetPos, botRoot.position).multiplyScalar(stiffness);
    state.velocity.addScaledVector(tmpVecA, delta);
    state.velocity.multiplyScalar(Math.exp(-damping * delta));
    botRoot.position.addScaledVector(state.velocity, delta);

    bodyGroup.lookAt(state.lookTargetPos);
    const localVelX = state.velocity.x;
    const speed = state.velocity.length();
    bodyGroup.rotateZ(THREE.MathUtils.clamp(-localVelX * 0.22, -0.35, 0.35));
    bodyGroup.rotateX(THREE.MathUtils.clamp(speed * 0.12, -0.22, 0.22));

    const wingFlutter = Math.sin(now * 4.5) * 0.06 + Math.min(0.28, speed * 0.25);
    wingL.rotation.z = wingFlutter;
    wingR.rotation.z = -wingFlutter;

    blinkTimer -= delta;
    let eyeScaleY = 1.0;
    if (blinkTimer <= 0) {
      if (blinkTimer > -0.11) eyeScaleY = 0.12;
      else blinkTimer = 2.6 + Math.random() * 3.8;
    }
    eyeL.scale.y = eyeScaleY;
    eyeR.scale.y = eyeScaleY;

    reactorCore.rotation.y = now * 1.4;
    reactorCore.rotation.x = now * 0.8;
    gyroRing1.rotation.z = now * 1.2;
    gyroRing2.rotation.x = now * 0.9;

    updateTentacleIK(delta, now);
    stepCapePhysics(delta, now);

    const targetExt = boardState.visible ? 1.0 : 0.0;
    boardState.extend += (targetExt - boardState.extend) * Math.min(1, delta * 8.5);
    if (boardState.extend > 0.02) {
      boardContainer.visible = true;
      boomArm.visible = true;
      const offsetX = -0.58 * boardState.extend;
      const offsetY = 0.06 * boardState.extend;
      const offsetZ = 0.12 * boardState.extend;
      boardContainer.position.set(offsetX, offsetY, offsetZ);
      boardContainer.scale.setScalar(Math.max(0.05, boardState.extend));
      boardPivot.lookAt(hp.x, botRoot.position.y, hp.z);

      const armStart = new THREE.Vector3(-0.14, 0, 0);
      const armEnd = new THREE.Vector3(offsetX + BOARD_W * 0.45 * boardState.extend, offsetY, offsetZ);
      const armLen = Math.max(0.05, armStart.distanceTo(armEnd));
      boomArm.position.lerpVectors(armStart, armEnd, 0.5);
      boomArm.scale.set(1, armLen, 1);
      boomArm.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), armEnd.clone().sub(armStart).normalize());
    } else {
      boardContainer.visible = false;
      boomArm.visible = false;
    }

    if (state.heldObj && (state.mode === 'holding' || state.mode === 'handing')) {
      const forwardDir = new THREE.Vector3(0, 0, 1).applyQuaternion(clawGroup.quaternion);
      tmpVecA.copy(clawGroup.position).addScaledVector(forwardDir, 0.045 + state.heldRadius * 0.35);
      state.heldObj.position.lerp(tmpVecA, 1 - Math.exp(-16 * delta));
      if (state.heldEnt?.physics?.vel) state.heldEnt.physics.vel.set(0, 0, 0);
    }

    const targetEnergy = (APP.botSpeaking || APP.ttsSpeaking)
      ? 0.55 + Math.abs(Math.sin(now * 16.0)) * 0.45
      : (APP.micLevel || 0);
    audioEnergy += (targetEnergy - audioEnergy) * Math.min(1, delta * 12);
    plasmaMat.emissiveIntensity = 1.1 + audioEnergy * 2.4;
    botLight.intensity = 0.6 + audioEnergy * 1.2;
  }

  function buildSceneMemorySnapshot() {
    const ents = APP.entities || [];
    if (!ents.length) return 'Scene is currently empty.';
    const lines = [];
    for (let i = Math.max(0, ents.length - 12); i < ents.length; i++) {
      const e = ents[i];
      if (!e || !e.group) continue;
      e.group.updateMatrixWorld(true);
      const box = new THREE.Box3().setFromObject(e.group);
      const size = new THREE.Vector3();
      box.getSize(size);
      const partNames = [];
      e.group.traverse((c) => {
        if (c !== e.group && c.name) partNames.push(c.name);
      });
      const wp = e.group.position;
      lines.push(
        `• id="${e.id}" name="${e.name}" pos=[${wp.x.toFixed(2)},${wp.y.toFixed(2)},${wp.z.toFixed(2)}] ` +
        `localSize=[w:${size.x.toFixed(2)},h:${size.y.toFixed(2)},d:${size.z.toFixed(2)}] ` +
        `topY=${size.y.toFixed(2)} parts=[${partNames.slice(0, 10).join(',')}]` +
        (e.meta?.summary ? ` (${e.meta.summary})` : '')
      );
    }
    if (APP.lastEntityId) lines.push(`Most recently referenced entity: "${APP.lastEntityId}".`);
    return lines.join('\n');
  }

  function modifyExistingEntity(entityId, action, jsCode, summaryNote) {
    const ent = getEntityRecord(entityId || APP.lastEntityId || 'latest');
    if (!ent || !ent.group) {
      return { ok: false, error: `Entity "${entityId}" not found.` };
    }
    const targetGroup = ent.group;
    targetGroup.updateMatrixWorld(true);

    const localBox = new THREE.Box3();
    const invRoot = targetGroup.matrixWorld.clone().invert();
    targetGroup.traverse((c) => {
      if (c.isMesh && c.geometry) {
        c.geometry.computeBoundingBox();
        const b = c.geometry.boundingBox.clone();
        b.applyMatrix4(invRoot.clone().multiply(c.matrixWorld));
        localBox.union(b);
      }
    });
    const size = new THREE.Vector3();
    const center = new THREE.Vector3();
    if (!localBox.isEmpty()) {
      localBox.getSize(size);
      localBox.getCenter(center);
    }

    const info = {
      box: localBox,
      size,
      center,
      minY: isFinite(localBox.min.y) ? localBox.min.y : 0,
      maxY: isFinite(localBox.max.y) ? localBox.max.y : size.y,
      topY: isFinite(localBox.max.y) ? localBox.max.y : size.y,
      findPart: (q) => {
        const low = String(q || '').toLowerCase();
        let found = null;
        targetGroup.traverse((c) => {
          if (!found && c !== targetGroup && String(c.name || '').toLowerCase().includes(low)) found = c;
        });
        return found;
      }
    };

    try {
      const V3 = (x = 0, y = 0, z = 0) => new THREE.Vector3(x, y, z);
      const clean = String(jsCode || '').trim().replace(/^```(?:javascript|js)?/i, '').replace(/```$/i, '').trim();
      const fn = new Function('APP', 'targetGroup', 'THREE', 'V3', 'info', clean);
      fn(APP, targetGroup, THREE, V3, info);

      targetGroup.traverse((c) => {
        if (c.isMesh) {
          c.castShadow = true;
          c.receiveShadow = true;
          c.userData.entity = ent;
        }
      });

      if (summaryNote) {
        ent.meta = ent.meta || {};
        ent.meta.summary = ((ent.meta.summary || '') + '; ' + summaryNote).slice(-180);
      }
      APP.lastEntityId = ent.id;
      APP.visual?.applyDissolveShader?.(targetGroup, 'materialize');
      tentacle_show(ent);
      return { ok: true, id: ent.id, name: ent.name, message: `Modified ${ent.name} (${action}).` };
    } catch (err) {
      return { ok: false, error: `${err.name}: ${err.message}` };
    }
  }

  async function resolveBoardImage(name) {
    const key = String(name || '').replace(/\.[^.]+$/, '').toLowerCase().trim();
    if (!key) return null;
    let url = null;
    if (APP.importedImages) {
      const rec = APP.importedImages.get(key) || Array.from(APP.importedImages.values()).find((r) => r.cleanName.includes(key));
      if (rec) url = rec.url;
    }
    if (!url) {
      for (const p of [`./assets/${key}.jpg`, `./assets/${key}.png`, `./assets/${key}.webp`, `./${key}.jpg`, `./${key}.png`]) {
        try { const h = await fetch(p, { method: 'HEAD' }); if (h.ok) { url = p; break; } } catch (_) {}
      }
    }
    if (!url) return null;
    return new Promise((resolve) => {
      const img = new Image();
      img.crossOrigin = 'anonymous';
      img.onload = () => resolve(img);
      img.onerror = () => resolve(null);
      img.src = url;
    });
  }

  function ensureUnlockedAudioContext() {
    if (!audioCtx) {

  function floatTo16BitPCMBase64(float32, inputSampleRate) {
    const targetRate = 16000;
    const ratio = inputSampleRate / targetRate;
    const outLen = Math.max(1, Math.floor(float32.length / ratio));
    const buf = new ArrayBuffer(outLen * 2);
    const view = new DataView(buf);
    let sumSq = 0;
    for (let i = 0; i < outLen; i++) {
      const pos = i * ratio;
      const i0 = Math.floor(pos);
      const i1 = Math.min(float32.length - 1, i0 + 1);
      const frac = pos - i0;
      const s = Math.max(-1, Math.min(1, float32[i0] * (1 - frac) + float32[i1] * frac));
      sumSq += s * s;
      view.setInt16(i * 2, s < 0 ? s * 0x8000 : s * 0x7FFF, true);
    }
    APP.micLevel = Math.min(1, Math.sqrt(sumSq / outLen) * 5.0);
    const bytes = new Uint8Array(buf);
    let bin = '';
    for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
    return btoa(bin);
  }

  function stopLiveAudioPlayback() {
    activePlaySources.forEach((src) => { try { src.stop(); } catch (_) {} });

  function playPCM24kBase64(b64) {
    if (!b64) return;
    const ctx = ensureUnlockedAudioContext();
    const bin = atob(b64);
    const sampleCount = Math.floor(bin.length / 2);
    if (sampleCount <= 0) return;

    const floatBuf = new Float32Array(sampleCount);
    for (let i = 0; i < sampleCount; i++) {
      const lo = bin.charCodeAt(i * 2);
      const hi = bin.charCodeAt(i * 2 + 1);
      let val = (hi << 8) | lo;
      if (val >= 0x8000) val -= 0x10000;
      floatBuf[i] = val / 32768;
    }

    const audioBuf = ctx.createBuffer(1, sampleCount, 24000);
    audioBuf.getChannelData(0).set(floatBuf);
    const src = ctx.createBufferSource();
    src.buffer = audioBuf;
    src.connect(ctx.destination);

    const now = ctx.currentTime;
    if (nextPlayTime < now) nextPlayTime = now + 0.025;
    src.start(nextPlayTime);
    nextPlayTime += audioBuf.duration;

    APP.botSpeaking = true;
    activePlaySources.add(src);
    src.onended = function () {
      activePlaySources.delete(src);
      if (activePlaySources.size === 0) APP.botSpeaking = false;
    };
  }

  async function startMicStreamingForLive() {
    if (!navigator.mediaDevices?.getUserMedia) return;
    const ctx = ensureUnlockedAudioContext();
    stopMicStreamingForLive();

    micStream = await navigator.mediaDevices.getUserMedia({
      audio: { echoCancellation: true, noiseSuppression: true, autoGainControl: true, channelCount: 1 }
    });
    const source = ctx.createMediaStreamSource(micStream);
    const muteGain = ctx.createGain();
    muteGain.gain.value = 0;
    muteGain.connect(ctx.destination);

    let usedWorklet = false;
    if (ctx.audioWorklet && typeof Blob !== 'undefined') {
      try {
        const workletCode = [
          'class PCM16Cap extends AudioWorkletProcessor {',
          '  process(inputs) { const ch = inputs[0]?.[0]; if (ch) this.port.postMessage(ch); return true; }',
          '}',
          'registerProcessor("wxr-geeny-pcm16-v2", PCM16Cap);'
        ].join('\n');
        const url = URL.createObjectURL(new Blob([workletCode], { type: 'application/javascript' }));
        await ctx.audioWorklet.addModule(url);
        URL.revokeObjectURL(url);
        micWorkletNode = new AudioWorkletNode(ctx, 'wxr-geeny-pcm16-v2');
        micWorkletNode.port.onmessage = function (ev) {
          if (!liveWS || liveWS.readyState !== 1 || APP.botSpeaking) return;
          const b64 = floatTo16BitPCMBase64(ev.data, ctx.sampleRate);
          liveWS.send(JSON.stringify({ realtimeInput: { audio: { mimeType: 'audio/pcm;rate=16000', data: b64 } } }));
        };
        source.connect(micWorkletNode);
        micWorkletNode.connect(muteGain);
        usedWorklet = true;
      } catch (_) {}
    }

    if (!usedWorklet) {
      micScriptNode = ctx.createScriptProcessor(2048, 1, 1);
      micScriptNode.onaudioprocess = function (e) {
        if (!liveWS || liveWS.readyState !== 1 || APP.botSpeaking) return;
        const b64 = floatTo16BitPCMBase64(e.inputBuffer.getChannelData(0), ctx.sampleRate);
        liveWS.send(JSON.stringify({ realtimeInput: { audio: { mimeType: 'audio/pcm;rate=16000', data: b64 } } }));
      };
      source.connect(micScriptNode);
      micScriptNode.connect(muteGain);
    }
  }

  function stopMicStreamingForLive() {

        if (msg.setupComplete) {

                realtimeInput: { video: { mimeType: 'image/jpeg', data: String(b64).replace(/^data:image\/\w+;base64,/, '') } }
              }));
            }
          }, 1500);
          if (!settled) { settled = true; resolve({ ok: true }); }
          return;
        }

        if (msg.serverContent) {
          if (msg.serverContent.interrupted) stopLiveAudioPlayback();
          const turn = msg.serverContent.modelTurn;
          if (turn?.parts) {
            for (const part of turn.parts) {
              if (part.inlineData?.data && (!part.inlineData.mimeType || part.inlineData.mimeType.includes('pcm') || part.inlineData.mimeType.startsWith('audio/'))) {
                playPCM24kBase64(part.inlineData.data);

          if (msg.serverContent.outputTranscription?.text) {

          }
        }

        if (msg.toolCall?.functionCalls) {

            responses.push({ id: fc.id, name: fc.name, response: out || { ok: true } });
          }
          if (liveWS && liveWS.readyState === 1) {
            liveWS.send(JSON.stringify({ toolResponse: { functionResponses: responses } }));
          }
        }
      };

      liveWS.onerror = function () {
        if (!settled) { settled = true; resolve({ ok: false, error: `Gemini Live connection failed for "${model.id}".` }); }
      };
      liveWS.onclose = function (ev) {
        liveConnected = false;
        stopMicStreamingForLive();
        if (visionTimer) { clearInterval(visionTimer); visionTimer = null; }
        if (!settled) { settled = true; resolve({ ok: false, error: `Gemini Live closed (${ev.code})` }); }
      };
    });
  }

  // Check if user wants to change cape texture to a specific picture
  const texMatch = raw.match(/(?:texture|picture|image|photo|to|صورة|علم)\s+(?:to\s+|of\s+|the\s+)?([a-zA-Z0-9_\-\.\u0600-\u06FF]+)$/i);
  if ((low.includes('texture') || low.includes('picture') || low.includes('image') || low.includes('change cape') || low.includes('صورة')) && texMatch && texMatch[1]) {
    const res = await APP.bot.setCapeTexture(texMatch[1]);
    if (res?.message) APP.ui?.toast?.(res.message);
    if (!res?.ok && res?.error) APP.ui?.toast?.(res.error, 5000);
    return res;
  }

  if (low.includes('off') || low.includes('hide') || low.includes('remove') || low.includes('اخف')) {
    APP.bot.setCapeEnabled(false);
    APP.ui?.toast?.('Geeny Cape: OFF');
    return { ok: true, message: 'Geeny Cape turned OFF.' };
  }
  if (low.includes('on') || low.includes('show') || low.includes('اظهر')) {
    APP.bot.setCapeEnabled(true);
    APP.ui?.toast?.('Geeny Cape: ON');
    return { ok: true, message: 'Geeny Cape turned ON.' };
  }

  const next = APP.bot.toggleCape();
  APP.ui?.toast?.('Geeny Cape: ' + (next ? 'ON' : 'OFF'));
  return { ok: true, message: 'Geeny Cape is now ' + (next ? 'ON' : 'OFF') + '.' };
})();
