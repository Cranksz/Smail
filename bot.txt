  function drawBoardRoundedRect(x, y, w, h, r, fill, stroke) {
    bCtx.beginPath();
    bCtx.moveTo(x + r, y);
    bCtx.arcTo(x + w, y, x + w, y + h, r);
    bCtx.arcTo(x + w, y + h, x, y + h, r);
    bCtx.arcTo(x, y + h, x, y, r);
    bCtx.arcTo(x, y, x + w, y, r);
    bCtx.closePath();
    if (fill) { bCtx.fillStyle = fill; bCtx.fill(); }
    if (stroke) { bCtx.strokeStyle = stroke; bCtx.lineWidth = 3; bCtx.stroke(); }
  }

  function wrapCanvasText(ctx, text, x, y, maxWidth, lineHeight) {
    const words = String(text || '').split(/\s+/);
    let line = '';
    let cy = y;
    for (let i = 0; i < words.length; i++) {
      const test = line ? line + ' ' + words[i] : words[i];
      if (ctx.measureText(test).width > maxWidth && line) {
        ctx.fillText(line, x, cy);
        line = words[i];
        cy += lineHeight;
      } else {
        line = test;
      }
    }
    if (line) ctx.fillText(line, x, cy);
    return cy + lineHeight;
  }

  function executeDrawCommands(cmds, area) {
    if (!Array.isArray(cmds)) return;
    const ox = area.x, oy = area.y, aw = area.w, ah = area.h;
    const px = (nx) => ox + Number(nx || 0) * aw;
    const py = (ny) => oy + Number(ny || 0) * ah;

    for (const c of cmds) {
      if (!c || !c.type) continue;
      bCtx.save();
      bCtx.strokeStyle = c.color || '#00f0b4';
      bCtx.fillStyle = c.fill || c.color || 'rgba(0,240,180,0.18)';
      bCtx.lineWidth = c.width || 5;
      bCtx.lineCap = 'round';
      bCtx.lineJoin = 'round';

      if (c.type === 'circle') {
        bCtx.beginPath();
        bCtx.arc(px(c.cx ?? 0.5), py(c.cy ?? 0.5), (c.r ?? 0.15) * Math.min(aw, ah), 0, Math.PI * 2);
        if (c.filled) bCtx.fill();
        bCtx.stroke();
      } else if (c.type === 'ellipse') {
        bCtx.beginPath();
        bCtx.ellipse(px(c.cx ?? 0.5), py(c.cy ?? 0.5), (c.rx ?? 0.2) * aw, (c.ry ?? 0.15) * ah, c.rot || 0, 0, Math.PI * 2);
        if (c.filled) bCtx.fill();
        bCtx.stroke();
      } else if (c.type === 'line') {
        bCtx.beginPath();
        bCtx.moveTo(px(c.x1), py(c.y1));
        bCtx.lineTo(px(c.x2), py(c.y2));
        bCtx.stroke();
      } else if (c.type === 'arc') {
        bCtx.beginPath();
        bCtx.arc(px(c.cx ?? 0.5), py(c.cy ?? 0.5), (c.r ?? 0.1) * Math.min(aw, ah), c.start ?? 0, c.end ?? Math.PI);
        bCtx.stroke();
      } else if (c.type === 'poly' && Array.isArray(c.points) && c.points.length >= 2) {
        bCtx.beginPath();
        bCtx.moveTo(px(c.points[0][0]), py(c.points[0][1]));
        for (let i = 1; i < c.points.length; i++) bCtx.lineTo(px(c.points[i][0]), py(c.points[i][1]));
        if (c.closed) bCtx.closePath();
        if (c.filled) bCtx.fill();
        bCtx.stroke();
      } else if (c.type === 'curve') {
        bCtx.beginPath();
        bCtx.moveTo(px(c.x1), py(c.y1));
        bCtx.quadraticCurveTo(px(c.cx), py(c.cy), px(c.x2), py(c.y2));
        bCtx.stroke();
      } else if (c.type === 'text') {
        bCtx.fillStyle = c.color || '#e2f8ff';
        bCtx.font = 'bold ' + (c.size || 24) + 'px system-ui, sans-serif';
        bCtx.textAlign = 'center';
        bCtx.fillText(String(c.text || ''), px(c.x ?? 0.5), py(c.y ?? 0.5));
      }
      bCtx.restore();
    }
  }

      action: () => hideBoard()
    });

    if (boardState.mode === 'tutorial' && boardState.steps.length > 0) {
      const idx = Math.max(0, Math.min(boardState.steps.length - 1, boardState.currentStep));
      const step = boardState.steps[idx];

      drawBoardRoundedRect(sketchBox.x, sketchBox.y, sketchBox.w, sketchBox.h, 18, 'rgba(3, 8, 18, 0.85)', '#1e3a5f');

      // Accumulate drawing steps 0..idx so the drawing builds up progressively!
      for (let s = 0; s <= idx; s++) {
        const st = boardState.steps[s];
        if (st && Array.isArray(st.draw_commands)) {
          executeDrawCommands(st.draw_commands, {
            x: sketchBox.x + 24,

      bCtx.fillStyle = '#38bdf8';
      bCtx.font = 'bold 28px system-ui, sans-serif';
      bCtx.textAlign = 'left';
      bCtx.fillText(`Step ${idx + 1} / ${boardState.steps.length}`, rx + 24, 146);

      bCtx.fillStyle = '#00f0b4';
      bCtx.font = 'bold 30px system-ui, sans-serif';
      let nextY = wrapCanvasText(bCtx, step.title || '', rx + 24, 198, rw - 48, 38);

      bCtx.fillStyle = '#e2e8f0';
      bCtx.font = '25px system-ui, sans-serif';
      wrapCanvasText(bCtx, step.caption || '', rx + 24, nextY + 18, rw - 48, 36);
    } else if (boardState.mode === 'image' && boardState.imageObj) {
      const img = boardState.imageObj;

      drawBoardRoundedRect(box.x, box.y, box.w, box.h, 18, 'rgba(4, 9, 18, 0.9)', '#254a78');
      const aspect = (img.width || 1) / (img.height || 1);
      let dw = box.w - 30, dh = dw / aspect;
      if (dh > box.h - 30) { dh = box.h - 30; dw = dh * aspect; }

        bCtx.font = '26px system-ui, sans-serif';
        bCtx.textAlign = 'center';
        bCtx.fillText(boardState.caption.slice(0, 76), W / 2, 718);

    } else {

      let cy = 148;
      bCtx.textAlign = 'left';

      for (let i = 0; i < lines.length; i++) {
        bCtx.fillStyle = '#00f0b4';
        bCtx.font = 'bold 28px system-ui, sans-serif';
        bCtx.fillText('•', 62, cy);

        bCtx.font = '26px system-ui, sans-serif';

      const fy = H - 84;
      const prev = { x: 36, y: fy, w: 200, h: 56 };
      const next = { x: W - 236, y: fy, w: 200, h: 56 };
      drawBoardRoundedRect(prev.x, prev.y, prev.w, prev.h, 14, '#132640', '#38bdf8');
      drawBoardRoundedRect(next.x, next.y, next.w, next.h, 14, '#0f5c52', '#00f0b4');
      bCtx.fillStyle = '#e0f2fe';
      bCtx.font = 'bold 24px system-ui, sans-serif';
      bCtx.textAlign = 'center';
      bCtx.fillText('< Prev Step', prev.x + prev.w / 2, prev.y + prev.h / 2);
      bCtx.fillText('Next Step >', next.x + next.w / 2, next.y + next.h / 2);
      boardState.hitZones.push({
        id: 'prev', u0: prev.x / W, u1: (prev.x + prev.w) / W, v0: 1 - (prev.y + prev.h) / H, v1: 1 - prev.y / H,
        action: () => stepBoard(-1)
      });
      boardState.hitZones.push({
        id: 'next', u0: next.x / W, u1: (next.x + next.w) / W, v0: 1 - (next.y + next.h) / H, v1: 1 - next.y / H,
        action: () => stepBoard(1)

  boardScreen.userData.onPoke = function (hit) {
    if (!boardState.visible || !hit || !hit.uv) return;
    const now = performance.now();
    if (now - boardState.lastClick < 280) return;
    const uv = hit.uv;
    for (const z of boardState.hitZones) {
      if (uv.x >= z.u0 && uv.x <= z.u1 && uv.y >= z.v0 && uv.y <= z.v1) {
        boardState.lastClick = now;
        z.action();
        break;
      }
    }
  };
  APP.pokeTargets?.add(boardScreen);

  function stepBoard(delta) {
    if (!boardState.steps.length) return;
    boardState.currentStep = Math.max(0, Math.min(boardState.steps.length - 1, boardState.currentStep + delta));
    redrawBoard();
    const st = boardState.steps[boardState.currentStep];
    if (st?.caption) APP.speakTTS?.(`${st.title}. ${st.caption}`);
  }

  function showBoard(opts = {}) {
    if (!botEnabled) setEnabled(true);
    boardState.visible = true;
    boardState.title = opts.title || 'GEENY HOLO-BOARD';
    boardState.mode = opts.mode || (Array.isArray(opts.steps) && opts.steps.length ? 'tutorial' : opts.imageObj ? 'image' : 'notes');
    boardState.steps = Array.isArray(opts.steps) ? opts.steps : [];
    boardState.currentStep = Math.max(0, opts.stepIndex || 0);
    boardState.notes = Array.isArray(opts.notes) ? opts.notes : [];
    boardState.imageObj = opts.imageObj || null;
    boardState.caption = opts.caption || '';

    boardContainer.visible = true;
    boomArm.visible = true;
    redrawBoard();

    // Bring Geeny comfortably into view so user can read the board easily
    const hp = APP.headPos || new THREE.Vector3(0, 1.5, 0);
    const hf = APP.headFwd || new THREE.Vector3(0, 0, -1);
    const flat = new THREE.Vector3(hf.x, 0, hf.z).normalize();
    if (flat.lengthSq() < 0.01) flat.set(0, 0, -1);
    const right = new THREE.Vector3().crossVectors(flat, new THREE.Vector3(0, 1, 0)).normalize();
    state.worldAnchor.copy(hp).addScaledVector(flat, 1.05).addScaledVector(right, 0.36);
    state.worldAnchor.y = Math.max(1.05, hp.y + 0.04);

    return { ok: true, message: 'Geeny extended holographic board: ' + boardState.title };
  }

  function hideBoard() {
    boardState.visible = false;
    return { ok: true, message: 'Geeny retracted holographic board.' };
  }

            { type: 'ellipse', cx: 0.57, cy: 0.82, rx: 0.045, ry: 0.03, color: '#00f0b4', width: 5 },
            { type: 'curve', x1: 0.66, y1: 0.72, cx: 0.88, cy: 0.68, x2: 0.80, y2: 0.42, color: '#00f0b4', width: 7 }
          ]
        }
      ]
    };
  }

  // ====================================================================
  // 3. LONG 18-SEGMENT DYNAMIC TENTACLE + SERVO CLAW
  // ====================================================================
  const TENTACLE_SEGMENTS = 18;
  const tentacleGroup = new THREE.Group();
  tentacleGroup.name = 'APP_GEENY_TENTACLE';
  scene.add(tentacleGroup);

  const segMatDark = new THREE.MeshStandardMaterial({ color: 0x152030, metalness: 0.88, roughness: 0.24 });
  const segMatGlow = new THREE.MeshStandardMaterial({
    color: 0x00f0b4,
    emissive: 0x0ea5e9,
    emissiveIntensity: 0.95,
    roughness: 0.22
  });

  const segments = [];
  for (let i = 0; i < TENTACLE_SEGMENTS; i++) {
    const t = i / (TENTACLE_SEGMENTS - 1);
    const radius = THREE.MathUtils.lerp(0.028, 0.0095, t);
    const segGroup = new THREE.Group();
    const body = new THREE.Mesh(new THREE.CylinderGeometry(radius * 0.88, radius, 0.068, 12), segMatDark);
    body.rotation.x = Math.PI / 2;
    segGroup.add(body);
    const ring = new THREE.Mesh(new THREE.TorusGeometry(radius * 0.98, radius * 0.16, 6, 16), segMatGlow);
    segGroup.add(ring);
    tentacleGroup.add(segGroup);
    segments.push(segGroup);
  }

  const clawGroup = new THREE.Group();
  tentacleGroup.add(clawGroup);
  const palmMesh = new THREE.Mesh(new THREE.CylinderGeometry(0.017, 0.013, 0.028, 14), segMatDark);
  palmMesh.rotation.x = Math.PI / 2;
  clawGroup.add(palmMesh);

  const prongs = [];
  for (let p = 0; p < 3; p++) {
    const pivot = new THREE.Group();
    pivot.rotation.z = (p / 3) * Math.PI * 2;
    const fp = new THREE.Group();
    fp.position.set(0.015, 0, 0.012);
    const finger = new THREE.Mesh(new THREE.BoxGeometry(0.007, 0.008, 0.040), segMatDark);
    finger.position.set(0.004, 0, 0.020);
    const tipGlow = new THREE.Mesh(new THREE.SphereGeometry(0.005, 8, 8), segMatGlow);
    tipGlow.position.set(-0.002, 0, 0.041);
    fp.add(finger, tipGlow);
    pivot.add(fp);
    clawGroup.add(pivot);
    prongs.push(fp);
  }

  const showBeam = new THREE.Mesh(
    new THREE.ConeGeometry(0.14, 0.60, 24, 1, true),
    new THREE.MeshBasicMaterial({
      color: 0x00f0b4,
      transparent: true,
      opacity: 0.24,
      blending: THREE.AdditiveBlending,
      depthWrite: false,
      side: THREE.DoubleSide
    })
  );
  showBeam.rotation.x = -Math.PI / 2;
  showBeam.position.z = 0.32;
  showBeam.visible = false;
  clawGroup.add(showBeam);

  // ====================================================================
  // 4. SPRING-DAMPED FLIGHT PHYSICS + LONG TENTACLE IK
  // ====================================================================
  let botEnabled = true;
  let audioEnergy = 0;
  let clawClosure = 0.2;
  let clawTargetClosure = 0.2;
  let blinkTimer = 3.0;

  const state = {
    mode: 'idle',
    targetObj: null,
    heldObj: null,
    heldEnt: null,
    heldRadius: 0.12,
    worldAnchor: new THREE.Vector3(0.46, 1.48, -1.05),
    hoverTargetPos: new THREE.Vector3(0.46, 1.48, -1.05),
    velocity: new THREE.Vector3(),
    tipTargetPos: new THREE.Vector3(0.46, 1.02, -1.05),
    currentTipPos: new THREE.Vector3(0.46, 1.02, -1.05),
    showTimer: 0,
    grabProgress: 0,
    nextThoughtTime: 5.0,
    lookTargetPos: new THREE.Vector3(0, 1.5, 0)
  };

  const vP0 = new THREE.Vector3();
  const vC1 = new THREE.Vector3();
  const vC2 = new THREE.Vector3();
  const vP3 = new THREE.Vector3();
  const vPt = new THREE.Vector3();
  const vNext = new THREE.Vector3();
  const vObjCenter = new THREE.Vector3();
  const vObjSize = new THREE.Vector3();
  const tmpBox = new THREE.Box3();
  const tmpVecA = new THREE.Vector3();
  const tmpVecB = new THREE.Vector3();

  function getEntityRecord(target) {
    if (!target) return null;
    if (typeof target === 'object' && target.group) return target;
    if (target.isObject3D && target.userData?.entity) return target.userData.entity;
    return APP.findEntity ? APP.findEntity(target) : null;
  }

  function getEntityObject3D(target) {
    if (!target) return null;
    if (target.isObject3D) return target;
    if (target.group?.isObject3D) return target.group;
    if (target.mesh?.isObject3D) return target.mesh;
    const rec = getEntityRecord(target);
    return rec ? (rec.group || rec.mesh) : null;
  }

  function computeObjCenterAndRadius(obj) {
    obj.updateMatrixWorld(true);
    tmpBox.setFromObject(obj);
    if (tmpBox.isEmpty()) {
      obj.getWorldPosition(vObjCenter);
      return { center: vObjCenter.clone(), radius: 0.12 };
    }
    tmpBox.getCenter(vObjCenter);
    tmpBox.getSize(vObjSize);
    const radius = Math.max(0.06, Math.min(0.32, Math.max(vObjSize.x, vObjSize.z) * 0.45));
    return { center: vObjCenter.clone(), radius };
  }

  function releaseHeldEntity() {
    if (state.heldEnt) { state.heldEnt.heldByBot = false; state.heldEnt = null; }
    if (state.heldObj) {
      state.heldObj.userData = state.heldObj.userData || {};
      state.heldObj.userData.heldByBot = false;
      state.heldObj = null;
    }
  }

  function tentacle_release() {
    releaseHeldEntity();
    state.mode = 'idle';
    state.targetObj = null;
    state.worldAnchor.copy(botRoot.position);
    clawTargetClosure = 0.08;
    showBeam.visible = false;
    return { ok: true, message: 'Geeny released the object.' };
  }

  function tentacle_grab(target) {
    if (!botEnabled) setEnabled(true);
    const obj = getEntityObject3D(target || 'latest');
    if (!obj) return { ok: false, error: 'Target object not found to grab: ' + String(target || '') };
    releaseHeldEntity();
    state.mode = 'grabbing';
    state.targetObj = obj;
    state.grabProgress = 0;
    clawTargetClosure = 0.0;
    showBeam.visible = false;
    return { ok: true, message: 'Geeny extending long tentacle to grab ' + (obj.userData?.entity?.name || obj.name || 'object') + '.' };
  }

  function tentacle_hand_to_user(target) {
    if (!botEnabled) setEnabled(true);
    let obj = getEntityObject3D(target || 'latest');
    if (!obj && state.heldObj) obj = state.heldObj;
    if (!obj) return { ok: false, error: 'No target object found to hand to user.' };
    if (state.heldObj !== obj) {
      releaseHeldEntity();
      state.targetObj = obj;
      const info = computeObjCenterAndRadius(obj);
      state.heldObj = obj;
      state.heldEnt = getEntityRecord(obj);
      state.heldRadius = info.radius;
      if (state.heldEnt) state.heldEnt.heldByBot = true;
      obj.userData = obj.userData || {};
      obj.userData.heldByBot = true;
    }
    state.mode = 'handing';
    clawTargetClosure = 0.85;
    showBeam.visible = false;
    return { ok: true, message: 'Geeny bringing ' + (state.heldEnt?.name || obj.name || 'object') + ' to your hand.' };
  }

  function tentacle_show(target) {
    if (!botEnabled) setEnabled(true);
    const obj = getEntityObject3D(target || 'latest');
    if (!obj) return { ok: false, error: 'Target object not found to showcase.' };
    if (!state.heldObj || state.heldObj === obj) releaseHeldEntity();
    state.mode = 'showing';
    state.targetObj = obj;
    state.showTimer = 4.5;
    clawTargetClosure = 0.25;
    showBeam.visible = true;
    return { ok: true, message: 'Geeny showcasing ' + (obj.userData?.entity?.name || obj.name || 'object') + '.' };
  }

  function evalCubicBezier(p0, c1, c2, p3, t, out) {
    const u = 1 - t;
    out.set(0, 0, 0)
      .addScaledVector(p0, u * u * u)
      .addScaledVector(c1, 3 * u * u * t)
      .addScaledVector(c2, 3 * u * t * t)
      .addScaledVector(p3, t * t * t);
    return out;
  }

  function updateTentacleIK(dt, elapsed) {
    vP0.set(0, -0.14, 0);
    botRoot.localToWorld(vP0);

    const tipLerp = 1 - Math.exp(-9.5 * dt);
    state.currentTipPos.lerp(state.tipTargetPos, tipLerp);
    vP3.copy(state.currentTipPos);

    const dist = vP0.distanceTo(vP3);
    // Velocity lag curves the long tentacle organically behind flight motion
    vC1.copy(vP0)
      .add(new THREE.Vector3(0, -Math.max(0.24, dist * 0.48), 0))
      .addScaledVector(state.velocity, -0.18);

    tmpVecA.subVectors(vP3, vP0).normalize();
    vC2.copy(vP3)
      .addScaledVector(tmpVecA, -Math.max(0.18, dist * 0.36))
      .addScaledVector(state.velocity, -0.12);
    vC2.y += Math.sin(elapsed * 2.2) * 0.045 + 0.08;

    for (let i = 0; i < TENTACLE_SEGMENTS; i++) {
      const t = i / TENTACLE_SEGMENTS;
      const tNext = (i + 1) / TENTACLE_SEGMENTS;
      evalCubicBezier(vP0, vC1, vC2, vP3, t, vPt);
      evalCubicBezier(vP0, vC1, vC2, vP3, tNext, vNext);
      segments[i].position.copy(vPt);
      segments[i].lookAt(vNext);
    }

    clawGroup.position.copy(vP3);
    if (state.mode === 'showing' && state.targetObj) {
      const info = computeObjCenterAndRadius(state.targetObj);
      clawGroup.lookAt(info.center);
      const beamDist = Math.max(0.15, Math.min(0.95, vP3.distanceTo(info.center)));
      showBeam.scale.set(1, 1, beamDist / 0.60);
      showBeam.position.z = beamDist * 0.52;
    } else {
      evalCubicBezier(vP0, vC1, vC2, vP3, 0.9, tmpVecB);
      tmpVecA.subVectors(vP3, tmpVecB).normalize();
      if (tmpVecA.lengthSq() > 0.001) clawGroup.lookAt(tmpVecB.copy(vP3).add(tmpVecA));
    }

    clawClosure += (clawTargetClosure - clawClosure) * Math.min(1, dt * 10);
    const openAngle = THREE.MathUtils.lerp(0.62, -0.14, clawClosure);
    for (let p = 0; p < prongs.length; p++) prongs[p].rotation.y = openAngle;
  }

  function updateRobot(dt, elapsed) {
    if (!botEnabled) return;
    const delta = Math.min(0.05, dt || 0.016);
    const now = elapsed || (performance.now() * 0.001);

    const hp = APP.headPos || new THREE.Vector3(0, 1.5, 0);
    const hf = APP.headFwd || new THREE.Vector3(0, 0, -1);
    const flatFwd = new THREE.Vector3(hf.x, 0, hf.z).normalize();
    if (flatFwd.lengthSq() < 0.01) flatFwd.set(0, 0, -1);
    const right = new THREE.Vector3().crossVectors(flatFwd, new THREE.Vector3(0, 1, 0)).normalize();

    if (state.mode === 'idle') {
      const distToUser = state.worldAnchor.distanceTo(hp);
      if (distToUser > 3.0) {
        state.worldAnchor.copy(hp).addScaledVector(flatFwd, 1.15).addScaledVector(right, 0.45);
        state.worldAnchor.y = Math.max(1.05, hp.y + 0.04);
      } else if (distToUser < 0.45) {
        tmpVecA.subVectors(state.worldAnchor, hp).setY(0).normalize();
        if (tmpVecA.lengthSq() < 0.01) tmpVecA.copy(flatFwd);
        state.worldAnchor.copy(hp).addScaledVector(tmpVecA, 0.72);
        state.worldAnchor.y = Math.max(1.05, hp.y + 0.04);
      }

      if (now > state.nextThoughtTime && !boardState.visible) {
        state.nextThoughtTime = now + 7 + Math.random() * 5;
        const ents = Array.isArray(APP.entities) ? APP.entities : [];
        if (ents.length > 0 && Math.random() < 0.58) {
          const pick = ents[Math.floor(Math.random() * ents.length)];
          const obj = pick?.group || pick?.mesh;
          if (obj && !pick.isHeld) {
            const info = computeObjCenterAndRadius(obj);
            if (info.center.distanceTo(hp) < 2.6) {
              state.worldAnchor.copy(info.center).add(new THREE.Vector3((Math.random() - 0.5) * 0.45, 0.38, (Math.random() - 0.5) * 0.45));
              state.lookTargetPos.copy(info.center);
            }
          }
        } else {
          state.lookTargetPos.copy(hp);
        }
      } else if (APP.botSpeaking || APP.micLevel > 0.12 || boardState.visible) {
        state.lookTargetPos.copy(hp);
      }

      state.hoverTargetPos.copy(state.worldAnchor).add(new THREE.Vector3(
        Math.sin(now * 0.7) * 0.055,
        Math.sin(now * 1.7) * 0.035,
        Math.cos(now * 0.6) * 0.045
      ));

      // Long tentacle gracefully sweeps underneath Geeny in idle
      state.tipTargetPos.copy(botRoot.position).add(new THREE.Vector3(
        Math.sin(now * 1.25) * 0.14,
        -0.46 + Math.cos(now * 1.55) * 0.05,
        -0.08 + Math.cos(now * 0.9) * 0.10
      ));
    } else if (state.mode === 'grabbing') {
      if (!state.targetObj || !state.targetObj.parent) {
        tentacle_release();
      } else {
        const info = computeObjCenterAndRadius(state.targetObj);
        state.hoverTargetPos.copy(info.center).add(new THREE.Vector3(0.28, 0.46, 0.22));
        state.tipTargetPos.copy(info.center);
        state.lookTargetPos.copy(info.center);

        if (state.currentTipPos.distanceTo(info.center) < 0.16) {
          state.grabProgress += delta;
          clawTargetClosure = 0.88;
          if (state.grabProgress > 0.22) {
            state.heldObj = state.targetObj;
            state.heldEnt = getEntityRecord(state.heldObj);
            state.heldRadius = info.radius;
            if (state.heldEnt) state.heldEnt.heldByBot = true;
            state.heldObj.userData = state.heldObj.userData || {};
            state.heldObj.userData.heldByBot = true;
            state.mode = 'holding';
          }
        }
      }
    } else if (state.mode === 'holding') {
      if (!state.heldObj || !state.heldObj.parent) {
        tentacle_release();
      } else {
        state.hoverTargetPos.copy(state.worldAnchor).add(new THREE.Vector3(0, Math.sin(now * 1.5) * 0.025, 0));
        state.tipTargetPos.copy(botRoot.position).add(new THREE.Vector3(-0.10, -0.44, 0.22));
        state.lookTargetPos.copy(hp);
      }
    } else if (state.mode === 'handing') {
      if (!state.heldObj || !state.heldObj.parent) {
        tentacle_release();
      } else {
        state.hoverTargetPos.copy(hp).addScaledVector(flatFwd, 0.72);
        state.hoverTargetPos.y = Math.max(0.92, hp.y + 0.02);

        let handAnchor = null;
        let handGrabbing = false;
        const hands = APP.hands || {};
        for (const k of Object.keys(hands)) {
          const h = hands[k];
          if (!h || !h.active) continue;
          const pos = h.isHand ? h.fingerTips?.index : h.pos;
          if (pos) {
            if (!handAnchor || pos.distanceTo(state.currentTipPos) < handAnchor.distanceTo(state.currentTipPos)) {
              handAnchor = pos;
            }
            if ((h.pinching || h.trigger || h.gripping) && pos.distanceTo(state.currentTipPos) < 0.22) {
              handGrabbing = true;
            }
          }
        }

        if (handAnchor && handAnchor.distanceTo(hp) < 1.1) {
          state.tipTargetPos.copy(handAnchor).addScaledVector(flatFwd, -0.05);
        } else {
          state.tipTargetPos.copy(hp).addScaledVector(flatFwd, 0.42);
          state.tipTargetPos.y = Math.max(0.75, hp.y - 0.20);
        }
        state.lookTargetPos.copy(hp);

        if (handGrabbing || state.heldEnt?.grabbedByHand) tentacle_release();
      }
    } else if (state.mode === 'showing') {
      if (!state.targetObj || !state.targetObj.parent) {
        tentacle_release();
      } else {
        const info = computeObjCenterAndRadius(state.targetObj);
        state.hoverTargetPos.copy(info.center).add(new THREE.Vector3(0.36, 0.40 + Math.sin(now * 2.0) * 0.03, 0.22));
        state.tipTargetPos.copy(info.center).add(new THREE.Vector3(0.18, 0.12, 0.12));
        state.worldAnchor.copy(state.hoverTargetPos);
        state.lookTargetPos.copy(info.center);

        state.showTimer -= delta;
        if (state.showTimer <= 0) tentacle_release();
      }
    }

    // Spring-damper flight physics + dynamic banking tilt
    const stiffness = 22.0;
    const damping = 8.4;
    tmpVecA.subVectors(state.hoverTargetPos, botRoot.position).multiplyScalar(stiffness);
    state.velocity.addScaledVector(tmpVecA, delta);
    state.velocity.multiplyScalar(Math.exp(-damping * delta));
    botRoot.position.addScaledVector(state.velocity, delta);

    // Orient Geeny chassis smoothly toward lookTargetPos + bank into lateral motion
    bodyGroup.lookAt(state.lookTargetPos);
    const localVelX = state.velocity.x;
    const speed = state.velocity.length();
    bodyGroup.rotateZ(THREE.MathUtils.clamp(-localVelX * 0.22, -0.35, 0.35));
    bodyGroup.rotateX(THREE.MathUtils.clamp(speed * 0.12, -0.22, 0.22));

    // Wing stabilizer dynamic flare
    const wingFlutter = Math.sin(now * 4.5) * 0.06 + Math.min(0.28, speed * 0.25);
    wingL.rotation.z = wingFlutter;
    wingR.rotation.z = -wingFlutter;

    // Eye blink & audio reaction
    blinkTimer -= delta;
    let eyeScaleY = 1.0;
    if (blinkTimer <= 0) {
      if (blinkTimer > -0.11) eyeScaleY = 0.12;
      else blinkTimer = 2.6 + Math.random() * 3.8;
    }
    eyeL.scale.y = eyeScaleY;
    eyeR.scale.y = eyeScaleY;

    reactorCore.rotation.y = now * 1.4;
    reactorCore.rotation.x = now * 0.8;
    gyroRing1.rotation.z = now * 1.2;
    gyroRing2.rotation.x = now * 0.9;

    updateTentacleIK(delta, now);

    // Animate extendable holographic board & telescoping arm
    const targetExt = boardState.visible ? 1.0 : 0.0;
    boardState.extend += (targetExt - boardState.extend) * Math.min(1, delta * 8.5);
    if (boardState.extend > 0.02) {
      boardContainer.visible = true;
      boomArm.visible = true;
      const offsetX = -0.58 * boardState.extend;
      const offsetY = 0.06 * boardState.extend;
      const offsetZ = 0.12 * boardState.extend;
      boardContainer.position.set(offsetX, offsetY, offsetZ);
      boardContainer.scale.setScalar(Math.max(0.05, boardState.extend));
      boardPivot.lookAt(hp.x, botRoot.position.y, hp.z);

      // Connect telescoping arm from chassis edge to board frame edge
      const armStart = new THREE.Vector3(-0.14, 0, 0);
      const armEnd = new THREE.Vector3(offsetX + BOARD_W * 0.45 * boardState.extend, offsetY, offsetZ);
      const armLen = Math.max(0.05, armStart.distanceTo(armEnd));
      boomArm.position.lerpVectors(armStart, armEnd, 0.5);
      boomArm.scale.set(1, armLen, 1);
      boomArm.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), armEnd.clone().sub(armStart).normalize());
    } else {
      boardContainer.visible = false;
      boomArm.visible = false;
    }

    if (state.heldObj && (state.mode === 'holding' || state.mode === 'handing')) {
      const forwardDir = new THREE.Vector3(0, 0, 1).applyQuaternion(clawGroup.quaternion);
      tmpVecA.copy(clawGroup.position).addScaledVector(forwardDir, 0.045 + state.heldRadius * 0.35);
      state.heldObj.position.lerp(tmpVecA, 1 - Math.exp(-16 * delta));
      if (state.heldEnt?.physics?.vel) state.heldEnt.physics.vel.set(0, 0, 0);
    }

    const targetEnergy = (APP.botSpeaking || APP.ttsSpeaking)
      ? 0.55 + Math.abs(Math.sin(now * 16.0)) * 0.45
      : (APP.micLevel || 0);
    audioEnergy += (targetEnergy - audioEnergy) * Math.min(1, delta * 12);
    plasmaMat.emissiveIntensity = 1.1 + audioEnergy * 2.4;
    botLight.intensity = 0.6 + audioEnergy * 1.2;
  }

  function setEnabled(val) {
    botEnabled = !!val;
    botRoot.visible = botEnabled;
    tentacleGroup.visible = botEnabled;
    if (!botEnabled) {
      tentacle_release();
      hideBoard();
      disconnectLive();
    }
    APP.emit?.('bot:state_changed', botEnabled);
    APP.toolbar?.syncState?.();
    return botEnabled;
  }

  function buildSceneMemorySnapshot() {
    const ents = APP.entities || [];
    if (!ents.length) return 'Scene is currently empty.';
    const lines = [];
    for (let i = Math.max(0, ents.length - 12); i < ents.length; i++) {
      const e = ents[i];
      if (!e || !e.group) continue;
      e.group.updateMatrixWorld(true);
      const box = new THREE.Box3().setFromObject(e.group);
      const size = new THREE.Vector3();
      box.getSize(size);
      const partNames = [];
      e.group.traverse((c) => {
        if (c !== e.group && c.name) partNames.push(c.name);
      });
      const wp = e.group.position;
      lines.push(
        `• id="${e.id}" name="${e.name}" pos=[${wp.x.toFixed(2)},${wp.y.toFixed(2)},${wp.z.toFixed(2)}] ` +
        `localSize=[w:${size.x.toFixed(2)},h:${size.y.toFixed(2)},d:${size.z.toFixed(2)}] ` +
        `topY=${size.y.toFixed(2)} parts=[${partNames.slice(0, 10).join(',')}]` +
        (e.meta?.summary ? ` (${e.meta.summary})` : '')
      );
    }
    if (APP.lastEntityId) lines.push(`Most recently referenced entity: "${APP.lastEntityId}".`);
    return lines.join('\n');
  }

  function modifyExistingEntity(entityId, action, jsCode, summaryNote) {
    const ent = getEntityRecord(entityId || APP.lastEntityId || 'latest');
    if (!ent || !ent.group) {
      return { ok: false, error: `Entity "${entityId}" not found in scene. Existing entities: ${(APP.entities || []).map(e => e.id + ':' + e.name).join(', ')}` };
    }
    const targetGroup = ent.group;
    targetGroup.updateMatrixWorld(true);

    // Compute local bounding box relative to targetGroup so attachments align accurately
    const localBox = new THREE.Box3();
    const invRoot = targetGroup.matrixWorld.clone().invert();
    targetGroup.traverse((c) => {
      if (c.isMesh && c.geometry) {
        c.geometry.computeBoundingBox();
        const b = c.geometry.boundingBox.clone();
        const relMat = invRoot.clone().multiply(c.matrixWorld);
        b.applyMatrix4(relMat);
        localBox.union(b);
      }
    });
    const size = new THREE.Vector3();
    const center = new THREE.Vector3();
    if (!localBox.isEmpty()) {
      localBox.getSize(size);
      localBox.getCenter(center);
    }

    const info = {
      box: localBox,
      size,
      center,
      minY: isFinite(localBox.min.y) ? localBox.min.y : 0,
      maxY: isFinite(localBox.max.y) ? localBox.max.y : size.y,
      topY: isFinite(localBox.max.y) ? localBox.max.y : size.y,
      findPart: (q) => {
        const low = String(q || '').toLowerCase();
        let found = null;
        targetGroup.traverse((c) => {
          if (!found && c !== targetGroup && String(c.name || '').toLowerCase().includes(low)) found = c;
        });
        return found;
      }
    };

    try {
      const V3 = (x = 0, y = 0, z = 0) => new THREE.Vector3(x, y, z);
      const clean = String(jsCode || '').trim().replace(/^```(?:javascript|js)?/i, '').replace(/```$/i, '').trim();
      const fn = new Function('APP', 'targetGroup', 'THREE', 'V3', 'info', clean);
      fn(APP, targetGroup, THREE, V3, info);

      targetGroup.traverse((c) => {
        if (c.isMesh) {
          c.castShadow = true;
          c.receiveShadow = true;
          c.userData.entity = ent;
        }
      });

      if (summaryNote) {
        ent.meta = ent.meta || {};
        ent.meta.summary = ((ent.meta.summary || '') + '; ' + summaryNote).slice(-180);
      }
      APP.lastEntityId = ent.id;
      APP.visual?.applyDissolveShader?.(targetGroup, 'materialize');
      tentacle_show(ent);
      return { ok: true, id: ent.id, name: ent.name, message: `Modified ${ent.name} (${action}).` };
    } catch (err) {
      return { ok: false, error: `${err.name}: ${err.message}` };
    }
  }

  function ensureUnlockedAudioContext() {
    if (!audioCtx) {
      const AC = window.AudioContext || window.webkitAudioContext;
      audioCtx = new AC();
    }
    if (audioCtx.state === 'suspended') audioCtx.resume().catch(() => {});
    try {
      const buf = audioCtx.createBuffer(1, 128, audioCtx.sampleRate);
      const src = audioCtx.createBufferSource();
      src.buffer = buf;
      src.connect(audioCtx.destination);
      src.start(0);
    } catch (_) {}
    return audioCtx;
  }

  function floatTo16BitPCMBase64(float32, inputSampleRate) {
    const targetRate = 16000;
    const ratio = inputSampleRate / targetRate;
    const outLen = Math.max(1, Math.floor(float32.length / ratio));
    const buf = new ArrayBuffer(outLen * 2);
    const view = new DataView(buf);
    let sumSq = 0;
    for (let i = 0; i < outLen; i++) {
      const pos = i * ratio;
      const i0 = Math.floor(pos);
      const i1 = Math.min(float32.length - 1, i0 + 1);
      const frac = pos - i0;
      const s = Math.max(-1, Math.min(1, float32[i0] * (1 - frac) + float32[i1] * frac));
      sumSq += s * s;
      view.setInt16(i * 2, s < 0 ? s * 0x8000 : s * 0x7FFF, true);
    }
    APP.micLevel = Math.min(1, Math.sqrt(sumSq / outLen) * 5.0);
    const bytes = new Uint8Array(buf);
    let bin = '';
    for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
    return btoa(bin);
  }

  function stopLiveAudioPlayback() {
    activePlaySources.forEach((src) => { try { src.stop(); } catch (_) {} });
    activePlaySources.clear();
    nextPlayTime = audioCtx ? audioCtx.currentTime : 0;
    APP.botSpeaking = false;
  }

  function playPCM24kBase64(b64) {
    if (!b64) return;
    const ctx = ensureUnlockedAudioContext();
    const bin = atob(b64);
    const sampleCount = Math.floor(bin.length / 2);
    if (sampleCount <= 0) return;

    const floatBuf = new Float32Array(sampleCount);
    for (let i = 0; i < sampleCount; i++) {
      const lo = bin.charCodeAt(i * 2);
      const hi = bin.charCodeAt(i * 2 + 1);
      let val = (hi << 8) | lo;
      if (val >= 0x8000) val -= 0x10000;
      floatBuf[i] = val / 32768;
    }

    const audioBuf = ctx.createBuffer(1, sampleCount, 24000);
    audioBuf.getChannelData(0).set(floatBuf);
    const src = ctx.createBufferSource();
    src.buffer = audioBuf;
    src.connect(ctx.destination);

    const now = ctx.currentTime;
    if (nextPlayTime < now) nextPlayTime = now + 0.025;
    src.start(nextPlayTime);
    nextPlayTime += audioBuf.duration;

    APP.botSpeaking = true;
    activePlaySources.add(src);
    src.onended = function () {
      activePlaySources.delete(src);
      if (activePlaySources.size === 0) APP.botSpeaking = false;
    };
  }

  async function startMicStreamingForLive() {
    if (!navigator.mediaDevices?.getUserMedia) return;
    const ctx = ensureUnlockedAudioContext();
    stopMicStreamingForLive();

    micStream = await navigator.mediaDevices.getUserMedia({
      audio: { echoCancellation: true, noiseSuppression: true, autoGainControl: true, channelCount: 1 }
    });
    const source = ctx.createMediaStreamSource(micStream);
    const muteGain = ctx.createGain();
    muteGain.gain.value = 0;
    muteGain.connect(ctx.destination);

    let usedWorklet = false;
    if (ctx.audioWorklet && typeof Blob !== 'undefined') {
      try {
        const workletCode = [
          'class PCM16Cap extends AudioWorkletProcessor {',
          '  process(inputs) { const ch = inputs[0]?.[0]; if (ch) this.port.postMessage(ch); return true; }',
          '}',
          'registerProcessor("wxr-geeny-pcm16", PCM16Cap);'
        ].join('\n');
        const url = URL.createObjectURL(new Blob([workletCode], { type: 'application/javascript' }));
        await ctx.audioWorklet.addModule(url);
        URL.revokeObjectURL(url);
        micWorkletNode = new AudioWorkletNode(ctx, 'wxr-geeny-pcm16');
        micWorkletNode.port.onmessage = function (ev) {
          if (!liveWS || liveWS.readyState !== 1 || APP.botSpeaking) return;
          const b64 = floatTo16BitPCMBase64(ev.data, ctx.sampleRate);
          liveWS.send(JSON.stringify({ realtimeInput: { audio: { mimeType: 'audio/pcm;rate=16000', data: b64 } } }));
        };
        source.connect(micWorkletNode);
        micWorkletNode.connect(muteGain);
        usedWorklet = true;
      } catch (_) {}
    }

    if (!usedWorklet) {
      micScriptNode = ctx.createScriptProcessor(2048, 1, 1);
      micScriptNode.onaudioprocess = function (e) {
        if (!liveWS || liveWS.readyState !== 1 || APP.botSpeaking) return;
        const b64 = floatTo16BitPCMBase64(e.inputBuffer.getChannelData(0), ctx.sampleRate);
        liveWS.send(JSON.stringify({ realtimeInput: { audio: { mimeType: 'audio/pcm;rate=16000', data: b64 } } }));
      };
      source.connect(micScriptNode);
      micScriptNode.connect(muteGain);
    }
  }

  function stopMicStreamingForLive() {
    if (micWorkletNode) { try { micWorkletNode.disconnect(); } catch (_) {} micWorkletNode = null; }
    if (micScriptNode) { try { micScriptNode.disconnect(); } catch (_) {} micScriptNode = null; }
    if (micStream) { micStream.getTracks().forEach((t) => { try { t.stop(); } catch (_) {} }); micStream = null; }
    APP.micLevel = 0;
  }

  async function connectLive(enableMicStream) {
    ensureUnlockedAudioContext();
    if (!APP.ai) return { ok: false, error: 'APP.ai unavailable.' };
    const model = APP.ai.getSelectedModel();
    if (!model.supportsLiveWS) return { ok: false, error: `"${model.id}" is not a Live WebSocket model.` };
    const apiKey = APP.ai.getKey('gemini');
    if (!apiKey) return { ok: false, error: 'Missing Google AI Studio key (AIza...).' };

    disconnectLive();
    const wsUrl = 'wss://generativelanguage.googleapis.com/ws/google.ai.generativelanguage.v1beta.GenerativeService.BidiGenerateContent?key=' +
      encodeURIComponent(apiKey);

    return new Promise((resolve) => {
      try { liveWS = new WebSocket(wsUrl); }
      catch (e) { resolve({ ok: false, error: 'WebSocket error: ' + e.message }); return; }

      let settled = false;
      liveWS.onopen = function () {
        liveWS.send(JSON.stringify({
          setup: {
            model: 'models/' + model.id,
            generationConfig: {
              responseModalities: ['AUDIO'],
              speechConfig: { voiceConfig: { prebuiltVoiceConfig: { voiceName: 'Aoede' } } }
            },
            systemInstruction: { parts: [{ text: buildSystemPrompt(APP.execMode === 'ai') }] },
            tools: [{ functionDeclarations: BOT_TOOLS }],
            inputAudioTranscription: {},
            outputAudioTranscription: {}
          }
        }));
      };

      liveWS.onmessage = async function (evt) {
        let msg;
        try {
          msg = JSON.parse(typeof evt.data === 'string' ? evt.data : await evt.data.text());
        } catch (_) { return; }

        if (msg.setupComplete) {
          liveConnected = true;
          APP.ui?.setStatus?.(`Geeny Live (${model.id})`, 'ok');
          APP.ui?.toast?.(`Geeny Live Voice connected (${model.id})`);
          if (enableMicStream) {
            startMicStreamingForLive().catch((err) => APP.ui?.toast?.('Live Mic Error: ' + err.message));
          }
          visionTimer = setInterval(() => {
            if (!liveWS || liveWS.readyState !== 1) return;
            const b64 = APP.captureVisionFrame?.() || APP.latestVisionB64;
            if (b64) {
              liveWS.send(JSON.stringify({
                realtimeInput: { video: { mimeType: 'image/jpeg', data: String(b64).replace(/^data:image\/\w+;base64,/, '') } }
              }));
            }
          }, 1500);
          if (!settled) { settled = true; resolve({ ok: true }); }
          return;
        }

        if (msg.serverContent) {
          if (msg.serverContent.interrupted) stopLiveAudioPlayback();
          const turn = msg.serverContent.modelTurn;
          if (turn?.parts) {
            for (const part of turn.parts) {
              if (part.inlineData?.data && (!part.inlineData.mimeType || part.inlineData.mimeType.includes('pcm') || part.inlineData.mimeType.startsWith('audio/'))) {
                playPCM24kBase64(part.inlineData.data);
              }
              if (part.text) APP.ui?.toast?.('Geeny: ' + part.text.slice(0, 140));
            }
          }
          if (msg.serverContent.outputTranscription?.text) {
            APP.ui?.toast?.('Geeny: ' + msg.serverContent.outputTranscription.text.slice(0, 140));
          }
        }

        if (msg.toolCall?.functionCalls) {
          const responses = [];
          for (const fc of msg.toolCall.functionCalls) {
            const out = await executeToolCall(fc.name, fc.args || {});
            responses.push({ id: fc.id, name: fc.name, response: out || { ok: true } });
          }
          if (liveWS && liveWS.readyState === 1) {
            liveWS.send(JSON.stringify({ toolResponse: { functionResponses: responses } }));
          }
        }
      };

      liveWS.onerror = function () {
        if (!settled) { settled = true; resolve({ ok: false, error: `Gemini Live connection failed for "${model.id}".` }); }
      };
      liveWS.onclose = function (ev) {
        liveConnected = false;
        stopMicStreamingForLive();
        if (visionTimer) { clearInterval(visionTimer); visionTimer = null; }
        if (!settled) { settled = true; resolve({ ok: false, error: `Gemini Live closed (${ev.code})` }); }
      };
    });
  }

  function disconnectLive() {
    liveConnected = false;
    stopMicStreamingForLive();
    stopLiveAudioPlayback();
    if (visionTimer) { clearInterval(visionTimer); visionTimer = null; }
    if (liveWS) { try { liveWS.close(); } catch (_) {} liveWS = null; }
  }

  async function setLiveMic(enabled) {
    ensureUnlockedAudioContext();
    if (enabled) {
      if (!liveConnected) {
        const res = await connectLive(true);
        if (!res.ok) {
          APP.ui?.toast?.(res.error, 6000);
          APP.micActive = false;
          APP.toolbar?.syncState?.();
        }
      } else {
        await startMicStreamingForLive();
      }
    } else {
      stopMicStreamingForLive();
    }
  }

  async function sendLiveText(text) {
    ensureUnlockedAudioContext();

      const res = await connectLive(!!APP.micActive);
      if (!res.ok) throw new Error(res.error);
    }
    liveWS.send(JSON.stringify({
      clientContent: { turns: [{ role: 'user', parts: [{ text: String(text) }] }], turnComplete: true }
    }));
  }

    const visionB64 = model.supportsVision ? (APP.captureVisionFrame?.() || APP.latestVisionB64) : undefined;
    chatHistory.push({ role: 'user', content: String(userText) });
    if (chatHistory.length > 20) chatHistory.splice(0, chatHistory.length - 20);

    APP.ui?.setStatus?.(`Geeny thinking (${model.id})...`, 'busy');

    const res = await APP.ai.chat({
      systemPrompt: buildSystemPrompt(!!opts.ignoreInstructions),
      messages: chatHistory,
      tools: BOT_TOOLS,
      imageB64: visionB64,
      onToolCall: executeToolCall,
      maxSteps: 6
    });

    if (!res.ok) {
      APP.ui?.setStatus?.('AI Error', 'err');
      APP.ui?.toast?.(res.error, 6500);
      return res;
    }

    // Record tool actions into episodic memory so Geeny remembers exact created IDs & modifications
    const toolSummary = (res.toolTrace || [])
      .map((t) => `[${t.name}: ${JSON.stringify(t.result)}]`)
      .join(' ');
    const replyText = res.text || (toolSummary ? 'Done!' : 'Ready.');
    chatHistory.push({
      role: 'assistant',
      content: toolSummary ? `${replyText}\n(Executed: ${toolSummary})` : replyText
    });

    APP.ui?.setStatus?.('Ready', 'ok');
    APP.ui?.toast?.('Geeny: ' + replyText);
    APP.speakTTS?.(replyText);
    return res;
  }

  // ====================================================================
  // 9. HOOKS & EXPORTS
  // ====================================================================
  const unsubFrame = APP.onUpdate ? APP.onUpdate(updateRobot) : null;
  if (!unsubFrame && APP.frameHooks) APP.frameHooks.add(updateRobot);
  const unsubBotToggle = APP.on?.('bot:toggle', () => setEnabled(!botEnabled));

  APP._botCleanup = function () {
    if (typeof unsubFrame === 'function') unsubFrame();
    else APP.frameHooks?.delete(updateRobot);
    if (typeof unsubBotToggle === 'function') unsubBotToggle();
    APP.pokeTargets?.delete(boardScreen);
    disconnectLive();
    if (botRoot.parent) botRoot.parent.remove(botRoot);
    if (tentacleGroup.parent) tentacleGroup.parent.remove(tentacleGroup);
  };

  APP.bot = {
    name: 'Geeny',
    root: botRoot,
    tentacleGroup,
    get heldEntity() { return state.heldEnt; },
    set heldEntity(v) { if (!v) releaseHeldEntity(); },
    isEnabled: () => botEnabled,
    setEnabled,
    toggle: () => setEnabled(!botEnabled),
    tentacle_grab,
    tentacle_hand_to_user,
    tentacle_show,
    tentacle_release,
    showBoard,
    hideBoard,
    stepBoard,
    modifyEntity: modifyExistingEntity,
    onEntityCreated: function (ent) {
      if (!ent) return;
      APP.lastEntityId = ent.id;
      if (botEnabled && state.mode === 'idle') tentacle_show(ent);
    },
    setLiveMic,
    connectLive,
    disconnectLive,
    isLiveConnected: () => liveConnected,
    sendLiveText,
    handleCommand,
    update: updateRobot
  };

  APP.registerAbilityHandler?.('robot_companion_sphere', function (text) {
    const low = String(text || '').toLowerCase();
    if (low.includes('hide board') || low.includes('close board') || low.includes('اخف اللوح')) return hideBoard();
    if (low.includes('draw') && low.includes('cat')) {
      const r = showBoard(getCatDrawingTutorial());
      APP.speakTTS?.('Here is how to draw a cat! Step 1: start with a circle for the head and an oval for the body.');
      return r;
    }
    if (low.includes('board') || low.includes('notes') || low.includes('اللوح')) {
      return showBoard({
        title: 'Geeny Notes',
        mode: 'notes',
        notes: [
          'Ask me: "Hey Geeny, show me how to draw a cat"',
          'Build and attach: "make a cat" then "give it a hat"',
          'Say "show a portal to the forest" to open a 3D world portal'
        ]
      });
    }
    if (low.includes('release') || low.includes('drop') || low.includes('اترك')) return tentacle_release();
    if (low.includes('hand me') || low.includes('give me') || low.includes('اعطني')) return tentacle_hand_to_user('latest');
    if (low.includes('grab') || low.includes('امسك')) return tentacle_grab('latest');
    if (low.includes('show') || low.includes('ارني')) return tentacle_show('latest');
    if (low.includes('come here') || low.includes('تعال')) {
      const hp = APP.headPos || new THREE.Vector3(0, 1.5, 0);
      const hf = APP.headFwd || new THREE.Vector3(0, 0, -1);
      const flat = new THREE.Vector3(hf.x, 0, hf.z).normalize();
      state.worldAnchor.copy(hp).addScaledVector(flat, 0.95);
      state.worldAnchor.y = Math.max(1.0, hp.y + 0.04);
      state.mode = 'idle';
      return { ok: true, message: 'Geeny flew closer.' };
    }
    if (low.includes('off') || low.includes('hide')) {
      setEnabled(false);
      return { ok: true, message: 'Geeny hidden.' };
    }
    setEnabled(true);
    return { ok: true, message: 'Geeny online.' };
  });
})();
