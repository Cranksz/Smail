(function () {
  const APP = window.APP || (window.APP = {});
  const THREE = APP.THREE;
  if (!THREE || !APP.scene) return;

  if (typeof APP._botCleanup === 'function') {
    try { APP._botCleanup(); } catch (_) {}
  }

  const scene = APP.scene;

  // ====================================================================
  // 1. 3D ROBOT SPHERE + 12-SEGMENT CUBIC-BEZIER IK TENTACLE RIG
  // ====================================================================
  const botRoot = new THREE.Group();
  botRoot.name = 'APP_ROBOT_SPHERE';
  botRoot.position.set(0.46, 1.48, -1.05);
  scene.add(botRoot);

  const bodyGroup = new THREE.Group();
  botRoot.add(bodyGroup);

  const shellMat = new THREE.MeshStandardMaterial({
    color: 0x1a2536,
    metalness: 0.88,
    roughness: 0.20
  });
  const shellMesh = new THREE.Mesh(new THREE.SphereGeometry(0.145, 32, 24), shellMat);
  shellMesh.castShadow = true;
  bodyGroup.add(shellMesh);

  const coreMat = new THREE.MeshStandardMaterial({
    color: 0x00f0b4,
    emissive: 0x00ff88,
    emissiveIntensity: 1.0,
    roughness: 0.18,
    metalness: 0.2
  });
  const seamRing = new THREE.Mesh(new THREE.TorusGeometry(0.146, 0.0075, 12, 48), coreMat);
  seamRing.rotation.x = Math.PI / 2;
  bodyGroup.add(seamRing);

  const haloMat1 = new THREE.MeshBasicMaterial({
    color: 0x00f0b4,
    transparent: true,
    opacity: 0.72,
    blending: THREE.AdditiveBlending
  });
  const haloRing1 = new THREE.Mesh(new THREE.TorusGeometry(0.205, 0.004, 10, 56), haloMat1);
  haloRing1.rotation.x = Math.PI * 0.42;
  bodyGroup.add(haloRing1);

  const haloMat2 = new THREE.MeshBasicMaterial({
    color: 0x38bdf8,
    transparent: true,
    opacity: 0.52,
    blending: THREE.AdditiveBlending
  });
  const haloRing2 = new THREE.Mesh(new THREE.TorusGeometry(0.235, 0.0032, 10, 56), haloMat2);
  haloRing2.rotation.y = Math.PI * 0.3;
  bodyGroup.add(haloRing2);

  // Expressive Optic Eye Pivot (tracks user or target cleanly without moving body)
  const eyePivot = new THREE.Group();
  bodyGroup.add(eyePivot);

  const visorHousing = new THREE.Mesh(
    new THREE.CylinderGeometry(0.056, 0.062, 0.036, 24),
    new THREE.MeshStandardMaterial({ color: 0x080d17, metalness: 0.92, roughness: 0.14 })
  );
  visorHousing.rotation.x = Math.PI / 2;
  visorHousing.position.z = 0.132;
  eyePivot.add(visorHousing);

  const opticMat = new THREE.MeshStandardMaterial({
    color: 0x67e8f9,
    emissive: 0x00f0b4,
    emissiveIntensity: 1.8,
    roughness: 0.1
  });
  const opticLens = new THREE.Mesh(new THREE.SphereGeometry(0.041, 20, 16), opticMat);
  opticLens.scale.set(1, 1, 0.45);
  opticLens.position.z = 0.146;
  eyePivot.add(opticLens);

  const botLight = new THREE.PointLight(0x00f0b4, 0.55, 2.4);
  botLight.position.set(0, 0, 0.16);
  bodyGroup.add(botLight);

  // 12-Segment Tapered Robotic Tentacle
  const TENTACLE_SEGMENTS = 12;
  const tentacleGroup = new THREE.Group();
  tentacleGroup.name = 'APP_ROBOT_TENTACLE';
  scene.add(tentacleGroup);

  const segMatDark = new THREE.MeshStandardMaterial({ color: 0x152030, metalness: 0.86, roughness: 0.24 });
  const segMatGlow = new THREE.MeshStandardMaterial({
    color: 0x00f0b4,
    emissive: 0x0ea5e9,
    emissiveIntensity: 0.9,
    roughness: 0.25
  });

  const segments = [];
  for (let i = 0; i < TENTACLE_SEGMENTS; i++) {
    const t = i / (TENTACLE_SEGMENTS - 1);
    const radius = THREE.MathUtils.lerp(0.030, 0.011, t);
    const segGroup = new THREE.Group();
    const body = new THREE.Mesh(new THREE.CylinderGeometry(radius * 0.88, radius, 0.055, 14), segMatDark);
    body.rotation.x = Math.PI / 2;
    segGroup.add(body);
    const ring = new THREE.Mesh(new THREE.TorusGeometry(radius * 0.98, radius * 0.18, 8, 18), segMatGlow);
    segGroup.add(ring);
    tentacleGroup.add(segGroup);
    segments.push(segGroup);
  }

  // 3-Prong Mechanical Gripper Claw
  const clawGroup = new THREE.Group();
  tentacleGroup.add(clawGroup);

  const palmMesh = new THREE.Mesh(new THREE.CylinderGeometry(0.018, 0.014, 0.028, 14), segMatDark);
  palmMesh.rotation.x = Math.PI / 2;
  clawGroup.add(palmMesh);

  const prongs = [];
  for (let p = 0; p < 3; p++) {
    const pivot = new THREE.Group();
    pivot.rotation.z = (p / 3) * Math.PI * 2;
    const fingerPivot = new THREE.Group();
    fingerPivot.position.set(0.016, 0, 0.012);
    const finger = new THREE.Mesh(new THREE.BoxGeometry(0.008, 0.009, 0.038), segMatDark);
    finger.position.set(0.004, 0, 0.019);
    const tipGlow = new THREE.Mesh(new THREE.SphereGeometry(0.0055, 10, 10), segMatGlow);
    tipGlow.position.set(-0.002, 0, 0.039);
    fingerPivot.add(finger, tipGlow);
    pivot.add(fingerPivot);
    clawGroup.add(pivot);
    prongs.push(fingerPivot);
  }

  // Holographic Show Beam Cone
  const showBeam = new THREE.Mesh(
    new THREE.ConeGeometry(0.14, 0.55, 24, 1, true),
    new THREE.MeshBasicMaterial({
      color: 0x00f0b4,
      transparent: true,
      opacity: 0.22,
      blending: THREE.AdditiveBlending,
      depthWrite: false,
      side: THREE.DoubleSide
    })
  );
  showBeam.rotation.x = -Math.PI / 2;
  showBeam.position.z = 0.30;
  showBeam.visible = false;
  clawGroup.add(showBeam);

  // ====================================================================
  // 2. AUTONOMOUS WORLD-SPACE BRAIN + CUBIC-BEZIER IK + TENTACLE ACTIONS
  // ====================================================================
  let botEnabled = true;
  let audioEnergy = 0;
  let clawClosure = 0.2;
  let clawTargetClosure = 0.2;

  const state = {
    mode: 'idle', // 'idle' | 'grabbing' | 'holding' | 'handing' | 'showing'
    targetObj: null,
    heldObj: null,
    heldEnt: null,
    heldRadius: 0.12,
    worldAnchor: new THREE.Vector3(0.46, 1.48, -1.05),
    hoverTargetPos: new THREE.Vector3(0.46, 1.48, -1.05),
    tipTargetPos: new THREE.Vector3(0.46, 1.18, -1.05),
    currentTipPos: new THREE.Vector3(0.46, 1.18, -1.05),
    showTimer: 0,
    grabProgress: 0,
    nextThoughtTime: 5.0,
    lookTargetPos: new THREE.Vector3(0, 1.5, 0)
  };

  const vP0 = new THREE.Vector3();
  const vC1 = new THREE.Vector3();
  const vC2 = new THREE.Vector3();
  const vP3 = new THREE.Vector3();
  const vPt = new THREE.Vector3();
  const vNext = new THREE.Vector3();
  const vObjCenter = new THREE.Vector3();
  const vObjSize = new THREE.Vector3();
  const tmpBox = new THREE.Box3();
  const tmpVecA = new THREE.Vector3();
  const tmpVecB = new THREE.Vector3();

  function getEntityRecord(target) {
    if (!target) return null;
    if (typeof target === 'object' && target.group) return target;
    if (target.isObject3D && target.userData?.entity) return target.userData.entity;
    if (typeof APP.findEntity === 'function') {
      return APP.findEntity(target);
    }
    return null;
  }

  function getEntityObject3D(target) {
    if (!target) return null;
    if (target.isObject3D) return target;
    if (target.group && target.group.isObject3D) return target.group;
    if (target.mesh && target.mesh.isObject3D) return target.mesh;
    const rec = getEntityRecord(target);
    return rec ? (rec.group || rec.mesh) : null;
  }

  function computeObjCenterAndRadius(obj) {
    obj.updateMatrixWorld(true);
    tmpBox.setFromObject(obj);
    if (tmpBox.isEmpty()) {
      obj.getWorldPosition(vObjCenter);
      return { center: vObjCenter.clone(), radius: 0.12 };
    }
    tmpBox.getCenter(vObjCenter);
    tmpBox.getSize(vObjSize);
    const radius = Math.max(0.06, Math.min(0.32, Math.max(vObjSize.x, vObjSize.z) * 0.45));
    return { center: vObjCenter.clone(), radius };
  }

  function releaseHeldEntity() {
    if (state.heldEnt) {
      state.heldEnt.heldByBot = false;
      state.heldEnt = null;
    }
    if (state.heldObj) {
      state.heldObj.userData = state.heldObj.userData || {};
      state.heldObj.userData.heldByBot = false;
      state.heldObj = null;
    }
  }

  function tentacle_release() {
    releaseHeldEntity();
    state.mode = 'idle';
    state.targetObj = null;
    state.worldAnchor.copy(botRoot.position);
    clawTargetClosure = 0.08;
    showBeam.visible = false;
    return { ok: true, message: 'Tentacle released object and resumed autonomous hover.' };
  }

  function tentacle_grab(target) {
    if (!botEnabled) setEnabled(true);
    const obj = getEntityObject3D(target || 'latest');
    if (!obj) {
      return { ok: false, error: 'Could not find target object to grab: ' + String(target || '') };
    }
    releaseHeldEntity();
    state.mode = 'grabbing';
    state.targetObj = obj;
    state.grabProgress = 0;
    clawTargetClosure = 0.0;
    showBeam.visible = false;
    return { ok: true, message: 'Flying to grab ' + (obj.userData?.entity?.name || obj.name || 'object') + '.' };
  }

  function tentacle_hand_to_user(target) {
    if (!botEnabled) setEnabled(true);
    let obj = getEntityObject3D(target || 'latest');
    if (!obj && state.heldObj) obj = state.heldObj;
    if (!obj) {
      return { ok: false, error: 'No target object found to hand to user.' };
    }
    if (state.heldObj !== obj) {
      releaseHeldEntity();
      state.targetObj = obj;
      const info = computeObjCenterAndRadius(obj);
      state.heldObj = obj;
      state.heldEnt = getEntityRecord(obj);
      state.heldRadius = info.radius;
      if (state.heldEnt) state.heldEnt.heldByBot = true;
      obj.userData = obj.userData || {};
      obj.userData.heldByBot = true;
    }
    state.mode = 'handing';
    clawTargetClosure = 0.85;
    showBeam.visible = false;
    return { ok: true, message: 'Bringing ' + (state.heldEnt?.name || obj.name || 'object') + ' directly to your hand.' };
  }

  function tentacle_show(target) {
    if (!botEnabled) setEnabled(true);
    const obj = getEntityObject3D(target || 'latest');
    if (!obj) {
      return { ok: false, error: 'No target object found to showcase.' };
    }
    if (!state.heldObj || state.heldObj === obj) {
      releaseHeldEntity();
    }
    state.mode = 'showing';
    state.targetObj = obj;
    state.showTimer = 4.8;
    clawTargetClosure = 0.25;
    showBeam.visible = true;
    return { ok: true, message: 'Showcasing ' + (obj.userData?.entity?.name || obj.name || 'object') + '.' };
  }

  function comeNearUser() {
    const hp = APP.headPos || new THREE.Vector3(0, 1.5, 0);
    const hf = APP.headFwd || new THREE.Vector3(0, 0, -1);
    const flat = new THREE.Vector3(hf.x, 0, hf.z).normalize();
    if (flat.lengthSq() < 0.01) flat.set(0, 0, -1);
    const right = new THREE.Vector3().crossVectors(flat, new THREE.Vector3(0, 1, 0)).normalize();
    state.worldAnchor.copy(hp).addScaledVector(flat, 0.92).addScaledVector(right, 0.35);
    state.worldAnchor.y = Math.max(0.95, hp.y + 0.02);
    state.mode = 'idle';
  }

  function evalCubicBezier(p0, c1, c2, p3, t, out) {
    const u = 1 - t;
    const tt = t * t;
    const uu = u * u;
    out.set(0, 0, 0)
      .addScaledVector(p0, uu * u)
      .addScaledVector(c1, 3 * uu * t)
      .addScaledVector(c2, 3 * u * tt)
      .addScaledVector(p3, tt * t);
    return out;
  }

  function updateTentacleIK(dt, elapsed) {
    vP0.set(0, -0.14, 0);
    botRoot.localToWorld(vP0);

    const tipLerp = 1 - Math.exp(-11 * dt);
    state.currentTipPos.lerp(state.tipTargetPos, tipLerp);
    vP3.copy(state.currentTipPos);

    const dist = vP0.distanceTo(vP3);
    vC1.copy(vP0).add(new THREE.Vector3(0, -Math.max(0.16, dist * 0.40), 0));

    tmpVecA.subVectors(vP3, vP0).normalize();
    vC2.copy(vP3).addScaledVector(tmpVecA, -Math.max(0.12, dist * 0.34));
    vC2.y += Math.sin(elapsed * 2.1) * 0.025 + 0.05;

    for (let i = 0; i < TENTACLE_SEGMENTS; i++) {
      const t = i / TENTACLE_SEGMENTS;
      const tNext = (i + 1) / TENTACLE_SEGMENTS;
      evalCubicBezier(vP0, vC1, vC2, vP3, t, vPt);
      evalCubicBezier(vP0, vC1, vC2, vP3, tNext, vNext);
      segments[i].position.copy(vPt);
      segments[i].lookAt(vNext);
    }

    clawGroup.position.copy(vP3);
    if (state.mode === 'showing' && state.targetObj) {
      const info = computeObjCenterAndRadius(state.targetObj);
      clawGroup.lookAt(info.center);
      const beamDist = Math.max(0.15, Math.min(0.85, vP3.distanceTo(info.center)));
      showBeam.scale.set(1, 1, beamDist / 0.55);
      showBeam.position.z = beamDist * 0.52;
    } else {
      evalCubicBezier(vP0, vC1, vC2, vP3, 0.9, tmpVecB);
      tmpVecA.subVectors(vP3, tmpVecB).normalize();
      if (tmpVecA.lengthSq() > 0.001) {
        clawGroup.lookAt(tmpVecB.copy(vP3).add(tmpVecA));
      }
    }

    clawClosure += (clawTargetClosure - clawClosure) * Math.min(1, dt * 10);
    const openAngle = THREE.MathUtils.lerp(0.62, -0.14, clawClosure);
    for (let p = 0; p < prongs.length; p++) {
      prongs[p].rotation.y = openAngle;
    }
  }

  function updateRobot(dt, elapsed) {
    if (!botEnabled) return;
    const delta = Math.min(0.05, dt || 0.016);
    const now = elapsed || (performance.now() * 0.001);

    const hp = APP.headPos || new THREE.Vector3(0, 1.5, 0);
    const hf = APP.headFwd || new THREE.Vector3(0, 0, -1);
    const flatFwd = new THREE.Vector3(hf.x, 0, hf.z).normalize();
    if (flatFwd.lengthSq() < 0.01) flatFwd.set(0, 0, -1);
    const right = new THREE.Vector3().crossVectors(flatFwd, new THREE.Vector3(0, 1, 0)).normalize();

    if (state.mode === 'idle') {
      // Autonomous world-space curiosity: NEVER slides away when you look at it!
      const distToUser = state.worldAnchor.distanceTo(hp);
      if (distToUser > 2.9) {
        // User walked far away -> politely catch up to a nearby workspace spot
        state.worldAnchor.copy(hp).addScaledVector(flatFwd, 1.15).addScaledVector(right, 0.45);
        state.worldAnchor.y = Math.max(1.05, hp.y + 0.02);
      } else if (distToUser < 0.46) {
        // Too close to user's face -> gentle step back along away vector
        tmpVecA.subVectors(state.worldAnchor, hp).setY(0).normalize();
        if (tmpVecA.lengthSq() < 0.01) tmpVecA.copy(flatFwd);
        state.worldAnchor.copy(hp).addScaledVector(tmpVecA, 0.72);
        state.worldAnchor.y = Math.max(1.05, hp.y + 0.02);
      }

      // Autonomous thinking tick every 7-11s: inspect a scene object or subtly patrol workspace
      if (now > state.nextThoughtTime) {
        state.nextThoughtTime = now + 7 + Math.random() * 5;
        const ents = Array.isArray(APP.entities) ? APP.entities : [];
        if (ents.length > 0 && Math.random() < 0.6) {
          const pick = ents[Math.floor(Math.random() * ents.length)];
          const obj = pick?.group || pick?.mesh;
          if (obj && !pick.isHeld) {
            const info = computeObjCenterAndRadius(obj);
            if (info.center.distanceTo(hp) < 2.6) {
              state.worldAnchor.copy(info.center).add(new THREE.Vector3(
                (Math.random() - 0.5) * 0.45,
                0.38 + Math.random() * 0.16,
                (Math.random() - 0.5) * 0.45
              ));
              state.lookTargetPos.copy(info.center);
            }
          }
        } else {
          state.lookTargetPos.copy(hp);
        }
      } else if (APP.botSpeaking || APP.micLevel > 0.12) {
        state.lookTargetPos.copy(hp);
      }

      // Subtle organic zero-G hover around worldAnchor (head orientation does NOT push worldAnchor)
      state.hoverTargetPos.copy(state.worldAnchor).add(new THREE.Vector3(
        Math.sin(now * 0.65) * 0.06,
        Math.sin(now * 1.65) * 0.035,
        Math.cos(now * 0.55) * 0.05
      ));

      state.tipTargetPos.copy(botRoot.position).add(new THREE.Vector3(
        Math.sin(now * 1.3) * 0.04,
        -0.29 + Math.cos(now * 1.6) * 0.025,
        0.04
      ));

      // Smooth eye tracking without rotating/moving body away
      eyePivot.lookAt(state.lookTargetPos.lengthSq() > 0 ? state.lookTargetPos : hp);
    } else if (state.mode === 'grabbing') {
      if (!state.targetObj || !state.targetObj.parent) {
        tentacle_release();
      } else {
        const info = computeObjCenterAndRadius(state.targetObj);
        state.hoverTargetPos.copy(info.center).add(new THREE.Vector3(0.22, 0.36, 0.18));
        state.tipTargetPos.copy(info.center);
        eyePivot.lookAt(info.center);

        if (state.currentTipPos.distanceTo(info.center) < 0.15) {
          state.grabProgress += delta;
          clawTargetClosure = 0.88;
          if (state.grabProgress > 0.22) {
            state.heldObj = state.targetObj;
            state.heldEnt = getEntityRecord(state.heldObj);
            state.heldRadius = info.radius;
            if (state.heldEnt) state.heldEnt.heldByBot = true;
            state.heldObj.userData = state.heldObj.userData || {};
            state.heldObj.userData.heldByBot = true;
            state.mode = 'holding';
          }
        }
      }
    } else if (state.mode === 'holding') {
      if (!state.heldObj || !state.heldObj.parent) {
        tentacle_release();
      } else {
        state.hoverTargetPos.copy(state.worldAnchor).add(new THREE.Vector3(0, Math.sin(now * 1.5) * 0.025, 0));
        state.tipTargetPos.copy(botRoot.position).add(new THREE.Vector3(-0.08, -0.32, 0.16));
        eyePivot.lookAt(hp);
      }
    } else if (state.mode === 'handing') {
      if (!state.heldObj || !state.heldObj.parent) {
        tentacle_release();
      } else {
        state.hoverTargetPos.copy(hp).addScaledVector(flatFwd, 0.62);
        state.hoverTargetPos.y = Math.max(0.88, hp.y - 0.04);

        let handAnchor = null;
        let handGrabbing = false;
        const hands = APP.hands || {};
        for (const k of Object.keys(hands)) {
          const h = hands[k];
          if (!h || !h.active) continue;
          const pos = h.isHand ? h.fingerTips?.index : h.pos;
          if (pos) {
            if (!handAnchor || pos.distanceTo(state.currentTipPos) < handAnchor.distanceTo(state.currentTipPos)) {
              handAnchor = pos;
            }
            if ((h.pinching || h.trigger || h.gripping) && pos.distanceTo(state.currentTipPos) < 0.22) {
              handGrabbing = true;
            }
          }
        }

        if (handAnchor && handAnchor.distanceTo(hp) < 1.1) {
          state.tipTargetPos.copy(handAnchor).addScaledVector(flatFwd, -0.05);
        } else {
          state.tipTargetPos.copy(hp).addScaledVector(flatFwd, 0.42);
          state.tipTargetPos.y = Math.max(0.75, hp.y - 0.22);
        }
        eyePivot.lookAt(hp);

        if (handGrabbing || state.heldEnt?.grabbedByHand) {
          tentacle_release();
        }
      }
    } else if (state.mode === 'showing') {
      if (!state.targetObj || !state.targetObj.parent) {
        tentacle_release();
      } else {
        const info = computeObjCenterAndRadius(state.targetObj);
        state.hoverTargetPos.copy(info.center).add(new THREE.Vector3(0.32, 0.34 + Math.sin(now * 2.0) * 0.03, 0.18));
        state.tipTargetPos.copy(info.center).add(new THREE.Vector3(0.16, 0.10, 0.10));
        state.worldAnchor.copy(state.hoverTargetPos);
        eyePivot.lookAt(info.center);

        state.showTimer -= delta;
        if (state.showTimer <= 0) {
          tentacle_release();
        }
      }
    }

    botRoot.position.lerp(state.hoverTargetPos, 1 - Math.exp(-4.8 * delta));
    updateTentacleIK(delta, now);

    if (state.heldObj && (state.mode === 'holding' || state.mode === 'handing')) {
      const forwardDir = new THREE.Vector3(0, 0, 1).applyQuaternion(clawGroup.quaternion);
      tmpVecA.copy(clawGroup.position).addScaledVector(forwardDir, 0.045 + state.heldRadius * 0.35);
      state.heldObj.position.lerp(tmpVecA, 1 - Math.exp(-16 * delta));
      if (state.heldEnt?.physics?.vel) state.heldEnt.physics.vel.set(0, 0, 0);
    }

    // Audio-reactive core glow
    const targetEnergy = (APP.botSpeaking || APP.ttsSpeaking)
      ? 0.55 + Math.abs(Math.sin(now * 16.0)) * 0.45
      : (APP.micLevel || 0);
    audioEnergy += (targetEnergy - audioEnergy) * Math.min(1, delta * 12);

    coreMat.emissiveIntensity = 0.85 + audioEnergy * 2.2;
    opticMat.emissiveIntensity = 1.4 + audioEnergy * 1.8;
    botLight.intensity = 0.5 + audioEnergy * 1.1;

    const haloScale = 1 + audioEnergy * 0.14;
    haloRing1.scale.setScalar(haloScale);
    haloRing1.rotation.z = now * 1.1;
    haloRing2.rotation.x = now * 0.85;
    haloRing2.rotation.y = now * 0.65;
  }

  function setEnabled(val) {
    botEnabled = !!val;
    botRoot.visible = botEnabled;
    tentacleGroup.visible = botEnabled;
    if (!botEnabled) {
      tentacle_release();
      disconnectLive();
    }
    APP.emit?.('bot:state_changed', botEnabled);
    APP.toolbar?.syncState?.();
    return botEnabled;
  }

  async function executeToolCall(name, args) {
    const a = args || {};

      if (res?.ok && res.entity) {
        return { ok: true, id: res.entity.id, name: res.entity.name };
      }
      return res;
    }
    if (name === 'tentacle_grab') return tentacle_grab(a.target || 'latest');
    if (name === 'tentacle_hand_to_user') return tentacle_hand_to_user(a.target || 'latest');
    if (name === 'tentacle_show') return tentacle_show(a.target || 'latest');
    if (name === 'tentacle_release') return tentacle_release();

      const out = await APP.runInstructionMatch?.(a.command || '');

    if (name === 'spawn_repo_asset') {

      if (!spawner) return { ok: false, error: 'Asset spawner unavailable.' };
      return await spawner(a.name || '');
    }
    return { ok: false, error: 'Unknown tool: ' + name };

  // ====================================================================
  // 4. GEMINI 3.8 LIVE WEBSOCKET (Unlocked AudioContext + 16k/24k PCM)
  // ====================================================================
  let liveWS = null;
  let liveConnected = false;
  let audioCtx = null;
  let micStream = null;
  let micWorkletNode = null;
  let micScriptNode = null;
  let visionTimer = null;
  let nextPlayTime = 0;
  const activePlaySources = new Set();

  function ensureUnlockedAudioContext() {
    if (!audioCtx) {
      const AC = window.AudioContext || window.webkitAudioContext;
      audioCtx = new AC();
    }
    if (audioCtx.state === 'suspended') {
      audioCtx.resume().catch(() => {});
    }
    // Play a 1-sample silent buffer inside user gesture so Quest/Chrome/Safari unlocks audio out
    try {
      const buf = audioCtx.createBuffer(1, 128, audioCtx.sampleRate);
      const src = audioCtx.createBufferSource();
      src.buffer = buf;
      src.connect(audioCtx.destination);
      src.start(0);
    } catch (_) {}
    return audioCtx;
  }

  // Linear-interpolated resampler to 16kHz Int16 PCM Base64
  function floatTo16BitPCMBase64(float32, inputSampleRate) {
    const targetRate = 16000;
    const ratio = inputSampleRate / targetRate;
    const outLen = Math.max(1, Math.floor(float32.length / ratio));
    const buf = new ArrayBuffer(outLen * 2);
    const view = new DataView(buf);
    let sumSq = 0;

    for (let i = 0; i < outLen; i++) {
      const pos = i * ratio;
      const i0 = Math.floor(pos);
      const i1 = Math.min(float32.length - 1, i0 + 1);
      const frac = pos - i0;
      const sample = float32[i0] * (1 - frac) + float32[i1] * frac;
      const s = Math.max(-1, Math.min(1, sample));
      sumSq += s * s;
      view.setInt16(i * 2, s < 0 ? s * 0x8000 : s * 0x7FFF, true);
    }
    APP.micLevel = Math.min(1, Math.sqrt(sumSq / outLen) * 5.0);

    const bytes = new Uint8Array(buf);
    let bin = '';
    for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
    return btoa(bin);
  }

  function stopLiveAudioPlayback() {
    activePlaySources.forEach((src) => { try { src.stop(); } catch (_) {} });
    activePlaySources.clear();
    nextPlayTime = audioCtx ? audioCtx.currentTime : 0;
    APP.botSpeaking = false;
  }

  function playPCM24kBase64(b64) {
    if (!b64) return;
    const ctx = ensureUnlockedAudioContext();
    const bin = atob(b64);
    const sampleCount = Math.floor(bin.length / 2);
    if (sampleCount <= 0) return;

    const floatBuf = new Float32Array(sampleCount);
    for (let i = 0; i < sampleCount; i++) {
      const lo = bin.charCodeAt(i * 2);
      const hi = bin.charCodeAt(i * 2 + 1);
      let val = (hi << 8) | lo;
      if (val >= 0x8000) val -= 0x10000;
      floatBuf[i] = val / 32768;
    }

    const audioBuf = ctx.createBuffer(1, sampleCount, 24000);
    audioBuf.getChannelData(0).set(floatBuf);

    const src = ctx.createBufferSource();
    src.buffer = audioBuf;
    src.connect(ctx.destination);

    const now = ctx.currentTime;
    if (nextPlayTime < now) nextPlayTime = now + 0.025;
    src.start(nextPlayTime);
    nextPlayTime += audioBuf.duration;

    APP.botSpeaking = true;
    activePlaySources.add(src);
    src.onended = function () {
      activePlaySources.delete(src);
      if (activePlaySources.size === 0) APP.botSpeaking = false;
    };
  }

  async function startMicStreamingForLive() {
    if (!navigator.mediaDevices?.getUserMedia) return;
    const ctx = ensureUnlockedAudioContext();
    stopMicStreamingForLive();

    micStream = await navigator.mediaDevices.getUserMedia({
      audio: { echoCancellation: true, noiseSuppression: true, autoGainControl: true, channelCount: 1 }
    });
    const source = ctx.createMediaStreamSource(micStream);

    // Silent sink so ScriptProcessor/Worklet runs without feeding mic back to speakers
    const muteGain = ctx.createGain();
    muteGain.gain.value = 0;
    muteGain.connect(ctx.destination);

    let usedWorklet = false;
    if (ctx.audioWorklet && typeof Blob !== 'undefined') {
      try {
        const workletCode = [
          'class PCM16Cap extends AudioWorkletProcessor {',
          '  process(inputs) {',
          '    const ch = inputs[0] && inputs[0][0];',
          '    if (ch) this.port.postMessage(ch);',
          '    return true;',
          '  }',
          '}',
          'registerProcessor("wxr-pcm16-cap-v2", PCM16Cap);'
        ].join('\n');
        const blobUrl = URL.createObjectURL(new Blob([workletCode], { type: 'application/javascript' }));
        await ctx.audioWorklet.addModule(blobUrl);
        URL.revokeObjectURL(blobUrl);
        micWorkletNode = new AudioWorkletNode(ctx, 'wxr-pcm16-cap-v2');
        micWorkletNode.port.onmessage = function (ev) {
          if (!liveWS || liveWS.readyState !== 1 || APP.botSpeaking) return;
          const b64 = floatTo16BitPCMBase64(ev.data, ctx.sampleRate);
          liveWS.send(JSON.stringify({
            realtimeInput: { audio: { mimeType: 'audio/pcm;rate=16000', data: b64 } }
          }));
        };
        source.connect(micWorkletNode);
        micWorkletNode.connect(muteGain);
        usedWorklet = true;
      } catch (_) {}
    }

    if (!usedWorklet) {
      micScriptNode = ctx.createScriptProcessor(2048, 1, 1);
      micScriptNode.onaudioprocess = function (e) {
        if (!liveWS || liveWS.readyState !== 1 || APP.botSpeaking) return;
        const b64 = floatTo16BitPCMBase64(e.inputBuffer.getChannelData(0), ctx.sampleRate);
        liveWS.send(JSON.stringify({
          realtimeInput: { audio: { mimeType: 'audio/pcm;rate=16000', data: b64 } }
        }));
      };
      source.connect(micScriptNode);
      micScriptNode.connect(muteGain);
    }
  }

  function stopMicStreamingForLive() {
    if (micWorkletNode) { try { micWorkletNode.disconnect(); } catch (_) {} micWorkletNode = null; }
    if (micScriptNode) { try { micScriptNode.disconnect(); } catch (_) {} micScriptNode = null; }
    if (micStream) {
      micStream.getTracks().forEach((t) => { try { t.stop(); } catch (_) {} });
      micStream = null;
    }
    APP.micLevel = 0;
  }

  async function connectLive(enableMicStream) {
    ensureUnlockedAudioContext();
    if (!APP.ai) return { ok: false, error: 'APP.ai unavailable.' };
    const model = APP.ai.getSelectedModel();
    if (!model.supportsLiveWS) {
      return { ok: false, error: `Selected model "${model.id}" is not a Gemini Live WebSocket model.` };
    }
    const apiKey = APP.ai.getKey('gemini');
    if (!apiKey) {
      return { ok: false, error: `Missing Google AI Studio key (AIza...) for ${model.id}.` };
    }

    disconnectLive();

    const wsUrl = 'wss://generativelanguage.googleapis.com/ws/google.ai.generativelanguage.v1beta.GenerativeService.BidiGenerateContent?key=' +
      encodeURIComponent(apiKey);

    return new Promise((resolve) => {
      try {
        liveWS = new WebSocket(wsUrl);
      } catch (e) {
        resolve({ ok: false, error: 'WebSocket init error: ' + e.message });
        return;
      }

      let settled = false;
      liveWS.onopen = function () {
        const setupPayload = {
          setup: {
            model: 'models/' + model.id,
            generationConfig: {
              responseModalities: ['AUDIO'],
              speechConfig: {
                voiceConfig: {
                  prebuiltVoiceConfig: { voiceName: 'Aoede' }
                }
              }
            },
            systemInstruction: { parts: [{ text: buildSystemPrompt(APP.execMode === 'ai') }] },
            tools: [{ functionDeclarations: BOT_TOOLS }],
            inputAudioTranscription: {},
            outputAudioTranscription: {}
          }
        };
        liveWS.send(JSON.stringify(setupPayload));
      };

      liveWS.onmessage = async function (evt) {
        let msg;
        try {
          const text = typeof evt.data === 'string' ? evt.data : await evt.data.text();
          msg = JSON.parse(text);
        } catch (_) { return; }

        if (msg.setupComplete) {
          liveConnected = true;
          APP.ui?.setStatus?.(`Gemini Live Ready (${model.id})`, 'ok');
          APP.ui?.toast?.(`Gemini Live Voice connected (${model.id})`);

          if (enableMicStream) {
            startMicStreamingForLive().catch((err) => {
              APP.ui?.toast?.('Live Mic Error: ' + err.message);
            });
          }

          visionTimer = setInterval(() => {
            if (!liveWS || liveWS.readyState !== 1) return;
            const b64 = APP.captureVisionFrame?.() || APP.latestVisionB64;
            if (b64) {
              const clean = String(b64).replace(/^data:image\/\w+;base64,/, '');
              liveWS.send(JSON.stringify({
                realtimeInput: { video: { mimeType: 'image/jpeg', data: clean } }
              }));
            }
          }, 1400);

          if (!settled) { settled = true; resolve({ ok: true }); }
          return;
        }

        if (msg.serverContent) {
          if (msg.serverContent.interrupted) {
            stopLiveAudioPlayback();
          }
          const turn = msg.serverContent.modelTurn;
          if (turn && Array.isArray(turn.parts)) {
            for (const part of turn.parts) {
              if (part.inlineData?.data && (!part.inlineData.mimeType || part.inlineData.mimeType.includes('pcm') || part.inlineData.mimeType.startsWith('audio/'))) {
                playPCM24kBase64(part.inlineData.data);
              }
              if (part.text) {
                APP.ui?.toast?.('Bot: ' + part.text.slice(0, 130));
              }
            }
          }
          if (msg.serverContent.outputTranscription?.text) {
            APP.ui?.toast?.('Bot: ' + msg.serverContent.outputTranscription.text.slice(0, 130));
          }
        }

        if (msg.toolCall && Array.isArray(msg.toolCall.functionCalls)) {
          const responses = [];
          for (const fc of msg.toolCall.functionCalls) {
            const out = await executeToolCall(fc.name, fc.args || {});
            responses.push({ id: fc.id, name: fc.name, response: out || { ok: true } });
          }
          if (liveWS && liveWS.readyState === 1) {
            liveWS.send(JSON.stringify({ toolResponse: { functionResponses: responses } }));
          }
        }
      };

      liveWS.onerror = function () {
        if (!settled) {
          settled = true;
          resolve({ ok: false, error: `Gemini Live WebSocket failed for "${model.id}". Verify AIza key or try gemini-3.8-flash.` });
        }
      };

      liveWS.onclose = function (ev) {
        liveConnected = false;
        stopMicStreamingForLive();
        if (visionTimer) { clearInterval(visionTimer); visionTimer = null; }
        if (!settled) {
          settled = true;
          resolve({ ok: false, error: `Gemini Live closed (${ev.code} ${ev.reason || ''})` });
        }
      };
    });
  }

  function disconnectLive() {
    liveConnected = false;
    stopMicStreamingForLive();
    stopLiveAudioPlayback();
    if (visionTimer) { clearInterval(visionTimer); visionTimer = null; }
    if (liveWS) {
      try { liveWS.close(); } catch (_) {}
      liveWS = null;
    }
  }

  async function setLiveMic(enabled) {
    ensureUnlockedAudioContext();
    if (enabled) {
      if (!liveConnected) {
        const res = await connectLive(true);
        if (!res.ok) {
          APP.ui?.toast?.(res.error, 6000);
          APP.micActive = false;
          APP.toolbar?.syncState?.();
        }
      } else {
        await startMicStreamingForLive();
      }
    } else {
      stopMicStreamingForLive();
    }
  }

  async function sendLiveText(text) {
    ensureUnlockedAudioContext();
    if (!liveConnected || !liveWS || liveWS.readyState !== 1) {
      const res = await connectLive(!!APP.micActive);
      if (!res.ok) throw new Error(res.error);
    }
    liveWS.send(JSON.stringify({
      clientContent: {
        turns: [{ role: 'user', parts: [{ text: String(text) }] }],
        turnComplete: true
      }
    }));
  }

  // ====================================================================
  // 5. TURN-BASED MULTIMODAL AI HANDLER
  // ====================================================================
  const chatHistory = [];

  async function handleCommand(userText, opts = {}) {
    if (!APP.ai) return { ok: false, error: 'AI module (ai_models.txt) not loaded.' };
    const model = APP.ai.getSelectedModel();

    if (model.supportsLiveWS && !model.supportsRest) {
      try {
        await sendLiveText(userText);
        return { ok: true, text: '(Sent to Gemini Live)' };
      } catch (e) {
        APP.ui?.setStatus?.('Live Error', 'err');
        APP.ui?.toast?.(e.message, 6000);
        return { ok: false, error: e.message };
      }
    }

    const visionB64 = model.supportsVision
      ? (APP.captureVisionFrame?.() || APP.latestVisionB64)
      : undefined;

    chatHistory.push({ role: 'user', content: String(userText) });
    if (chatHistory.length > 10) chatHistory.splice(0, chatHistory.length - 10);

    APP.ui?.setStatus?.(`Thinking (${model.id})...`, 'busy');

    const res = await APP.ai.chat({
      systemPrompt: buildSystemPrompt(!!opts.ignoreInstructions),
      messages: chatHistory,
      tools: BOT_TOOLS,
      imageB64: visionB64,
      onToolCall: executeToolCall,
      maxSteps: 6
    });

    if (!res.ok) {
      APP.ui?.setStatus?.('AI Error', 'err');
      APP.ui?.toast?.(res.error, 6500);
      return res;
    }

    const replyText = res.text || (res.toolTrace?.length ? 'Done.' : 'Ready.');
    chatHistory.push({ role: 'assistant', content: replyText });
    APP.ui?.setStatus?.('Ready', 'ok');
    APP.ui?.toast?.('Bot: ' + replyText);
    APP.speakTTS?.(replyText);
    return res;
  }

    if (typeof unsubFrame === 'function') unsubFrame();
    else APP.frameHooks?.delete(updateRobot);
    if (typeof unsubBotToggle === 'function') unsubBotToggle();
    disconnectLive();
    if (botRoot.parent) botRoot.parent.remove(botRoot);
    if (tentacleGroup.parent) tentacleGroup.parent.remove(tentacleGroup);
  };

  APP.bot = {

    connectLive,

    isLiveConnected: () => liveConnected,

    handleCommand,
    update: updateRobot
  };

  APP.registerAbilityHandler?.('robot_companion_sphere', function (text) {
    const low = String(text || '').toLowerCase();
    if (low.includes('release') || low.includes('drop') || low.includes('اترك')) return tentacle_release();

    if (low.includes('grab') || low.includes('امسك')) return tentacle_grab('latest');

    }
    if (low.includes('off') || low.includes('hide')) {
      setEnabled(false);
      return { ok: true, message: 'Robot sphere disabled.' };
    }
    setEnabled(true);
    return { ok: true, message: 'Robot sphere active.' };
  });
