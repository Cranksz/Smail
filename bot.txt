  function parsePersonalityText(rawTxt) {
    const txt = String(rawTxt || '').trim();
    if (!txt) return;
    personality.raw = txt;
    const grabField = (key) => {
      const m = txt.match(new RegExp('^' + key + ':\\s*(.+)$', 'mi'));
      return m ? m[1].trim() : '';
    };
    const nm = grabField('Name'); if (nm) personality.name = nm;
    const lv = grabField('Live_Voice'); if (lv) personality.liveVoice = lv;
    const pt = parseFloat(grabField('TTS_Pitch')); if (isFinite(pt)) personality.ttsPitch = THREE.MathUtils.clamp(pt, 0.5, 1.8);
    const rt = parseFloat(grabField('TTS_Rate')); if (isFinite(rt)) personality.ttsRate = THREE.MathUtils.clamp(rt, 0.6, 1.6);
    const em = grabField('Default_Emotion'); if (em) personality.defaultEmotion = em.toLowerCase();
  }

  async function loadPersonalityFile() {
    if (APP.modules && APP.modules['personality.txt']) {
      parsePersonalityText(APP.modules['personality.txt']);
      return;

      const res = await fetch('./personality.txt', { cache: 'no-store' });
      if (res.ok) {
        const txt = await res.text();
        if (APP.modules) APP.modules['personality.txt'] = txt;
        parsePersonalityText(txt);
      }
    } catch (_) {}
  }

  function hangEntityOnHook(entOrQuery) {
    const ent = getEntityRecord(entOrQuery || 'latest');
    if (!ent || !ent.group || ent === boardEnt || ent === boardStylusEnt) {
      return { ok: false, error: 'No suitable object found to hang on hook.' };
    }
    if (hookState.hungEnt && hookState.hungEnt !== ent) unhookEntity();
    ent.group.updateMatrixWorld(true);
    const box = new THREE.Box3().setFromObject(ent.group);
    const sz = new THREE.Vector3(); box.getSize(sz);
    const topOffset = Math.min(0.28, Math.max(0.04, sz.y * 0.45));
    hookState.hungEnt = ent;
    hookState.hungOffset.set(0, -topOffset, 0);
    ent.hungOnHook = true;
    ent.heldByBot = false;
    if (ent.physics) ent.physics.enabled = false;
    hookLatchGlow.material.opacity = 0.92;
    hookState.velZ += 2.2;
    setEmotion('happy');
    return { ok: true, id: ent.id, name: ent.name, message: `Hung "${ent.name}" onto Geeny's hook!` };
  }

  function unhookEntity(suppressCooldown) {
    const ent = hookState.hungEnt;
    if (!ent) return { ok: false, message: 'Nothing is hanging on Geeny\'s hook.' };
    ent.hungOnHook = false;
    if (ent.physics) {
      ent.physics.enabled = true;
      if (ent.physics.vel) ent.physics.vel.set(state.velocity.x * 0.4, -0.1, state.velocity.z * 0.4);
    }
    hookState.hungEnt = null;
    if (!suppressCooldown) hookState.cooldownUntil = performance.now() + 1400;
    hookLatchGlow.material.opacity = 0.45;
    return { ok: true, name: ent.name, message: `Unhooked "${ent.name}".` };
  }

  function updateDanglingHook(dt, elapsed) {
    const safeDt = Math.min(0.035, Math.max(0.004, dt || 0.016));
    const accelX = (state.velocity.x - hookState.prevBotVel.x) / safeDt;
    const accelZ = (state.velocity.z - hookState.prevBotVel.z) / safeDt;
    hookState.prevBotVel.copy(state.velocity);

    const kSpring = 26.0, cDamp = 3.2;
    const torX = -kSpring * hookState.angX - cDamp * hookState.velX - accelZ * 0.085;
    const torZ = -kSpring * hookState.angZ - cDamp * hookState.velZ + accelX * 0.085;
    hookState.velX += torX * safeDt; hookState.velZ += torZ * safeDt;

    hookState.angZ = THREE.MathUtils.clamp(hookState.angZ + hookState.velZ * safeDt, -0.85, 0.85);

  function stepCapePhysics(dt, elapsed) {
    capeMesh.visible = capeEnabled && botEnabled;
    capeCollarGroup.visible = capeEnabled && botEnabled;
    if (!capeMesh.visible) return;

    const safeDt = Math.min(0.025, Math.max(0.004, dt || 0.016));
    bodyGroup.updateMatrixWorld(true);

    tmpCapeW.copy(capePinnedLocal[Math.floor(CAPE_COLS / 2)]);
    bodyGroup.localToWorld(tmpCapeW);
    if (!capeInitialized || !isFinite(capeCur[0].x) || capeCur[0].distanceTo(tmpCapeW) > 1.2) snapCapeToRest();

    for (let c = 0; c < CAPE_COLS; c++) {
      tmpCapeW.copy(capePinnedLocal[c]);
      bodyGroup.localToWorld(tmpCapeW);
      capeCur[c].copy(tmpCapeW);
      capePrev[c].copy(tmpCapeW);
    }

    if (capeGrabCornerActive) {
      capeCur[CAPE_GRAB_IDX].lerp(clawGroup.position, 0.72);
      capePrev[CAPE_GRAB_IDX].copy(capeCur[CAPE_GRAB_IDX]);
    }

    let userStillHoldingCape = false;
    for (const [hk, h] of Object.entries(APP.hands || {})) {
      if (!h || !h.active) continue;
      const tip = h.isHand ? h.fingerTips?.index : h.pos;
      if (!tip) continue;
      const prevH = userCapePrevHands.get(hk);
      const hVel = prevH ? tmpCapeDelta.copy(tip).sub(prevH) : tmpCapeDelta.set(0, 0, 0);
      userCapePrevHands.set(hk, tip.clone());

      const pinchingOrTrig = !!(h.pinching || h.trigger);
      if (pinchingOrTrig && userCapeGrabIdx === -1) {
        let bestI = -1, bestD = 0.14;
        for (let idx = CAPE_COLS; idx < CAPE_COUNT; idx++) {
          const d = capeCur[idx].distanceTo(tip);
          if (d < bestD) { bestD = d; bestI = idx; }
        }
        if (bestI !== -1) { userCapeGrabIdx = bestI; userCapeGrabHand = hk; setEmotion('playful'); }
      }
      if (userCapeGrabHand === hk && pinchingOrTrig && userCapeGrabIdx !== -1) {
        capeCur[userCapeGrabIdx].lerp(tip, 0.78);
        capePrev[userCapeGrabIdx].copy(capeCur[userCapeGrabIdx]);
        userStillHoldingCape = true;
      } else {
        const spd = hVel.length();
        for (let idx = CAPE_COLS; idx < CAPE_COUNT; idx++) {
          const d = capeCur[idx].distanceTo(tip);
          if (d < 0.085) {
            const push = capeCur[idx].clone().sub(tip).normalize().multiplyScalar(0.085 - d);
            capeCur[idx].add(push);
            if (spd > 0.005) capePrev[idx].addScaledVector(hVel, -0.45);
          }
        }
      }
    }
    if (!userStillHoldingCape) { userCapeGrabIdx = -1; userCapeGrabHand = null; }

    tmpCapeBack.set(0, 0, -1).applyQuaternion(bodyGroup.getWorldQuaternion(new THREE.Quaternion())).normalize();
    const gravStep = -4.6 * safeDt * safeDt;
    const maxStepSpeed = 0.045;

    for (let r = 1; r < CAPE_ROWS; r++) {
      for (let c = 0; c < CAPE_COLS; c++) {
        const idx = r * CAPE_COLS + c;
        if ((capeGrabCornerActive && idx === CAPE_GRAB_IDX) || idx === userCapeGrabIdx) continue;
        const cur = capeCur[idx], prev = capePrev[idx];
        let vx = (cur.x - prev.x) * 0.965, vy = (cur.y - prev.y) * 0.965, vz = (cur.z - prev.z) * 0.965;
        const spdSq = vx * vx + vy * vy + vz * vz;
        if (spdSq > maxStepSpeed * maxStepSpeed) { const s = maxStepSpeed / Math.sqrt(spdSq); vx *= s; vy *= s; vz *= s; }
        prev.copy(cur);
        const flutter = (Math.sin(elapsed * 4.2 + c * 0.55 + r * 0.38) * 0.5 + 0.5) * 0.00048;
        cur.x += vx + tmpCapeBack.x * flutter; cur.y += vy + gravStep; cur.z += vz + tmpCapeBack.z * flutter;
      }
    }

    const bodyCenter = botRoot.position, R_BODY = 0.164, R_BODY_SQ = R_BODY * R_BODY;
    for (let iter = 0; iter < 5; iter++) {
      for (let k = 0; k < capeConstraints.length; k++) {
        const con = capeConstraints[k], ia = con[0], ib = con[1], restLen = con[2], stiff = con[3];
        const pa = capeCur[ia], pb = capeCur[ib];
        tmpCapeDelta.subVectors(pb, pa);
        const d = tmpCapeDelta.length();
        if (d < 1e-6) continue;
        const diff = ((d - restLen) / d) * stiff;
        const pinA = ia < CAPE_COLS || (capeGrabCornerActive && ia === CAPE_GRAB_IDX) || ia === userCapeGrabIdx;
        const pinB = ib < CAPE_COLS || (capeGrabCornerActive && ib === CAPE_GRAB_IDX) || ib === userCapeGrabIdx;
        if (pinA && pinB) continue;
        if (pinA) pb.addScaledVector(tmpCapeDelta, -diff);
        else if (pinB) pa.addScaledVector(tmpCapeDelta, diff);
        else { pa.addScaledVector(tmpCapeDelta, diff * 0.5); pb.addScaledVector(tmpCapeDelta, -diff * 0.5); }
      }
      for (let idx = CAPE_COLS; idx < CAPE_COUNT; idx++) {
        if ((capeGrabCornerActive && idx === CAPE_GRAB_IDX) || idx === userCapeGrabIdx) continue;
        const p = capeCur[idx];
        tmpCapeDelta.subVectors(p, bodyCenter);
        const dSq = tmpCapeDelta.lengthSq();
        if (dSq < R_BODY_SQ) {
          const d = Math.sqrt(dSq);
          if (d > 1e-5) p.copy(bodyCenter).addScaledVector(tmpCapeDelta, R_BODY / d);
          else p.set(bodyCenter.x, bodyCenter.y, bodyCenter.z - R_BODY);
        }
        if (p.y < 0.02) p.y = 0.02;
      }
    }

    const arr = capeGeo.attributes.position.array;
    for (let i = 0; i < CAPE_COUNT; i++) {
      const p = capeCur[i];
      if (!isFinite(p.x) || !isFinite(p.y) || !isFinite(p.z)) { snapCapeToRest(); break; }
      arr[i * 3] = p.x; arr[i * 3 + 1] = p.y; arr[i * 3 + 2] = p.z;
    }
    capeGeo.attributes.position.needsUpdate = true;
    capeGeo.computeVertexNormals();
  }

  async function setCapeTexture(nameOrUrl) {
    const q = String(nameOrUrl || '').trim();
    const low = q.toLowerCase();
    if (!q || low === 'algeria' || low === 'algerian' || low === 'flag' || low === 'default' || low.includes('الجزائر')) {
      capeMat.map = defaultCapeTex; capeMat.needsUpdate = true; setCapeEnabled(true);
      return { ok: true, texture: 'Algerian Flag', message: 'Geeny cape set to Algerian Flag.' };
    }
    const clean = q.replace(/\.[^.]+$/, '').toLowerCase();
    let url = q.startsWith('http') || q.startsWith('blob:') || q.startsWith('./') || q.startsWith('data:') ? q : null;
    if (!url && APP.importedImages) {
      const rec = APP.importedImages.get(clean) || Array.from(APP.importedImages.values()).find((r) => r.cleanName.includes(clean));
      if (rec) url = rec.url;
    }
    if (!url) {
      for (const p of [`./assets/${clean}.png`, `./assets/${clean}.jpg`, `./assets/${clean}.jpeg`, `./assets/${clean}.webp`, `./${clean}.png`, `./${clean}.jpg`]) {
        try { const h = await fetch(p, { method: 'HEAD' }); if (h.ok) { url = p; break; } } catch (_) {}
      }
    }
    if (!url) return { ok: false, error: `Picture "${q}" not found in /assets/ or imported images.` };
    try {
      const tex = await new THREE.TextureLoader().loadAsync(url);
      tex.colorSpace = THREE.SRGBColorSpace; tex.anisotropy = 4;
      if (customCapeTex && customCapeTex !== defaultCapeTex) { try { customCapeTex.dispose(); } catch (_) {} }
      customCapeTex = tex; capeMat.map = tex; capeMat.needsUpdate = true; setCapeEnabled(true);
      return { ok: true, texture: q, message: `Updated Geeny cape texture to "${q}".` };
    } catch (e) {
      return { ok: false, error: 'Failed to load cape texture: ' + e.message };
    }
  }

  function drawInkAtCanvasXY(cx, cy, isStart) {
    if (isStart) saveInkUndoSnapshot();

      inkCtx.lineWidth = boardInkState.brushWidth;

    inkCtx.beginPath();
    if (boardInkState.lastPt && !isStart) inkCtx.moveTo(boardInkState.lastPt.x, boardInkState.lastPt.y);

    inkCtx.lineTo(cx + 0.01, cy + 0.01); inkCtx.stroke();
    inkCtx.restore();
    inkTex.needsUpdate = true; boardInkState.lastPt = { x: cx, y: cy };

  function drawBoardRoundedRect(x, y, w, h, r, fill, stroke, lw = 3) {
    bCtx.beginPath(); bCtx.moveTo(x + r, y); bCtx.arcTo(x + w, y, x + w, y + h, r);
    bCtx.arcTo(x + w, y + h, x, y + h, r); bCtx.arcTo(x, y + h, x, y, r); bCtx.arcTo(x, y, x + w, y, r);
    bCtx.closePath();
    if (fill) { bCtx.fillStyle = fill; bCtx.fill(); }
    if (stroke) { bCtx.strokeStyle = stroke; bCtx.lineWidth = lw; bCtx.stroke(); }
  }

  function wrapCanvasText(ctx, text, x, y, maxWidth, lineHeight, maxY) {
    const paragraphs = String(text || '').split('\n');
    let cy = y;
    for (const para of paragraphs) {
      const words = para.split(/\s+/);
      let line = '';
      for (let i = 0; i < words.length; i++) {
        const test = line ? line + ' ' + words[i] : words[i];
        if (ctx.measureText(test).width > maxWidth && line) {
          ctx.fillText(line, x, cy); line = words[i]; cy += lineHeight;
          if (maxY && cy > maxY) return cy;
        } else { line = test; }
      }
      if (line) { ctx.fillText(line, x, cy); cy += lineHeight; }
      if (maxY && cy > maxY) return cy;
    }
    return cy;
  }

  function executeDrawCommands(cmds, area) {
    if (!Array.isArray(cmds)) return;
    const ox = area.x, oy = area.y, aw = area.w, ah = area.h;
    const px = (nx) => ox + Number(nx || 0) * aw, py = (ny) => oy + Number(ny || 0) * ah;
    for (const c of cmds) {
      if (!c || !c.type) continue;
      bCtx.save();
      bCtx.strokeStyle = c.color || '#00f0b4'; bCtx.fillStyle = c.fill || c.color || 'rgba(0,240,180,0.18)';
      bCtx.lineWidth = c.width || 4; bCtx.lineCap = 'round'; bCtx.lineJoin = 'round';
      if (c.dashed) bCtx.setLineDash([8, 6]);
      if (c.type === 'grid') {
        bCtx.strokeStyle = c.color || 'rgba(56,189,248,0.18)'; bCtx.lineWidth = 1.5;
        const n = c.divisions || 6;
        for (let i = 1; i < n; i++) { bCtx.beginPath(); bCtx.moveTo(px(i / n), py(0)); bCtx.lineTo(px(i / n), py(1)); bCtx.stroke(); bCtx.beginPath(); bCtx.moveTo(px(0), py(i / n)); bCtx.lineTo(px(1), py(i / n)); bCtx.stroke(); }
      } else if (c.type === 'circle') {
        bCtx.beginPath(); bCtx.arc(px(c.cx ?? 0.5), py(c.cy ?? 0.5), (c.r ?? 0.15) * Math.min(aw, ah), 0, Math.PI * 2); if (c.filled) bCtx.fill(); bCtx.stroke();
      } else if (c.type === 'ellipse') {
        bCtx.beginPath(); bCtx.ellipse(px(c.cx ?? 0.5), py(c.cy ?? 0.5), (c.rx ?? 0.2) * aw, (c.ry ?? 0.15) * ah, c.rot || 0, 0, Math.PI * 2); if (c.filled) bCtx.fill(); bCtx.stroke();
      } else if (c.type === 'line' || c.type === 'arrow') {
        const x1 = px(c.x1), y1 = py(c.y1), x2 = px(c.x2), y2 = py(c.y2);
        bCtx.beginPath(); bCtx.moveTo(x1, y1); bCtx.lineTo(x2, y2); bCtx.stroke();
        if (c.type === 'arrow') {
          const ang = Math.atan2(y2 - y1, x2 - x1), hl = 14;
          bCtx.beginPath(); bCtx.moveTo(x2, y2); bCtx.lineTo(x2 - hl * Math.cos(ang - 0.4), y2 - hl * Math.sin(ang - 0.4));
          bCtx.moveTo(x2, y2); bCtx.lineTo(x2 - hl * Math.cos(ang + 0.4), y2 - hl * Math.sin(ang + 0.4)); bCtx.stroke();
        }
      } else if (c.type === 'poly' && Array.isArray(c.points) && c.points.length >= 2) {
        bCtx.beginPath(); bCtx.moveTo(px(c.points[0][0]), py(c.points[0][1]));
        for (let i = 1; i < c.points.length; i++) bCtx.lineTo(px(c.points[i][0]), py(c.points[i][1]));
        if (c.closed) bCtx.closePath(); if (c.filled) bCtx.fill(); bCtx.stroke();
      } else if (c.type === 'curve') {
        bCtx.beginPath(); bCtx.moveTo(px(c.x1), py(c.y1)); bCtx.quadraticCurveTo(px(c.cx), py(c.cy), px(c.x2), py(c.y2)); bCtx.stroke();
      } else if (c.type === 'text') {
        bCtx.fillStyle = c.color || '#e2f8ff'; bCtx.font = 'bold ' + (c.size || 22) + 'px system-ui, sans-serif';
        bCtx.textAlign = 'center'; bCtx.fillText(String(c.text || ''), px(c.x ?? 0.5), py(c.y ?? 0.5));
      }
      bCtx.restore();
    }
  }

  function updateBoardPenDrawing() {
    if (!boardState.visible || !boardRoot.visible) { stylusGroup.visible = false; return; }
    ensureBoardAttachedToScene();
    stylusGroup.visible = true;

    if (boardInkState.docked && !boardStylusEnt.isHeld && !boardStylusEnt.isGrabbed && !boardStylusEnt.heldBy) {

      boardRoot.getWorldQuaternion(tmpDockQ);
      stylusGroup.position.lerp(tmpDockW, 0.35);
      stylusGroup.quaternion.slerp(tmpDockQ, 0.35);
    }

    let drewThisFrame = false;
    if (!boardInkState.docked) {
      stTipAnchor.getWorldPosition(tmpPenTipW);
      tmpBoardLocal.copy(tmpPenTipW);
      boardScreen.worldToLocal(tmpBoardLocal);
      const inWritingArea = Math.abs(tmpBoardLocal.x) <= BOARD_W * 0.48 && tmpBoardLocal.y >= -BOARD_H * 0.47 && tmpBoardLocal.y <= BOARD_H * 0.36;
      if (inWritingArea && Math.abs(tmpBoardLocal.z) < 0.036) {

        drewThisFrame = true;
      }
    }

    if (!drewThisFrame) {
      for (const h of Object.values(APP.hands || {})) {
        if (!h || !h.active || !(h.pinching || h.trigger)) continue;
        const tip = h.isHand ? h.fingerTips?.index : h.pos;
        if (!tip) continue;
        tmpBoardLocal.copy(tip); boardScreen.worldToLocal(tmpBoardLocal);
        const inWritingArea = Math.abs(tmpBoardLocal.x) <= BOARD_W * 0.48 && tmpBoardLocal.y >= -BOARD_H * 0.47 && tmpBoardLocal.y <= BOARD_H * 0.36;
        if (inWritingArea && Math.abs(tmpBoardLocal.z) < 0.028) {
          drawInkAtCanvasXY((tmpBoardLocal.x / BOARD_W + 0.5) * inkCanvas.width, (0.5 - tmpBoardLocal.y / BOARD_H) * inkCanvas.height, !boardInkState.lastPt);
          drewThisFrame = true; break;
        }
      }
    }
    if (!drewThisFrame && !desktopBoardDrawing) boardInkState.lastPt = null;
  }

  function loadCORSImage(url) {
    return new Promise((resolve) => {

  async function fetchWikipediaData(topic) {
    const clean = String(topic || '').trim();
    if (!clean) return null;
    const langHost = /[\u0600-\u06FF]/.test(clean) ? 'ar.wikipedia.org' : 'en.wikipedia.org';
    try {
      const sJson = await (await fetch(`https://${langHost}/w/api.php?action=opensearch&search=${encodeURIComponent(clean)}&limit=1&namespace=0&format=json&origin=*`)).json();
      const title = sJson?.[1]?.[0] || clean;
      const sumRes = await fetch(`https://${langHost}/api/rest_v1/page/summary/${encodeURIComponent(title)}`);
      if (!sumRes.ok) return null;
      const sum = await sumRes.json();
      const eJson = await (await fetch(`https://${langHost}/w/api.php?action=query&prop=extracts&exintro=false&explaintext=true&exchars=2400&titles=${encodeURIComponent(sum.title || title)}&format=json&origin=*`)).json();
      const firstPage = Object.values(eJson?.query?.pages || {})[0];
      const fullText = firstPage?.extract || sum.extract || '';
      let imgObj = null;
      const imgUrl = sum.thumbnail?.source || sum.originalimage?.source;
      if (imgUrl) imgObj = await loadCORSImage(imgUrl);

  function getCompassAndRelativeDir(worldPt) {
    const { hp, hf } = getReliableHeadPose();
    const flatFwd = new THREE.Vector3(hf.x, 0, hf.z).normalize();
    if (flatFwd.lengthSq() < 0.01) flatFwd.set(0, 0, -1);
    const right = new THREE.Vector3().crossVectors(flatFwd, new THREE.Vector3(0, 1, 0)).normalize();
    const delta = new THREE.Vector3().subVectors(worldPt, hp);
    const dist = delta.length(), fwdDot = delta.dot(flatFwd), rightDot = delta.dot(right), dy = delta.y;
    const fb = fwdDot > 0.25 ? 'in front' : (fwdDot < -0.25 ? 'behind' : '');
    const lr = rightDot > 0.25 ? 'right' : (rightDot < -0.25 ? 'left' : '');
    const ud = dy > 0.35 ? 'above eye-level' : (dy < -0.45 ? 'low near floor' : 'at eye/chest level');
    const rel = [fb, lr].filter(Boolean).join('-') || 'right beside you';
    let compass = 'North(-Z)';
    const ang = Math.atan2(delta.x, -delta.z) * (180 / Math.PI);
    if (ang >= -45 && ang < 45) compass = 'North(-Z)';
    else if (ang >= 45 && ang < 135) compass = 'East(+X)';
    else if (ang >= -135 && ang < -45) compass = 'West(-X)';
    else compass = 'South(+Z)';
    return { dist, rel, ud, compass };
  }

'ON' : 'OFF'}, board=${boardState.visible ? 'OPEN' : 'CLOSED'}, hook=${hookState.hungEnt ? `holding "${hookState.hungEnt.name}"` : 'empty'}`,
      `SCENE OBJECTS (${objList.length}):`,

      state.mode = 'idle'; state.idleRoutine = 'holo_projector'; state.idleRoutineTimer = Number(params.duration || 10.0);
      lensProjGroup.visible = true; setEmotion('focused');
      return { ok: true, message: `Geeny projecting hologram.` };
    }
    if (act === 'stop_hologram' || act === 'hide_hologram') { clearIdleEffects(); state.idleRoutine = 'none'; return { ok: true, message: 'Stopped hologram.' }; }
    if (act === 'move_relative') { state.mode = 'idle'; state.worldAnchor.add(new THREE.Vector3(Number(params.dx || 0), Number(params.dy || 0), Number(params.dz || 0))); state.worldAnchor.y = Math.max(0.4, state.worldAnchor.y); return { ok: true, message: 'Geeny moved.' }; }
    if (act === 'inspect_object' && params.target) return tentacle_show(params.target);
    return { ok: false, error: 'Unknown control action: ' + action };
  }

  function evalCubicBezier(p0, c1, c2, p3, t, out) {
    const u = 1 - t;
    out.set(0, 0, 0).addScaledVector(p0, u * u * u).addScaledVector(c1, 3 * u * u * t).addScaledVector(c2, 3 * u * t * t).addScaledVector(p3, t * t * t);
    return out;
  }

  function updateTentacleIK(dt, elapsed) {
    vP0.set(0, -0.14, 0);
    botRoot.localToWorld(vP0);

    let holdingClaw = false;
    for (const [hk, h] of Object.entries(APP.hands || {})) {
      if (!h || !h.active) continue;
      const tip = h.isHand ? h.fingerTips?.index : h.pos;
      if (!tip) continue;
      const dist = tip.distanceTo(clawGroup.position);
      if ((h.pinching || h.trigger) && (state.userClawGrabHand === hk || (!state.userClawGrabHand && dist < 0.14))) {
        state.userClawGrabHand = hk; state.tipTargetPos.copy(tip); clawTargetClosure = 0.85; holdingClaw = true; setEmotion('happy'); break;
      } else if (dist < 0.085 && state.mode === 'idle') {
        state.tipTargetPos.copy(tip).add(new THREE.Vector3(0, 0.04, 0)); clawTargetClosure = 0.05; setEmotion('playful');
      }
    }
    if (!holdingClaw) state.userClawGrabHand = null;

    const tipLerp = 1 - Math.exp(-13.0 * dt);
    state.currentTipPos.lerp(state.tipTargetPos, tipLerp);
    vP3.copy(state.currentTipPos);

    const dist = vP0.distanceTo(vP3);
    vC1.copy(vP0).add(new THREE.Vector3(0, -Math.max(0.22, dist * 0.46), 0)).addScaledVector(state.velocity, -0.18);
    tmpVecA.subVectors(vP3, vP0).normalize();
    vC2.copy(vP3).addScaledVector(tmpVecA, -Math.max(0.16, dist * 0.34)).addScaledVector(state.velocity, -0.14);
    vC2.y += Math.sin(elapsed * 2.2) * 0.035 + 0.065;

    for (let i = 0; i < TENTACLE_SEGMENTS; i++) {
      const t = i / TENTACLE_SEGMENTS, tNext = (i + 1) / TENTACLE_SEGMENTS;
      evalCubicBezier(vP0, vC1, vC2, vP3, t, vPt);
      evalCubicBezier(vP0, vC1, vC2, vP3, tNext, vNext);
      segments[i].position.copy(vPt);
      segments[i].lookAt(vNext);
    }

    clawGroup.position.copy(vP3);
    if (state.mode === 'showing' && state.targetObj) {
      const info = computeObjCenterAndRadius(state.targetObj);
      clawGroup.lookAt(info.center);
      const beamDist = Math.max(0.15, Math.min(0.95, vP3.distanceTo(info.center)));
      showBeam.scale.set(1, 1, beamDist / 0.60); showBeam.position.z = beamDist * 0.52;
    } else if (state.mode === 'pointing') {
      const tgt = state.targetObj ? computeObjCenterAndRadius(state.targetObj).center : state.pointTargetPos;
      clawGroup.lookAt(tgt);
      const beamDist = Math.max(0.25, Math.min(1.6, vP3.distanceTo(tgt)));
      showBeam.scale.set(0.65, 0.65, beamDist / 0.60); showBeam.position.z = beamDist * 0.52;
    } else {
      evalCubicBezier(vP0, vC1, vC2, vP3, 0.9, tmpVecB);
      tmpVecA.subVectors(vP3, tmpVecB).normalize();
      if (tmpVecA.lengthSq() > 0.001) clawGroup.lookAt(tmpVecB.copy(vP3).add(tmpVecA));
    }

    clawClosure += (clawTargetClosure - clawClosure) * Math.min(1, dt * 10);
    const openAngle = THREE.MathUtils.lerp(0.62, -0.14, clawClosure);
    for (let p = 0; p < prongs.length; p++) prongs[p].rotation.y = openAngle;
  }

  function onDesktopPointerDown(e) {
    if (e.button !== 0 || !APP.camera || !botEnabled) return;
    const el = APP.renderer?.domElement;
    const rect = el ? el.getBoundingClientRect() : { left: 0, top: 0, width: window.innerWidth, height: window.innerHeight };
    deskNdc.x = ((e.clientX - rect.left) / Math.max(1, rect.width)) * 2 - 1;
    deskNdc.y = -((e.clientY - rect.top) / Math.max(1, rect.height)) * 2 + 1;
    deskRay.setFromCamera(deskNdc, APP.camera);

    if (boardState.visible && boardRoot.visible) {
      const bHits = deskRay.intersectObject(boardScreen, false);
      if (bHits.length && bHits[0].uv) {
        const uv = bHits[0].uv;
        let hitBtn = false;
        for (const z of boardState.hitZones) {
          if (uv.x >= z.u0 && uv.x <= z.u1 && uv.y >= z.v0 && uv.y <= z.v1) { z.action(); hitBtn = true; break; }
        }
        if (!hitBtn && uv.y < 0.88) { desktopBoardDrawing = true; drawInkAtCanvasXY(uv.x * inkCanvas.width, (1 - uv.y) * inkCanvas.height, true); }
        return;
      }
    }

    if (capeMesh.visible) {
      const cHits = deskRay.intersectObject(capeMesh, false);
      if (cHits.length) {
        let bestI = -1, bestD = 0.18;
        for (let i = CAPE_COLS; i < CAPE_COUNT; i++) {
          const d = capeCur[i].distanceTo(cHits[0].point);
          if (d < bestD) { bestD = d; bestI = i; }
        }
        if (bestI !== -1) {
          deskGrabCapeIdx = bestI;
          deskCapePlane.setFromNormalAndCoplanarPoint(APP.camera.getWorldDirection(new THREE.Vector3()).negate(), cHits[0].point);
          setEmotion('playful'); return;
        }
      }
    }

    const bodyHits = deskRay.intersectObjects([topShell, visorGlass, botShell, palmMesh], false);
    if (bodyHits.length) { state.nodPhase = Math.PI * 2; state.petGlowBoost = 1.6; setEmotion('happy'); APP.ui?.toast?.(`${personality.name} acknowledges your tap.`, 1400); }
  }

  function updateRobot(dt, elapsed) {
    if (!botEnabled) return;
    const delta = Math.min(0.05, dt || 0.016);
    const now = elapsed || (performance.now() * 0.001);
    const nowMs = performance.now();

    const { hp, hf } = getReliableHeadPose();
    const flatFwd = new THREE.Vector3(hf.x, 0, hf.z).normalize();
    if (flatFwd.lengthSq() < 0.01) flatFwd.set(0, 0, -1);
    const right = new THREE.Vector3().crossVectors(flatFwd, new THREE.Vector3(0, 1, 0)).normalize();

    for (const h of Object.values(APP.hands || {})) {
      if (!h || !h.active) continue;
      const tip = h.isHand ? h.fingerTips?.index : h.pos;
      if (!tip) continue;
      const dBody = tip.distanceTo(botRoot.position);
      if (dBody < 0.22 && dBody > 1e-4) {
        if (tip.y > botRoot.position.y + 0.06 && now - state.lastPetTime > 1.2) {
          state.lastPetTime = now; state.nodPhase = Math.PI * 2; state.petGlowBoost = 1.6; setEmotion('happy');
        } else {
          tmpVecA.subVectors(botRoot.position, tip).normalize();
          state.velocity.addScaledVector(tmpVecA, (0.22 - dBody) * 18.0 * delta);
        }
      }
    }
    state.petGlowBoost = Math.max(0, state.petGlowBoost - delta * 1.4);

    if (state.behaviorTimer > 0) {
      state.behaviorTimer -= delta;
      if (state.behaviorTimer <= 0 && (state.behavior === 'orbit' || state.behavior === 'dance')) state.behavior = 'autonomous';
    }

    const isSpeakingNow = !!(APP.botSpeaking || APP.ttsSpeaking || nowMs < speechState.until);
    const speechAmp = APP.botSpeaking ? (APP.botSpeechEnergy || (0.45 + Math.abs(Math.sin(now * 14.0)) * 0.55))
      : (isSpeakingNow ? (0.4 + Math.abs(Math.sin(now * 12.0)) * 0.5) : 0.0);

    if (state.mode === 'idle') {
      if (state.generatingActive) {
        // Watch & gesture toward AI 3D Hologram Generation Site while building!
        state.lookTargetPos.lerp(state.generatingPos, delta * 5.5);
        if (!state.userClawGrabHand) {
          const dirToGen = tmpVecA.subVectors(state.generatingPos, botRoot.position).normalize();
          state.tipTargetPos.copy(botRoot.position).addScaledVector(dirToGen, 0.52).add(new THREE.Vector3(0, Math.sin(now * 6.0) * 0.04, 0));
          clawTargetClosure = 0.25 + Math.sin(now * 7.0) * 0.15;
        }
        setEmotion('focused');
      } else if (isSpeakingNow && !state.userClawGrabHand) {
        // Speech-Aligned Head & Tentacle Choreography!
        const spT = (nowMs - speechState.startedAt) * 0.001;
        if (speechState.intent === 'spatial' && speechState.targetDirHint) {
          state.lookTargetPos.lerp(speechState.targetDirHint, delta * 5.0);
          const dirSp = tmpVecA.subVectors(speechState.targetDirHint, botRoot.position).normalize();
          state.tipTargetPos.copy(botRoot.position).addScaledVector(dirSp, 0.54);
          clawTargetClosure = 0.85;
        } else if (speechState.intent === 'question') {
          state.lookTargetPos.lerp(hp, delta * 6.0);
          state.tipTargetPos.copy(botRoot.position).add(right.clone().multiplyScalar(0.14)).add(new THREE.Vector3(0, -0.22 + speechAmp * 0.08, 0.24));
          clawTargetClosure = 0.06; // Open inquisitive claw
        } else {
          // Expressive rhythmic conversational hand-gestures synced to speech cadence!
          state.lookTargetPos.lerp(hp, delta * 6.0);
          const gestX = Math.sin(spT * 5.2) * (0.08 + speechAmp * 0.12);
          const gestY = -0.26 + Math.sin(spT * 8.4) * (0.04 + speechAmp * 0.09);
          const gestZ = 0.24 + Math.cos(spT * 4.1) * 0.06;
          state.tipTargetPos.copy(botRoot.position).addScaledVector(right, gestX).addScaledVector(flatFwd, gestZ).add(new THREE.Vector3(0, gestY, 0));
          clawTargetClosure = 0.20 + Math.sin(spT * 9.0) * 0.25;
        }
      } else {
        if (state.behavior === 'autonomous' && !boardState.visible) {
          if (state.idleRoutine === 'none') {
            state.nextIdleSwitch -= delta;
            if (state.nextIdleSwitch <= 0) startNextIdleRoutine();
          } else {
            state.idleRoutineTimer -= delta;
            if (state.idleRoutineTimer <= 0) {
              clearIdleEffects(); state.idleRoutine = 'none'; state.nextIdleSwitch = 4.5 + Math.random() * 4.0;
              setEmotion(personality.defaultEmotion || 'curious');
            }
          }
        } else if (state.idleRoutine !== 'holo_projector') {
          clearIdleEffects(); state.idleRoutine = 'none';
        }

        if (state.behavior === 'follow') {
          state.worldAnchor.copy(hp).addScaledVector(flatFwd, 1.05).addScaledVector(right, 0.42);
          state.worldAnchor.y = Math.max(1.0, hp.y + 0.05); state.lookTargetPos.copy(hp);
        } else if (state.behavior === 'orbit') {
          const ang = now * 1.1;
          state.worldAnchor.set(hp.x + Math.cos(ang) * 1.15, Math.max(1.0, hp.y + 0.08), hp.z + Math.sin(ang) * 1.15);
          state.lookTargetPos.copy(hp);
        } else if (state.behavior === 'dance') {
          state.hoverTargetPos.copy(state.worldAnchor).add(new THREE.Vector3(Math.sin(now * 4.5) * 0.22, Math.abs(Math.sin(now * 6.0)) * 0.16, Math.cos(now * 3.5) * 0.15));
          state.targetYawOffset += delta * 2.4;

          if (distToUser > 3.2) {

            state.worldAnchor.y = Math.max(1.05, hp.y + 0.04);
          } else if (distToUser < 0.52) {
            tmpVecA.subVectors(state.worldAnchor, hp).setY(0).normalize();
            if (tmpVecA.lengthSq() < 0.01) tmpVecA.copy(flatFwd);
            state.worldAnchor.copy(hp).addScaledVector(tmpVecA, 0.78);
            state.worldAnchor.y = Math.max(1.05, hp.y + 0.04);
          }

          if (state.idleRoutine === 'cape_flaunt' && capeEnabled) {

            const cornerTarget = botRoot.position.clone().addScaledVector(right, 0.26).add(new THREE.Vector3(0, -0.22 + Math.sin(now * 2.4) * 0.05, 0.16));
            state.tipTargetPos.copy(cornerTarget); clawTargetClosure = 0.86;

          } else if (state.idleRoutine === 'holo_projector') {
            state.lookTargetPos.copy(hp);
            const dialWorld = new THREE.Vector3(0.04, 0.02, 0.22); bodyGroup.localToWorld(dialWorld);
            state.tipTargetPos.copy(dialWorld); clawTargetClosure = 0.72;
            lensDial.rotation.z = now * 2.2; lensProjGroup.visible = true;
            holoSpinObj.rotation.y = now * 1.8; holoMeshA.rotation.x = now * 1.2; holoMeshB.rotation.z = -now * 2.0;
          } else if (state.idleRoutine === 'curious_scan') {
            const ents = (APP.entities || []).filter((e) => e && e !== boardEnt && e !== boardStylusEnt && !e.isHeld && e.category !== 'world_fx');
            if (ents.length) {
              const pick = ents[Math.floor(now * 0.18) % ents.length];
              const info = computeObjCenterAndRadius(pick.group);
              if (info.center.distanceTo(hp) < 3.0) {
                state.lookTargetPos.lerp(info.center, delta * 3.0);
                state.tipTargetPos.copy(botRoot.position).add(new THREE.Vector3(Math.sin(now * 1.2) * 0.10, -0.44, 0.05));
              }
            } else {
              state.lookTargetPos.copy(hp);
              state.tipTargetPos.copy(botRoot.position).add(new THREE.Vector3(Math.sin(now * 1.4) * 0.16, -0.44, 0.08));
            }
          } else {
            if (state.behavior === 'autonomous') state.targetYawOffset = 0;
            let focusPt = hp;
            for (const h of Object.values(APP.hands || {})) {

            state.lookTargetPos.lerp(focusPt, delta * 5.0);
            if (!state.userClawGrabHand) {
              state.tipTargetPos.copy(botRoot.position).add(new THREE.Vector3(Math.sin(now * 1.25) * 0.14, -0.46 + Math.cos(now * 1.55) * 0.05, -0.08 + Math.cos(now * 0.9) * 0.10));
              clawTargetClosure = 0.15;
            }
          }
        }
      }

      if (state.behavior !== 'dance') {
        state.hoverTargetPos.copy(state.worldAnchor).add(new THREE.Vector3(Math.sin(now * 0.7) * 0.055, Math.sin(now * 1.7) * 0.035, Math.cos(now * 0.6) * 0.045));
      }
    } else if (state.mode === 'pointing') {
      const tgt = state.targetObj ? computeObjCenterAndRadius(state.targetObj).center : state.pointTargetPos;
      const dirToTgt = tmpVecA.subVectors(tgt, botRoot.position).normalize();
      state.tipTargetPos.copy(botRoot.position).addScaledVector(dirToTgt, 0.62);

      state.pointTimer -= delta;

      if (!state.targetObj || !state.targetObj.parent) tentacle_release();
      else {
        const info = computeObjCenterAndRadius(state.targetObj);
        state.hoverTargetPos.copy(info.center).add(new THREE.Vector3(0.28, 0.46, 0.22));
        state.tipTargetPos.copy(info.center); state.lookTargetPos.copy(info.center);
        if (state.currentTipPos.distanceTo(info.center) < 0.16) {
          state.grabProgress += delta; clawTargetClosure = 0.88;
          if (state.grabProgress > 0.22) {
            state.heldObj = state.targetObj; state.heldEnt = getEntityRecord(state.heldObj); state.heldRadius = info.radius;
            if (state.heldEnt) state.heldEnt.heldByBot = true;
            state.heldObj.userData = state.heldObj.userData || {}; state.heldObj.userData.heldByBot = true;
            state.mode = 'holding'; setEmotion('happy');
          }
        }
      }
    } else if (state.mode === 'holding') {
      if (!state.heldObj || !state.heldObj.parent) tentacle_release();
      else {
        state.hoverTargetPos.copy(state.worldAnchor).add(new THREE.Vector3(0, Math.sin(now * 1.5) * 0.025, 0));
        state.tipTargetPos.copy(botRoot.position).add(new THREE.Vector3(-0.10, -0.44, 0.22)); state.lookTargetPos.copy(hp);
      }
    } else if (state.mode === 'handing') {
      if (!state.heldObj || !state.heldObj.parent) tentacle_release();
      else {
        state.hoverTargetPos.copy(hp).addScaledVector(flatFwd, 0.72); state.hoverTargetPos.y = Math.max(0.92, hp.y + 0.02);
        let handAnchor = null, handGrabbing = false;
        for (const h of Object.values(APP.hands || {})) {
          if (!h || !h.active) continue;
          const pos = h.isHand ? h.fingerTips?.index : h.pos;
          if (pos) {
            if (!handAnchor || pos.distanceTo(state.currentTipPos) < handAnchor.distanceTo(state.currentTipPos)) handAnchor = pos;
            if ((h.pinching || h.trigger || h.gripping) && pos.distanceTo(state.currentTipPos) < 0.22) handGrabbing = true;
          }
        }
        if (handAnchor && handAnchor.distanceTo(hp) < 1.1) state.tipTargetPos.copy(handAnchor).addScaledVector(flatFwd, -0.05);
        else { state.tipTargetPos.copy(hp).addScaledVector(flatFwd, 0.42); state.tipTargetPos.y = Math.max(0.75, hp.y - 0.20); }
        state.lookTargetPos.copy(hp);
        if (handGrabbing || state.heldEnt?.grabbedByHand) tentacle_release();
      }
    } else if (state.mode === 'showing') {
      if (!state.targetObj || !state.targetObj.parent) tentacle_release();
      else {
        const info = computeObjCenterAndRadius(state.targetObj);
        state.hoverTargetPos.copy(info.center).add(new THREE.Vector3(0.36, 0.40 + Math.sin(now * 2.0) * 0.03, 0.22));
        state.tipTargetPos.copy(info.center).add(new THREE.Vector3(0.18, 0.12, 0.12));
        state.worldAnchor.copy(state.hoverTargetPos); state.lookTargetPos.copy(info.center);
        state.showTimer -= delta;
        if (state.showTimer <= 0) tentacle_release();
      }

    const stiffness = 30.0, damping = 8.2;
    tmpVecA.subVectors(state.hoverTargetPos, botRoot.position).multiplyScalar(stiffness);
    state.velocity.addScaledVector(tmpVecA, delta);
    state.velocity.multiplyScalar(Math.exp(-damping * delta));
    botRoot.position.addScaledVector(state.velocity, delta);

    bodyGroup.lookAt(state.lookTargetPos);
    state.yawOffset += (state.targetYawOffset - state.yawOffset) * Math.min(1, delta * 6.5);
    bodyGroup.rotateY(state.yawOffset);

    // Speech-driven head nods & inquisitive tilts
    if (isSpeakingNow) {
      if (speechState.intent === 'question') {
        bodyGroup.rotateZ(0.18 + Math.sin(now * 4.0) * 0.04);
        bodyGroup.rotateX(-0.08 + speechAmp * 0.10);
      } else {
        bodyGroup.rotateX(Math.sin(now * 9.5) * (0.04 + speechAmp * 0.12));
        bodyGroup.rotateY(Math.sin(now * 4.2) * 0.07);
      }
    }
    if (state.nodPhase > 0) {
      state.nodPhase = Math.max(0, state.nodPhase - delta * 12.0);
      bodyGroup.rotateX(Math.sin(state.nodPhase) * 0.28);
    }

    const speed = state.velocity.length();
    bodyGroup.rotateZ(THREE.MathUtils.clamp(-state.velocity.x * 0.24, -0.38, 0.38));
    bodyGroup.rotateX(THREE.MathUtils.clamp(speed * 0.12, -0.22, 0.22));

    const wingSpeed = state.emotion === 'excited' || state.emotion === 'playful' ? 9.5 : 4.5;
    const wingFlutter = Math.sin(now * wingSpeed) * 0.07 + Math.min(0.30, speed * 0.26) + speechAmp * 0.09;
    wingL.rotation.z = wingFlutter;
    wingR.rotation.z = -wingFlutter;

    // Dynamic Eye-Lens Pupil Dilation & Contraction (like a living iris!)
    let targetPupil = 1.0;
    if (state.emotion === 'curious' || speechState.intent === 'question') targetPupil = 1.38;
    else if (state.emotion === 'excited' || state.emotion === 'happy') targetPupil = 1.24;
    else if (state.emotion === 'focused' || state.mode === 'pointing' || state.generatingActive) targetPupil = 0.56;
    if (APP.visual?.getEnvironment?.() === 'matrix_black') targetPupil *= 1.16;
    targetPupil += Math.sin(now * 2.2) * 0.06 + speechAmp * 0.22;
    targetPupil = THREE.MathUtils.clamp(targetPupil, 0.45, 1.55);
    pupilScale += (targetPupil - pupilScale) * Math.min(1, delta * 9.0);
    pupilIrisRing.scale.setScalar(pupilScale);
    pupilCoreDisc.scale.setScalar(0.78 + (pupilScale - 0.5) * 0.65);

    blinkTimer -= delta;
    let eyeScaleY = state.emotion === 'happy' ? 0.65 : state.emotion === 'focused' ? 0.78 : 1.0;
    let eyeScaleX = state.emotion === 'happy' ? 1.18 : 1.0;
    if (blinkTimer <= 0) {
      if (blinkTimer > -0.11) eyeScaleY = 0.10;
      else blinkTimer = 2.4 + Math.random() * 3.8;
    }
    eyeL.scale.set(eyeScaleX, eyeScaleY, 1);
    eyeR.scale.set(eyeScaleX, eyeScaleY, 1);

    const emHex = state.emotion === 'excited' ? 0xfbbf24 : state.emotion === 'happy' ? 0x00ff9d : state.emotion === 'playful' ? 0xf472b6 : 0x5efce8;
    eyeMat.color.setHex(emHex);
    pupilIrisMat.color.setHex(emHex);

    reactorCore.rotation.y = now * 1.4;
    reactorCore.rotation.x = now * 0.8;

    updateTentacleIK(delta, now);
    stepCapePhysics(delta, now);
    updateDanglingHook(delta, now);

    if (boardState.visible && boardRoot.visible) {
      if (!boardDetached && !boardEnt.isHeld) {
        const targetBoardPos = botRoot.position.clone().addScaledVector(right, -0.66).add(new THREE.Vector3(0, -0.02, 0.08));
        boardRoot.position.lerp(targetBoardPos, 1 - Math.exp(-8 * delta));
        boardRoot.lookAt(hp.x, boardRoot.position.y, hp.z);

        boomArm.visible = true;
        const armStart = botRoot.position.clone().addScaledVector(right, -0.14);
        const armEnd = boardRoot.position.clone().addScaledVector(right, BOARD_W * 0.48);
        const armLen = Math.max(0.05, armStart.distanceTo(armEnd));
        boomArm.position.lerpVectors(armStart, armEnd, 0.5);
        boomArm.scale.set(1, armLen, 1);
        boomArm.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), armEnd.sub(armStart).normalize());
      } else {
        boomArm.visible = false;
      }
      updateBoardPenDrawing();
    } else {
      boomArm.visible = false;
      stylusGroup.visible = false;
    }

    if (state.heldObj && (state.mode === 'holding' || state.mode === 'handing')) {
      const forwardDir = new THREE.Vector3(0, 0, 1).applyQuaternion(clawGroup.quaternion);
      tmpVecA.copy(clawGroup.position).addScaledVector(forwardDir, 0.045 + state.heldRadius * 0.35);
      state.heldObj.position.lerp(tmpVecA, 1 - Math.exp(-16 * delta));
      if (state.heldEnt?.physics?.vel) state.heldEnt.physics.vel.set(0, 0, 0);
    }

    const targetEnergy = isSpeakingNow ? (0.45 + speechAmp * 0.65) : (APP.micLevel || 0);
    audioEnergy += (targetEnergy - audioEnergy) * Math.min(1, delta * 12);
    plasmaMat.emissiveIntensity = 1.1 + audioEnergy * 2.4 + state.petGlowBoost;
    botLight.intensity = 0.6 + audioEnergy * 1.2 + state.petGlowBoost * 0.8;
  }

  }

  function floatTo16BitPCMBase64(float32, inputSampleRate) {
    const ratio = inputSampleRate / 16000, outLen = Math.max(1, Math.floor(float32.length / ratio));
    const buf = new ArrayBuffer(outLen * 2), view = new DataView(buf); let sumSq = 0;
    for (let i = 0; i < outLen; i++) {
      const pos = i * ratio, i0 = Math.floor(pos), i1 = Math.min(float32.length - 1, i0 + 1), frac = pos - i0;
      const s = Math.max(-1, Math.min(1, float32[i0] * (1 - frac) + float32[i1] * frac));
      sumSq += s * s; view.setInt16(i * 2, s < 0 ? s * 0x8000 : s * 0x7FFF, true);
    }
    APP.micLevel = Math.min(1, Math.sqrt(sumSq / outLen) * 5.0);
    const bytes = new Uint8Array(buf); let bin = '';
    for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
    return btoa(bin);
  }

  function playPCM24kBase64(b64) {
    if (!b64) return;
    const ctx = ensureUnlockedAudioContext(), bin = atob(b64), sampleCount = Math.floor(bin.length / 2);
    if (sampleCount <= 0) return;
    const floatBuf = new Float32Array(sampleCount); let sumSq = 0;
    for (let i = 0; i < sampleCount; i++) {
      let val = (bin.charCodeAt(i * 2 + 1) << 8) | bin.charCodeAt(i * 2);
      if (val >= 0x8000) val -= 0x10000;
      const s = val / 32768; floatBuf[i] = s; sumSq += s * s;
    }
    APP.botSpeechEnergy = Math.min(1.0, Math.sqrt(sumSq / sampleCount) * 4.5);
    const audioBuf = ctx.createBuffer(1, sampleCount, 24000);
    audioBuf.getChannelData(0).set(floatBuf);
    const src = ctx.createBufferSource(); src.buffer = audioBuf; src.connect(ctx.destination);
    const now = ctx.currentTime; if (nextPlayTime < now) nextPlayTime = now + 0.025;
    src.start(nextPlayTime); nextPlayTime += audioBuf.duration;
    APP.botSpeaking = true; activePlaySources.add(src);
    src.onended = () => { activePlaySources.delete(src); if (activePlaySources.size === 0) { APP.botSpeaking = false; APP.botSpeechEnergy = 0; } };
  }
