        const idx = r * CAPE_COLS + c;
        if (capeGrabCornerActive && idx === CAPE_GRAB_IDX) continue;
        const cur = capeCur[idx];
        const prev = capePrev[idx];

        let vx = (cur.x - prev.x) * 0.965;
        let vy = (cur.y - prev.y) * 0.965;
        let vz = (cur.z - prev.z) * 0.965;
        const spdSq = vx * vx + vy * vy + vz * vz;
        if (spdSq > maxStepSpeed * maxStepSpeed) {
          const s = maxStepSpeed / Math.sqrt(spdSq);
          vx *= s; vy *= s; vz *= s;
        }
        prev.copy(cur);
        const flutter = (Math.sin(elapsed * 4.2 + c * 0.55 + r * 0.38) * 0.5 + 0.5) * 0.00052;
        cur.x += vx + tmpCapeBack.x * flutter;
        cur.y += vy + gravStep;
        cur.z += vz + tmpCapeBack.z * flutter;
      }
    }

    const bodyCenter = botRoot.position;
    const R_BODY = 0.176;
    const R_BODY_SQ = R_BODY * R_BODY;

    for (let iter = 0; iter < 5; iter++) {
      for (let k = 0; k < capeConstraints.length; k++) {
        const con = capeConstraints[k];
        const ia = con[0], ib = con[1], restLen = con[2], stiff = con[3];
        const pa = capeCur[ia], pb = capeCur[ib];
        tmpCapeDelta.subVectors(pb, pa);
        const d = tmpCapeDelta.length();
        if (d < 1e-6) continue;
        const diff = ((d - restLen) / d) * stiff;
        const pinA = ia < CAPE_COLS || (capeGrabCornerActive && ia === CAPE_GRAB_IDX);
        const pinB = ib < CAPE_COLS || (capeGrabCornerActive && ib === CAPE_GRAB_IDX);
        if (pinA && pinB) continue;
        if (pinA) pb.addScaledVector(tmpCapeDelta, -diff);
        else if (pinB) pa.addScaledVector(tmpCapeDelta, diff);
        else { pa.addScaledVector(tmpCapeDelta, diff * 0.5); pb.addScaledVector(tmpCapeDelta, -diff * 0.5); }
      }

      for (let idx = CAPE_COLS; idx < CAPE_COUNT; idx++) {
        if (capeGrabCornerActive && idx === CAPE_GRAB_IDX) continue;
        const p = capeCur[idx];
        tmpCapeDelta.subVectors(p, bodyCenter);
        let dSq = tmpCapeDelta.lengthSq();
        if (dSq < R_BODY_SQ) {
          const d = Math.sqrt(dSq);
          if (d > 1e-5) p.copy(bodyCenter).addScaledVector(tmpCapeDelta, R_BODY / d);
          else p.set(bodyCenter.x, bodyCenter.y, bodyCenter.z - R_BODY);
        }

        if (!capeGrabCornerActive) {
          for (let s = 0; s < TENTACLE_SEGMENTS; s++) {
            const segPos = segments[s].position;
            const rTent = THREE.MathUtils.lerp(0.052, 0.032, s / (TENTACLE_SEGMENTS - 1));
            tmpCapeDelta.subVectors(p, segPos);
            dSq = tmpCapeDelta.lengthSq();
            if (dSq < rTent * rTent) {
              const d = Math.sqrt(dSq);
              if (d > 1e-5) p.copy(segPos).addScaledVector(tmpCapeDelta, rTent / d);
              else p.copy(segPos).add(new THREE.Vector3(0, 0, -rTent));
            }
          }
          tmpCapeDelta.subVectors(p, clawGroup.position);
          dSq = tmpCapeDelta.lengthSq();
          if (dSq < 0.045 * 0.045) {
            const d = Math.sqrt(dSq);
            if (d > 1e-5) p.copy(clawGroup.position).addScaledVector(tmpCapeDelta, 0.045 / d);
          }
        }
        if (p.y < 0.02) p.y = 0.02;
      }
    }

    const arr = capeGeo.attributes.position.array;
    for (let i = 0; i < CAPE_COUNT; i++) {
      const p = capeCur[i];
      if (!isFinite(p.x) || !isFinite(p.y) || !isFinite(p.z)) { snapCapeToRest(); break; }
      arr[i * 3] = p.x; arr[i * 3 + 1] = p.y; arr[i * 3 + 2] = p.z;
    }
    capeGeo.attributes.position.needsUpdate = true;
    capeGeo.computeVertexNormals();
  }

  async function setCapeTexture(nameOrUrl) {
    const q = String(nameOrUrl || '').trim();
    const low = q.toLowerCase();
    if (!q || low === 'algeria' || low === 'algerian' || low === 'flag' || low === 'default' || low.includes('الجزائر')) {
      capeMat.map = defaultCapeTex;
      capeMat.needsUpdate = true;
      setCapeEnabled(true);
      return { ok: true, texture: 'Algerian Flag', message: 'Geeny cape set to Algerian Flag.' };
    }
    const clean = q.replace(/\.[^.]+$/, '').toLowerCase();
    let url = q.startsWith('http') || q.startsWith('blob:') || q.startsWith('./') || q.startsWith('data:') ? q : null;
    if (!url && APP.importedImages) {
      const rec = APP.importedImages.get(clean) || Array.from(APP.importedImages.values()).find((r) => r.cleanName.includes(clean));
      if (rec) url = rec.url;
    }
    if (!url) {
      for (const p of [`./assets/${clean}.png`, `./assets/${clean}.jpg`, `./assets/${clean}.jpeg`, `./assets/${clean}.webp`, `./${clean}.png`, `./${clean}.jpg`]) {
        try { const h = await fetch(p, { method: 'HEAD' }); if (h.ok) { url = p; break; } } catch (_) {}
      }
    }
    if (!url) return { ok: false, error: `Picture "${q}" not found in /assets/ or imported images.` };
    try {
      const tex = await new THREE.TextureLoader().loadAsync(url);
      tex.colorSpace = THREE.SRGBColorSpace;
      tex.anisotropy = 4;
      if (customCapeTex && customCapeTex !== defaultCapeTex) { try { customCapeTex.dispose(); } catch (_) {} }
      customCapeTex = tex;
      capeMat.map = tex;
      capeMat.needsUpdate = true;
      setCapeEnabled(true);
      return { ok: true, texture: q, message: `Updated Geeny cape texture to "${q}".` };
    } catch (e) {
      return { ok: false, error: 'Failed to load cape texture: ' + e.message };
    }
  }

  function drawBoardRoundedRect(x, y, w, h, r, fill, stroke) {
    bCtx.beginPath();
    bCtx.moveTo(x + r, y);
    bCtx.arcTo(x + w, y, x + w, y + h, r);
    bCtx.arcTo(x + w, y + h, x, y + h, r);
    bCtx.arcTo(x, y + h, x, y, r);
    bCtx.arcTo(x, y, x + w, y, r);
    bCtx.closePath();
    if (fill) { bCtx.fillStyle = fill; bCtx.fill(); }
    if (stroke) { bCtx.strokeStyle = stroke; bCtx.lineWidth = 3; bCtx.stroke(); }

  function wrapCanvasText(ctx, text, x, y, maxWidth, lineHeight, maxY) {
    const paragraphs = String(text || '').split('\n');
    let cy = y;
    for (const para of paragraphs) {
      const words = para.split(/\s+/);
      let line = '';
      for (let i = 0; i < words.length; i++) {
        const test = line ? line + ' ' + words[i] : words[i];
        if (ctx.measureText(test).width > maxWidth && line) {
          ctx.fillText(line, x, cy); line = words[i]; cy += lineHeight;
          if (maxY && cy > maxY) return cy;
        } else { line = test; }
      }
      if (line) { ctx.fillText(line, x, cy); cy += lineHeight; }
      if (maxY && cy > maxY) return cy;
    }
    return cy;
  }

  function executeDrawCommands(cmds, area) {
    if (!Array.isArray(cmds)) return;
    const ox = area.x, oy = area.y, aw = area.w, ah = area.h;
    const px = (nx) => ox + Number(nx || 0) * aw;
    const py = (ny) => oy + Number(ny || 0) * ah;
    for (const c of cmds) {
      if (!c || !c.type) continue;
      bCtx.save();
      bCtx.strokeStyle = c.color || '#00f0b4';
      bCtx.fillStyle = c.fill || c.color || 'rgba(0,240,180,0.18)';
      bCtx.lineWidth = c.width || 4;
      bCtx.lineCap = 'round'; bCtx.lineJoin = 'round';
      if (c.dashed) bCtx.setLineDash([8, 6]);
      if (c.type === 'grid') {
        bCtx.strokeStyle = c.color || 'rgba(56,189,248,0.18)'; bCtx.lineWidth = 1.5;
        const n = c.divisions || 6;
        for (let i = 1; i < n; i++) {
          bCtx.beginPath(); bCtx.moveTo(px(i / n), py(0)); bCtx.lineTo(px(i / n), py(1)); bCtx.stroke();
          bCtx.beginPath(); bCtx.moveTo(px(0), py(i / n)); bCtx.lineTo(px(1), py(i / n)); bCtx.stroke();
        }
      } else if (c.type === 'circle') {
        bCtx.beginPath(); bCtx.arc(px(c.cx ?? 0.5), py(c.cy ?? 0.5), (c.r ?? 0.15) * Math.min(aw, ah), 0, Math.PI * 2);
        if (c.filled) bCtx.fill(); bCtx.stroke();
      } else if (c.type === 'ellipse') {
        bCtx.beginPath(); bCtx.ellipse(px(c.cx ?? 0.5), py(c.cy ?? 0.5), (c.rx ?? 0.2) * aw, (c.ry ?? 0.15) * ah, c.rot || 0, 0, Math.PI * 2);
        if (c.filled) bCtx.fill(); bCtx.stroke();
      } else if (c.type === 'line' || c.type === 'arrow') {
        const x1 = px(c.x1), y1 = py(c.y1), x2 = px(c.x2), y2 = py(c.y2);
        bCtx.beginPath(); bCtx.moveTo(x1, y1); bCtx.lineTo(x2, y2); bCtx.stroke();
        if (c.type === 'arrow') {
          const ang = Math.atan2(y2 - y1, x2 - x1), hl = 14;
          bCtx.beginPath();
          bCtx.moveTo(x2, y2); bCtx.lineTo(x2 - hl * Math.cos(ang - 0.4), y2 - hl * Math.sin(ang - 0.4));
          bCtx.moveTo(x2, y2); bCtx.lineTo(x2 - hl * Math.cos(ang + 0.4), y2 - hl * Math.sin(ang + 0.4));
          bCtx.stroke();
        }
      } else if (c.type === 'arc') {
        bCtx.beginPath(); bCtx.arc(px(c.cx ?? 0.5), py(c.cy ?? 0.5), (c.r ?? 0.1) * Math.min(aw, ah), c.start ?? 0, c.end ?? Math.PI); bCtx.stroke();
      } else if (c.type === 'poly' && Array.isArray(c.points) && c.points.length >= 2) {
        bCtx.beginPath(); bCtx.moveTo(px(c.points[0][0]), py(c.points[0][1]));
        for (let i = 1; i < c.points.length; i++) bCtx.lineTo(px(c.points[i][0]), py(c.points[i][1]));
        if (c.closed) bCtx.closePath(); if (c.filled) bCtx.fill(); bCtx.stroke();
      } else if (c.type === 'curve') {
        bCtx.beginPath(); bCtx.moveTo(px(c.x1), py(c.y1)); bCtx.quadraticCurveTo(px(c.cx), py(c.cy), px(c.x2), py(c.y2)); bCtx.stroke();
      } else if (c.type === 'text') {
        bCtx.fillStyle = c.color || '#e2f8ff'; bCtx.font = 'bold ' + (c.size || 22) + 'px system-ui, sans-serif';
        bCtx.textAlign = 'center'; bCtx.fillText(String(c.text || ''), px(c.x ?? 0.5), py(c.y ?? 0.5));
      }
      bCtx.restore();
    }
  }

    pen.nibMesh.getWorldPosition(tmpPenTipW);

  async function fetchWikipediaData(topic) {
    const clean = String(topic || '').trim();
    if (!clean) return null;
    const langHost = /[\u0600-\u06FF]/.test(clean) ? 'ar.wikipedia.org' : 'en.wikipedia.org';
    try {
      const sJson = await (await fetch(`https://${langHost}/w/api.php?action=opensearch&search=${encodeURIComponent(clean)}&limit=1&namespace=0&format=json&origin=*`)).json();
      const title = sJson?.[1]?.[0] || clean;
      const sumRes = await fetch(`https://${langHost}/api/rest_v1/page/summary/${encodeURIComponent(title)}`);
      if (!sumRes.ok) return null;
      const sum = await sumRes.json();
      const eJson = await (await fetch(`https://${langHost}/w/api.php?action=query&prop=extracts&exintro=false&explaintext=true&exchars=2400&titles=${encodeURIComponent(sum.title || title)}&format=json&origin=*`)).json();
      const firstPage = Object.values(eJson?.query?.pages || {})[0];
      const fullText = firstPage?.extract || sum.extract || '';
      let imgObj = null;
      const imgUrl = sum.thumbnail?.source || sum.originalimage?.source;
      if (imgUrl) imgObj = await loadCORSImage(imgUrl);

      const chunks = [];
      let rem = fullText.trim();
      while (rem.length > 0) {
        if (rem.length <= 540) { chunks.push(rem); break; }
        let cut = rem.lastIndexOf('.', 540);
        if (cut < 250) cut = 540;
        chunks.push(rem.slice(0, cut + 1).trim());
        rem = rem.slice(cut + 1).trim();
      }
      return { title: sum.title || title, summary: sum.extract || '', pages: chunks.length ? chunks : [sum.description || ''], imageObj: imgObj };
    } catch (_) { return null; }
  }

  async function fetchOnlinePhoto(query) {
    const clean = String(query || '').trim();
    if (!clean) return null;
    const key = clean.replace(/\.[^.]+$/, '').toLowerCase();
    if (APP.importedImages) {
      const rec = APP.importedImages.get(key) || Array.from(APP.importedImages.values()).find((r) => r.cleanName.includes(key));
      if (rec) { const localImg = await loadCORSImage(rec.url); if (localImg) return localImg; }
    }
    const wiki = await fetchWikipediaData(clean);
    if (wiki?.imageObj) return wiki.imageObj;
    try {
      const data = await (await fetch(`https://commons.wikimedia.org/w/api.php?action=query&generator=search&gsrnamespace=6&gsrsearch=${encodeURIComponent(clean)}&gsrlimit=4&prop=imageinfo&iiprop=url&iiurlwidth=1024&format=json&origin=*`)).json();
      for (const p of Object.values(data?.query?.pages || {})) {

    } catch (_) {}
    return null;
  }

  function controlBotAction(action, params = {}) {
    if (!botEnabled) setEnabled(true);
    const act = String(action || '').toLowerCase();
    const hp = APP.headPos || new THREE.Vector3(0, 1.5, 0);
    const hf = APP.headFwd || new THREE.Vector3(0, 0, -1);
    const flat = new THREE.Vector3(hf.x, 0, hf.z).normalize();
    if (flat.lengthSq() < 0.01) flat.set(0, 0, -1);
    const right = new THREE.Vector3().crossVectors(flat, new THREE.Vector3(0, 1, 0)).normalize();

    if (act === 'come_here' || act === 'come') {
      clearIdleEffects();
      state.mode = 'idle'; state.idleRoutine = 'none'; state.behavior = 'autonomous'; state.targetYawOffset = 0;
      state.worldAnchor.copy(hp).addScaledVector(flat, Number(params.distance || 0.85)).addScaledVector(right, 0.22);

      state.lookTargetPos.copy(hp);
      setEmotion('happy');
      return { ok: true, message: 'Geeny flew right over to you.' };
    }
    if (act === 'turn_around' || act === 'turn_back') {
      state.behavior = 'turned'; state.targetYawOffset += Math.PI;
      setEmotion('playful');
      return { ok: true, message: 'Geeny turned around.' };
    }
    if (act === 'spin' || act === 'spin_360') {
      state.behavior = 'turned'; state.targetYawOffset += Math.PI * 2;
      setEmotion('excited');
      return { ok: true, message: 'Geeny spun 360 degrees.' };
    }
    if (act === 'face_user' || act === 'look_at_user') {
      state.behavior = 'autonomous'; state.targetYawOffset = 0; state.lookTargetPos.copy(hp);
      setEmotion('curious');
      return { ok: true, message: 'Geeny facing you.' };
    }
    if (act === 'follow_on') { state.mode = 'idle'; state.behavior = 'follow'; state.targetYawOffset = 0; setEmotion('happy'); return { ok: true, message: 'Geeny following you.' }; }
    if (act === 'follow_off' || act === 'stay') { state.behavior = 'autonomous'; state.worldAnchor.copy(botRoot.position); return { ok: true, message: 'Geeny staying here.' }; }
    if (act === 'orbit_user') { state.mode = 'idle'; state.behavior = 'orbit'; state.behaviorTimer = 8.0; state.targetYawOffset = 0; setEmotion('playful'); return { ok: true, message: 'Geeny orbiting you.' }; }
    if (act === 'dance') { state.mode = 'idle'; state.behavior = 'dance'; state.behaviorTimer = 6.0; setEmotion('excited'); return { ok: true, message: 'Geeny dancing!' }; }
    if (act === 'nod') { state.nodPhase = Math.PI * 4; setEmotion('happy'); return { ok: true, message: 'Geeny nodded.' }; }
    if (act === 'cape_flaunt' || act === 'show_cape') {
      setCapeEnabled(true); state.mode = 'idle'; state.idleRoutine = 'cape_flaunt'; state.idleRoutineTimer = 7.0;
      setEmotion('happy');
      return { ok: true, message: 'Geeny showing off its cape!' };
    }
    if (act === 'holo_projector' || act === 'project_hologram') {
      state.mode = 'idle'; state.idleRoutine = 'holo_projector'; state.idleRoutineTimer = 8.0; lensProjGroup.visible = true;
      setEmotion('curious');
      return { ok: true, message: 'Geeny rotating eye lens to project holograms!' };
    }
    if (act === 'juggle' || act === 'play') {
      state.mode = 'idle'; startJuggleRoutine(8.5);
      return { ok: true, message: 'Geeny playing by tossing and catching!' };
    }
    if (act === 'move_relative') {
      state.mode = 'idle';
      state.worldAnchor.add(new THREE.Vector3(Number(params.dx || 0), Number(params.dy || 0), Number(params.dz || 0)));
      state.worldAnchor.y = Math.max(0.4, state.worldAnchor.y);
      return { ok: true, message: 'Geeny moved.' };
    }
    if (act === 'inspect_object' && params.target) return tentacle_show(params.target);
    return { ok: false, error: 'Unknown control action: ' + action };
  }

  function evalCubicBezier(p0, c1, c2, p3, t, out) {
    const u = 1 - t;
    out.set(0, 0, 0).addScaledVector(p0, u * u * u).addScaledVector(c1, 3 * u * u * t).addScaledVector(c2, 3 * u * t * t).addScaledVector(p3, t * t * t);
    return out;
  }

  function updateTentacleIK(dt, elapsed) {
    vP0.set(0, -0.14, 0);
    botRoot.localToWorld(vP0);

    const tipLerp = 1 - Math.exp(-11.0 * dt);
    state.currentTipPos.lerp(state.tipTargetPos, tipLerp);
    vP3.copy(state.currentTipPos);

    const dist = vP0.distanceTo(vP3);
    vC1.copy(vP0).add(new THREE.Vector3(0, -Math.max(0.22, dist * 0.46), 0)).addScaledVector(state.velocity, -0.16);
    tmpVecA.subVectors(vP3, vP0).normalize();
    vC2.copy(vP3).addScaledVector(tmpVecA, -Math.max(0.16, dist * 0.34)).addScaledVector(state.velocity, -0.12);
    vC2.y += Math.sin(elapsed * 2.2) * 0.04 + 0.07;

    for (let i = 0; i < TENTACLE_SEGMENTS; i++) {
      const t = i / TENTACLE_SEGMENTS, tNext = (i + 1) / TENTACLE_SEGMENTS;
      evalCubicBezier(vP0, vC1, vC2, vP3, t, vPt);
      evalCubicBezier(vP0, vC1, vC2, vP3, tNext, vNext);
      segments[i].position.copy(vPt);
      segments[i].lookAt(vNext);
    }

    clawGroup.position.copy(vP3);
    if (state.mode === 'showing' && state.targetObj) {
      const info = computeObjCenterAndRadius(state.targetObj);
      clawGroup.lookAt(info.center);
      const beamDist = Math.max(0.15, Math.min(0.95, vP3.distanceTo(info.center)));
      showBeam.scale.set(1, 1, beamDist / 0.60);
      showBeam.position.z = beamDist * 0.52;
    } else if (state.mode === 'pointing') {
      const tgt = state.targetObj ? computeObjCenterAndRadius(state.targetObj).center : state.pointTargetPos;
      clawGroup.lookAt(tgt);
      const beamDist = Math.max(0.25, Math.min(1.6, vP3.distanceTo(tgt)));
      showBeam.scale.set(0.65, 0.65, beamDist / 0.60);
      showBeam.position.z = beamDist * 0.52;
    } else {
      evalCubicBezier(vP0, vC1, vC2, vP3, 0.9, tmpVecB);
      tmpVecA.subVectors(vP3, tmpVecB).normalize();
      if (tmpVecA.lengthSq() > 0.001) clawGroup.lookAt(tmpVecB.copy(vP3).add(tmpVecA));
    }

    clawClosure += (clawTargetClosure - clawClosure) * Math.min(1, dt * 10);
    const openAngle = THREE.MathUtils.lerp(0.62, -0.14, clawClosure);
    for (let p = 0; p < prongs.length; p++) prongs[p].rotation.y = openAngle;
  }

  function updateRobot(dt, elapsed) {
    if (!botEnabled) return;
    const delta = Math.min(0.05, dt || 0.016);
    const now = elapsed || (performance.now() * 0.001);

    const hp = APP.headPos || new THREE.Vector3(0, 1.5, 0);
    const hf = APP.headFwd || new THREE.Vector3(0, 0, -1);
    const flatFwd = new THREE.Vector3(hf.x, 0, hf.z).normalize();
    if (flatFwd.lengthSq() < 0.01) flatFwd.set(0, 0, -1);
    const right = new THREE.Vector3().crossVectors(flatFwd, new THREE.Vector3(0, 1, 0)).normalize();

    if (state.behaviorTimer > 0) {
      state.behaviorTimer -= delta;
      if (state.behaviorTimer <= 0 && (state.behavior === 'orbit' || state.behavior === 'dance')) state.behavior = 'autonomous';
    }

    if (state.mode === 'idle') {
      // Idle routine lifecycle
      if (state.behavior === 'autonomous' && !boardState.visible) {
        if (state.idleRoutine === 'none') {
          state.nextIdleSwitch -= delta;
          if (state.nextIdleSwitch <= 0) startNextIdleRoutine();
        } else {
          state.idleRoutineTimer -= delta;
          if (state.idleRoutineTimer <= 0) {
            clearIdleEffects();
            state.idleRoutine = 'none';
            state.nextIdleSwitch = 3.5 + Math.random() * 3.5;
            setEmotion('curious');
          }
        }
      } else {
        clearIdleEffects();
        state.idleRoutine = 'none';
      }

      if (state.behavior === 'follow') {
        state.worldAnchor.copy(hp).addScaledVector(flatFwd, 1.05).addScaledVector(right, 0.42);
        state.worldAnchor.y = Math.max(1.0, hp.y + 0.05);
        state.lookTargetPos.copy(hp);
      } else if (state.behavior === 'orbit') {
        const ang = now * 1.1;
        state.worldAnchor.set(hp.x + Math.cos(ang) * 1.15, Math.max(1.0, hp.y + 0.08), hp.z + Math.sin(ang) * 1.15);
        state.lookTargetPos.copy(hp);
      } else if (state.behavior === 'dance') {
        state.hoverTargetPos.copy(state.worldAnchor).add(new THREE.Vector3(Math.sin(now * 4.5) * 0.22, Math.abs(Math.sin(now * 6.0)) * 0.16, Math.cos(now * 3.5) * 0.15));
        state.targetYawOffset += delta * 2.4;
      } else {
        const distToUser = state.worldAnchor.distanceTo(hp);
        if (distToUser > 3.2) {
          state.worldAnchor.copy(hp).addScaledVector(flatFwd, 1.15).addScaledVector(right, 0.45);
          state.worldAnchor.y = Math.max(1.05, hp.y + 0.04);
        } else if (distToUser < 0.44) {
          tmpVecA.subVectors(state.worldAnchor, hp).setY(0).normalize();
          if (tmpVecA.lengthSq() < 0.01) tmpVecA.copy(flatFwd);
          state.worldAnchor.copy(hp).addScaledVector(tmpVecA, 0.72);
          state.worldAnchor.y = Math.max(1.05, hp.y + 0.04);
        }

        // Execute active idle routine animations!
        if (state.idleRoutine === 'cape_flaunt' && capeEnabled) {
          // Turn slightly sideways & lift bottom cape edge toward user to show off flag/texture
          state.targetYawOffset = Math.sin(now * 1.6) * 0.55 + 0.85;
          state.lookTargetPos.copy(hp);
          const cornerTarget = botRoot.position.clone()
            .addScaledVector(right, 0.26)
            .add(new THREE.Vector3(0, -0.22 + Math.sin(now * 2.4) * 0.05, 0.16));
          state.tipTargetPos.copy(cornerTarget);
          clawTargetClosure = 0.86;
          capeGrabCornerActive = state.currentTipPos.distanceTo(cornerTarget) < 0.18;
        } else if (state.idleRoutine === 'holo_projector') {
          state.lookTargetPos.copy(hp);
          // Tentacle reaches up to dial the front eye lens while hologram spins
          const dialWorld = new THREE.Vector3(0.04, 0.02, 0.22);
          bodyGroup.localToWorld(dialWorld);
          state.tipTargetPos.copy(dialWorld);
          clawTargetClosure = 0.72;
          lensDial.rotation.z = now * 2.2;
          lensProjGroup.visible = true;
          holoSpinObj.rotation.y = now * 1.8;
          holoMeshA.rotation.x = now * 1.2;
          holoMeshB.rotation.z = -now * 2.0;
        } else if (state.idleRoutine === 'juggle_toss') {
          state.juggleTimer += delta;
          const jTargetObj = state.juggleTargetEnt?.group || juggleOrbMesh;
          if (state.jugglePhase === 0) {
            // Reach & grab
            state.tipTargetPos.copy(state.juggleObjPos);
            state.lookTargetPos.copy(state.juggleObjPos);
            clawTargetClosure = 0.85;
            if (state.juggleTimer > 0.45) {
              state.jugglePhase = 1;
              state.juggleTimer = 0;
              state.juggleObjVel.set((Math.random() - 0.5) * 0.15, 2.55, (Math.random() - 0.5) * 0.15);
              clawTargetClosure = 0.05;
            }
          } else if (state.jugglePhase === 1) {
            // Airborne arc + eye tracks the tossed object!
            state.juggleObjVel.y -= 6.2 * delta;
            state.juggleObjPos.addScaledVector(state.juggleObjVel, delta);
            jTargetObj.position.copy(state.juggleObjPos);
            jTargetObj.rotation.x += delta * 4.0;
            jTargetObj.rotation.y += delta * 3.0;
            state.lookTargetPos.copy(state.juggleObjPos);
            state.tipTargetPos.copy(botRoot.position).add(new THREE.Vector3(0, -0.32, 0.22));
            if (state.juggleObjVel.y < 0 && state.juggleObjPos.y <= botRoot.position.y - 0.26) {
              state.jugglePhase = 2;
              state.juggleTimer = 0;
              clawTargetClosure = 0.90;
            }
          } else {
            // Caught back in claw!
            state.tipTargetPos.copy(botRoot.position).add(new THREE.Vector3(0.08, -0.34, 0.20));
            state.juggleObjPos.copy(clawGroup.position);
            jTargetObj.position.copy(state.juggleObjPos);
            state.lookTargetPos.copy(hp);
            if (state.juggleTimer > 0.55) {
              state.jugglePhase = 0;
              state.juggleTimer = 0;
            }
          }
        } else if (state.idleRoutine === 'curious_patrol') {
          const ents = (APP.entities || []).filter((e) => e && e !== boardEnt && !e.isHeld && e.category !== 'world_fx');
          if (ents.length) {
            const pick = ents[Math.floor(now * 0.2) % ents.length];
            const info = computeObjCenterAndRadius(pick.group);
            if (info.center.distanceTo(hp) < 2.8) {
              state.worldAnchor.lerp(info.center.clone().add(new THREE.Vector3(0.32, 0.42, 0.28)), delta * 0.8);
              state.lookTargetPos.copy(info.center);
              state.tipTargetPos.copy(info.center).add(new THREE.Vector3(0.12, 0.14, 0.12));
            }
          } else {
            state.lookTargetPos.copy(hp);
            state.tipTargetPos.copy(botRoot.position).add(new THREE.Vector3(Math.sin(now * 1.4) * 0.18, -0.44, 0.08));
          }
        } else {
          if (state.behavior === 'autonomous') state.targetYawOffset = 0;
          state.lookTargetPos.copy(hp);
          state.tipTargetPos.copy(botRoot.position).add(new THREE.Vector3(Math.sin(now * 1.25) * 0.14, -0.46 + Math.cos(now * 1.55) * 0.05, -0.08 + Math.cos(now * 0.9) * 0.10));
          clawTargetClosure = 0.15;
        }
      }

      if (state.behavior !== 'dance') {
        state.hoverTargetPos.copy(state.worldAnchor).add(new THREE.Vector3(Math.sin(now * 0.7) * 0.055, Math.sin(now * 1.7) * 0.035, Math.cos(now * 0.6) * 0.045));
      }
    } else if (state.mode === 'pointing') {
      const tgt = state.targetObj ? computeObjCenterAndRadius(state.targetObj).center : state.pointTargetPos;
      const dirToTgt = tmpVecA.subVectors(tgt, botRoot.position).normalize();
      state.tipTargetPos.copy(botRoot.position).addScaledVector(dirToTgt, 0.62);
      state.lookTargetPos.copy(tgt);
      state.pointTimer -= delta;
      if (state.pointTimer <= 0) tentacle_release();
    } else if (state.mode === 'grabbing') {
      if (!state.targetObj || !state.targetObj.parent) {
        tentacle_release();
      } else {
        const info = computeObjCenterAndRadius(state.targetObj);
        state.hoverTargetPos.copy(info.center).add(new THREE.Vector3(0.28, 0.46, 0.22));
        state.tipTargetPos.copy(info.center);
        state.lookTargetPos.copy(info.center);
        if (state.currentTipPos.distanceTo(info.center) < 0.16) {
          state.grabProgress += delta; clawTargetClosure = 0.88;
          if (state.grabProgress > 0.22) {
            state.heldObj = state.targetObj; state.heldEnt = getEntityRecord(state.heldObj); state.heldRadius = info.radius;
            if (state.heldEnt) state.heldEnt.heldByBot = true;
            state.heldObj.userData = state.heldObj.userData || {}; state.heldObj.userData.heldByBot = true;
            state.mode = 'holding'; setEmotion('happy');
          }
        }
      }
    } else if (state.mode === 'holding') {
      if (!state.heldObj || !state.heldObj.parent) tentacle_release();
      else {
        state.hoverTargetPos.copy(state.worldAnchor).add(new THREE.Vector3(0, Math.sin(now * 1.5) * 0.025, 0));
        state.tipTargetPos.copy(botRoot.position).add(new THREE.Vector3(-0.10, -0.44, 0.22));
        state.lookTargetPos.copy(hp);
      }
    } else if (state.mode === 'handing') {
      if (!state.heldObj || !state.heldObj.parent) tentacle_release();
      else {
        state.hoverTargetPos.copy(hp).addScaledVector(flatFwd, 0.72);
        state.hoverTargetPos.y = Math.max(0.92, hp.y + 0.02);
        let handAnchor = null, handGrabbing = false;
        const hands = APP.hands || {};
        for (const k of Object.keys(hands)) {
          const h = hands[k]; if (!h || !h.active) continue;
          const pos = h.isHand ? h.fingerTips?.index : h.pos;
          if (pos) {
            if (!handAnchor || pos.distanceTo(state.currentTipPos) < handAnchor.distanceTo(state.currentTipPos)) handAnchor = pos;
            if ((h.pinching || h.trigger || h.gripping) && pos.distanceTo(state.currentTipPos) < 0.22) handGrabbing = true;
          }
        }
        if (handAnchor && handAnchor.distanceTo(hp) < 1.1) state.tipTargetPos.copy(handAnchor).addScaledVector(flatFwd, -0.05);
        else { state.tipTargetPos.copy(hp).addScaledVector(flatFwd, 0.42); state.tipTargetPos.y = Math.max(0.75, hp.y - 0.20); }
        state.lookTargetPos.copy(hp);
        if (handGrabbing || state.heldEnt?.grabbedByHand) tentacle_release();
      }
    } else if (state.mode === 'showing') {
      if (!state.targetObj || !state.targetObj.parent) tentacle_release();
      else {
        const info = computeObjCenterAndRadius(state.targetObj);
        state.hoverTargetPos.copy(info.center).add(new THREE.Vector3(0.36, 0.40 + Math.sin(now * 2.0) * 0.03, 0.22));
        state.tipTargetPos.copy(info.center).add(new THREE.Vector3(0.18, 0.12, 0.12));
        state.worldAnchor.copy(state.hoverTargetPos);
        state.lookTargetPos.copy(info.center);
        state.showTimer -= delta;
        if (state.showTimer <= 0) tentacle_release();
      }
    }

    const stiffness = 28.0, damping = 8.4;
    tmpVecA.subVectors(state.hoverTargetPos, botRoot.position).multiplyScalar(stiffness);
    state.velocity.addScaledVector(tmpVecA, delta);
    state.velocity.multiplyScalar(Math.exp(-damping * delta));
    botRoot.position.addScaledVector(state.velocity, delta);

    bodyGroup.lookAt(state.lookTargetPos);
    state.yawOffset += (state.targetYawOffset - state.yawOffset) * Math.min(1, delta * 6.5);
    bodyGroup.rotateY(state.yawOffset);
    if (state.nodPhase > 0) {
      state.nodPhase = Math.max(0, state.nodPhase - delta * 12.0);
      bodyGroup.rotateX(Math.sin(state.nodPhase) * 0.28);
    }

    const speed = state.velocity.length();
    bodyGroup.rotateZ(THREE.MathUtils.clamp(-state.velocity.x * 0.22, -0.35, 0.35));
    bodyGroup.rotateX(THREE.MathUtils.clamp(speed * 0.10, -0.20, 0.20));

    // Emotional expressions (eyes color + squint/bounce + wing flare)
    const wingSpeed = state.emotion === 'excited' || state.emotion === 'playful' ? 9.0 : 4.5;
    const wingFlutter = Math.sin(now * wingSpeed) * 0.07 + Math.min(0.28, speed * 0.25);
    wingL.rotation.z = wingFlutter;
    wingR.rotation.z = -wingFlutter;

    blinkTimer -= delta;
    let eyeScaleY = state.emotion === 'happy' ? 0.65 : state.emotion === 'focused' ? 0.78 : 1.0;
    let eyeScaleX = state.emotion === 'happy' ? 1.18 : 1.0;
    if (blinkTimer <= 0) {
      if (blinkTimer > -0.11) eyeScaleY = 0.10;
      else blinkTimer = 2.4 + Math.random() * 3.8;
    }
    eyeL.scale.set(eyeScaleX, eyeScaleY, 1);
    eyeR.scale.set(eyeScaleX, eyeScaleY, 1);
    eyeMat.color.setHex(
      state.emotion === 'excited' ? 0xfbbf24 :
      state.emotion === 'happy' ? 0x00ff9d :
      state.emotion === 'playful' ? 0xf472b6 : 0x5efce8
    );

    reactorCore.rotation.y = now * 1.4;
    reactorCore.rotation.x = now * 0.8;

    updateTentacleIK(delta, now);
    stepCapePhysics(delta, now);

    if (boardState.visible && boardRoot.visible) {
      if (!boardDetached && !boardEnt.isHeld) {
        const targetBoardPos = botRoot.position.clone().addScaledVector(right, -0.62).add(new THREE.Vector3(0, 0.04, 0.12));
        boardRoot.position.lerp(targetBoardPos, 1 - Math.exp(-8 * delta));
        boardRoot.lookAt(hp.x, boardRoot.position.y, hp.z);

        boomArm.visible = true;
        const armStart = botRoot.position.clone().addScaledVector(right, -0.14);
        const armEnd = boardRoot.position.clone().addScaledVector(right, BOARD_W * 0.48);
        const armLen = Math.max(0.05, armStart.distanceTo(armEnd));
        boomArm.position.lerpVectors(armStart, armEnd, 0.5);
        boomArm.scale.set(1, armLen, 1);
        boomArm.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), armEnd.sub(armStart).normalize());
      } else {
        boomArm.visible = false;
      }
      updateBoardPenDrawing();
    } else {
      boomArm.visible = false;
    }

    if (state.heldObj && (state.mode === 'holding' || state.mode === 'handing')) {
      const forwardDir = new THREE.Vector3(0, 0, 1).applyQuaternion(clawGroup.quaternion);
      tmpVecA.copy(clawGroup.position).addScaledVector(forwardDir, 0.045 + state.heldRadius * 0.35);
      state.heldObj.position.lerp(tmpVecA, 1 - Math.exp(-16 * delta));
      if (state.heldEnt?.physics?.vel) state.heldEnt.physics.vel.set(0, 0, 0);
    }

    const targetEnergy = (APP.botSpeaking || APP.ttsSpeaking)
      ? 0.55 + Math.abs(Math.sin(now * 16.0)) * 0.45 : (APP.micLevel || 0);
    audioEnergy += (targetEnergy - audioEnergy) * Math.min(1, delta * 12);
    plasmaMat.emissiveIntensity = 1.1 + audioEnergy * 2.4;
    botLight.intensity = 0.6 + audioEnergy * 1.2;
  }

  function setEnabled(val) {
    botEnabled = !!val;
    botRoot.visible = botEnabled;
    tentacleGroup.visible = botEnabled;
    capeMesh.visible = botEnabled && capeEnabled;
    if (!botEnabled) { tentacle_release(); hideBoard(); disconnectLive(); }
    else if (capeEnabled) snapCapeToRest();
    APP.emit?.('bot:state_changed', botEnabled);
    APP.toolbar?.syncState?.();
    return botEnabled;
  }

  function buildSceneMemorySnapshot() {
    const ents = (APP.entities || []).filter((e) => e && e !== boardEnt);
    if (!ents.length) return 'Scene is currently empty.';
    const lines = [];
    for (let i = Math.max(0, ents.length - 12); i < ents.length; i++) {
      const e = ents[i];
      if (!e || !e.group) continue;
      e.group.updateMatrixWorld(true);
      const box = new THREE.Box3().setFromObject(e.group);
      const size = new THREE.Vector3(); box.getSize(size);
      const partNames = [];
      e.group.traverse((c) => { if (c !== e.group && c.name) partNames.push(c.name); });
      const wp = e.group.position;
      lines.push(
        `• id="${e.id}" name="${e.name}" pos=[${wp.x.toFixed(2)},${wp.y.toFixed(2)},${wp.z.toFixed(2)}] ` +
        `localSize=[w:${size.x.toFixed(2)},h:${size.y.toFixed(2)},d:${size.z.toFixed(2)}] topY=${size.y.toFixed(2)} ` +
        `parts=[${partNames.slice(0, 10).join(',')}]` + (e.meta?.summary ? ` (${e.meta.summary})` : '')
      );
    }
    if (APP.lastEntityId) lines.push(`Most recently referenced entity: "${APP.lastEntityId}".`);
    return lines.join('\n');
  }

  function modifyExistingEntity(entityId, action, jsCode, summaryNote) {
    const ent = getEntityRecord(entityId || APP.lastEntityId || 'latest');
    if (!ent || !ent.group) return { ok: false, error: `Entity "${entityId}" not found.` };
    const targetGroup = ent.group;
    targetGroup.updateMatrixWorld(true);

    const localBox = new THREE.Box3();
    const invRoot = targetGroup.matrixWorld.clone().invert();
    targetGroup.traverse((c) => {
      if (c.isMesh && c.geometry) {
        c.geometry.computeBoundingBox();
        const b = c.geometry.boundingBox.clone();
        b.applyMatrix4(invRoot.clone().multiply(c.matrixWorld));
        localBox.union(b);
      }
    });
    const size = new THREE.Vector3(), center = new THREE.Vector3();
    if (!localBox.isEmpty()) { localBox.getSize(size); localBox.getCenter(center); }

    const info = {
      box: localBox, size, center,
      minY: isFinite(localBox.min.y) ? localBox.min.y : 0,
      maxY: isFinite(localBox.max.y) ? localBox.max.y : size.y,
      topY: isFinite(localBox.max.y) ? localBox.max.y : size.y,
      findPart: (q) => {
        const low = String(q || '').toLowerCase();
        let found = null;
        targetGroup.traverse((c) => { if (!found && c !== targetGroup && String(c.name || '').toLowerCase().includes(low)) found = c; });
        return found;
      },
      scatterTufts: (tuftMat, tuftCount = 180, tuftHeight = 0.032, tuftRadius = 0.008) => {
        const meshes = [];
        targetGroup.traverse((c) => { if (c.isMesh && c.geometry?.attributes?.position) meshes.push(c); });
        if (!meshes.length) return 0;
        const coneGeo = new THREE.ConeGeometry(tuftRadius, tuftHeight, 5);
        coneGeo.translate(0, tuftHeight * 0.5, 0);
        const inst = new THREE.InstancedMesh(coneGeo, tuftMat, tuftCount);
        const dummy = new THREE.Object3D();
        const vPos = new THREE.Vector3(), vNorm = new THREE.Vector3(), qAlign = new THREE.Quaternion();
        const up = new THREE.Vector3(0, 1, 0);
        let placed = 0;
        for (let i = 0; i < tuftCount; i++) {
          const m = meshes[i % meshes.length];
          const posA = m.geometry.attributes.position;
          const normA = m.geometry.attributes.normal;
          const vi = Math.floor(Math.random() * posA.count);
          vPos.fromBufferAttribute(posA, vi);
          m.updateMatrixWorld(true);
          const relMat = invRoot.clone().multiply(m.matrixWorld);
          vPos.applyMatrix4(relMat);
          if (normA) {
            vNorm.fromBufferAttribute(normA, vi).transformDirection(relMat).normalize();
          } else {
            vNorm.copy(vPos).sub(center).normalize();
          }
          qAlign.setFromUnitVectors(up, vNorm);
          dummy.position.copy(vPos);
          dummy.quaternion.copy(qAlign);
          dummy.rotateX((Math.random() - 0.5) * 0.45);
          dummy.rotateZ((Math.random() - 0.5) * 0.45);
          const sc = 0.7 + Math.random() * 0.65;
          dummy.scale.set(sc, sc, sc);
          dummy.updateMatrix();
          inst.setMatrixAt(placed++, dummy.matrix);
        }
        inst.instanceMatrix.needsUpdate = true;
        inst.name = 'fur_tufts';
        targetGroup.add(inst);
        return placed;
      }
    };

    try {
      const V3 = (x = 0, y = 0, z = 0) => new THREE.Vector3(x, y, z);
      const clean = String(jsCode || '').trim().replace(/^```(?:javascript|js)?/i, '').replace(/```$/i, '').trim();
      const fn = new Function('APP', 'targetGroup', 'THREE', 'V3', 'info', clean);
      fn(APP, targetGroup, THREE, V3, info);

      targetGroup.traverse((c) => { if (c.isMesh) { c.castShadow = true; c.receiveShadow = true; c.userData.entity = ent; } });
      if (summaryNote) {
        ent.meta = ent.meta || {};
        ent.meta.summary = ((ent.meta.summary || '') + '; ' + summaryNote).slice(-180);
      }
      APP.lastEntityId = ent.id;
      APP.visual?.applyDissolveShader?.(targetGroup, 'materialize');
      tentacle_show(ent);
      return { ok: true, id: ent.id, name: ent.name, message: `Modified ${ent.name} (${action}).` };
    } catch (err) {
      return { ok: false, error: `${err.name}: ${err.message}` };
    }
  }

  function runPhysicsSandboxPreset() {
    const hp = APP.headPos || new THREE.Vector3(0, 1.5, 0);
    const hf = APP.headFwd || new THREE.Vector3(0, 0, -1);
    const flat = new THREE.Vector3(hf.x, 0, hf.z).normalize();
    if (flat.lengthSq() < 0.01) flat.set(0, 0, -1);
    const center = hp.clone().addScaledVector(flat, 1.25);

    const colors = [0x00f0b4, 0x38bdf8, 0xf472b6, 0xfbbf24, 0xa78bfa, 0xfb7185];
    const bodies = [];
    for (let i = 0; i < 8; i++) {
      const r = 0.065 + Math.random() * 0.04;
      const g = new THREE.Group();
      const mesh = new THREE.Mesh(
        i % 3 === 0 ? new THREE.BoxGeometry(r * 1.8, r * 1.8, r * 1.8) : new THREE.SphereGeometry(r, 20, 16),
        new THREE.MeshStandardMaterial({ color: colors[i % colors.length], metalness: 0.72, roughness: 0.22 })
      );
      g.add(mesh);
      g.position.set(center.x + (Math.random() - 0.5) * 0.65, 1.1 + i * 0.18, center.z + (Math.random() - 0.5) * 0.65);
      const ent = APP.registerEntity(g, {
        name: 'Physics Body ' + (i + 1),
        category: 'physics_sim',
        holdable: true,
        hasGravity: true,
        restitution: 0.78,
        velocity: new THREE.Vector3((Math.random() - 0.5) * 0.8, 0.4, (Math.random() - 0.5) * 0.8),
        physics: { gravity: true, bounce: 0.78, radius: r }
      });
      bodies.push(ent);
    }

    // Add elastic body-to-body & hand-punch collision manager
    const simMgr = new THREE.Group();
    simMgr.name = 'Physics Simulation Solver';
    const tmpDiff = new THREE.Vector3();
    APP.registerEntity(simMgr, {
      name: 'Physics Collision Solver',
      category: 'world_fx',
      grabbable: false,
      physics: { enabled: false },
      onUpdate: function () {
        const active = (APP.entities || []).filter((e) => e && e.category === 'physics_sim' && e.group?.parent);
        if (!active.length) return;
        for (let i = 0; i < active.length; i++) {
          const a = active[i], ra = a.physics.radius || 0.08;
          // Hand poke/punch push!
          for (const h of Object.values(APP.hands || {})) {
            if (!h || !h.active) continue;
            const hpPos = h.isHand ? h.fingerTips?.index : h.pos;
            if (!hpPos) continue;
            tmpDiff.subVectors(a.group.position, hpPos);
            const d = tmpDiff.length();
            if (d < ra + 0.09 && d > 1e-4) {
              tmpDiff.normalize();
              a.group.position.addScaledVector(tmpDiff, (ra + 0.09 - d));
              a.physics.vel.addScaledVector(tmpDiff, 2.2);
            }
          }
          // Sphere-sphere elastic collisions
          for (let j = i + 1; j < active.length; j++) {
            const b = active[j], rb = b.physics.radius || 0.08;
            const minD = ra + rb;
            tmpDiff.subVectors(b.group.position, a.group.position);
            const d = tmpDiff.length();
            if (d < minD && d > 1e-5) {
              tmpDiff.divideScalar(d);
              const overlap = (minD - d) * 0.5;
              a.group.position.addScaledVector(tmpDiff, -overlap);
              b.group.position.addScaledVector(tmpDiff, overlap);
              const rv = tmpVecA.subVectors(b.physics.vel, a.physics.vel).dot(tmpDiff);
              if (rv < 0) {
                const jImp = -(1 + 0.78) * rv * 0.5;
                a.physics.vel.addScaledVector(tmpDiff, -jImp);
                b.physics.vel.addScaledVector(tmpDiff, jImp);
              }
            }
          }
        }
      }
    });
    return { ok: true, count: bodies.length, message: 'Spawned interactive colliding Physics Simulation (grab or punch the bodies!).' };
  }

  async function executeToolCall(name, args) {
    const a = args || {};
    if (name === 'control_bot') {
      if (a.emotion) setEmotion(a.emotion);
      return controlBotAction(a.action, a);
    }
    if (name === 'tentacle_point') return tentacle_point(a.target, a.coords);
    if (name === 'create_procedural_object') {
      if (typeof APP.createProcedural !== 'function') return { ok: false, error: 'APP.createProcedural missing.' };
      const res = APP.createProcedural(a.js_code, { name: a.name || 'AI Creation', holdable: a.holdable !== false });
      if (res?.ok && res.entity) { APP.lastEntityId = res.entity.id; return { ok: true, id: res.entity.id, name: res.entity.name }; }
      return res;
    }
    if (name === 'modify_entity') return modifyExistingEntity(a.entity_id, a.action || 'attach', a.js_code, a.summary || a.action);
    if (name === 'run_world_fx') {
      const p = String(a.preset || '').toLowerCase();
      if (p === 'physics' || (!a.js_code && /\b(physics|bounce|collid)\b/i.test(a.name || ''))) {
        return runPhysicsSandboxPreset();
      }
      if (APP.ai?.executeWorldFX) return APP.ai.executeWorldFX(a);
      if (p === 'water' && APP.fx?.water) return APP.fx.water(a);
      if (p === 'rain' && APP.fx?.rain) return APP.fx.rain(a);
      if (p === 'snow' && APP.fx?.snow) return APP.fx.snow(a);
      if (p === 'lava' && APP.fx?.lavaFloor) return APP.fx.lavaFloor(a);
      return { ok: false, error: 'World FX engine not available.' };
    }
    if (name === 'set_cape') {
      if (typeof a.enabled === 'boolean') setCapeEnabled(a.enabled);
      if (a.texture) return await setCapeTexture(a.texture);
      return { ok: true, capeEnabled, message: 'Geeny cape is now ' + (capeEnabled ? 'ON' : 'OFF') + '.' };
    }
    if (name === 'show_holo_board') {
      const m = String(a.mode || '').toLowerCase();
      if (m === 'wikipedia') {
        const wiki = await fetchWikipediaData(a.topic || a.title);
        if (!wiki) return { ok: false, error: `Could not fetch Wikipedia article for "${a.topic || a.title}".` };
        showBoard({ title: 'Wikipedia: ' + wiki.title, mode: 'wikipedia', pages: wiki.pages, imageObj: wiki.imageObj, caption: wiki.summary });
        return { ok: true, title: wiki.title, summary: wiki.summary.slice(0, 320), pages: wiki.pages.length };
      }
      if (m === 'photo' || m === 'image') {
        const q = a.topic || a.image_name || a.title;
        const img = await fetchOnlinePhoto(q);
        if (!img) return { ok: false, error: `Could not find photo for "${q}".` };
        return showBoard({ title: a.title || q, mode: 'image', imageObj: img, caption: 'Photo: ' + q });
      }
      if (m === 'tutorial') {
        const refImg = a.photo_query ? await fetchOnlinePhoto(a.photo_query) : await fetchOnlinePhoto(a.title);
        if ((!Array.isArray(a.steps) || !a.steps.length) && /cat|قط/i.test(a.title || '')) return showBoard(getAdvancedCatTutorial(refImg));
        return showBoard(Object.assign({}, a, { imageObj: refImg }));
      }
      return showBoard(a);
    }
    if (name === 'hide_holo_board') return hideBoard();
    if (name === 'tentacle_grab') return tentacle_grab(a.target || 'latest');
    if (name === 'tentacle_hand_to_user') return tentacle_hand_to_user(a.target || 'latest');
    if (name === 'tentacle_show') return tentacle_show(a.target || 'latest');
    if (name === 'tentacle_release') return tentacle_release();
    if (name === 'open_portal') {
      if (typeof APP.openPortal === 'function') return await APP.openPortal(a.world || '');
      return await APP.runInstructionMatch?.('portal ' + (a.world || '')) || { ok: false, error: 'Portal unavailable.' };
    }
    if (name === 'set_environment') {
      return APP.visual?.setEnvironment ? { ok: true, environment: APP.visual.setEnvironment(a.environment || 'default') } : { ok: false, error: 'Visual unavailable.' };
    }
    if (name === 'run_instruction') return await APP.runInstructionMatch?.(a.command || '') || { ok: false, error: 'No matching instruction.' };
    if (name === 'spawn_repo_asset') {
      const spawner = APP.spawnAsset || APP.spawnAssetByName;
      return spawner ? await spawner(a.name || '') : { ok: false, error: 'Spawner unavailable.' };
    }
    if (name === 'list_txt_files') {
      const files = Object.entries(APP.modules || {}).map(([fn, txt]) => {
        const abs = APP.parseModuleText ? APP.parseModuleText(fn, txt) : [];
        return { file: fn, lines: txt.split('\n').length, abilities: abs.map((x) => x.id) };
      });
      return { ok: true, files, images: APP.importedImages ? Array.from(APP.importedImages.keys()) : [], glbs: APP.importedGLBs ? Array.from(APP.importedGLBs.keys()) : [] };
    }
    if (name === 'read_txt_file') {
      const fn = String(a.filename || '').trim(), raw = APP.modules?.[fn];
      if (!raw) return { ok: false, error: `File "${fn}" not loaded.` };
      const lines = raw.split('\n');
      const s = Math.max(1, Number(a.start_line || 1)) - 1, e = Math.min(lines.length, Number(a.end_line || lines.length));
      return { ok: true, filename: fn, totalLines: lines.length, startLine: s + 1, endLine: e, content: lines.slice(s, e).join('\n') };
    }
    if (name === 'edit_txt_file') {
      const fn = String(a.filename || '').trim();
      if (!fn.endsWith('.txt')) return { ok: false, error: 'Filename must end with .txt' };
      const prev = APP.modules?.[fn] || '';
      const next = APP.toolbar?.applyAIModuleOutput ? APP.toolbar.applyAIModuleOutput(a.patch_or_content, prev, fn) : String(a.patch_or_content || '').trim();
      const check = APP.toolbar?.validateModuleSyntax ? APP.toolbar.validateModuleSyntax(fn, next) : { ok: true };
      if (!check.ok) return { ok: false, error: check.error };
      if (!txtSnapshots.has(fn)) txtSnapshots.set(fn, prev);
      APP.toolbar?.stageEditInEditor?.(fn, prev, next, `Geeny edited ${fn}.`);
      if (a.run_sandbox !== false) await APP.reloadModuleText?.(fn, next);
      return { ok: true, filename: fn, abilities: check.count, message: `Updated ${fn} live in Sandbox.` };
    }
    if (name === 'rollback_txt_file') {
      const fn = String(a.filename || '').trim();
      if (!txtSnapshots.has(fn)) return { ok: false, error: `No snapshot for "${fn}".` };
      const prev = txtSnapshots.get(fn);
      await APP.reloadModuleText?.(fn, prev);
      APP.toolbar?.stageEditInEditor?.(fn, prev, prev, `Rolled back ${fn}.`);
      txtSnapshots.delete(fn);
      return { ok: true, message: `Rolled back ${fn}.` };
    }
    return { ok: false, error: 'Unknown tool: ' + name };
  }

  function ensureUnlockedAudioContext() {
    if (!audioCtx) { const AC = window.AudioContext || window.webkitAudioContext; audioCtx = new AC(); }
    if (audioCtx.state === 'suspended') audioCtx.resume().catch(() => {});
    try {
      const buf = audioCtx.createBuffer(1, 128, audioCtx.sampleRate);
      const src = audioCtx.createBufferSource(); src.buffer = buf; src.connect(audioCtx.destination); src.start(0);
    } catch (_) {}
    return audioCtx;
  }

  function floatTo16BitPCMBase64(float32, inputSampleRate) {
    const ratio = inputSampleRate / 16000;
    const outLen = Math.max(1, Math.floor(float32.length / ratio));
    const buf = new ArrayBuffer(outLen * 2), view = new DataView(buf);
    let sumSq = 0;
    for (let i = 0; i < outLen; i++) {
      const pos = i * ratio, i0 = Math.floor(pos), i1 = Math.min(float32.length - 1, i0 + 1), frac = pos - i0;
      const s = Math.max(-1, Math.min(1, float32[i0] * (1 - frac) + float32[i1] * frac));
      sumSq += s * s;
      view.setInt16(i * 2, s < 0 ? s * 0x8000 : s * 0x7FFF, true);
    }
    APP.micLevel = Math.min(1, Math.sqrt(sumSq / outLen) * 5.0);
    const bytes = new Uint8Array(buf); let bin = '';
    for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
    return btoa(bin);
  }

  function stopLiveAudioPlayback() {
    activePlaySources.forEach((src) => { try { src.stop(); } catch (_) {} });
    activePlaySources.clear();
    nextPlayTime = audioCtx ? audioCtx.currentTime : 0;
    APP.botSpeaking = false;
  }

  function playPCM24kBase64(b64) {
    if (!b64) return;
    const ctx = ensureUnlockedAudioContext(), bin = atob(b64), sampleCount = Math.floor(bin.length / 2);
    if (sampleCount <= 0) return;
    const floatBuf = new Float32Array(sampleCount);
    for (let i = 0; i < sampleCount; i++) {
      let val = (bin.charCodeAt(i * 2 + 1) << 8) | bin.charCodeAt(i * 2);
      if (val >= 0x8000) val -= 0x10000;
      floatBuf[i] = val / 32768;
    }
    const audioBuf = ctx.createBuffer(1, sampleCount, 24000);
    audioBuf.getChannelData(0).set(floatBuf);
    const src = ctx.createBufferSource(); src.buffer = audioBuf; src.connect(ctx.destination);
    const now = ctx.currentTime;
    if (nextPlayTime < now) nextPlayTime = now + 0.025;
    src.start(nextPlayTime); nextPlayTime += audioBuf.duration;
    APP.botSpeaking = true; activePlaySources.add(src);
    src.onended = function () { activePlaySources.delete(src); if (activePlaySources.size === 0) APP.botSpeaking = false; };
  }

  async function startMicStreamingForLive() {
    if (!navigator.mediaDevices?.getUserMedia) return;
    const ctx = ensureUnlockedAudioContext(); stopMicStreamingForLive();
    micStream = await navigator.mediaDevices.getUserMedia({ audio: { echoCancellation: true, noiseSuppression: true, autoGainControl: true, channelCount: 1 } });
    const source = ctx.createMediaStreamSource(micStream);
    const muteGain = ctx.createGain(); muteGain.gain.value = 0; muteGain.connect(ctx.destination);

    let usedWorklet = false;
    if (ctx.audioWorklet && typeof Blob !== 'undefined') {
      try {
        const workletCode = 'class PCM16Cap extends AudioWorkletProcessor { process(inputs) { const ch = inputs[0]?.[0]; if (ch) this.port.postMessage(ch); return true; } }\nregisterProcessor("wxr-geeny-pcm16-v4", PCM16Cap);';
        const url = URL.createObjectURL(new Blob([workletCode], { type: 'application/javascript' }));
        await ctx.audioWorklet.addModule(url); URL.revokeObjectURL(url);
        micWorkletNode = new AudioWorkletNode(ctx, 'wxr-geeny-pcm16-v4');
        micWorkletNode.port.onmessage = function (ev) {
          if (!liveWS || liveWS.readyState !== 1 || APP.botSpeaking) return;
          liveWS.send(JSON.stringify({ realtimeInput: { audio: { mimeType: 'audio/pcm;rate=16000', data: floatTo16BitPCMBase64(ev.data, ctx.sampleRate) } } }));
        };
        source.connect(micWorkletNode); micWorkletNode.connect(muteGain); usedWorklet = true;
      } catch (_) {}
    }
    if (!usedWorklet) {
      micScriptNode = ctx.createScriptProcessor(2048, 1, 1);
      micScriptNode.onaudioprocess = function (e) {
        if (!liveWS || liveWS.readyState !== 1 || APP.botSpeaking) return;
        liveWS.send(JSON.stringify({ realtimeInput: { audio: { mimeType: 'audio/pcm;rate=16000', data: floatTo16BitPCMBase64(e.inputBuffer.getChannelData(0), ctx.sampleRate) } } }));
      };
      source.connect(micScriptNode); micScriptNode.connect(muteGain);
    }
  }

  function stopMicStreamingForLive() {
    if (micWorkletNode) { try { micWorkletNode.disconnect(); } catch (_) {} micWorkletNode = null; }
    if (micScriptNode) { try { micScriptNode.disconnect(); } catch (_) {} micScriptNode = null; }
    if (micStream) { micStream.getTracks().forEach((t) => { try { t.stop(); } catch (_) {} }); micStream = null; }
    APP.micLevel = 0;
  }

  async function connectLive(enableMicStream) {
    ensureUnlockedAudioContext();
    if (!APP.ai) return { ok: false, error: 'APP.ai unavailable.' };
    const model = APP.ai.getSelectedModel();
    if (!model.supportsLiveWS) return { ok: false, error: `"${model.id}" is not a Live WebSocket model.` };
    const apiKey = APP.ai.getKey('gemini');
    if (!apiKey) return { ok: false, error: 'Missing Google AI Studio key (AIza...).' };
    disconnectLive();
    const wsUrl = 'wss://generativelanguage.googleapis.com/ws/google.ai.generativelanguage.v1beta.GenerativeService.BidiGenerateContent?key=' + encodeURIComponent(apiKey);

    return new Promise((resolve) => {
      try { liveWS = new WebSocket(wsUrl); } catch (e) { resolve({ ok: false, error: e.message }); return; }
      let settled = false;
      liveWS.onopen = function () {
        liveWS.send(JSON.stringify({
          setup: {
            model: 'models/' + model.id,
            generationConfig: { responseModalities: ['AUDIO'], speechConfig: { voiceConfig: { prebuiltVoiceConfig: { voiceName: 'Aoede' } } } },
            systemInstruction: { parts: [{ text: buildSystemPrompt(APP.instWeight <= 0, APP.instWeight) }] },
            tools: [{ functionDeclarations: BOT_TOOLS }],
            inputAudioTranscription: {}, outputAudioTranscription: {}
          }
        }));
      };
      liveWS.onmessage = async function (evt) {
        let msg;
        try { msg = JSON.parse(typeof evt.data === 'string' ? evt.data : await evt.data.text()); } catch (_) { return; }
        if (msg.setupComplete) {
          liveConnected = true; APP.ui?.setStatus?.(`Geeny Live (${model.id})`, 'ok'); APP.ui?.toast?.(`Geeny Live Voice connected`);
          if (enableMicStream) startMicStreamingForLive().catch((err) => APP.ui?.toast?.('Live Mic Error: ' + err.message));
          visionTimer = setInterval(() => {
            if (!liveWS || liveWS.readyState !== 1) return;
            const b64 = APP.captureVisionFrame?.() || APP.latestVisionB64;
            if (b64) liveWS.send(JSON.stringify({ realtimeInput: { video: { mimeType: 'image/jpeg', data: String(b64).replace(/^data:image\/\w+;base64,/, '') } } }));
          }, 1500);
          if (!settled) { settled = true; resolve({ ok: true }); }
          return;
        }
        if (msg.serverContent) {
          if (msg.serverContent.interrupted) stopLiveAudioPlayback();
          const turn = msg.serverContent.modelTurn;
          if (turn?.parts) {
            for (const part of turn.parts) {
              if (part.inlineData?.data && (!part.inlineData.mimeType || part.inlineData.mimeType.includes('pcm') || part.inlineData.mimeType.startsWith('audio/'))) playPCM24kBase64(part.inlineData.data);
              if (part.text) APP.ui?.toast?.('Geeny: ' + part.text.slice(0, 140));
            }
          }
          if (msg.serverContent.outputTranscription?.text) APP.ui?.toast?.('Geeny: ' + msg.serverContent.outputTranscription.text.slice(0, 140));
        }
        if (msg.toolCall?.functionCalls) {
          const responses = [];
          for (const fc of msg.toolCall.functionCalls) {
            const out = await executeToolCall(fc.name, fc.args || {});
            responses.push({ id: fc.id, name: fc.name, response: out || { ok: true } });
          }
          if (liveWS && liveWS.readyState === 1) liveWS.send(JSON.stringify({ toolResponse: { functionResponses: responses } }));
        }
      };
      liveWS.onerror = function () { if (!settled) { settled = true; resolve({ ok: false, error: `Gemini Live connection failed.` }); } };
      liveWS.onclose = function (ev) {
        liveConnected = false; stopMicStreamingForLive();
        if (visionTimer) { clearInterval(visionTimer); visionTimer = null; }
        if (!settled) { settled = true; resolve({ ok: false, error: `Gemini Live closed (${ev.code})` }); }
      };
    });
  }

  function disconnectLive() {
    liveConnected = false; stopMicStreamingForLive(); stopLiveAudioPlayback();
    if (visionTimer) { clearInterval(visionTimer); visionTimer = null; }
    if (liveWS) { try { liveWS.close(); } catch (_) {} liveWS = null; }
