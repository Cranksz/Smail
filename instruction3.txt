  function hash2i(ix, iy) {
    let n = Math.imul(ix, 374761393) + Math.imul(iy, 668265263);
    n = Math.imul(n ^ (n >>> 13), 1274126177);
    return ((n ^ (n >>> 16)) >>> 0) / 4294967295;
  }

  function tileValueNoise(u, v, period, seed) {
    const x = u * period, y = v * period;
    const x0 = Math.floor(x), y0 = Math.floor(y);
    const fx = x - x0, fy = y - y0;
    const sx = fx * fx * (3 - 2 * fx), sy = fy * fy * (3 - 2 * fy);
    const ix0 = ((x0 % period) + period) % period;
    const iy0 = ((y0 % period) + period) % period;
    const ix1 = (ix0 + 1) % period;
    const iy1 = (iy0 + 1) % period;
    const a = hash2i(ix0 + seed, iy0 + seed * 3);
    const b = hash2i(ix1 + seed, iy0 + seed * 3);
    const c = hash2i(ix0 + seed, iy1 + seed * 3);
    const d = hash2i(ix1 + seed, iy1 + seed * 3);
    return (a + (b - a) * sx) + ((c + (d - c) * sx) - (a + (b - a) * sx)) * sy;
  }

  function fbmTile(u, v, seed) {
    let val = 0, amp = 0.5, freq = 6;
    for (let o = 0; o < 4; o++) {
      val += tileValueNoise(u, v, freq, seed + o * 97) * amp;
      freq *= 2; amp *= 0.5;
    }
    return val;
  }

  function wrapStroke(ctx, x1, y1, cx, cy, x2, y2, S) {
    for (let ox = -S; ox <= S; ox += S) {
      for (let oy = -S; oy <= S; oy += S) {
        if (x1 + ox < -75 || x1 + ox > S + 75 || y1 + oy < -75 || y1 + oy > S + 75) continue;
        ctx.beginPath();
        ctx.moveTo(x1 + ox, y1 + oy);
        ctx.quadraticCurveTo(cx + ox, cy + oy, x2 + ox, y2 + oy);
        ctx.stroke();
      }
    }
  }

  function calibrateProceduralColor(col, archetype) {
    if (!cfg.calibrateProceduralColors || !col || archetype === 'glass') return col;
    const hsl = {};
    col.getHSL(hsl);
    if (hsl.s > 0.62) hsl.s = 0.62 + (hsl.s - 0.62) * 0.25;
    if (hsl.l > 0.86) hsl.l = 0.86;
    if (hsl.l < 0.05 && hsl.s > 0.1) hsl.l = 0.06;
    return new THREE.Color().setHSL(hsl.h, hsl.s, hsl.l);
  }

  function detectMaterialArchetype(mat, meshName, entityName) {
    const label = `${meshName || ''} ${entityName || ''} ${mat.name || ''}`.toLowerCase();
    const hasAlphaCutout = !!(mat.alphaMap || (mat.alphaTest && mat.alphaTest > 0) || (mat.transparent && mat.map && mat.opacity >= 0.88));

    if (!hasAlphaCutout && (
      /\b(glass|crystal|ice|lens|window|bottle|diamond|gem|water|visor|windshield)\b/i.test(label) ||
      (mat.transmission && mat.transmission > 0.25) ||
      (mat.transparent && !mat.map && mat.opacity < 0.85 && mat.opacity > 0.08)
    )) {
      return 'glass';
    }
    if (/\b(cloth|fabric|linen|silk|cotton|velvet|wool|denim|leather|carpet|curtain|shirt|cape|coat)\b/i.test(label) ||
        (mat.sheen && mat.sheen > 0.35 && !/\b(leaf|petal|skin)\b/i.test(label))) {
      return 'cloth';
    }
    if (/\b(skin|flesh|face|hand|ear|leaf|leaves|foliage|plant|flower|petal|wax|candle|jade|fruit|apple|grape|coral|jelly|organic|mushroom)\b/i.test(label)) {
      return 'sss';
    }
    if (/\b(metal|steel|iron|gold|silver|chrome|brass|copper|bronze|armor|blade|sword|gun|chassis|mech|coin|alloy|titanium|aluminum)\b/i.test(label) ||
        (mat.metalness !== undefined && mat.metalness >= 0.62 && !mat.metalnessMap)) {
      return 'metal';
    }
    if (/\b(stone|rock|wood|bark|brick|wall|concrete|clay|ceramic|terracotta|trunk|marble|granite|crate|barrel|statue)\b/i.test(label)) {
      return 'rough_organic';
    }
    return 'standard_real';
  }

  function inferSSSColor(mat, meshName, entityName) {
    const label = `${meshName || ''} ${entityName || ''}`.toLowerCase();
    if (/\b(leaf|leaves|foliage|plant|moss|jade|fern|grass|tree)\b/i.test(label)) return new THREE.Color(0x4ade80);
    if (/\b(wax|candle|amber|honey)\b/i.test(label)) return new THREE.Color(0xfbbf24);
    if (/\b(ice|crystal|snow)\b/i.test(label)) return new THREE.Color(0x7dd3fc);
    if (mat.color && !mat.vertexColors) return mat.color.clone().lerp(new THREE.Color(0xff5232), 0.45);
    return new THREE.Color(0xff6b4a);
  }

  function upgradeMaterialRealistic(mat, archetype, meshName, entityName, objSeed = 0, isGLB = false) {
    if (!mat || mat.isShaderMaterial || mat.userData?.nxrSkip) return mat;
    if (mat.userData?.nxrRealUpgraded && mat.userData?.nxrArchetype === archetype) return mat;

    const hasOriginalPBRMaps = !!(mat.map || mat.normalMap || mat.roughnessMap || mat.metalnessMap || mat.aoMap);
    const treatAsTexturedAsset = isGLB || hasOriginalPBRMaps;

    let targetMat = mat;
    if (!mat.isMeshStandardMaterial) {
      targetMat = new THREE.MeshStandardMaterial({
        color: mat.color ? mat.color.clone() : new THREE.Color(0xcbd5e1),
        map: mat.map || null,
        vertexColors: !!mat.vertexColors,
        transparent: !!mat.transparent,
        opacity: mat.opacity !== undefined ? mat.opacity : 1.0,
        alphaTest: mat.alphaTest || 0,
        side: mat.side !== undefined ? mat.side : THREE.FrontSide,
        roughness: 0.52,
        metalness: 0.04
      });
    } else if (archetype === 'glass' && !(targetMat instanceof THREE.MeshPhysicalMaterial)) {
      const pm = new THREE.MeshPhysicalMaterial();
      THREE.MeshStandardMaterial.prototype.copy.call(pm, targetMat);
      pm.userData = Object.assign({}, targetMat.userData || {});
      pm.onBeforeCompile = targetMat.onBeforeCompile;
      pm.customProgramCacheKey = targetMat.customProgramCacheKey;
      targetMat = pm;
    }

    if (!treatAsTexturedAsset && targetMat.color && !targetMat.vertexColors) {
      targetMat.color.copy(calibrateProceduralColor(targetMat.color, archetype));
    }

    targetMat.envMap = (APP.isAR ? scene.environment : (studioEnvTex || APP.roomEnvTexture || scene.environment)) || null;
    if (!targetMat.envMapIntensity || targetMat.envMapIntensity === 1) {
      targetMat.envMapIntensity = archetype === 'metal' ? 2.0 : archetype === 'glass' ? 2.5 : 1.25;
    }

    let modeId = 0; // 0:standard, 1:metal, 2:glass, 3:sss, 4:rough_organic, 5:cloth
    let sssStrength = 0.0;
    let sssCol = new THREE.Color(0xff5533);

    if (archetype === 'metal') {
      modeId = 1;
      if (!treatAsTexturedAsset) {
        targetMat.metalness = Math.max(0.86, targetMat.metalness || 0.90);
        targetMat.roughness = THREE.MathUtils.clamp(targetMat.roughness || 0.28, 0.14, 0.44);
      }
    } else if (archetype === 'glass') {
      modeId = 2;
      targetMat.transparent = false;
      targetMat.opacity = 1.0;
      targetMat.transmission = Math.max(0.92, targetMat.transmission || 0.94);
      targetMat.ior = targetMat.ior || 1.52;
      targetMat.thickness = targetMat.thickness || 0.34;
      targetMat.roughness = THREE.MathUtils.clamp(targetMat.roughness || 0.12, 0.04, 0.24);
      targetMat.metalness = 0.0;
      targetMat.reflectivity = 1.0;
      targetMat.clearcoat = 1.0;
      targetMat.clearcoatRoughness = 0.04;
      if (!targetMat.normalMap) {
        targetMat.normalMap = baked.glassNormalTex;
        targetMat.normalScale = new THREE.Vector2(0.22, 0.22);
      }
      if (targetMat.color) targetMat.attenuationColor = targetMat.color.clone().lerp(new THREE.Color(0xdcf6ff), 0.72);
    } else if (archetype === 'sss') {
      modeId = 3;
      sssStrength = 0.95;
      sssCol = inferSSSColor(targetMat, meshName, entityName);
      if (!treatAsTexturedAsset) {
        targetMat.roughness = THREE.MathUtils.clamp(targetMat.roughness || 0.48, 0.30, 0.74);
        targetMat.metalness = 0.0;

  // 5. PROCEDURAL SILHOUETTE BEVELING (SKIPPED FOR GLBs & SCULPTED!)
  // ====================================================================
  function createBeveledCylinderGeometry(rT, rB, h, segs) {
    const hh = h * 0.5;
    const bT = rT > 0.004 ? Math.min(rT * 0.16, h * 0.09) : 0;
    const bB = rB > 0.004 ? Math.min(rB * 0.16, h * 0.09) : 0;
    const pts = [new THREE.Vector2(0.0001, -hh)];
    if (bB > 0) {
      pts.push(
        new THREE.Vector2(Math.max(0.0008, rB - bB), -hh),
        new THREE.Vector2(rB - bB * 0.22, -hh + bB * 0.3),
        new THREE.Vector2(rB, -hh + bB)
      );
    } else {
      pts.push(new THREE.Vector2(Math.max(0.0001, rB), -hh));
    }
    if (bT > 0) {
      pts.push(
        new THREE.Vector2(rT, hh - bT),
        new THREE.Vector2(rT - bT * 0.22, hh - bT * 0.3),
        new THREE.Vector2(Math.max(0.0008, rT - bT), hh),
        new THREE.Vector2(0.0001, hh)
      );
    } else {
      pts.push(new THREE.Vector2(Math.max(0.0001, rT), hh));
    }
    const geo = new THREE.LatheGeometry(pts, Math.max(14, segs || 20));
    geo.computeVertexNormals();
    return geo;
  }

  function bevelAndSealProceduralParts(group) {
    if (group.userData?.nxrSealed) return;
    group.userData.nxrSealed = true;

    group.traverse((c) => {
      if (!c.isMesh || !c.geometry || c.userData?.nxrDecal) return;
      const p = c.geometry.parameters;
      if (c.geometry.type === 'BoxGeometry' && p && APP.RoundedBoxGeometry) {
        const w = p.width || 1, h = p.height || 1, d = p.depth || 1;
        const r = Math.min(w, h, d) * 0.065;
        if (r > 0.0004) {
          const old = c.geometry;
          c.geometry = new APP.RoundedBoxGeometry(w, h, d, 2, r);
          old.dispose();
        }
      } else if (c.geometry.type === 'CylinderGeometry' && p && !p.openEnded) {
        const rT = p.radiusTop ?? 1, rB = p.radiusBottom ?? 1, h = p.height ?? 1;
        if (Math.min(Math.max(rT, rB), h) > 0.008) {
          const old = c.geometry;
          c.geometry = createBeveledCylinderGeometry(rT, rB, h, p.radialSegments);
          old.dispose();
        }
      }
    });

    group.updateMatrixWorld(true);
    const invGroup = group.matrixWorld.clone().invert();
    const parts = [];
    group.traverse((c) => {
      if (!c.isMesh || !c.geometry || c.userData?.nxrDecal) return;
      c.geometry.computeBoundingBox();
      const relM = invGroup.clone().multiply(c.matrixWorld);
      const box = c.geometry.boundingBox.clone().applyMatrix4(relM);
      const sz = new THREE.Vector3(), ctr = new THREE.Vector3();
      box.getSize(sz); box.getCenter(ctr);
      parts.push({ mesh: c, box, ctr, vol: sz.x * sz.y * sz.z });
    });

    if (parts.length > 1 && parts.length <= 42) {
      parts.sort((a, b) => b.vol - a.vol);
      const closestA = new THREE.Vector3(), closestB = new THREE.Vector3(), delta = new THREE.Vector3();
      for (let i = 1; i < parts.length; i++) {
        const p = parts[i];
        let bestDist = Infinity, bestDelta = null;
        for (let j = 0; j < parts.length; j++) {
          if (i === j || parts[j].vol < p.vol * 0.5) continue;
          parts[j].box.clampPoint(p.ctr, closestA);
          p.box.clampPoint(closestA, closestB);
          const d = closestA.distanceTo(closestB);
          if (d < bestDist) {
            bestDist = d;
            delta.subVectors(closestA, closestB);
            bestDelta = delta.clone();
          }
        }
        if (bestDelta && bestDist > 0.0015 && bestDist < 0.024) {
          const shift = bestDelta.length() + 0.003;
          bestDelta.normalize().multiplyScalar(shift);
          if (p.mesh.parent) {
            const worldShift = bestDelta.clone().applyMatrix4(new THREE.Matrix4().extractRotation(group.matrixWorld));
            const invParent = new THREE.Matrix4().extractRotation(p.mesh.parent.matrixWorld).invert();
            worldShift.applyMatrix4(invParent);
            p.mesh.position.add(worldShift);
          }
        }
      }
    }
  }

  function bakeFastCreviceAndHeight(group) {
    group.updateMatrixWorld(true);
    const invGroup = group.matrixWorld.clone().invert();
    const meshes = [], meshBoxes = [];

    group.traverse((c) => {
      if (c.isMesh && c.geometry && !c.userData?.nxrDecal) {
        if (!c.geometry.attributes.normal) c.geometry.computeVertexNormals();
        c.geometry.computeBoundingBox();
        const boxG = c.geometry.boundingBox.clone();
        const relM = invGroup.clone().multiply(c.matrixWorld);
        boxG.applyMatrix4(relM);
        meshes.push({ mesh: c, relM, boxG });
        meshBoxes.push(boxG);
      }
    });
    if (!meshes.length) return;

    const minYGroup = meshes.reduce((m, x) => Math.min(m, x.boxG.min.y), Infinity);
    const maxYGroup = meshes.reduce((m, x) => Math.max(m, x.boxG.max.y), -Infinity);
    const spanY = Math.max(0.05, maxYGroup - minYGroup);

    const vPos = new THREE.Vector3(), vNorm = new THREE.Vector3(), closest = new THREE.Vector3();
    const normalMat = new THREE.Matrix3();
    for (let mi = 0; mi < meshes.length; mi++) {
      const { mesh, relM } = meshes[mi];
      const geo = mesh.geometry, posAttr = geo.attributes.position, normAttr = geo.attributes.normal;
      if (!posAttr) continue;
      const count = posAttr.count;
      const crevArr = new Float32Array(count);
      const botArr = new Float32Array(count);
      const stride = count > 4500 ? 3 : 1;
      normalMat.getNormalMatrix(relM);

      for (let i = 0; i < count; i++) {
        if (stride > 1 && i % stride !== 0) {
          crevArr[i] = crevArr[i - 1] || 0;
          botArr[i] = botArr[i - 1] || 0.5;
          continue;
        }
        vPos.fromBufferAttribute(posAttr, i).applyMatrix4(relM);
        let occ = 0;
        for (let oi = 0; oi < meshBoxes.length; oi++) {
          if (oi === mi) continue;
          meshBoxes[oi].clampPoint(vPos, closest);
          const d = vPos.distanceTo(closest);
          if (d < 0.095) occ = Math.max(occ, (1.0 - d / 0.095) * 0.88);
        }
        const floorDist = vPos.y - minYGroup;
        if (floorDist < 0.065) occ = Math.max(occ, (1.0 - Math.max(0, floorDist) / 0.065) * 0.55);

        if (normAttr) {
          vNorm.fromBufferAttribute(normAttr, i).applyMatrix3(normalMat).normalize();
          if (vNorm.y < 0) occ = Math.max(occ, Math.min(0.35, -vNorm.y * 0.35));
        }
        crevArr[i] = THREE.MathUtils.clamp(occ, 0, 1);
        botArr[i] = THREE.MathUtils.clamp(floorDist / spanY, 0, 1);
      }
      geo.setAttribute('aNxrCrevice', new THREE.BufferAttribute(crevArr, 1));
      geo.setAttribute('aNxrBottom', new THREE.BufferAttribute(botArr, 1));
    }
  }

  function attachLiteHeatShimmerIfHot(ent) {
    if (!ent || !ent.group || ent.group.getObjectByName('NXR_HeatShimmer')) return;
    const label = `${ent.name || ''} ${ent.category || ''}`.toLowerCase();
    if (ent.category !== 'torch' && !/\b(fire|flame|torch|lava|magma|furnace|forge|candle)\b/i.test(label)) return;

    const mat = new THREE.ShaderMaterial({
      transparent: true, depthWrite: false, blending: THREE.AdditiveBlending, side: THREE.DoubleSide,
      uniforms: { uTime: sharedUniforms.uNxrTime },
      vertexShader: 'uniform float uTime; varying vec2 vUv; void main(){ vUv=uv; vec3 p=position; p.x+=sin(uTime*10.0+position.y*22.0)*0.012*uv.y; gl_Position=projectionMatrix*modelViewMatrix*vec4(p,1.0); }',
      fragmentShader: 'uniform float uTime; varying vec2 vUv; void main(){ float wave=sin(vUv.y*28.0-uTime*14.0+sin(vUv.x*12.0)*2.0)*0.5+0.5; float fade=sin(vUv.x*3.14159)*(1.0-vUv.y)*smoothstep(0.0,0.25,vUv.y); gl_FragColor=vec4(vec3(1.0,0.65,0.28)*wave,fade*0.14); }'
    });
    const m = new THREE.Mesh(new THREE.CylinderGeometry(0.14, 0.09, 0.42, 12, 8, true), mat);
    m.name = 'NXR_HeatShimmer'; m.userData.nxrDecal = true; m.position.set(0, 0.68, 0);
    ent.group.add(m); heatMirages.push(m);
  }

  const causticDecals = [];
  function attachCausticsIfRefractive(ent, hasGlass) {
    if (!hasGlass && !/\b(water|glass|crystal|pool|ice)\b/i.test(ent.name || '')) return;
    if (ent.group.getObjectByName('NXR_CausticsProjector')) return;

    const cMat = new THREE.ShaderMaterial({
      transparent: true, depthWrite: false, blending: THREE.AdditiveBlending,
      uniforms: { uTime: sharedUniforms.uNxrTime },
      vertexShader: 'varying vec2 vUv; void main(){ vUv=uv; gl_Position=projectionMatrix*modelViewMatrix*vec4(position,1.0); }',
      fragmentShader: 'uniform float uTime; varying vec2 vUv; void main(){ float d=length(vUv-0.5)*2.0; float mask=pow(max(0.0,1.0-d),1.8); float c1=sin(vUv.x*24.0+uTime*2.2)*cos(vUv.y*24.0-uTime*1.8); float c2=sin((vUv.x+vUv.y)*30.0-uTime*2.8); float c=pow(clamp(0.5+0.25*(c1+c2),0.0,1.0),4.0); gl_FragColor=vec4(vec3(0.45,0.92,1.0)*c*1.4,mask*clamp(c*1.3,0.0,0.75)); }'
    });
    const cMesh = new THREE.Mesh(new THREE.PlaneGeometry(0.88, 0.88), cMat);
    cMesh.name = 'NXR_CausticsProjector'; cMesh.userData.nxrDecal = true;
    cMesh.rotation.x = -Math.PI / 2; cMesh.position.set(0, 0.012, 0);
    ent.group.add(cMesh); causticDecals.push(cMesh);
  }

  function attachSoftContactShadow(ent) {
    if (!ent || !ent.group || ent.category === 'world_fx' || ent.category === 'portal' || ent.category === 'board' || ent.category === 'cloth' || ent.isPen) return;
    if (ent.group.getObjectByName('NXR_ContactShadow')) return;

    ent.group.updateMatrixWorld(true);
    const box = new THREE.Box3().setFromObject(ent.group);
    if (box.isEmpty()) return;
    const size = new THREE.Vector3(); box.getSize(size);
    const spanX = Math.max(0.22, Math.min(2.6, size.x * 1.60));
    const spanZ = Math.max(0.22, Math.min(2.6, size.z * 1.60));

    const sMat = new THREE.MeshBasicMaterial({ map: baked.contactShadowTex, transparent: true, opacity: 0.72, depthWrite: false });
    sMat.userData.nxrSkip = true;
    const sMesh = new THREE.Mesh(new THREE.PlaneGeometry(spanX, spanZ), sMat);
    sMesh.name = 'NXR_ContactShadow'; sMesh.userData.nxrDecal = true;
    sMesh.rotation.x = -Math.PI / 2; sMesh.position.set(0, 0.004, 0);
    ent.group.add(sMesh);
  }

  function enhanceEntityRealism(ent) {
    if (!cfg.enabled || !ent || !ent.group) return;
    if (ent.category === 'board' || ent.category === 'portal' || ent.category === 'world_fx' || ent.category === 'cloth' || ent.category === 'asset_image' || ent.category === 'asset_video') return;

      bevelAndSealProceduralParts(ent.group);
    }
    bakeFastCreviceAndHeight(ent.group);

    if (!ent.group.userData.nxrSeed) ent.group.userData.nxrSeed = (entitySeedCounter++) * 1.37;
    if (!ent.group.userData.nxrPrevPos) ent.group.userData.nxrPrevPos = ent.group.position.clone();
    if (!ent.group.userData.nxrVel) ent.group.userData.nxrVel = new THREE.Vector3();

    let hasGlass = false;
    const objSeed = ent.group.userData.nxrSeed;
    ent.group.traverse((c) => {
      if (!c.isMesh || c.userData?.nxrDecal || !c.material) return;
      const mats = Array.isArray(c.material) ? c.material : [c.material];
      const nextMats = mats.map((m) => {
        const arch = detectMaterialArchetype(m, c.name, ent.name);
        if (arch === 'glass') hasGlass = true;
        return upgradeMaterialRealistic(m, arch, c.name, ent.name, objSeed, isGLB);
      });
      c.material = Array.isArray(c.material) ? nextMats : nextMats[0];
      c.castShadow = true;
      c.receiveShadow = true;
    });

    attachLiteHeatShimmerIfHot(ent);
    attachCausticsIfRefractive(ent, hasGlass);
    attachSoftContactShadow(ent);
  }

  APP.craft.blendedJoint = function (pA, pB, rA, rB, color, matOpts = {}) {
    const curve = new THREE.LineCurve3(new THREE.Vector3(...pA), new THREE.Vector3(...pB));
    const segs = 12, radSegs = 18;
    const geo = new THREE.TubeGeometry(curve, segs, 1.0, radSegs, false);
    const pos = geo.attributes.position;
    const v = new THREE.Vector3(), cPt = new THREE.Vector3();
    for (let i = 0; i < pos.count; i++) {
      const ring = Math.floor(i / (radSegs + 1));
      const t = Math.min(1, ring / segs);
      curve.getPoint(t, cPt);
      v.fromBufferAttribute(pos, i).sub(cPt);
      const r = THREE.MathUtils.lerp(rA, rB, t) * (1.0 + Math.sin(t * Math.PI) * 0.08);
      v.multiplyScalar(r).add(cPt);
      pos.setXYZ(i, v.x, v.y, v.z);
    }
    geo.computeVertexNormals();
    return new THREE.Mesh(geo, APP.craft.mat(color, matOpts));
  };

  const tmpVel = new THREE.Vector3();
  function updateRealism(dt, elapsed) {
    const delta = Math.max(0.004, Math.min(0.05, dt || 0.016));
    sharedUniforms.uNxrTime.value = elapsed || (performance.now() * 0.001);

    if (!APP.isAR && studioEnvTex && scene.environment !== studioEnvTex) {
      scene.environment = studioEnvTex;
    }

    const hasRainOrWater = (APP.entities || []).some((e) => e && e.category === 'world_fx' && /rain|water/i.test(e.name || ''));
    const targetWet = hasRainOrWater ? Math.max(cfg.wetness, 0.85) : cfg.wetness;
    sharedUniforms.uNxrWetness.value += (targetWet - sharedUniforms.uNxrWetness.value) * Math.min(1, delta * 2.5);

    if (APP.dirLight) sharedUniforms.uNxrLightDir.value.copy(APP.dirLight.position).normalize();

    const ents = APP.entities || [];
    for (let i = 0; i < ents.length; i++) {
      const e = ents[i];
      if (!e || !e.group || !e.group.parent) continue;

      const cs = e.group.getObjectByName('NXR_ContactShadow');
      if (cs) {
        const lift = Math.max(0, e.group.position.y);
        cs.position.y = -lift + 0.004;
        cs.material.opacity = Math.max(0, 0.72 - lift * 0.95);
        cs.visible = cs.material.opacity > 0.03;
      }

      if (cfg.motionBlur && e.group.userData.nxrPrevPos) {
        tmpVel.subVectors(e.group.position, e.group.userData.nxrPrevPos).divideScalar(delta).clampLength(0, 6.0);
        e.group.userData.nxrPrevPos.copy(e.group.position);
        const smoothV = e.group.userData.nxrVel;
        smoothV.lerp(tmpVel, Math.min(1, delta * 18.0));
        if (smoothV.lengthSq() > 0.02 || e.group.userData.nxrHadVel) {
          const active = smoothV.lengthSq() > 0.02;
          e.group.userData.nxrHadVel = active;
          e.group.traverse((c) => {
            if (!c.isMesh || !c.material) return;
            const mats = Array.isArray(c.material) ? c.material : [c.material];
            for (let m = 0; m < mats.length; m++) {
              if (mats[m]?.userData?.nxrVelUniform) {
                if (active) mats[m].userData.nxrVelUniform.value.copy(smoothV);
                else mats[m].userData.nxrVelUniform.value.set(0, 0, 0);
              }
            }
          });
        }
      }
    }
  }
