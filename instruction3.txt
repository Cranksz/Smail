(function () {
  const APP = window.APP || (window.APP = {});
  const THREE = APP.THREE;
  if (!THREE || !APP.scene || !APP.renderer) return;

  }

  const scene = APP.scene;
  const renderer = APP.renderer;

  // ====================================================================
  // 1. LITE REALISM CONFIG & SHADOW / TONE-MAPPING SETUP
  // ====================================================================
  const cfg = {
    enabled: true,
    calibrateColors: true,
    wetness: 0.14,
    parallaxScale: 0.018,
    edgeWear: 0.68,
    causticsStrength: 0.85

    APP.dirLight.shadow.normalBias = 0.015;

  // Soft sky-bounce fill light (eliminates harsh black undersides)

  // ====================================================================
  // 2. PRE-BAKED TEXTURE SUITE (GENERATED ONCE, ZERO PER-PIXEL NOISE LOOPS)
  // ====================================================================
  function buildBakedDetailTextures() {
    const S = 256;

    // A. Detail Atlas: R = micro height/grain, G = brushed scratches, B = rust/wear blotches
    const dCan = document.createElement('canvas');
    dCan.width = dCan.height = S;
    const dCtx = dCan.getContext('2d');
    const img = dCtx.createImageData(S, S);
    for (let y = 0; y < S; y++) {
      for (let x = 0; x < S; x++) {
        const u = (x / S) * Math.PI * 2;
        const v = (y / S) * Math.PI * 2;
        const grain = 0.5
          + 0.22 * Math.sin(u * 13 + Math.cos(v * 11))
          + 0.16 * Math.cos(v * 23 - Math.sin(u * 19))
          + (Math.random() - 0.5) * 0.24;
        const brush = 0.5 + 0.35 * Math.sin(v * 47 + Math.sin(u * 3) * 2.0) + (Math.random() - 0.5) * 0.18;
        const rust = 0.5
          + 0.28 * Math.sin(u * 3 + v * 2) * Math.cos(v * 4 - u * 2)
          + 0.22 * Math.sin(u * 7 - v * 5);
        const i = (y * S + x) * 4;
        img.data[i] = THREE.MathUtils.clamp(grain * 255, 0, 255);
        img.data[i + 1] = THREE.MathUtils.clamp(brush * 255, 0, 255);
        img.data[i + 2] = THREE.MathUtils.clamp(rust * 255, 0, 255);
        img.data[i + 3] = 255;
      }
    }
    dCtx.putImageData(img, 0, 0);

    // Paint sharp crisp metallic micro-scratches into Green channel
    dCtx.globalCompositeOperation = 'lighter';
    for (let k = 0; k < 140; k++) {
      const x1 = Math.random() * S, y1 = Math.random() * S;
      const ang = Math.random() * Math.PI * 2, len = 14 + Math.random() * 58;
      dCtx.strokeStyle = 'rgba(0, 160, 0, 0.45)';
      dCtx.lineWidth = 0.7 + Math.random() * 0.8;
      dCtx.beginPath();
      dCtx.moveTo(x1, y1);
      dCtx.lineTo(x1 + Math.cos(ang) * len, y1 + Math.sin(ang) * len);
      dCtx.stroke();
    }
    const detailTex = new THREE.CanvasTexture(dCan);
    detailTex.wrapS = detailTex.wrapT = THREE.RepeatWrapping;
    detailTex.needsUpdate = true;

    // B. Crisp Tangent Normal Map derived from height/scratches for tactile surface relief
    const nCan = document.createElement('canvas');
    nCan.width = nCan.height = S;
    const nCtx = nCan.getContext('2d');
    const src = dCtx.getImageData(0, 0, S, S).data;
    const nImg = nCtx.createImageData(S, S);
    for (let y = 0; y < S; y++) {
      for (let x = 0; x < S; x++) {
        const xm = (x - 1 + S) % S, xp = (x + 1) % S;
        const ym = (y - 1 + S) % S, yp = (y + 1) % S;
        const hL = (src[(y * S + xm) * 4] + src[(y * S + xm) * 4 + 1] * 0.6) / 408;
        const hR = (src[(y * S + xp) * 4] + src[(y * S + xp) * 4 + 1] * 0.6) / 408;
        const hU = (src[(ym * S + x) * 4] + src[(ym * S + x) * 4 + 1] * 0.6) / 408;
        const hD = (src[(yp * S + x) * 4] + src[(yp * S + x) * 4 + 1] * 0.6) / 408;
        const dx = (hL - hR) * 2.2, dy = (hU - hD) * 2.2;
        const len = Math.hypot(dx, dy, 1.0);
        const i = (y * S + x) * 4;
        nImg.data[i] = Math.round(((dx / len) * 0.5 + 0.5) * 255);
        nImg.data[i + 1] = Math.round(((dy / len) * 0.5 + 0.5) * 255);
        nImg.data[i + 2] = Math.round(((1.0 / len) * 0.5 + 0.5) * 255);
        nImg.data[i + 3] = 255;
      }
    }
    nCtx.putImageData(nImg, 0, 0);
    const surfaceNormalTex = new THREE.CanvasTexture(nCan);
    surfaceNormalTex.wrapS = surfaceNormalTex.wrapT = THREE.RepeatWrapping;
    surfaceNormalTex.needsUpdate = true;

    // C. Smooth optical ripple normal map for Glass refraction distortion
    const gCan = document.createElement('canvas');
    gCan.width = gCan.height = 128;
    const gCtx = gCan.getContext('2d');
    const gImg = gCtx.createImageData(128, 128);
    for (let y = 0; y < 128; y++) {
      for (let x = 0; x < 128; x++) {
        const u = (x / 128) * Math.PI * 2, v = (y / 128) * Math.PI * 2;
        const dx = Math.sin(u * 3 + Math.cos(v * 2)) * 0.28;
        const dy = Math.cos(v * 3 - Math.sin(u * 2)) * 0.28;
        const len = Math.hypot(dx, dy, 1.0);
        const i = (y * 128 + x) * 4;
        gImg.data[i] = Math.round(((dx / len) * 0.5 + 0.5) * 255);
        gImg.data[i + 1] = Math.round(((dy / len) * 0.5 + 0.5) * 255);
        gImg.data[i + 2] = Math.round(((1.0 / len) * 0.5 + 0.5) * 255);
        gImg.data[i + 3] = 255;
      }
    }
    gCtx.putImageData(gImg, 0, 0);
    const glassNormalTex = new THREE.CanvasTexture(gCan);
    glassNormalTex.wrapS = glassNormalTex.wrapT = THREE.RepeatWrapping;
    glassNormalTex.needsUpdate = true;

    // D. Soft Contact Shadow Texture
    const sCan = document.createElement('canvas');
    sCan.width = sCan.height = 128;
    const sCtx = sCan.getContext('2d');
    const g = sCtx.createRadialGradient(64, 64, 4, 64, 64, 62);
    g.addColorStop(0, 'rgba(0, 0, 0, 0.58)');
    g.addColorStop(0.5, 'rgba(0, 0, 0, 0.22)');
    g.addColorStop(1, 'rgba(0, 0, 0, 0.0)');
    sCtx.fillStyle = g;
    sCtx.fillRect(0, 0, 128, 128);
    const contactShadowTex = new THREE.CanvasTexture(sCan);
    contactShadowTex.needsUpdate = true;

    return { detailTex, surfaceNormalTex, glassNormalTex, contactShadowTex };
  }

  const baked = buildBakedDetailTextures();

  const sharedUniforms = {
    uNxrTime: { value: 0 },
    uNxrWetness: { value: cfg.wetness },
    uNxrParallax: { value: cfg.parallaxScale },
    uNxrEdgeWear: { value: cfg.edgeWear },
    uNxrCausticsGlobal: { value: cfg.causticsStrength },
    uNxrDetailTex: { value: baked.detailTex },
    uNxrLightDir: { value: new THREE.Vector3(0.45, 0.82, 0.35).normalize() }
  };

  // ====================================================================
  // 3. COLOR CALIBRATION & MATERIAL ARCHETYPES
  // ====================================================================
  function calibrateNaturalColor(col, archetype) {
    if (!cfg.calibrateColors || !col || archetype === 'glass') return col;
    const hsl = {};
    col.getHSL(hsl);
    if (hsl.s > 0.62) hsl.s = 0.62 + (hsl.s - 0.62) * 0.25;
    if (hsl.l > 0.86) hsl.l = 0.86;
    if (hsl.l < 0.05 && hsl.s > 0.1) hsl.l = 0.06;
    return new THREE.Color().setHSL(hsl.h, hsl.s, hsl.l);
  }

  function detectMaterialArchetype(mat, meshName, entityName) {
    const label = `${meshName || ''} ${entityName || ''} ${mat.name || ''}`.toLowerCase();
    if (/\b(glass|crystal|ice|lens|window|bottle|diamond|gem|water|transparent|visor)\b/i.test(label) ||
        (mat.transparent && mat.opacity < 0.88 && mat.opacity > 0.08) ||
        (mat.transmission && mat.transmission > 0.2)) {
      return 'glass';
    }
    if (/\b(metal|steel|iron|gold|silver|chrome|brass|copper|bronze|armor|blade|sword|gun|robot|chassis|mech|coin|pen|clip)\b/i.test(label) ||
        (mat.metalness !== undefined && mat.metalness >= 0.55)) {
      return 'metal';
    }
    if (/\b(skin|flesh|face|hand|ear|leaf|leaves|foliage|plant|flower|petal|wax|candle|jade|fruit|apple|grape|coral|jelly|organic|mushroom)\b/i.test(label)) {
      return 'sss';
    }
    if (/\b(stone|rock|wood|bark|brick|wall|ground|sand|fur|cloth|leather|fabric|ceramic|trunk)\b/i.test(label)) {
      return 'rough_organic';
    }
    return 'standard_real';
  }

  function inferSSSColor(mat, meshName, entityName) {
    const label = `${meshName || ''} ${entityName || ''}`.toLowerCase();
    if (/\b(leaf|leaves|foliage|plant|moss|jade|fern|grass|tree)\b/i.test(label)) return new THREE.Color(0x4ade80);
    if (/\b(wax|candle|amber|honey)\b/i.test(label)) return new THREE.Color(0xfbbf24);
    if (/\b(ice|crystal|snow)\b/i.test(label)) return new THREE.Color(0x7dd3fc);
    if (mat.color && !mat.vertexColors) return mat.color.clone().lerp(new THREE.Color(0xff4d2e), 0.45);
    return new THREE.Color(0xff6b4a);
  }

  // ====================================================================
  // 4. LITE HARDWARE-FAST REALISM SHADER (ZERO HEAVY NOISE LOOPS!)
  // ====================================================================
  function upgradeMaterialRealistic(mat, archetype, meshName, entityName) {
    if (!mat || mat.isShaderMaterial || mat.userData?.nxrSkip) return mat;
    if (mat.userData?.nxrRealUpgraded && mat.userData?.nxrArchetype === archetype) return mat;

    let targetMat = mat;
    if (!mat.isMeshStandardMaterial) {
      targetMat = new THREE.MeshPhysicalMaterial({
        color: mat.color ? mat.color.clone() : new THREE.Color(0xcbd5e1),
        map: mat.map || null,
        vertexColors: !!mat.vertexColors,
        transparent: !!mat.transparent,
        opacity: mat.opacity !== undefined ? mat.opacity : 1.0,
        side: mat.side !== undefined ? mat.side : THREE.FrontSide,
        roughness: 0.45,
        metalness: 0.08
      });
    } else if (!(targetMat instanceof THREE.MeshPhysicalMaterial)) {
      const pm = new THREE.MeshPhysicalMaterial();
      THREE.MeshStandardMaterial.prototype.copy.call(pm, targetMat);
      targetMat = pm;
    }

    if (targetMat.color && !targetMat.vertexColors) {
      targetMat.color.copy(calibrateNaturalColor(targetMat.color, archetype));
    }

    targetMat.envMap = (APP.isAR ? scene.environment : (studioEnvTex || APP.roomEnvTexture || scene.environment)) || null;
    targetMat.envMapIntensity = archetype === 'metal' ? 2.1 : archetype === 'glass' ? 2.4 : 1.25;

    let modeId = 0; // 0:standard, 1:metal, 2:glass, 3:sss, 4:rough_organic
    let sssStrength = 0.0;
    let sssCol = new THREE.Color(0xff5533);

    if (archetype === 'metal') {
      modeId = 1;
      targetMat.metalness = Math.max(0.88, targetMat.metalness || 0.92);
      targetMat.roughness = THREE.MathUtils.clamp(targetMat.roughness || 0.28, 0.14, 0.44);
      targetMat.normalMap = baked.surfaceNormalTex;
      targetMat.normalScale = new THREE.Vector2(0.32, 0.32);
      targetMat.clearcoat = 0.18;
      targetMat.clearcoatRoughness = 0.25;
    } else if (archetype === 'glass') {
      modeId = 2;
      targetMat.transparent = false;
      targetMat.opacity = 1.0;
      targetMat.transmission = 0.95;
      targetMat.ior = 1.52;
      targetMat.thickness = 0.28;
      targetMat.roughness = 0.12; // Realistic blurred + distorted background
      targetMat.metalness = 0.0;
      targetMat.reflectivity = 1.0;
      targetMat.clearcoat = 1.0;
      targetMat.clearcoatRoughness = 0.04;
      targetMat.normalMap = baked.glassNormalTex;
      targetMat.normalScale = new THREE.Vector2(0.14, 0.14);
      targetMat.attenuationDistance = 1.2;
      if (targetMat.color) targetMat.attenuationColor = targetMat.color.clone().lerp(new THREE.Color(0xdcf6ff), 0.72);
    } else if (archetype === 'sss') {
      modeId = 3;
      sssStrength = 0.95;
      sssCol = inferSSSColor(targetMat, meshName, entityName);
      targetMat.roughness = THREE.MathUtils.clamp(targetMat.roughness || 0.46, 0.28, 0.72);
      targetMat.metalness = 0.02;
      targetMat.normalMap = baked.surfaceNormalTex;
      targetMat.normalScale = new THREE.Vector2(0.24, 0.24);
      targetMat.sheen = 0.35;
      targetMat.sheenRoughness = 0.55;
      targetMat.sheenColor = sssCol.clone();
    } else {
      modeId = archetype === 'rough_organic' ? 4 : 0;
      targetMat.roughness = archetype === 'rough_organic'
        ? Math.max(0.55, targetMat.roughness || 0.74)
        : THREE.MathUtils.clamp(targetMat.roughness || 0.46, 0.22, 0.84);
      targetMat.normalMap = baked.surfaceNormalTex;
      targetMat.normalScale = new THREE.Vector2(0.36, 0.36);
    }

    const modeUniform = { value: modeId };
    const sssUniform = { value: sssStrength };
    const sssColUniform = { value: sssCol };

    targetMat.userData = targetMat.userData || {};
    targetMat.userData.nxrRealUpgraded = true;
    targetMat.userData.nxrArchetype = archetype;

    targetMat.customProgramCacheKey = () => `nxr_lite_real_v1_${modeId}`;
    targetMat.onBeforeCompile = function (shader) {
      shader.uniforms.uNxrTime = sharedUniforms.uNxrTime;
      shader.uniforms.uNxrWetness = sharedUniforms.uNxrWetness;
      shader.uniforms.uNxrParallax = sharedUniforms.uNxrParallax;
      shader.uniforms.uNxrEdgeWear = sharedUniforms.uNxrEdgeWear;
      shader.uniforms.uNxrCausticsGlobal = sharedUniforms.uNxrCausticsGlobal;
      shader.uniforms.uNxrDetailTex = sharedUniforms.uNxrDetailTex;
      shader.uniforms.uNxrLightDir = sharedUniforms.uNxrLightDir;
      shader.uniforms.uNxrMode = modeUniform;
      shader.uniforms.uNxrSSSStrength = sssUniform;
      shader.uniforms.uNxrSSSColor = sssColUniform;

      shader.vertexShader = shader.vertexShader
        .replace('#include <common>', [
          '#include <common>',
          'attribute float aNxrCrevice;',
          'varying vec3 vNxrLPos;',
          'varying vec3 vNxrWPos;',
          'varying vec3 vNxrWNorm;',
          'varying float vNxrCrevice;'
        ].join('\n'))
        .replace('#include <begin_vertex>', [
          '#include <begin_vertex>',
          'vNxrLPos = position;',
          'vNxrCrevice = aNxrCrevice;'
        ].join('\n'))
        .replace('#include <project_vertex>', [
          '#include <project_vertex>',
          'vec4 nxrWp = modelMatrix * vec4( transformed, 1.0 );',
          'vNxrWPos = nxrWp.xyz;',
          'vNxrWNorm = normalize( mat3( modelMatrix ) * objectNormal );'
        ].join('\n'));

      shader.fragmentShader = shader.fragmentShader
        .replace('#include <common>', [
          '#include <common>',
          'uniform float uNxrTime;',
          'uniform float uNxrWetness;',
          'uniform float uNxrParallax;',
          'uniform float uNxrEdgeWear;',
          'uniform float uNxrCausticsGlobal;',
          'uniform sampler2D uNxrDetailTex;',
          'uniform vec3 uNxrLightDir;',
          'uniform float uNxrMode;',
          'uniform float uNxrSSSStrength;',
          'uniform vec3 uNxrSSSColor;',
          'varying vec3 vNxrLPos;',
          'varying vec3 vNxrWPos;',
          'varying vec3 vNxrWNorm;',
          'varying float vNxrCrevice;',
          // Fast dominant-axis planar UV + 2-tap parallax lookup
          'vec2 nxrPlanarUV(vec3 p, vec3 n) {',
          '  vec3 an = abs(n);',
          '  if (an.x > an.y && an.x > an.z) return p.zy;',
          '  if (an.z > an.y) return p.xy;',
          '  return p.xz;',
          '}',
          'float nxrFastCaustic(vec2 p, float t) {',
          '  float a = sin(p.x * 12.0 + t * 2.1) * cos(p.y * 12.0 - t * 1.7);',
          '  float b = sin((p.x + p.y) * 16.0 - t * 2.6);',
          '  return pow(clamp(0.5 + 0.5 * (a + b) * 0.5, 0.0, 1.0), 4.0);',
          '}'
        ].join('\n'))
        .replace('#include <color_fragment>', [
          '#include <color_fragment>',
          'vec3 nxrVDirW = normalize(cameraPosition - vNxrWPos);',
          'vec3 nxrNW = normalize(vNxrWNorm);',
          'vec2 nxrDUV = nxrPlanarUV(vNxrLPos * 2.6, nxrNW);',
          'vec3 nxrDet0 = texture2D(uNxrDetailTex, nxrDUV).rgb;',
          // Lightweight 2-tap Parallax Depth offset
          'vec2 nxrPUV = nxrDUV - nxrPlanarUV(nxrVDirW, nxrNW) * (1.0 - nxrDet0.r) * uNxrParallax;',
          'vec3 nxrDet = texture2D(uNxrDetailTex, nxrPUV).rgb;',
          'float nxrGrain = nxrDet.r;',
          'float nxrScratch = smoothstep(0.68, 0.90, nxrDet.g);',
          'float nxrRustMask = (uNxrMode > 0.5 && uNxrMode < 1.5) ? smoothstep(0.56, 0.80, nxrDet.b + vNxrCrevice * 0.35) : 0.0;',
          // Fast hardware-derivative Edge Wear
          'float nxrEdgeCurv = clamp(length(fwidth(nxrNW)) * 2.8, 0.0, 1.0);',
          'float nxrEdgeMask = (uNxrMode < 1.5 || uNxrMode > 3.5) ? smoothstep(0.42, 0.76, nxrEdgeCurv * 0.85 + nxrDet.b * 0.35) * uNxrEdgeWear : 0.0;',
          // Surface Wetness
          'float nxrUp = clamp(nxrNW.y * 0.5 + 0.5, 0.0, 1.0);',
          'float nxrWetPatch = clamp(uNxrWetness * (0.45 + 0.80 * nxrUp * nxrDet.b), 0.0, 1.0);',
          'float nxrAO = 1.0 - clamp(vNxrCrevice, 0.0, 1.0) * 0.68;',
          'if (uNxrMode > 0.5 && uNxrMode < 1.5) {',
          '  diffuseColor.rgb = mix(diffuseColor.rgb, vec3(0.36, 0.14, 0.05), nxrRustMask * 0.80);',
          '  diffuseColor.rgb = mix(diffuseColor.rgb, vec3(0.88, 0.92, 0.97), nxrScratch * 0.48);',
          '} else if (uNxrMode < 0.5 || uNxrMode > 3.5) {',
          '  diffuseColor.rgb *= mix(0.84, 1.12, nxrGrain);',
          '  diffuseColor.rgb = mix(diffuseColor.rgb, vec3(0.76, 0.80, 0.86), nxrEdgeMask * 0.58);',
          '} else if (uNxrMode > 2.5 && uNxrMode < 3.5) {',
          '  diffuseColor.rgb *= mix(0.90, 1.08, nxrGrain);',
          '}',
          'diffuseColor.rgb *= mix(1.0, 0.56, nxrWetPatch * 0.68);',
          'diffuseColor.rgb *= nxrAO;'
        ].join('\n'))
        .replace('#include <roughnessmap_fragment>', [
          '#include <roughnessmap_fragment>',
          'if (uNxrMode > 0.5 && uNxrMode < 1.5) {',
          '  roughnessFactor = mix(roughnessFactor, 0.88, nxrRustMask * 0.85);',
          '  roughnessFactor = mix(roughnessFactor, 0.10, nxrScratch * 0.70);',
          '} else if (uNxrMode > 1.5 && uNxrMode < 2.5) {',
          '  roughnessFactor = clamp(0.10 + (nxrGrain - 0.5) * 0.08, 0.05, 0.20);',
          '} else {',
          '  roughnessFactor = clamp(roughnessFactor + (nxrGrain - 0.5) * 0.22, 0.08, 0.95);',
          '  roughnessFactor = mix(roughnessFactor, 0.16, nxrEdgeMask * 0.68);',
          '}',
          'roughnessFactor = mix(roughnessFactor, 0.05, nxrWetPatch * 0.82);'
        ].join('\n'))
        .replace('#include <metalnessmap_fragment>', [
          '#include <metalnessmap_fragment>',
          'if (uNxrMode > 0.5 && uNxrMode < 1.5) {',
          '  metalnessFactor = mix(metalnessFactor, 0.16, nxrRustMask * 0.82);',
          '} else if (uNxrMode < 0.5 || uNxrMode > 3.5) {',
          '  metalnessFactor = mix(metalnessFactor, 0.86, nxrEdgeMask * 0.72);',
          '}'
        ].join('\n'))
        .replace('#include <lights_fragment_end>', [
          '#include <lights_fragment_end>',
          'reflectedLight.indirectDiffuse *= nxrAO;',
          'reflectedLight.indirectSpecular *= mix(0.32, 1.0, nxrAO);',
          'if (uNxrSSSStrength > 0.01) {',
          '  vec3 nxrHBack = normalize(uNxrLightDir + nxrNW * 0.45);',
          '  float nxrBack = pow(clamp(dot(nxrVDirW, -nxrHBack), 0.0, 1.0), 2.5);',
          '  float nxrWrap = max(0.0, (dot(nxrNW, uNxrLightDir) + 0.55) / 1.55);',
          '  reflectedLight.directDiffuse += uNxrSSSColor * (nxrBack * 0.80 + nxrWrap * 0.24) * uNxrSSSStrength;',
          '}',
          'if (uNxrMode > 1.5 && uNxrMode < 2.5) {',
          '  float nxrFres = pow(1.0 - max(0.0, dot(nxrVDirW, nxrNW)), 3.2);',
          '  reflectedLight.indirectSpecular += vec3(0.78, 0.92, 1.0) * nxrFres * 0.78;',
          '}',
          'float nxrCausticsVis = (uNxrMode > 1.5 && uNxrMode < 2.5) ? 0.65 : (nxrWetPatch * nxrUp * 0.75);',
          'if (nxrCausticsVis > 0.02 && uNxrCausticsGlobal > 0.01) {',
          '  float nxrCVal = nxrFastCaustic(vNxrWPos.xz, uNxrTime);',
          '  reflectedLight.directDiffuse += vec3(0.45, 0.90, 1.0) * nxrCVal * nxrCausticsVis * uNxrCausticsGlobal * 0.95;',
          '}'
        ].join('\n'));
    };

    targetMat.needsUpdate = true;
    return targetMat;
  }

  // ====================================================================
  // 5. FAST BOUNDING-BOX CREVICE OCCLUSION + SUBTLE EDGE CHAMFER
  // ====================================================================
  function bevelSharpBoxes(group) {
    if (!APP.RoundedBoxGeometry) return;
    group.traverse((c) => {
      if (c.isMesh && c.geometry?.type === 'BoxGeometry' && c.geometry.parameters) {
        const p = c.geometry.parameters;
        const w = p.width || 1, h = p.height || 1, d = p.depth || 1;
        const r = Math.min(w, h, d) * 0.045;
        if (r > 0.0004) {
          const oldGeo = c.geometry;
          c.geometry = new APP.RoundedBoxGeometry(w, h, d, 2, r);
          oldGeo.dispose();
        }
      }
    });
  }

  function bakeFastCreviceAO(group) {
    group.updateMatrixWorld(true);
    const invGroup = group.matrixWorld.clone().invert();
    const meshes = [];
    const meshBoxes = [];

    group.traverse((c) => {
      if (c.isMesh && c.geometry && !c.userData?.nxrDecal) {
        if (!c.geometry.attributes.normal) c.geometry.computeVertexNormals();
        c.geometry.computeBoundingBox();
        const boxG = c.geometry.boundingBox.clone();
        const relM = invGroup.clone().multiply(c.matrixWorld);
        boxG.applyMatrix4(relM);
        meshes.push({ mesh: c, relM, boxG });
        meshBoxes.push(boxG);
      }
    });

    const vPos = new THREE.Vector3(), closest = new THREE.Vector3();
    for (let mi = 0; mi < meshes.length; mi++) {
      const { mesh, relM, boxG } = meshes[mi];
      const geo = mesh.geometry, posAttr = geo.attributes.position;
      if (!posAttr) continue;
      const count = posAttr.count;
      const crevArr = new Float32Array(count);
      const minYGroup = meshes.reduce((m, x) => Math.min(m, x.boxG.min.y), Infinity);
      const stride = count > 4000 ? 4 : 1;

      for (let i = 0; i < count; i++) {
        if (stride > 1 && i % stride !== 0) { crevArr[i] = crevArr[i - 1] || 0; continue; }
        vPos.fromBufferAttribute(posAttr, i).applyMatrix4(relM);
        let occ = 0;
        for (let oi = 0; oi < meshBoxes.length; oi++) {
          if (oi === mi) continue;
          meshBoxes[oi].clampPoint(vPos, closest);
          const d = vPos.distanceTo(closest);
          if (d < 0.085) occ = Math.max(occ, (1.0 - d / 0.085) * 0.90);
        }
        const floorDist = vPos.y - minYGroup;
        if (floorDist < 0.065) occ = Math.max(occ, (1.0 - Math.max(0, floorDist) / 0.065) * 0.55);
        crevArr[i] = THREE.MathUtils.clamp(occ, 0, 1);
      }
      geo.setAttribute('aNxrCrevice', new THREE.BufferAttribute(crevArr, 1));
    }
  }

  // ====================================================================
  // 6. LIGHTWEIGHT HEAT SHIMMER, CAUSTICS DECAL & CONTACT SHADOWS
  // ====================================================================
  const heatMirages = [];
  function attachLiteHeatShimmerIfHot(ent) {
    if (!ent || !ent.group || ent.group.getObjectByName('NXR_HeatShimmer')) return;
    const label = `${ent.name || ''} ${ent.category || ''}`.toLowerCase();
    const isHot = ent.category === 'torch' || /\b(fire|flame|torch|lava|magma|furnace|forge|candle)\b/i.test(label);
    if (!isHot) return;

    const mat = new THREE.ShaderMaterial({
      transparent: true,
      depthWrite: false,
      blending: THREE.AdditiveBlending,
      side: THREE.DoubleSide,
      uniforms: { uTime: sharedUniforms.uNxrTime },
      vertexShader: [
        'uniform float uTime; varying vec2 vUv;',
        'void main() {',
        '  vUv = uv;',
        '  vec3 p = position;',
        '  p.x += sin(uTime * 10.0 + position.y * 22.0) * 0.012 * uv.y;',
        '  gl_Position = projectionMatrix * modelViewMatrix * vec4(p, 1.0);',
        '}'
      ].join('\n'),
      fragmentShader: [
        'uniform float uTime; varying vec2 vUv;',
        'void main() {',
        '  float wave = sin(vUv.y * 28.0 - uTime * 14.0 + sin(vUv.x * 12.0) * 2.0) * 0.5 + 0.5;',
        '  float fade = sin(vUv.x * 3.14159) * (1.0 - vUv.y) * smoothstep(0.0, 0.25, vUv.y);',
        '  gl_FragColor = vec4(vec3(1.0, 0.65, 0.28) * wave, fade * 0.14);',
        '}'
      ].join('\n')
    });
    const m = new THREE.Mesh(new THREE.CylinderGeometry(0.14, 0.09, 0.42, 12, 8, true), mat);
    m.name = 'NXR_HeatShimmer';
    m.userData.nxrDecal = true;
    m.position.set(0, 0.68, 0);
    ent.group.add(m);
    heatMirages.push(m);
  }

  const causticDecals = [];
  function attachCausticsIfRefractive(ent, hasGlass) {
    if (!hasGlass && !/\b(water|glass|crystal|pool|ice)\b/i.test(ent.name || '')) return;
    if (ent.group.getObjectByName('NXR_CausticsProjector')) return;

    const cMat = new THREE.ShaderMaterial({
      transparent: true,
      depthWrite: false,
      blending: THREE.AdditiveBlending,
      uniforms: { uTime: sharedUniforms.uNxrTime },
      vertexShader: 'varying vec2 vUv; void main() { vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0); }',
      fragmentShader: [
        'uniform float uTime; varying vec2 vUv;',
        'void main() {',
        '  float d = length(vUv - 0.5) * 2.0;',
        '  float mask = pow(max(0.0, 1.0 - d), 1.8);',
        '  float c1 = sin(vUv.x * 24.0 + uTime * 2.2) * cos(vUv.y * 24.0 - uTime * 1.8);',
        '  float c2 = sin((vUv.x + vUv.y) * 30.0 - uTime * 2.8);',
        '  float c = pow(clamp(0.5 + 0.25 * (c1 + c2), 0.0, 1.0), 4.0);',
        '  gl_FragColor = vec4(vec3(0.45, 0.92, 1.0) * c * 1.4, mask * clamp(c * 1.3, 0.0, 0.8));',
        '}'
      ].join('\n')
    });
    const cMesh = new THREE.Mesh(new THREE.PlaneGeometry(0.85, 0.85), cMat);
    cMesh.name = 'NXR_CausticsProjector';
    cMesh.userData.nxrDecal = true;
    cMesh.rotation.x = -Math.PI / 2;
    cMesh.position.set(0, 0.012, 0);
    ent.group.add(cMesh);
    causticDecals.push(cMesh);
  }

  function attachSoftContactShadow(ent) {
    if (!ent || !ent.group || ent.category === 'world_fx' || ent.category === 'portal' || ent.category === 'board' || ent.isPen) return;
    if (ent.group.getObjectByName('NXR_ContactShadow')) return;

    ent.group.updateMatrixWorld(true);
    const box = new THREE.Box3().setFromObject(ent.group);
    if (box.isEmpty()) return;
    const size = new THREE.Vector3(); box.getSize(size);
    const span = Math.max(0.24, Math.min(2.2, Math.max(size.x, size.z) * 1.55));

    const sMat = new THREE.MeshBasicMaterial({ map: baked.contactShadowTex, transparent: true, opacity: 0.72, depthWrite: false });
    sMat.userData.nxrSkip = true;
    const sMesh = new THREE.Mesh(new THREE.PlaneGeometry(span, span), sMat);
    sMesh.name = 'NXR_ContactShadow';
    sMesh.userData.nxrDecal = true;
    sMesh.rotation.x = -Math.PI / 2;
    sMesh.position.set(0, 0.004, 0);
    ent.group.add(sMesh);
  }

  // ====================================================================
  // 7. ENTITY REALISM PROCESSOR
  // ====================================================================
  function enhanceEntityRealism(ent) {
    if (!cfg.enabled || !ent || !ent.group) return;
    if (ent.category === 'board' || ent.category === 'portal') return;

    bevelSharpBoxes(ent.group);
    bakeFastCreviceAO(ent.group);

    let hasGlass = false;
    ent.group.traverse((c) => {
      if (!c.isMesh || c.userData?.nxrDecal || !c.material) return;
      const mats = Array.isArray(c.material) ? c.material : [c.material];
      const nextMats = mats.map((m) => {
        const arch = detectMaterialArchetype(m, c.name, ent.name);
        if (arch === 'glass') hasGlass = true;
        return upgradeMaterialRealistic(m, arch, c.name, ent.name);
      });
      c.material = Array.isArray(c.material) ? nextMats : nextMats[0];
      c.castShadow = true;
      c.receiveShadow = true;
    });

    attachLiteHeatShimmerIfHot(ent);
    attachCausticsIfRefractive(ent, hasGlass);
    attachSoftContactShadow(ent);
  }

  // Lightweight per-frame update (zero heavy traversals!)
  function updateRealism(dt, elapsed) {
    const delta = Math.min(0.05, dt || 0.016);
    sharedUniforms.uNxrTime.value = elapsed || (performance.now() * 0.001);

    if (!APP.isAR && studioEnvTex && scene.environment !== studioEnvTex) {
      scene.environment = studioEnvTex;
    }

    const hasRainOrWater = (APP.entities || []).some((e) => e && e.category === 'world_fx' && /rain|water/i.test(e.name || ''));
    const targetWet = hasRainOrWater ? Math.max(cfg.wetness, 0.85) : cfg.wetness;
    sharedUniforms.uNxrWetness.value += (targetWet - sharedUniforms.uNxrWetness.value) * Math.min(1, delta * 2.5);

    if (APP.dirLight) sharedUniforms.uNxrLightDir.value.copy(APP.dirLight.position).normalize();

    const ents = APP.entities || [];
    for (let i = 0; i < ents.length; i++) {
      const e = ents[i];
      if (!e || !e.group || !e.group.parent) continue;
      const cs = e.group.getObjectByName('NXR_ContactShadow');
      if (cs) {
        const lift = Math.max(0, e.group.position.y);
        cs.position.y = -lift + 0.004;
        cs.material.opacity = Math.max(0, 0.72 - lift * 0.95);
        cs.visible = cs.material.opacity > 0.03;
      }
    }
  }

  const unsubFrame = APP.onUpdate ? APP.onUpdate(updateRealism) : null;
  if (!unsubFrame && APP.frameHooks) APP.frameHooks.add(updateRealism);

  APP._inst3RealismCleanup = function () {
    if (typeof unsubFrame === 'function') unsubFrame();
    else APP.frameHooks?.delete(updateRealism);
    if (typeof unsubCreate === 'function') unsubCreate();
    if (typeof unsubMods === 'function') unsubMods();
    if (bounceLight.parent) bounceLight.parent.remove(bounceLight);
    for (const m of heatMirages) { if (m.parent) m.parent.remove(m); m.material.dispose(); m.geometry.dispose(); }
    for (const c of causticDecals) { if (c.parent) c.parent.remove(c); c.material.dispose(); c.geometry.dispose(); }
    baked.detailTex.dispose();
    baked.surfaceNormalTex.dispose();
    baked.glassNormalTex.dispose();
    baked.contactShadowTex.dispose();
    if (studioEnvTex) studioEnvTex.dispose();
  };

    config: cfg,

      cfg.wetness = THREE.MathUtils.clamp(Number(val), 0, 1);

      return cfg.wetness;

    applyMaterialPreset(targetQuery, archetype) {
      const ent = APP.findEntity ? APP.findEntity(targetQuery || 'latest') : null;
      if (!ent || !ent.group) return { ok: false, error: 'Target entity not found.' };

        if (c.isMesh && !c.userData?.nxrDecal && c.material) {
          c.material = upgradeMaterialRealistic(c.material, archetype, c.name, ent.name);
        }
      });
      if (archetype === 'glass') attachCausticsIfRefractive(ent, true);
      return { ok: true, message: `Applied realistic ${archetype} shader to ${ent.name}.` };
    }
  };

    (APP.entities || []).forEach((e) => enhanceEntityRealism(e));
    APP.ui?.toast?.('Lite Realism Engine active!');
    return { ok: true };
  });

  APP.ui?.toast?.('Lite Realism Engine (instruction3.txt) loaded!', 3000);
