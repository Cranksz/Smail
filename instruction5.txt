  function getOutlineBBox(poly) {
    if (!poly || poly.length < 2) return null;
    let minX = 1, minY = 1, maxX = 0, maxY = 0;
    for (const p of poly) {
      if (p.x < minX) minX = p.x; if (p.y < minY) minY = p.y;
      if (p.x > maxX) maxX = p.x; if (p.y > maxY) maxY = p.y;
    }
    return { x: Math.max(0, minX), y: Math.max(0, minY), w: Math.max(0.03, Math.min(1, maxX) - minX), h: Math.max(0.03, Math.min(1, maxY) - minY) };
  }

  function pointInPolygon(x, y, poly) {
    let inside = false;
    for (let i = 0, j = poly.length - 1; i < poly.length; j = i++) {
      const xi = poly[i].x, yi = poly[i].y, xj = poly[j].x, yj = poly[j].y;
      if (((yi > y) !== (yj > y)) && (x < (xj - xi) * (y - yi) / ((yj - yi) || 1e-6) + xi)) inside = !inside;
    }
    return inside;
  }

  function computeInstantReliefDepthTexture() {
    if (!syncWorkCanvasFromSource(true)) return null;
    const S = 96;
    const sc = document.createElement('canvas'); sc.width = S; sc.height = S;
    const sCtx = sc.getContext('2d', { willReadFrequently: true });
    sCtx.drawImage(workCanvas, 0, 0, S, S);
    const px = sCtx.getImageData(0, 0, S, S).data;
    const out = new Uint8Array(S * S * 4);
    for (let y = 0; y < S; y++) {
      for (let x = 0; x < S; x++) {
        const i = y * S + x;
        const lum = (0.299 * px[i * 4] + 0.587 * px[i * 4 + 1] + 0.114 * px[i * 4 + 2]) / 255;
        const ir = y * S + Math.min(S - 1, x + 1), id = Math.min(S - 1, y + 1) * S + x;
        const lumR = (0.299 * px[ir * 4] + 0.587 * px[ir * 4 + 1] + 0.114 * px[ir * 4 + 2]) / 255;
        const lumD = (0.299 * px[id * 4] + 0.587 * px[id * 4 + 1] + 0.114 * px[id * 4 + 2]) / 255;
        const edge = Math.min(1, (Math.abs(lum - lumR) + Math.abs(lum - lumD)) * 2.8);
        const radial = Math.max(0, 1.0 - Math.hypot(x / S - 0.5, y / S - 0.5) * 1.4);
        const v = Math.min(255, Math.max(0, Math.round((lum * 0.48 + edge * 0.65 + radial * 0.32) * 255)));
        out[i * 4] = v; out[i * 4 + 1] = v; out[i * 4 + 2] = v; out[i * 4 + 3] = 255;
      }
    }
    const tex = new THREE.DataTexture(out, S, S, THREE.RGBAFormat);
    tex.minFilter = THREE.LinearFilter; tex.magFilter = THREE.LinearFilter;
    tex.flipY = true;
    tex.needsUpdate = true;
    fbiState.depthSource = 'InstantRelief3D';
    return tex;
  }

  function getOrCreateDepthWorker() {
    if (depthWorker) return depthWorker;
    const workerCode = `
      let pipe = null;
      self.onmessage = async (ev) => {
        const { id, dataUrl } = ev.data || {};
        try {
          if (!pipe) {
            self.postMessage({ id, status: 'progress', msg: 'LOADING DEPTH NET...' });
            const tfm = await import('https://cdn.jsdelivr.net/npm/@huggingface/transformers@3/+esm');
            if (tfm.env) tfm.env.allowLocalModels = false;
            pipe = await tfm.pipeline('depth-estimation', 'onnx-community/depth-anything-v2-small', {
              device: 'wasm',
              progress_callback: (p) => { if (p && typeof p.progress === 'number') self.postMessage({ id, status: 'progress', msg: 'DEPTH MODEL ' + Math.round(p.progress) + '%' }); }
            });
          }
          self.postMessage({ id, status: 'progress', msg: 'INFERENCING AI DEPTH...' });
          const out = await pipe(dataUrl);
          const d = out && out.depth;
          if (!d || !d.data) throw new Error('Empty depth tensor');
          const w = d.width, h = d.height, raw = d.data;
          let mn = Infinity, mx = -Infinity;
          for (let i = 0; i < raw.length; i++) { if (raw[i] < mn) mn = raw[i]; if (raw[i] > mx) mx = raw[i]; }
          const sp = Math.max(1e-5, mx - mn), rgba = new Uint8Array(w * h * 4);
          for (let i = 0; i < w * h; i++) {
            const v = Math.round(((raw[i] - mn) / sp) * 255);
            rgba[i * 4] = v; rgba[i * 4 + 1] = v; rgba[i * 4 + 2] = v; rgba[i * 4 + 3] = 255;
          }
          self.postMessage({ id, status: 'done', width: w, height: h, rgba }, [rgba.buffer]);
        } catch (err) {
          self.postMessage({ id, status: 'error', error: err.message || String(err) });
        }
      };
    `;
    const blobUrl = URL.createObjectURL(new Blob([workerCode], { type: 'application/javascript' }));
    depthWorker = new Worker(blobUrl, { type: 'module' });
    URL.revokeObjectURL(blobUrl);
    return depthWorker;
  }

        pushLog(`Enrolled "${rec.name}" (${rec.id}, 128-D).`);
        APP.ui?.toast?.(`Enrolled ${rec.name} into Watchlist!`, 2600);
      }

      fbiState.faceResults = dets.map((d, idx) => {
        const b = d.detection.box;
        let bestMatch = 'UNKNOWN SUBJECT', bestDist = 999, matchPct = 0;
        for (const entry of faceWatchlist) {
          const dist = faceapi.euclideanDistance(d.descriptor, new Float32Array(entry.descriptor));
          if (dist < bestDist) { bestDist = dist; if (dist < 0.58) { bestMatch = entry.name; matchPct = Math.max(55, Math.min(99, Math.round((1 - dist / 0.85) * 100))); } }
        }
        const exps = d.expressions ? Object.entries(d.expressions).sort((a, b) => b[1] - a[1]) : [];
        return {
          idx, box: { x: b.x / W, y: b.y / H, w: b.width / W, h: b.height / H },
          matchName: bestMatch, matchScore: matchPct,
          distance: bestDist < 10 ? Number(bestDist.toFixed(3)) : null,
          age: Math.round(d.age || 0),
          gender: `${d.gender || ''} (${Math.round((d.genderProbability || 0) * 100)}%)`,
          expression: exps[0] ? `${exps[0][0]} (${Math.round(exps[0][1] * 100)}%)` : 'neutral'
        };
      });

      fbiState.clues = fbiState.faceResults.map((fr, i) => ({
        id: `FACE-0${i + 1}`,
        label: fr.matchScore > 0 ? `${fr.matchName} ${fr.matchScore}%` : `UNKNOWN ~${fr.age}y ${fr.expression.split(' ')[0]}`,
        detail: `${fr.matchName} | Age~${fr.age} | ${fr.gender} | FACS: ${fr.expression}` + (fr.distance !== null ? ` | L2=${fr.distance}` : ''),
        confidence: fr.matchScore || 86,
        severity: fr.matchScore > 0 ? 'high' : 'med',
        bbox: [fr.box.x, fr.box.y, fr.box.w, fr.box.h]
      }));
      fbiState.activeClueIdx = 0;
      fbiState.deductionSummary = fbiState.clues.map((c) => c.detail).join(' // ');
      fbiState.statusText = `FACE RECOG: ${fbiState.faceResults.length} SUBJECT(S)`;
      playUiBeep(980, 0.06);
      return { ok: true, faces: fbiState.faceResults };
    } catch (e) {
      fbiState.statusText = 'FACE RECOG ERROR'; APP.ui?.toast?.(e.message, 4000);
      return { ok: false, error: e.message };
    } finally { fbiState.busy = false; refreshAllWindows(); }
  }

  function updateTracks(detections, nowMs) {
    const W = Math.max(1, workCanvas.width), H = Math.max(1, workCanvas.height);
    const COLS = ['#00f0b4', '#38bdf8', '#fbbf24', '#f472b6', '#a78bfa'];
    const items = (detections || []).map((d) => {
      const bx = d.boundingBox || {}, cat = d.categories?.[0] || {};
      const norm = { x: Math.max(0, (bx.originX || 0) / W), y: Math.max(0, (bx.originY || 0) / H), w: Math.min(1, (bx.width || 40) / W), h: Math.min(1, (bx.height || 40) / H) };
      let tag = 'neutral';
      try {
        const px = workCtx.getImageData(Math.floor(norm.x * W + norm.w * W * 0.25), Math.floor(norm.y * H + norm.h * H * 0.25), Math.max(4, Math.floor(norm.w * W * 0.5)), Math.max(4, Math.floor(norm.h * H * 0.5))).data;
        let r = 0, g = 0, b = 0, c = 0; for (let i = 0; i < px.length; i += 16) { r += px[i]; g += px[i + 1]; b += px[i + 2]; c++; }
        if (c) { r /= c; g /= c; b /= c; if (r > g + 30 && r > b + 30) tag = 'red'; else if (b > r + 30) tag = 'blue'; else if (g > r + 25) tag = 'green'; else if (r < 65 && g < 65 && b < 65) tag = 'dark'; else if (r > 190 && g > 190 && b > 190) tag = 'white'; }
      } catch (_) {}
      return { label: cat.categoryName || 'object', score: Number(cat.score || 0.5), bbox: norm, cx: norm.x + norm.w * 0.5, cy: norm.y + norm.h * 0.5, tag };
    });
    const used = new Set();
    for (const it of items) {
      let best = null, bd = 999;
      for (const t of fbiState.tracks) {
        if (used.has(t.id) || t.label !== it.label) continue;
        const d = Math.hypot(t.cx - it.cx, t.cy - it.cy);
        if (d < 0.22 && d < bd) { bd = d; best = t; }
      }
      if (best) {
        used.add(best.id); best.cx = it.cx; best.cy = it.cy; best.bbox = it.bbox; best.score = it.score; best.tag = it.tag; best.lastSeen = nowMs;
        best.trail.push({ x: it.cx, y: it.cy }); if (best.trail.length > 26) best.trail.shift();
      } else {
        const idNum = fbiState.nextTrackId++;
        const trk = { id: 'TRK-' + String(idNum).padStart(2, '0'), label: it.label, score: it.score, color: COLS[(idNum - 1) % COLS.length], bbox: it.bbox, cx: it.cx, cy: it.cy, tag: it.tag, trail: [{ x: it.cx, y: it.cy }], lastSeen: nowMs };
        used.add(trk.id); fbiState.tracks.push(trk);
      }
    }
    fbiState.tracks = fbiState.tracks.filter((t) => nowMs - t.lastSeen < 1600);
  }

  function runFastSelectiveEnhance() {
    if (!syncWorkCanvasFromSource(true)) return null;
    const W = workCanvas.width, H = workCanvas.height;
    const activePoly = fbiState.outlines[fbiState.outlines.length - 1] || null;
    const bb = activePoly ? getOutlineBBox(activePoly) : { x: 0, y: 0, w: 1, h: 1 };
    const rx = Math.floor(bb.x * W), ry = Math.floor(bb.y * H);
    const rw = Math.max(32, Math.floor(bb.w * W)), rh = Math.max(32, Math.floor(bb.h * H));
    const maxOut = 360;
    const sc = Math.min(2.4, maxOut / Math.max(rw, rh));
    const outW = Math.max(48, Math.round(rw * sc)), outH = Math.max(48, Math.round(rh * sc));
    const outC = document.createElement('canvas'); outC.width = outW; outC.height = outH;
    const oCtx = outC.getContext('2d', { willReadFrequently: true });
    oCtx.imageSmoothingEnabled = true; oCtx.imageSmoothingQuality = 'high';
    oCtx.drawImage(workCanvas, rx, ry, rw, rh, 0, 0, outW, outH);

    const img = oCtx.getImageData(0, 0, outW, outH), src = new Uint8ClampedArray(img.data), dst = img.data;
    let minL = 255, maxL = 0;
    for (let i = 0; i < src.length; i += 16) { const l = 0.299 * src[i] + 0.587 * src[i + 1] + 0.114 * src[i + 2]; if (l < minL) minL = l; if (l > maxL) maxL = l; }
    const span = Math.max(24, maxL - minL);
    for (let y = 1; y < outH - 1; y++) {
      for (let x = 1; x < outW - 1; x++) {
        const idx = (y * outW + x) * 4;
        for (let c = 0; c < 3; c++) {
          const cen = src[idx + c];
          const blur = (src[((y - 1) * outW + x) * 4 + c] + src[((y + 1) * outW + x) * 4 + c] + src[(y * outW + (x - 1)) * 4 + c] + src[(y * outW + (x + 1)) * 4 + c] + cen * 2) / 6;
          const sh = cen + (cen - blur) * 1.8;
          dst[idx + c] = Math.max(0, Math.min(255, Math.round(((sh - minL) / span) * 255 * 0.5 + sh * 0.5)));
        }
      }
    }
    oCtx.putImageData(img, 0, 0);
    return outC;
  }

  function analyzeFingerprintImage(imgOrCanvas) {
    const N = 160, fc = document.createElement('canvas'); fc.width = N; fc.height = N;
    const fCtx = fc.getContext('2d', { willReadFrequently: true }); fCtx.drawImage(imgOrCanvas, 0, 0, N, N);
    const d = fCtx.getImageData(0, 0, N, N).data, gray = new Float32Array(N * N);
    for (let i = 0; i < N * N; i++) gray[i] = 0.299 * d[i * 4] + 0.587 * d[i * 4 + 1] + 0.114 * d[i * 4 + 2];
    const bin = new Uint8Array(N * N);
    for (let y = 3; y < N - 3; y++) for (let x = 3; x < N - 3; x++) {
      let sum = 0; for (let dy = -3; dy <= 3; dy++) for (let dx = -3; dx <= 3; dx++) sum += gray[(y + dy) * N + (x + dx)];
      bin[y * N + x] = gray[y * N + x] < (sum / 49) - 4 ? 1 : 0;
    }
    const minutiae = []; let archC = 0, whorlC = 0;
    for (let y = 14; y < N - 14; y += 2) for (let x = 14; x < N - 14; x += 2) {
      if (!bin[y * N + x]) continue;
      const p = [bin[(y - 1) * N + x], bin[(y - 1) * N + (x + 1)], bin[y * N + (x + 1)], bin[(y + 1) * N + (x + 1)], bin[(y + 1) * N + x], bin[(y + 1) * N + (x - 1)], bin[y * N + (x - 1)], bin[(y - 1) * N + (x - 1)]];
      let cn = 0, nb = 0; for (let k = 0; k < 8; k++) { nb += p[k]; cn += Math.abs(p[k] - p[(k + 1) % 8]); }
      if (nb >= 1 && nb <= 5 && (cn * 0.5 === 1 || cn * 0.5 === 3) && !minutiae.some((m) => Math.hypot(m.x - x / N, m.y - y / N) < 0.055)) {
        minutiae.push({ x: x / N, y: y / N, type: cn * 0.5 === 1 ? 'ending' : 'bifurcation' });
      }
      if (Math.abs(gray[y * N + (x + 1)] - gray[y * N + (x - 1)]) > 18 && Math.abs(gray[(y + 1) * N + x] - gray[(y - 1) * N + x]) > 18) whorlC++; else archC++;
    }
    const endings = minutiae.filter((m) => m.type === 'ending').length, bifs = minutiae.filter((m) => m.type === 'bifurcation').length;
    const ratio = whorlC / Math.max(1, archC + whorlC);
    return { pattern: ratio > 0.34 ? 'Whorl (Spiral Core)' : ratio > 0.18 ? 'Ulnar/Radial Loop' : 'Plain/Tented Arch', minutiae: minutiae.slice(0, 44), endings, bifurcations: bifs, ridgeScore: ratio };
  }

  function createSyntheticDbPrint(seed, style) {
    const c = document.createElement('canvas'); c.width = 160; c.height = 160;
    const ctx = c.getContext('2d'); ctx.fillStyle = '#f8fafc'; ctx.fillRect(0, 0, 160, 160);
    ctx.strokeStyle = '#0f172a'; ctx.lineWidth = 2.2;
    for (let r = 8; r < 76; r += 6) {
      ctx.beginPath();
      for (let a = 0; a <= Math.PI * 2; a += 0.15) {
        const w = Math.sin(a * 3 + seed + r * 0.1) * 3;
        const rx = style === 'arch' ? 80 + (a - Math.PI) * 24 : 80 + Math.cos(a) * (r + w) * (style === 'loop' ? 0.85 : 1.0);
        const ry = style === 'arch' ? 130 - r - Math.cos(a * 0.5) * 18 : 84 + Math.sin(a) * (r + w) * 1.12;
        if (a === 0) ctx.moveTo(rx, ry); else ctx.lineTo(rx, ry);
      }
      ctx.stroke();
    }
    return c;
  }

  async function runForensicAnalyse(forcedDomain, customQuestion) {
    if (!fbiState.open) setWorkstationOpen(true);
    if (fbiState.busy) { APP.ui?.toast?.('Scan running...', 1400); return { ok: false }; }
    if (fbiState.evidenceType === 'none' && !customQuestion) { requireEvidenceOrPick(); return { ok: false }; }
    if (!APP.ai?.chat) { APP.ui?.toast?.('AI module not loaded.', 3000); return { ok: false }; }

    pushUndoSnapshot('before_analyse');
    fbiState.pickerOpen = false; fbiState.busy = true;
    if (forcedDomain) fbiState.mode = forcedDomain;
    fbiState.statusText = 'ANALYSING // CV + SHERLOCK AI';
    refreshAllWindows();
    await new Promise((r) => setTimeout(r, 20));

    try {
      await runMediaPipeOnFrame(false);
      updateDepthShaderMaskAndTexture();

      const activePoly = fbiState.outlines[fbiState.outlines.length - 1] || null;
      const b64 = getCroppedOutlineBase64(activePoly);
      const kbText = getRelevantKB((forcedDomain || '') + ' ' + (customQuestion || ''));

      const cvLines = [];
      if (fbiState.cvDetections.length) cvLines.push('OBJECTS: ' + fbiState.cvDetections.map((d) => `${d.categories?.[0]?.categoryName || 'obj'}(${Math.round((d.categories?.[0]?.score || 0) * 100)}%)`).join(', '));
      if (fbiState.cvPoses.length) cvLines.push(`POSES: ${fbiState.cvPoses.length} person(s)`);
      if (fbiState.faceResults.length) cvLines.push('FACES: ' + fbiState.faceResults.map((f) => `${f.matchName}(${f.matchScore}%) ${f.expression}`).join(', '));
      if (activePoly) cvLines.push('FOCUS: User drew a freehand outline mask isolating this target part.');

      const sysPrompt = [
        'You are the NEXUS-FBI Sherlock Holmes Forensic Deduction Engine.',
        'Inspect the evidence (especially the highlighted user-drawn region if masked) using detection.txt rules to uncover hidden secrets, body language, graphology, physiognomy, or crime scene evidence.',
        'Return ONLY JSON: { "subject_title": "...", "domain": "body|graphology|crime|physiognomy|mystery", "deduction_summary": "concise Sherlock deduction (max 4 sentences)", "clues": [{ "id": "CLUE-01", "label": "2-5 word tape", "detail": "...", "confidence": 90, "severity": "high|med|low", "bbox": [x,y,w,h] }], "links": [{ "from": "A", "to": "B", "relation": "..." }] }',
        '', kbText
      ].join('\n');

      const res = await APP.ai.chat({
        systemPrompt: sysPrompt,
        prompt: (customQuestion ? `QUESTION: ${customQuestion}\n` : `DOMAIN: ${forcedDomain || 'Full Multi-Domain Sherlock Analysis'}\n`) + cvLines.join('\n'),
        imageB64: fbiState.evidenceType !== 'none' ? b64 : undefined,
        imageMime: 'image/jpeg'
      });

      if (!res?.ok) throw new Error(res?.error || 'AI analysis failed.');
      let parsed = null;
      try { const m = String(res.text || '').match(/\{[\s\S]*\}/); if (m) parsed = JSON.parse(m[0]); } catch (_) {}

      if (parsed && Array.isArray(parsed.clues)) {
        const bb = activePoly ? getOutlineBBox(activePoly) : { x: 0, y: 0, w: 1, h: 1 };
        fbiState.clues = parsed.clues.map((c, idx) => {
          const rawB = Array.isArray(c.bbox) && c.bbox.length === 4 ? c.bbox.map((v) => Math.max(0, Math.min(1, Number(v) || 0))) : [0.2, 0.2, 0.4, 0.4];
          return {
            id: c.id || `CLUE-0${idx + 1}`, label: String(c.label || 'Clue').slice(0, 32),
            detail: String(c.detail || ''), confidence: Number(c.confidence || 85), severity: c.severity || 'med',
            bbox: [bb.x + rawB[0] * bb.w, bb.y + rawB[1] * bb.h, rawB[2] * bb.w, rawB[3] * bb.h]
          };
        });
        fbiState.deductionSummary = parsed.deduction_summary || res.text;
        fbiState.hypothesisLinks = Array.isArray(parsed.links) ? parsed.links : [];
        fbiState.activeClueIdx = 0;
        fbiState.lastScanSubject = parsed.subject_title || fbiState.evidenceTitle;
      } else {
        fbiState.deductionSummary = res.text || 'Analysis complete.';
      }

      fbiState.statusText = 'SHERLOCK SCAN COMPLETE';
      playUiBeep(1040, 0.06);
      APP.ui?.toast?.(`Found ${fbiState.clues.length} clues!`, 2800);
      if (typeof APP.speakTTS === 'function' && fbiState.deductionSummary) APP.speakTTS(fbiState.deductionSummary.slice(0, 240));
      return { ok: true, summary: fbiState.deductionSummary };
    } catch (e) {
      fbiState.statusText = 'SCAN ERROR'; APP.ui?.toast?.(e.message, 4500);
      return { ok: false, error: e.message };
    } finally { fbiState.busy = false; refreshAllWindows(); }
  }

  function uvToNormalizedImage(uv) {
    const cx = uv.x * winMain.canvas.width, cy = (1 - uv.y) * winMain.canvas.height;
    return {
      x: Math.max(0, Math.min(1, (cx - lastImgRect.x) / Math.max(1, lastImgRect.w))),
      y: Math.max(0, Math.min(1, (cy - lastImgRect.y) / Math.max(1, lastImgRect.h)))
    };

  function triggerCmdAtUV(uv) {
    const now = performance.now();
    if (now - lastPokeTs < 180) return;
    lastPokeTs = now;
    for (const b of cmdButtons) {
      if (uv.x >= b.u0 && uv.x <= b.u1 && uv.y >= b.v0 && uv.y <= b.v1) {
        fbiState.flashBtnId = b.id; fbiState.flashUntil = now + 180;
        playUiBeep(880, 0.04);
        redrawCmdDeck();
        b.action();
        refreshAllWindows();
        break;
      }
    }
  }

      if (uv.x >= b.u0 && uv.x <= b.u1 && uv.y >= b.v0 && uv.y <= b.v1) {

  function onPtrDown(e) {
    if (!fbiState.open || e.button !== 0 || !APP.camera) return;
    const el = APP.renderer?.domElement; if (e.target !== el) return;
    const rect = el.getBoundingClientRect();
    dNdc.x = ((e.clientX - rect.left) / Math.max(1, rect.width)) * 2 - 1;
    dNdc.y = -((e.clientY - rect.top) / Math.max(1, rect.height)) * 2 + 1;
    dRay.setFromCamera(dNdc, APP.camera);

    // 1. Grab any window's top drag bar
    const bh = dRay.intersectObjects(miniWindows.map((w) => w.dragBar), false);
    if (bh.length) {
      const win = miniWindows.find((w) => w.dragBar === bh[0].object);
      if (win) { startDraggingWin(win, bh[0].point); return; }
    }

    // 2. Allow dragging side/top non-button windows directly from their screen!
    const sideHits = dRay.intersectObjects([winRadar.screen, winScan.screen, winDed.screen, winComp.screen], false);
    if (sideHits.length) {
      const win = miniWindows.find((w) => w.screen === sideHits[0].object);
      if (win) { startDraggingWin(win, sideHits[0].point); return; }
    }

    // 3. Freehand outline drawing on winMain
    if (!fbiState.pickerOpen && fbiState.evidenceType !== 'none' && fbiState.drawOutlineMode) {
      const mh = dRay.intersectObject(winMain.screen, false);
      if (mh.length && mh[0].uv) {
        fbiState.activeStroke = [uvToNormalizedImage(mh[0].uv)];
        redrawOutlineOverlay();
      }
    }
  }

  function onPtrMove(e) {
    if (!fbiState.open || !APP.camera) return;
    const el = APP.renderer?.domElement; if (!el) return;
    const rect = el.getBoundingClientRect();
    dNdc.x = ((e.clientX - rect.left) / Math.max(1, rect.width)) * 2 - 1;
    dNdc.y = -((e.clientY - rect.top) / Math.max(1, rect.height)) * 2 + 1;
    dRay.setFromCamera(dNdc, APP.camera);

    if (activeDragWin) {
      const pt = new THREE.Vector3();
      if (dRay.ray.intersectPlane(dragPlane, pt)) {
        wsRoot.worldToLocal(pt);
        activeDragWin.group.position.copy(pt).add(dragOff);
        updateDynamicConnectors();
      }
      return;
    }

    if (fbiState.activeStroke) {
      const mh = dRay.intersectObject(winMain.screen, false);
      if (mh.length && mh[0].uv) {
        const pt = uvToNormalizedImage(mh[0].uv), last = fbiState.activeStroke[fbiState.activeStroke.length - 1];
        if (Math.hypot(pt.x - last.x, pt.y - last.y) > 0.012) {
          fbiState.activeStroke.push(pt);
          redrawOutlineOverlay(); // 0.05ms overlay update, ZERO lag!
        }
      }
      return;
    }

    const ch = dRay.intersectObject(winCmd.screen, false);
    let nh = null;
    if (ch.length && ch[0].uv) for (const b of cmdButtons) if (ch[0].uv.x >= b.u0 && ch[0].uv.x <= b.u1 && ch[0].uv.y >= b.v0 && ch[0].uv.y <= b.v1) { nh = b.id; break; }
    if (nh !== hoveredCmdId) { hoveredCmdId = nh; redrawCmdDeck(); }

    if (fbiState.pickerOpen || fbiState.evidenceType === 'none') {
      const mh = dRay.intersectObject(winMain.screen, false);
      let np = null;
      if (mh.length && mh[0].uv) for (const b of pickerButtons) if (mh[0].uv.x >= b.u0 && mh[0].uv.x <= b.u1 && mh[0].uv.y >= b.v0 && mh[0].uv.y <= b.v1) { np = b.id; break; }
      if (np !== hoveredPickerId) { hoveredPickerId = np; redrawMainViewport(); }
    }
  }

  function updateForensicSuite(dt, elapsed) {
    if (!fbiState.open || !wsRoot.visible) return;
    const now = elapsed || (performance.now() * 0.001);

    mainLineUniforms.uTime.value = now; mainLineUniforms.uBusy.value = fbiState.busy ? 1.0 : 0.0;
    radarShaderUniforms.uTime.value = now; radarShaderUniforms.uBusy.value = fbiState.busy ? 1.0 : 0.0;
    scanShaderUniforms.uTime.value = now;

    scanModelSpinGroup.rotation.y = Math.sin(now * 0.65) * 0.68;
    holoRingA.rotation.z = now * 0.5;
    holoSlice.position.y = Math.sin(now * 2.2) * 0.12;

    if (fbiState.flashBtnId && performance.now() > fbiState.flashUntil) { fbiState.flashBtnId = null; redrawCmdDeck(); }

    let showDot = false;
    if (APP.xrMode !== 'none') {
      const hands = APP.hands || {};
      if (dragHandKey && activeDragWin) {
        const dh = hands[dragHandKey];
        if (dh && dh.active && (dh.trigger || dh.gripping || dh.pinching)) {
          const targetPt = new THREE.Vector3();
          if (dh.isHand && dh.fingerTips?.index) targetPt.copy(dh.fingerTips.index);
          else if (dh.rayOrigin && dh.rayDir) targetPt.copy(dh.rayOrigin).addScaledVector(tmpDir.copy(dh.rayDir).normalize(), dragDist);
          wsRoot.worldToLocal(targetPt);
          activeDragWin.group.position.lerp(targetPt.add(dragOff), 0.55);
          updateDynamicConnectors();
        } else { dragHandKey = null; activeDragWin = null; }
      } else {
        for (const [hk, hand] of Object.entries(hands)) {
          if (!hand || !hand.active) continue;
          if (hand.rayOrigin && hand.rayDir) {
            tmpDir.copy(hand.rayDir).normalize(); dRay.set(hand.rayOrigin, tmpDir);
            const dragTargets = [...miniWindows.map((w) => w.dragBar), winRadar.screen, winScan.screen, winDed.screen, winComp.screen];
            const bh = dRay.intersectObjects(dragTargets, false);
            if (bh.length) {
              showDot = true; vrCursorDot.position.copy(bh[0].point); wsRoot.worldToLocal(vrCursorDot.position);
              if (hand.justTriggered || hand.justGripped || hand.justPinched) {
                activeDragWin = miniWindows.find((w) => w.dragBar === bh[0].object || w.screen === bh[0].object);
                if (activeDragWin) { dragHandKey = hk; dragDist = bh[0].distance; wsRoot.worldToLocal(dragOff.copy(bh[0].point)); dragOff.sub(activeDragWin.group.position).negate(); break; }
              }
            }
            const sh = dRay.intersectObjects([winCmd.screen, winMain.screen], false);
            if (sh.length && sh[0].uv) {
              showDot = true; vrCursorDot.position.copy(sh[0].point); wsRoot.worldToLocal(vrCursorDot.position);
              if (sh[0].object === winCmd.screen) {
                for (const b of cmdButtons) if (sh[0].uv.x >= b.u0 && sh[0].uv.x <= b.u1 && sh[0].uv.y >= b.v0 && sh[0].uv.y <= b.v1 && hoveredCmdId !== b.id) { hoveredCmdId = b.id; redrawCmdDeck(); break; }
              } else if (sh[0].object === winMain.screen && !fbiState.pickerOpen && fbiState.evidenceType !== 'none' && fbiState.drawOutlineMode) {
                if (hand.trigger || hand.pinching) {
                  const pt = uvToNormalizedImage(sh[0].uv);
                  if (!fbiState.activeStroke) fbiState.activeStroke = [pt];
                  else {
                    const last = fbiState.activeStroke[fbiState.activeStroke.length - 1];
                    if (Math.hypot(pt.x - last.x, pt.y - last.y) > 0.015) { fbiState.activeStroke.push(pt); redrawOutlineOverlay(); }
                  }
                } else if (fbiState.activeStroke) finishActiveOutline();
              }
            }
          }
        }
      }
    }
    vrCursorDot.visible = showDot;

    if (fbiState.evidenceType === 'video' && fbiState.sourceVideo && !fbiState.sourceVideo.paused && !fbiState.pickerOpen) {
      const nowMs = performance.now();
      if (fbiState.trackingActive && nowMs - fbiState.lastVideoDetectTs > 200) {
        fbiState.lastVideoDetectTs = nowMs;
        runMediaPipeOnFrame(true);
      }
      if (nowMs - fbiState.lastVideoDrawTs > 85) {
        fbiState.lastVideoDrawTs = nowMs;
        redrawMainViewport();
      }
    }
  }

    if (wsRoot.parent) wsRoot.parent.remove(wsRoot);
    if (drawerEl?.parentNode) drawerEl.parentNode.removeChild(drawerEl);
    domFileImg.remove(); domFileVid.remove(); domFilePrint.remove();
  };

  APP.detection = {
    reload: loadDetectionTxt,
    setKnowledgeText: (txt) => { parseKnowledgeBase(txt); pushLog('detection.txt synced.'); refreshAllWindows(); },
    getSections: () => kbState.sections
  };

  APP.forensics = {
    open: () => setWorkstationOpen(true),
    close: () => setWorkstationOpen(false),
    analyse: runForensicAnalyse,
    enhanceBlur: runEnhanceBlur,
    depth3D: runAIDepthInWorker,
    faceRecog: runRealFaceRecognition,
    trackPerson: togglePersonTracking,
    matchFingerprint: runFingerprintMatch,
    removeDisguise: runFaceDisguiseRemoval,
    undo: performUndo,
    reset: performReset,
    loadSnapshot: captureCameraSnapshot,
    runCommand: async function (text) {
      const low = String(text || '').toLowerCase().trim();
      setWorkstationOpen(true);
      if (!low || /^(fbi|forensic|forensics|detective|sherlock|open lab|investigation|تحقيق)$/i.test(low)) return { ok: true, message: 'FBI Studio ready.' };
      if (/undo|تراجع/.test(low)) return performUndo();
      if (/reset|اعادة/.test(low)) return performReset();
      if (/enhance|unblur|blur|وضح/.test(low)) return await runEnhanceBlur(text);
      if (/depth|3d scan/.test(low)) return runAIDepthInWorker();
      if (/face recog|recognize|identify face/.test(low)) return await runRealFaceRecognition(null);
      if (/fingerprint|بصمة/.test(low)) return await runFingerprintMatch();
      if (/mustache|beard|glasses|disguise|qwen edit|بدون شارب|بدون نظار/.test(low)) return await runFaceDisguiseRemoval(text);
      if (/track|find suspect|تتبع/.test(low)) return await togglePersonTracking(low.replace(/.*?(track|find)\s*(that|the|a)?\s*(person|suspect|guy)?\s*/i, '').trim());
      return await runForensicAnalyse('auto', text);
    }
  };

  APP.registerBotTool?.({
    name: 'forensic_investigate',
    description: 'Run FBI & Sherlock Holmes forensic tools: open, analyse, enhance_blur, depth_3d, face_recog, track_person, match_fingerprint, face_edit, undo, reset.',
    parameters: { type: 'object', properties: { action: { type: 'string' }, question: { type: 'string' } }, required: ['action'] }
  }, async (a) => {
    const act = String(a?.action || 'open').toLowerCase();
    setWorkstationOpen(true);
    if (act === 'open') return { ok: true };
    if (act === 'undo') return performUndo();
    if (act === 'reset') return performReset();
    if (act === 'depth_3d') return runAIDepthInWorker();
    if (act === 'face_recog') return await runRealFaceRecognition(null);
    if (act === 'enhance_blur') return await runEnhanceBlur(a.question);
    if (act === 'track_person') return await togglePersonTracking(a.question || '');
    if (act === 'match_fingerprint') return await runFingerprintMatch();
    if (act === 'face_edit') return await runFaceDisguiseRemoval(a.question);
    return await runForensicAnalyse(null, a.question);
  });

  loadDetectionTxt();
  APP.registerAbilityHandler?.('fbi_forensic_suite', async (t) => APP.forensics.runCommand(t));
  if (typeof command === 'string' && command.trim()) setWorkstationOpen(true);
})();
