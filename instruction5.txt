  function computeBBoxColorSignature(bbox) {
    try {
      const x = Math.max(0, Math.floor(bbox.x * workCanvas.width)), y = Math.max(0, Math.floor(bbox.y * workCanvas.height));
      const w = Math.max(4, Math.min(workCanvas.width - x, Math.floor(bbox.w * workCanvas.width))), h = Math.max(4, Math.min(workCanvas.height - y, Math.floor(bbox.h * workCanvas.height)));
      const data = workCtx.getImageData(x, y, w, h).data;
      let r = 0, g = 0, b = 0, cnt = 0;
      const step = Math.max(1, Math.floor((w * h) / 180));
      for (let i = 0; i < data.length; i += step * 4) { r += data[i]; g += data[i + 1]; b += data[i + 2]; cnt++; }
      if (!cnt) return { tag: 'neutral' };
      r = Math.round(r / cnt); g = Math.round(g / cnt); b = Math.round(b / cnt);
      let tag = 'neutral';
      if (r > g + 35 && r > b + 35) tag = 'red';
      else if (b > r + 35 && b > g + 25) tag = 'blue';
      else if (g > r + 25 && g > b + 25) tag = 'green';
      else if (r > 180 && g > 170 && b < 110) tag = 'yellow';
      else if (r > 195 && g > 195 && b > 195) tag = 'white';
      else if (r < 65 && g < 65 && b < 65) tag = 'dark';
      return { r, g, b, tag };
    } catch (_) { return { tag: 'unknown' }; }
  }

  function updateTrackersFromDetections(detections, nowMs) {
    const W = Math.max(1, workCanvas.width), H = Math.max(1, workCanvas.height);
    const items = (detections || []).map((d) => {
      const bx = d.boundingBox || {}, cat = d.categories?.[0] || {};
      const norm = { x: Math.max(0, (bx.originX || 0) / W), y: Math.max(0, (bx.originY || 0) / H), w: Math.min(1, (bx.width || 40) / W), h: Math.min(1, (bx.height || 40) / H) };
      return { label: cat.categoryName || 'object', score: Number(cat.score || 0.5), bbox: norm, cx: norm.x + norm.w * 0.5, cy: norm.y + norm.h * 0.5, col: computeBBoxColorSignature(norm) };
    });
    const usedTracks = new Set();
    const TRACK_COLORS = ['#00f0b4', '#38bdf8', '#fbbf24', '#f472b6', '#a78bfa', '#fb7185'];
    for (const item of items) {
      let bestTrk = null, bestScore = 999;
      for (const t of fbiState.tracks) {
        if (usedTracks.has(t.id) || t.label !== item.label) continue;
        const dist = Math.hypot(t.cx - item.cx, t.cy - item.cy);
        if (dist < 0.22 && dist < bestScore) { bestScore = dist; bestTrk = t; }
      }
      if (bestTrk) {
        usedTracks.add(bestTrk.id);
        bestTrk.cx = item.cx; bestTrk.cy = item.cy; bestTrk.bbox = item.bbox; bestTrk.score = item.score; bestTrk.col = item.col; bestTrk.lastSeen = nowMs;
        bestTrk.trail.push({ x: item.cx, y: item.cy }); if (bestTrk.trail.length > 28) bestTrk.trail.shift();
      } else {
        const idNum = fbiState.nextTrackId++;
        const trk = { id: 'TRK-' + String(idNum).padStart(2, '0'), label: item.label, score: item.score, color: TRACK_COLORS[(idNum - 1) % TRACK_COLORS.length], bbox: item.bbox, cx: item.cx, cy: item.cy, col: item.col, trail: [{ x: item.cx, y: item.cy }], lastSeen: nowMs };
        usedTracks.add(trk.id); fbiState.tracks.push(trk);
      }
    }
    fbiState.tracks = fbiState.tracks.filter((t) => nowMs - t.lastSeen < 1800);
  }

  async function runMediaPipeOnCurrentFrame(isVideoStream = false) {
    if (!syncWorkCanvasFromSource()) return;
    const mode = isVideoStream ? 'VIDEO' : 'IMAGE', ts = Math.max(1, Math.round(performance.now()));

        updateTrackersFromDetections(fbiState.cvDetections, ts);

    } catch (_) {}
    try {
      const poseD = await getPoseLandmarker(mode);
      if (poseD) {
        const res = isVideoStream ? poseD.detectForVideo(workCanvas, ts) : poseD.detect(workCanvas);
        fbiState.cvPoses = (res?.landmarks || []).map((lm, i) => ({ landmarks: lm, worldLandmarks: res.worldLandmarks?.[i] || null }));
      }
    } catch (_) {}
    if (!isVideoStream || fbiState.mode === 'physiognomy') {
      try {
        const faceD = await getFaceLandmarker(mode);
        if (faceD) {
          const res = isVideoStream ? faceD.detectForVideo(workCanvas, ts) : faceD.detect(workCanvas);
          fbiState.cvFaces = (res?.faceLandmarks || []).map((lm, i) => ({ landmarks: lm, blendshapes: res.faceBlendshapes?.[i]?.categories || [] }));
        }
      } catch (_) {}
    }
  }

  function computeGraphologyMetrics() {
    if (!syncWorkCanvasFromSource()) return null;
    const W = workCanvas.width, H = workCanvas.height;
    try {
      const data = workCtx.getImageData(0, 0, W, H).data;
      let darkPixels = 0, leftY = 0, leftCnt = 0, rightY = 0, rightCnt = 0, pressureHeavy = 0, edgeSum = 0;
      for (let y = 1; y < H - 1; y += 2) {
        for (let x = 1; x < W - 1; x += 2) {
          const i = (y * W + x) * 4, lum = 0.299 * data[i] + 0.587 * data[i + 1] + 0.114 * data[i + 2];
          if (lum < 135) {
            darkPixels++; if (lum < 65) pressureHeavy++;
            if (x < W * 0.4) { leftY += y; leftCnt++; } else if (x > W * 0.6) { rightY += y; rightCnt++; }
          }
          const ir = (y * W + (x + 1)) * 4;
          edgeSum += Math.abs(lum - (0.299 * data[ir] + 0.587 * data[ir + 1] + 0.114 * data[ir + 2]));
        }
      }
      return {
        darkPixels,

  function analyzeFingerprintImage(imgOrCanvas) {
    const N = 160, fc = document.createElement('canvas'); fc.width = N; fc.height = N;
    const fCtx = fc.getContext('2d', { willReadFrequently: true });
    fCtx.drawImage(imgOrCanvas, 0, 0, N, N);
    const d = fCtx.getImageData(0, 0, N, N).data, gray = new Float32Array(N * N);
    for (let i = 0; i < N * N; i++) gray[i] = 0.299 * d[i * 4] + 0.587 * d[i * 4 + 1] + 0.114 * d[i * 4 + 2];
    const bin = new Uint8Array(N * N);
    for (let y = 3; y < N - 3; y++) {
      for (let x = 3; x < N - 3; x++) {
        let sum = 0; for (let dy = -3; dy <= 3; dy++) for (let dx = -3; dx <= 3; dx++) sum += gray[(y + dy) * N + (x + dx)];
        bin[y * N + x] = gray[y * N + x] < (sum / 49) - 4 ? 1 : 0;
      }
    }
    const minutiae = []; let archCount = 0, whorlCount = 0;
    for (let y = 14; y < N - 14; y += 2) {
      for (let x = 14; x < N - 14; x += 2) {
        if (!bin[y * N + x]) continue;
        const p = [bin[(y - 1) * N + x], bin[(y - 1) * N + (x + 1)], bin[y * N + (x + 1)], bin[(y + 1) * N + (x + 1)], bin[(y + 1) * N + x], bin[(y + 1) * N + (x - 1)], bin[y * N + (x - 1)], bin[(y - 1) * N + (x - 1)]];
        let cn = 0, nb = 0; for (let k = 0; k < 8; k++) { nb += p[k]; cn += Math.abs(p[k] - p[(k + 1) % 8]); }
        cn *= 0.5;
        if (nb >= 1 && nb <= 5 && (cn === 1 || cn === 3) && !minutiae.some((m) => Math.hypot(m.x - x / N, m.y - y / N) < 0.055)) {
          minutiae.push({ x: x / N, y: y / N, type: cn === 1 ? 'ending' : 'bifurcation' });
        }
        const gx = gray[y * N + (x + 1)] - gray[y * N + (x - 1)], gy = gray[(y + 1) * N + x] - gray[(y - 1) * N + x];
        if (Math.abs(gx) > 18 && Math.abs(gy) > 18) whorlCount++; else archCount++;
      }
    }
    const endings = minutiae.filter((m) => m.type === 'ending').length, bifs = minutiae.filter((m) => m.type === 'bifurcation').length;
    const ratio = whorlCount / Math.max(1, archCount + whorlCount);
    const pattern = ratio > 0.34 ? 'Whorl (Spiral Core)' : ratio > 0.18 ? 'Ulnar/Radial Loop' : 'Plain/Tented Arch';
    return { pattern, minutiae: minutiae.slice(0, 42), endings, bifurcations: bifs, ridgeScore: ratio };
  }

            const lumR = (0.299 * px[ir * 4] + 0.587 * px[ir * 4 + 1] + 0.114 * px[ir * 4 + 2]) / 255;
            const lumD = (0.299 * px[id * 4] + 0.587 * px[id * 4 + 1] + 0.114 * px[id * 4 + 2]) / 255;
            edge = Math.min(1, (Math.abs(lum - lumR) + Math.abs(lum - lumD)) * 3.2);
          }
          const depthZ = (lum * 0.045 + edge * 0.042) * (0.35 + radial * 0.85);
          pAttr.setZ(i, depthZ);
          if (edge > 0.16 || (i % 2 === 0 && lum > 0.18)) {
            const vx = pAttr.getX(i), vy = pAttr.getY(i);
            ptVerts.push(vx, vy, depthZ + 0.004, vx, vy, -depthZ - 0.004);
            const c = edge > 0.35 ? [0.0, 0.96, 0.72] : [0.22, 0.74, 0.98];
            ptCols.push(c[0], c[1], c[2], c[0], c[1], c[2]);
          }
        }
      }
      planeGeo.computeVertexNormals();
      const wireMat = new THREE.MeshBasicMaterial({ color: 0x0ea5e9, wireframe: true, transparent: true, opacity: 0.34 });
      const frontMesh = new THREE.Mesh(planeGeo, wireMat), backMesh = new THREE.Mesh(planeGeo, wireMat);
      backMesh.scale.z = -1;
      scanModelSpinGroup.add(frontMesh, backMesh);

      if (ptVerts.length) {
        const ptGeo = new THREE.BufferGeometry();
        ptGeo.setAttribute('position', new THREE.Float32BufferAttribute(ptVerts, 3));
        ptGeo.setAttribute('color', new THREE.Float32BufferAttribute(ptCols, 3));
        scanModelSpinGroup.add(new THREE.Points(ptGeo, new THREE.PointsMaterial({ size: 0.006, vertexColors: true, transparent: true, opacity: 0.92, blending: THREE.AdditiveBlending, depthWrite: false })));
      }
    }

    if ((low.includes('body') || fbiState.cvPoses.length) && fbiState.cvPoses[0]?.worldLandmarks) {
      const wlm = fbiState.cvPoses[0].worldLandmarks;
      const POSE_PAIRS = [[11, 12], [11, 23], [12, 24], [23, 24], [11, 13], [13, 15], [12, 14], [14, 16], [23, 25], [25, 27], [24, 26], [26, 28], [0, 11], [0, 12]];
      const linePts = [];
      for (const [a, b] of POSE_PAIRS) if (wlm[a] && wlm[b]) linePts.push(new THREE.Vector3(wlm[a].x * 0.16, -wlm[a].y * 0.16, wlm[a].z * 0.16), new THREE.Vector3(wlm[b].x * 0.16, -wlm[b].y * 0.16, wlm[b].z * 0.16));
      if (linePts.length) {
        const skel = new THREE.LineSegments(new THREE.BufferGeometry().setFromPoints(linePts), new THREE.LineBasicMaterial({ color: 0x00f0b4, transparent: true, opacity: 0.95 }));
        skel.position.z = 0.05; scanModelSpinGroup.add(skel);
      }
    } else if (low.includes('fingerprint')) {
      for (let r = 0.025; r <= 0.11; r += 0.017) {
        const ring = new THREE.Mesh(new THREE.TorusGeometry(r, 0.0022, 6, 32), new THREE.MeshBasicMaterial({ color: r < 0.06 ? 0x00f0b4 : 0x38bdf8, transparent: true, opacity: 0.78 }));
        ring.position.z = 0.055 + (0.11 - r) * 0.25; scanModelSpinGroup.add(ring);
      }
    } else {
      const cage = new THREE.LineSegments(new THREE.EdgesGeometry(new THREE.BoxGeometry(0.24, 0.20, 0.11)), new THREE.LineBasicMaterial({ color: 0x00f0b4, transparent: true, opacity: 0.55 }));
      scanModelSpinGroup.add(cage);
    }
    redrawScanBackdrop();
  }

  function drawMiniTape(ctx, x1, y1, tx, ty, idStr, textStr, accentCol, isActive) {
    const label = `${idStr} // ${String(textStr || '').toUpperCase().slice(0, 32)}`;

  function cycleClue(delta = 1) {
    if (!fbiState.clues.length) { APP.ui?.toast?.('No clues yet — run Analyse first.', 2000); return; }

  async function runForensicAnalyse(forcedDomain, customQuestion) {
    if (!fbiState.open) setWorkstationOpen(true);
    if (fbiState.busy) { APP.ui?.toast?.('Scan already running...', 1600); return { ok: false }; }
    if (fbiState.evidenceType === 'none' && !customQuestion) {
      requireEvidenceOrPick('Load an image/video or click a Case in Evidence Bay first!');
      return { ok: false };
    }
    if (!APP.ai?.chat) { APP.ui?.toast?.('AI module not loaded.', 3500); return { ok: false }; }

    fbiState.pickerOpen = false;
    fbiState.busy = true;
    if (forcedDomain) fbiState.mode = forcedDomain;
    fbiState.statusText = 'SCANNING // MEDIAPIPE + SHERLOCK AI';
    refreshAllWindows();

    try {
      await runMediaPipeOnCurrentFrame(false);
      const graphMetrics = computeGraphologyMetrics();
      const kbText = getRelevantKB((forcedDomain || '') + ' ' + (customQuestion || ''));

      const cvLines = [];
      if (fbiState.cvDetections.length) {
        cvLines.push('OBJECTS: ' + fbiState.cvDetections.map((d) => {
          const cat = d.categories?.[0], b = d.boundingBox || {};
          return `${cat?.categoryName || 'obj'}(${Math.round((cat?.score || 0) * 100)}%)@[${(b.originX / workCanvas.width).toFixed(2)},${(b.originY / workCanvas.height).toFixed(2)},${(b.width / workCanvas.width).toFixed(2)},${(b.height / workCanvas.height).toFixed(2)}]`;
        }).join(', '));
      }
      if (fbiState.cvPoses.length) {
        const p = fbiState.cvPoses[0].landmarks;
        cvLines.push(`POSE: ${fbiState.cvPoses.length} person(s), shoulder_asym=${p[11] && p[12] ? Math.abs(p[11].y - p[12].y).toFixed(3) : 'n/a'}`);
      }
      if (fbiState.cvFaces.length) {
        const topB = (fbiState.cvFaces[0].blendshapes || []).filter((b) => b.score > 0.18 && !b.categoryName.startsWith('_')).sort((a, b) => b.score - a.score).slice(0, 6).map((b) => `${b.categoryName}:${(b.score * 100).toFixed(0)}%`).join(', ');
        cvLines.push(`FACE FACS: ${topB || 'neutral'}`);
      }
      if (graphMetrics) cvLines.push(`GRAPHOLOGY: slope=${graphMetrics.baselineSlope}%, heavy_pressure=${graphMetrics.pressureRatio}%, sharpness=${graphMetrics.sharpness}`);

      const hasVisual = fbiState.evidenceType !== 'none';
      const sysPrompt = [
        'You are the NEXUS-FBI Sherlock Holmes Forensic Deduction Engine.',
        'Use the FORENSIC KNOWLEDGE BASE (detection.txt) and CV metrics to uncover hidden secrets, body language, graphology, physiognomy, or crime scene evidence.',
        'Respond ONLY with valid JSON:',
        '{ "subject_title": "...", "domain": "body|graphology|crime|physiognomy|mystery", "deduction_summary": "max 4 sentences", "clues": [{ "id": "CLUE-01", "label": "2-5 words", "detail": "...", "confidence": 88, "severity": "high|med|low", "bbox": [x,y,w,h] }], "links": [{ "from": "A", "to": "B", "relation": "..." }] }',
        'bbox values must be 0.0..1.0 normalized [left, top, width, height].',
        '', kbText
      ].join('\n');

      pushLog('Querying Vision AI + detection.txt...');
      const res = await APP.ai.chat({
        systemPrompt: sysPrompt,
        prompt: (customQuestion ? `QUESTION: ${customQuestion}\n` : `DOMAIN: ${forcedDomain || 'Full Sherlock Scan'}\n`) + (cvLines.join('\n') || ''),
        imageB64: hasVisual ? getWorkBase64(false) : undefined,
        imageMime: 'image/jpeg'
      });

      if (!res?.ok) throw new Error(res?.error || 'AI analysis failed.');

      let parsed = null;
      try { const m = String(res.text || '').match(/\{[\s\S]*\}/); if (m) parsed = JSON.parse(m[0]); } catch (_) {}

      if (parsed && Array.isArray(parsed.clues)) {
        fbiState.clues = parsed.clues.map((c, idx) => ({
          id: c.id || `CLUE-0${idx + 1}`,
          label: String(c.label || 'Anomaly').slice(0, 32),
          detail: String(c.detail || ''),
          confidence: Number(c.confidence || 85),
          severity: c.severity || 'med',
          bbox: Array.isArray(c.bbox) && c.bbox.length === 4 ? c.bbox.map((v) => Math.max(0, Math.min(1, Number(v) || 0))) : [0.2 + idx * 0.12, 0.25, 0.2, 0.2]
        }));
        fbiState.deductionSummary = parsed.deduction_summary || res.text;
        fbiState.hypothesisLinks = Array.isArray(parsed.links) ? parsed.links : [];
        fbiState.activeClueIdx = 0;
        rebuild3DSubjectScan(parsed.subject_title || fbiState.evidenceTitle, parsed.domain || forcedDomain);
      } else {
        fbiState.deductionSummary = res.text || 'Analysis complete.';
        rebuild3DSubjectScan(fbiState.evidenceTitle, forcedDomain);
      }

      fbiState.statusText = 'ANALYSIS COMPLETE // READY';
      APP.ui?.toast?.('Sherlock Scan complete! ' + fbiState.clues.length + ' clues mapped.', 3200);
      if (APP.bot?.notifySpeech) APP.bot.notifySpeech(fbiState.deductionSummary);
      if (typeof APP.speakTTS === 'function' && fbiState.deductionSummary) APP.speakTTS(fbiState.deductionSummary.slice(0, 260));
      return { ok: true, summary: fbiState.deductionSummary, clues: fbiState.clues };
    } catch (err) {
      fbiState.statusText = 'SCAN ERROR';
      fbiState.deductionSummary = 'Error: ' + err.message;
      pushLog('Error: ' + err.message);
      APP.ui?.toast?.('Forensic Error: ' + err.message, 5500);
      return { ok: false, error: err.message };
    } finally {
      fbiState.busy = false;
      refreshAllWindows();
    }
  }

  async function runFingerprintMatch() {
    if (!fbiState.open) setWorkstationOpen(true);
    if (!requireEvidenceOrPick('Load a fingerprint image or pick CASE 03!')) return { ok: false };
    fbiState.pickerOpen = false; fbiState.busy = true; fbiState.mode = 'fingerprint';
    fbiState.statusText = 'EXTRACTING GALTON MINUTIAE...'; refreshAllWindows();
    try {
      syncWorkCanvasFromSource();
      const evPrint = analyzeFingerprintImage(workCanvas);
      fbiState.fingerprintReport = evPrint;
      let bestMatchName = '', bestScore = 0;
      if (fbiState.secondaryPrintImg) {
        const refPrint = analyzeFingerprintImage(fbiState.secondaryPrintImg);
        bestScore = Math.max(12, Math.min(99, Math.round(62 + (evPrint.pattern === refPrint.pattern ? 35 : 0) - (Math.abs(evPrint.endings - refPrint.endings) + Math.abs(evPrint.bifurcations - refPrint.bifurcations)) * 1.4)));
        bestMatchName = 'Reference Suspect Print';
      } else {
        for (const item of [
          { name: 'FBI-DB #A104 (Arch Profile)', c: createSyntheticDbPrint(1.2, 'arch') },
          { name: 'FBI-DB #L409 (Loop Profile)', c: createSyntheticDbPrint(2.5, 'loop') },
          { name: 'FBI-DB #W882 (Whorl Profile)', c: createSyntheticDbPrint(4.1, 'whorl') }
        ]) {
          const r = analyzeFingerprintImage(item.c);
          const sc = Math.max(20, Math.min(97, Math.round((r.pattern === evPrint.pattern ? 88 : 52) - Math.abs(evPrint.ridgeScore - r.ridgeScore) * 40)));
          if (sc > bestScore) { bestScore = sc; bestMatchName = item.name; }
        }
      }
      fbiState.clues = [
        { id: 'FP-PAT', label: evPrint.pattern, detail: `Henry class: ${evPrint.pattern}`, confidence: 93, severity: 'high', bbox: [0.22, 0.22, 0.56, 0.56] },
        { id: 'FP-MIN', label: `${evPrint.endings} END / ${evPrint.bifurcations} BIF`, detail: `Detected ${evPrint.minutiae.length} minutiae`, confidence: 89, severity: 'med', bbox: [0.30, 0.30, 0.40, 0.40] }
      ];
      fbiState.hypothesisLinks = [
        { from: evPrint.pattern, to: `${evPrint.minutiae.length} Minutiae`, relation: 'classified' },
        { from: `${evPrint.minutiae.length} Minutiae`, to: bestMatchName, relation: `${bestScore}% match` }
      ];
      fbiState.deductionSummary = `FINGERPRINT REPORT: Pattern=${evPrint.pattern} | Endings=${evPrint.endings} | Bifurcations=${evPrint.bifurcations} | Best Match: ${bestMatchName} (${bestScore}%).`;
      rebuild3DSubjectScan('3D Fingerprint Ridge Map', 'fingerprint');
      fbiState.statusText = `PRINT MATCHED (${bestScore}%)`;
      APP.ui?.toast?.(`Fingerprint Match: ${bestScore}% (${evPrint.pattern})`, 3200);
      return { ok: true, report: evPrint, match: bestMatchName, score: bestScore };
    } finally { fbiState.busy = false; refreshAllWindows(); }
  }

  let roiDragStart = null, lastPokeTs = 0;

  function triggerCmdAtUV(uv) {
    const now = performance.now();
    if (now - lastPokeTs < 240) return;
    lastPokeTs = now;
    for (const b of cmdButtons) {
      if (uv.x >= b.u0 && uv.x <= b.u1 && uv.y >= b.v0 && uv.y <= b.v1) { b.action(); break; }
    }
  }

  function triggerMainAtUV(uv) {
    const now = performance.now();
    if (fbiState.pickerOpen || fbiState.evidenceType === 'none') {
      if (now - lastPokeTs < 240) return;
      lastPokeTs = now;
      for (const b of pickerButtons) {
        if (uv.x >= b.u0 && uv.x <= b.u1 && uv.y >= b.v0 && uv.y <= b.v1) { b.action(); break; }
      }
      return;
    }
    const u = uv.x, v = 1 - uv.y;
    fbiState.roi = { x: Math.max(0, Math.min(0.75, u - 0.14)), y: Math.max(0, Math.min(0.75, v - 0.14)), w: 0.28, h: 0.28 };
    redrawMainViewport();

  const dRay = new THREE.Raycaster(), dNdc = new THREE.Vector2();
  let draggingWs = false, dragHandKey = null, dragDist = 1.25;
  const dragPlane = new THREE.Plane(), dragOff = new THREE.Vector3(), tmpDir = new THREE.Vector3(), tmpVec = new THREE.Vector3(), tmpLocal = new THREE.Vector3();

  function onPtrDown(e) {
    if (!fbiState.open || e.button !== 0 || !APP.camera) return;
    const el = APP.renderer?.domElement;
    if (e.target !== el) return;
    const rect = el.getBoundingClientRect();
    dNdc.x = ((e.clientX - rect.left) / Math.max(1, rect.width)) * 2 - 1;
    dNdc.y = -((e.clientY - rect.top) / Math.max(1, rect.height)) * 2 + 1;
    dRay.setFromCamera(dNdc, APP.camera);

    const dh = dRay.intersectObject(dragBar, false);
    if (dh.length) {
      draggingWs = true;
      dragPlane.setFromNormalAndCoplanarPoint(APP.camera.getWorldDirection(new THREE.Vector3()).negate(), dh[0].point);
      dragOff.copy(wsRoot.position).sub(dh[0].point);
      return;
    }
    if (!fbiState.pickerOpen && fbiState.evidenceType !== 'none') {
      const mh = dRay.intersectObject(mainScreen, false);
      if (mh.length && mh[0].uv) roiDragStart = { u: mh[0].uv.x, v: 1 - mh[0].uv.y };
    }
  }

  function onCanvasClick(e) {

  function onPtrMove(e) {
    if (!fbiState.open || !APP.camera) return;
    const el = APP.renderer?.domElement;
    if (!el) return;
    const rect = el.getBoundingClientRect();
    dNdc.x = ((e.clientX - rect.left) / Math.max(1, rect.width)) * 2 - 1;
    dNdc.y = -((e.clientY - rect.top) / Math.max(1, rect.height)) * 2 + 1;
    dRay.setFromCamera(dNdc, APP.camera);

    if (draggingWs) {
      const pt = new THREE.Vector3();
      if (dRay.ray.intersectPlane(dragPlane, pt)) wsRoot.position.copy(pt).add(dragOff);
      return;
    }
    if (roiDragStart) {
      const mh = dRay.intersectObject(mainScreen, false);
      if (mh.length && mh[0].uv) {
        const u2 = mh[0].uv.x, v2 = 1 - mh[0].uv.y;
        const x = Math.min(roiDragStart.u, u2), y = Math.min(roiDragStart.v, v2);
        const w = Math.abs(u2 - roiDragStart.u), h = Math.abs(v2 - roiDragStart.v);
        if (w > 0.03 && h > 0.03) { fbiState.roi = { x, y, w, h }; redrawMainViewport(); }
      }
      return;
    }

    const ch = dRay.intersectObject(cmdScreen, false);
    let newHov = null;
    if (ch.length && ch[0].uv) {
      const uv = ch[0].uv;
      for (const b of cmdButtons) if (uv.x >= b.u0 && uv.x <= b.u1 && uv.y >= b.v0 && uv.y <= b.v1) { newHov = b.id; break; }
    }
    if (newHov !== hoveredCmdId) { hoveredCmdId = newHov; redrawCmdDeck(); }

    if (fbiState.pickerOpen || fbiState.evidenceType === 'none') {
      const mh = dRay.intersectObject(mainScreen, false);
      let newPick = null;
      if (mh.length && mh[0].uv) {
        const uv = mh[0].uv;
        for (const b of pickerButtons) if (uv.x >= b.u0 && uv.x <= b.u1 && uv.y >= b.v0 && uv.y <= b.v1) { newPick = b.id; break; }
      }
      if (newPick !== hoveredPickerId) { hoveredPickerId = newPick; redrawMainViewport(); }
    }
  }

  function updateForensicSuite(dt, elapsed) {
    if (!fbiState.open || !wsRoot.visible) return;
    const now = elapsed || (performance.now() * 0.001), delta = Math.min(0.05, dt || 0.016);

    scanOverlayUniforms.uTime.value = now; scanOverlayUniforms.uBusy.value = fbiState.busy ? 1.0 : 0.0;
    scanModelSpinGroup.rotation.y = now * 0.65;
    holoRingBase.rotation.z = now * 0.4; holoRingInner.rotation.z = -now * 0.8;
    holoSliceRing.position.y = Math.sin(now * 2.2) * 0.13;

    const arr = ambGeo.attributes.position.array;
    for (let i = 0; i < ambCount; i++) { arr[i * 3 + 1] += 0.06 * delta; if (arr[i * 3 + 1] > 0.62) arr[i * 3 + 1] = -0.62; }
    ambGeo.attributes.position.needsUpdate = true;

    // VR Ray & Hand Drag / Hover Cursor Support
    let showDot = false;
    if (APP.xrMode !== 'none') {
      const hands = APP.hands || {};
      if (dragHandKey) {
        const dh = hands[dragHandKey];
        if (dh && dh.active && (dh.trigger || dh.gripping || dh.pinching)) {
          if (dh.isHand && dh.fingerTips?.index) wsRoot.position.lerp(tmpVec.copy(dh.fingerTips.index).add(dragOff), 0.55);
          else if (dh.rayOrigin && dh.rayDir) {
            tmpDir.copy(dh.rayDir).normalize();
            wsRoot.position.lerp(tmpVec.copy(dh.rayOrigin).addScaledVector(tmpDir, dragDist).add(dragOff), 0.55);
          }
          if (APP.headPos) wsRoot.lookAt(APP.headPos.x, wsRoot.position.y, APP.headPos.z);
        } else dragHandKey = null;
      } else {
        for (const [hk, hand] of Object.entries(hands)) {
          if (!hand || !hand.active) continue;
          if (hand.rayOrigin && hand.rayDir) {
            tmpDir.copy(hand.rayDir).normalize(); dRay.set(hand.rayOrigin, tmpDir);
            const dh = dRay.intersectObject(dragBar, false);
            if (dh.length) {
              showDot = true; vrCursorDot.position.copy(dh[0].point); wsRoot.worldToLocal(vrCursorDot.position);
              if (hand.justTriggered || hand.justGripped || hand.justPinched) { dragHandKey = hk; dragDist = dh[0].distance; dragOff.copy(wsRoot.position).sub(dh[0].point); break; }
            }
            const ch = dRay.intersectObjects([cmdScreen, mainScreen], false);
            if (ch.length && ch[0].uv) {
              showDot = true; vrCursorDot.position.copy(ch[0].point); wsRoot.worldToLocal(vrCursorDot.position);
              if (ch[0].object === cmdScreen) {
                for (const b of cmdButtons) if (ch[0].uv.x >= b.u0 && ch[0].uv.x <= b.u1 && ch[0].uv.y >= b.v0 && ch[0].uv.y <= b.v1 && hoveredCmdId !== b.id) { hoveredCmdId = b.id; redrawCmdDeck(); break; }
              }
            }
          }
          const tip = hand.isHand ? hand.fingerTips?.index : null;
          if (tip && (hand.justPinched)) {
            tmpLocal.copy(tip); cmdScreen.worldToLocal(tmpLocal);
            if (Math.abs(tmpLocal.z) < 0.045) {
              const u = tmpLocal.x / CMD_W + 0.5, v = tmpLocal.y / CMD_H + 0.5;
              if (u >= 0 && u <= 1 && v >= 0 && v <= 1) triggerCmdAtUV({ x: u, y: v });
            }
          }
        }
      }
    }
    vrCursorDot.visible = showDot;

    if (fbiState.clues.length > 1 && !fbiState.busy && !fbiState.pickerOpen) {
      fbiState.clueCycleTimer += delta;
      if (fbiState.clueCycleTimer > 3.8) {
        fbiState.clueCycleTimer = 0;
        fbiState.activeClueIdx = (fbiState.activeClueIdx + 1) % fbiState.clues.length;
        redrawMainViewport();
      }
    }

    if (fbiState.evidenceType === 'video' && fbiState.sourceVideo && !fbiState.sourceVideo.paused && !fbiState.pickerOpen) {
      if (fbiState.trackingActive && performance.now() - fbiState.lastVideoDetectTs > 140) {
        fbiState.lastVideoDetectTs = performance.now();
        runMediaPipeOnCurrentFrame(true);
      }
      redrawMainViewport();
    }

    codeRefreshTick += delta;
    if (codeRefreshTick > 0.22) { codeRefreshTick = 0; redrawCodeWindow(now); }
  }
