  function computeBBoxColorSignature(bbox) {
    try {
      const x = Math.max(0, Math.floor(bbox.x * workCanvas.width));
      const y = Math.max(0, Math.floor(bbox.y * workCanvas.height));
      const w = Math.max(4, Math.min(workCanvas.width - x, Math.floor(bbox.w * workCanvas.width)));
      const h = Math.max(4, Math.min(workCanvas.height - y, Math.floor(bbox.h * workCanvas.height)));
      const data = workCtx.getImageData(x, y, w, h).data;
      let r = 0, g = 0, b = 0, cnt = 0;
      const step = Math.max(1, Math.floor((w * h) / 180));
      for (let i = 0; i < data.length; i += step * 4) {
        r += data[i]; g += data[i + 1]; b += data[i + 2]; cnt++;
      }
      if (!cnt) return { r: 128, g: 128, b: 128, tag: 'neutral' };
      r = Math.round(r / cnt); g = Math.round(g / cnt); b = Math.round(b / cnt);
      let tag = 'neutral';
      if (r > g + 35 && r > b + 35) tag = 'red';
      else if (b > r + 35 && b > g + 25) tag = 'blue';
      else if (g > r + 25 && g > b + 25) tag = 'green';
      else if (r > 180 && g > 170 && b < 110) tag = 'yellow';
      else if (r > 195 && g > 195 && b > 195) tag = 'white';
      else if (r < 65 && g < 65 && b < 65) tag = 'dark';
      return { r, g, b, tag };
    } catch (_) {
      return { r: 128, g: 128, b: 128, tag: 'unknown' };
    }
  }

  function updateTrackersFromDetections(detections, nowMs) {
    const W = Math.max(1, workCanvas.width), H = Math.max(1, workCanvas.height);
    const items = (detections || []).map((d) => {
      const bx = d.boundingBox || {};
      const cat = d.categories?.[0] || {};
      const norm = {
        x: Math.max(0, (bx.originX || 0) / W),
        y: Math.max(0, (bx.originY || 0) / H),
        w: Math.min(1, (bx.width || 40) / W),
        h: Math.min(1, (bx.height || 40) / H)
      };
      return {
        label: cat.categoryName || 'object',
        score: Number(cat.score || 0.5),
        bbox: norm,
        cx: norm.x + norm.w * 0.5,
        cy: norm.y + norm.h * 0.5,
        col: computeBBoxColorSignature(norm)
      };
    });

    const usedTracks = new Set();
    const TRACK_COLORS = ['#00f0b4', '#38bdf8', '#fbbf24', '#f472b6', '#a78bfa', '#fb7185'];

    for (const item of items) {
      let bestTrk = null, bestScore = 999;
      for (const t of fbiState.tracks) {
        if (usedTracks.has(t.id) || t.label !== item.label) continue;
        const dist = Math.hypot(t.cx - item.cx, t.cy - item.cy);
        if (dist < 0.22 && dist < bestScore) { bestScore = dist; bestTrk = t; }
      }
      if (bestTrk) {
        usedTracks.add(bestTrk.id);
        const dtS = Math.max(0.03, (nowMs - bestTrk.lastSeen) * 0.001);
        bestTrk.vx = (item.cx - bestTrk.cx) / dtS;
        bestTrk.vy = (item.cy - bestTrk.cy) / dtS;
        bestTrk.cx = item.cx; bestTrk.cy = item.cy;
        bestTrk.bbox = item.bbox;
        bestTrk.score = item.score;
        bestTrk.col = item.col;
        bestTrk.lastSeen = nowMs;
        bestTrk.trail.push({ x: item.cx, y: item.cy });
        if (bestTrk.trail.length > 28) bestTrk.trail.shift();
      } else {
        const idNum = fbiState.nextTrackId++;
        const trk = {
          id: 'TRK-' + String(idNum).padStart(2, '0'),
          label: item.label,
          score: item.score,
          color: TRACK_COLORS[(idNum - 1) % TRACK_COLORS.length],
          bbox: item.bbox,
          cx: item.cx, cy: item.cy, vx: 0, vy: 0,
          col: item.col,
          trail: [{ x: item.cx, y: item.cy }],
          lastSeen: nowMs
        };
        usedTracks.add(trk.id);
        fbiState.tracks.push(trk);
      }
    }
    fbiState.tracks = fbiState.tracks.filter((t) => nowMs - t.lastSeen < 1800);
  }

  function computeGraphologyMetrics() {
    if (!syncWorkCanvasFromSource()) return null;
    const W = workCanvas.width, H = workCanvas.height;
    try {
      const data = workCtx.getImageData(0, 0, W, H).data;
      let darkPixels = 0, sumY = 0, leftY = 0, leftCnt = 0, rightY = 0, rightCnt = 0;
      let pressureHeavy = 0, edgeSum = 0;
      for (let y = 1; y < H - 1; y += 2) {
        for (let x = 1; x < W - 1; x += 2) {
          const i = (y * W + x) * 4;
          const lum = 0.299 * data[i] + 0.587 * data[i + 1] + 0.114 * data[i + 2];
          if (lum < 135) {
            darkPixels++; sumY += y;
            if (lum < 65) pressureHeavy++;
            if (x < W * 0.4) { leftY += y; leftCnt++; }
            else if (x > W * 0.6) { rightY += y; rightCnt++; }
          }
          const ir = (y * W + (x + 1)) * 4;
          const lumR = 0.299 * data[ir] + 0.587 * data[ir + 1] + 0.114 * data[ir + 2];
          edgeSum += Math.abs(lum - lumR);
        }
      }
      const avgLeftY = leftCnt ? leftY / leftCnt : H * 0.5;
      const avgRightY = rightCnt ? rightY / rightCnt : H * 0.5;
      const baselineSlope = ((avgLeftY - avgRightY) / H * 100).toFixed(2); // + = ascending
      const pressureRatio = darkPixels ? ((pressureHeavy / darkPixels) * 100).toFixed(1) : '0.0';
      const sharpness = (edgeSum / ((W * H) / 4)).toFixed(2);
      return { darkPixels, baselineSlope, pressureRatio, sharpness };
    } catch (_) { return null; }
  }

  function runLocalForensicEnhance(roiNorm) {
    if (!syncWorkCanvasFromSource()) return null;
    const W = workCanvas.width, H = workCanvas.height;
    const rx = roiNorm ? Math.max(0, Math.floor(roiNorm.x * W)) : 0;
    const ry = roiNorm ? Math.max(0, Math.floor(roiNorm.y * H)) : 0;
    const rw = roiNorm ? Math.max(32, Math.min(W - rx, Math.floor(roiNorm.w * W))) : W;
    const rh = roiNorm ? Math.max(32, Math.min(H - ry, Math.floor(roiNorm.h * H))) : H;

    const upScale = roiNorm ? Math.min(3, Math.max(1.5, 640 / Math.max(rw, rh))) : 1.0;
    const outW = Math.round(rw * upScale), outH = Math.round(rh * upScale);
    const outC = document.createElement('canvas');
    outC.width = outW; outC.height = outH;
    const oCtx = outC.getContext('2d', { willReadFrequently: true });
    oCtx.imageSmoothingEnabled = true;
    oCtx.imageSmoothingQuality = 'high';
    oCtx.drawImage(workCanvas, rx, ry, rw, rh, 0, 0, outW, outH);

    const img = oCtx.getImageData(0, 0, outW, outH);
    const src = new Uint8ClampedArray(img.data);
    const dst = img.data;

    // 1. Histogram Contrast Stretch (5%..95% luminance normalization)
    let minL = 255, maxL = 0;
    for (let i = 0; i < src.length; i += 16) {
      const l = 0.299 * src[i] + 0.587 * src[i + 1] + 0.114 * src[i + 2];
      if (l < minL) minL = l; if (l > maxL) maxL = l;
    }
    const span = Math.max(24, maxL - minL);

    // 2. 3x3 Unsharp Mask + Edge-Preserving Contrast Boost
    const amount = 1.65;
    for (let y = 1; y < outH - 1; y++) {
      for (let x = 1; x < outW - 1; x++) {
        const idx = (y * outW + x) * 4;
        for (let c = 0; c < 3; c++) {
          const center = src[idx + c];
          const blur = (
            src[((y - 1) * outW + (x - 1)) * 4 + c] + src[((y - 1) * outW + x) * 4 + c] * 2 + src[((y - 1) * outW + (x + 1)) * 4 + c] +
            src[(y * outW + (x - 1)) * 4 + c] * 2 + center * 4 + src[(y * outW + (x + 1)) * 4 + c] * 2 +
            src[((y + 1) * outW + (x - 1)) * 4 + c] + src[((y + 1) * outW + x) * 4 + c] * 2 + src[((y + 1) * outW + (x + 1)) * 4 + c]
          ) / 16.0;
          const sharpened = center + (center - blur) * amount;
          const stretched = ((sharpened - minL) / span) * 255;
          dst[idx + c] = Math.max(0, Math.min(255, Math.round(stretched * 0.55 + sharpened * 0.45)));
        }
      }
    }
    oCtx.putImageData(img, 0, 0);
    return outC;
  }

  // Real Fingerprint Ridge Minutiae Extractor (Crossing-Number Algorithm)
  function analyzeFingerprintImage(imgOrCanvas) {
    const N = 160;
    const fc = document.createElement('canvas'); fc.width = N; fc.height = N;
    const fCtx = fc.getContext('2d', { willReadFrequently: true });
    fCtx.drawImage(imgOrCanvas, 0, 0, N, N);
    const d = fCtx.getImageData(0, 0, N, N).data;
    const gray = new Float32Array(N * N);
    for (let i = 0; i < N * N; i++) gray[i] = 0.299 * d[i * 4] + 0.587 * d[i * 4 + 1] + 0.114 * d[i * 4 + 2];

    // Local Adaptive Binarization (7x7 mean)
    const bin = new Uint8Array(N * N);
    for (let y = 3; y < N - 3; y++) {
      for (let x = 3; x < N - 3; x++) {
        let sum = 0;
        for (let dy = -3; dy <= 3; dy++) for (let dx = -3; dx <= 3; dx++) sum += gray[(y + dy) * N + (x + dx)];
        const mean = sum / 49;
        bin[y * N + x] = gray[y * N + x] < mean - 4 ? 1 : 0;
      }
    }

    // Zhang-Suen-inspired Fast 2-pass Edge Thinning / Crossing-Number Minutiae
    const minutiae = [];
    let angleSum = 0, angleAbs = 0, archCount = 0, whorlCount = 0;
    for (let y = 14; y < N - 14; y += 2) {
      for (let x = 14; x < N - 14; x += 2) {
        const idx = y * N + x;
        if (!bin[idx]) continue;
        // 8-neighbor ring
        const p = [
          bin[(y - 1) * N + x], bin[(y - 1) * N + (x + 1)], bin[y * N + (x + 1)], bin[(y + 1) * N + (x + 1)],
          bin[(y + 1) * N + x], bin[(y + 1) * N + (x - 1)], bin[y * N + (x - 1)], bin[(y - 1) * N + (x - 1)]
        ];
        let cn = 0, nb = 0;
        for (let k = 0; k < 8; k++) { nb += p[k]; cn += Math.abs(p[k] - p[(k + 1) % 8]); }
        cn = cn * 0.5;
        if (nb >= 1 && nb <= 5 && (cn === 1 || cn === 3)) {
          if (!minutiae.some((m) => Math.hypot(m.x - x / N, m.y - y / N) < 0.055)) {
            minutiae.push({ x: x / N, y: y / N, type: cn === 1 ? 'ending' : 'bifurcation' });
          }
        }
        // Ridge gradient for pattern estimation
        const gx = gray[y * N + (x + 1)] - gray[y * N + (x - 1)];
        const gy = gray[(y + 1) * N + x] - gray[(y - 1) * N + x];
        const ang = Math.atan2(gy, gx);
        angleSum += ang; angleAbs += Math.abs(ang);
        if (Math.abs(gx) > 18 && Math.abs(gy) > 18) whorlCount++; else archCount++;
      }
    }
    const endings = minutiae.filter((m) => m.type === 'ending').length;
    const bifs = minutiae.filter((m) => m.type === 'bifurcation').length;
    const ratio = whorlCount / Math.max(1, archCount + whorlCount);
    const pattern = ratio > 0.34 ? 'Whorl (Spiral Core)' : ratio > 0.18 ? 'Ulnar/Radial Loop' : 'Plain/Tented Arch';
    return { pattern, minutiae: minutiae.slice(0, 42), endings, bifurcations: bifs, ridgeScore: ratio };
  }

  function createSyntheticDbPrint(seed, style) {

          if (gx < G - 1 && gy < G - 1) {
            const ir = i + 1, id = i + G;
            const lumR = (0.299 * px[ir * 4] + 0.587 * px[ir * 4 + 1] + 0.114 * px[ir * 4 + 2]) / 255;
            const lumD = (0.299 * px[id * 4] + 0.587 * px[id * 4 + 1] + 0.114 * px[id * 4 + 2]) / 255;
            edge = Math.min(1, (Math.abs(lum - lumR) + Math.abs(lum - lumD)) * 3.2);
          }
          const depthZ = (lum * 0.045 + edge * 0.042) * (0.35 + radial * 0.85);
          pAttr.setZ(i, depthZ);

          if (edge > 0.16 || (i % 2 === 0 && lum > 0.18)) {
            const vx = pAttr.getX(i), vy = pAttr.getY(i);
            ptVerts.push(vx, vy, depthZ + 0.004, vx, vy, -depthZ - 0.004);
            const c = edge > 0.35 ? [0.0, 0.96, 0.72] : [0.22, 0.74, 0.98];
            ptCols.push(c[0], c[1], c[2], c[0], c[1], c[2]);
          }
        }
      }
      planeGeo.computeVertexNormals();
      const wireMat = new THREE.MeshBasicMaterial({ color: 0x0ea5e9, wireframe: true, transparent: true, opacity: 0.34 });
      const frontMesh = new THREE.Mesh(planeGeo, wireMat);
      const backMesh = new THREE.Mesh(planeGeo, wireMat);
      backMesh.scale.z = -1;
      scanModelSpinGroup.add(frontMesh, backMesh);

      if (ptVerts.length) {
        const ptGeo = new THREE.BufferGeometry();
        ptGeo.setAttribute('position', new THREE.Float32BufferAttribute(ptVerts, 3));
        ptGeo.setAttribute('color', new THREE.Float32BufferAttribute(ptCols, 3));
        scanModelSpinGroup.add(new THREE.Points(ptGeo, new THREE.PointsMaterial({
          size: 0.006, vertexColors: true, transparent: true, opacity: 0.92, blending: THREE.AdditiveBlending, depthWrite: false
        })));
      }
    }

    // 2. SEMANTIC 3D BIOMETRIC / SCENE OVERLAY (Matched to what's actually being analyzed!)
    if ((low.includes('body') || fbiState.cvPoses.length) && fbiState.cvPoses[0]?.worldLandmarks) {
      // Real 3D Kinesics Pose Skeleton from MediaPipe worldLandmarks!
      const wlm = fbiState.cvPoses[0].worldLandmarks;
      const POSE_PAIRS = [
        [11, 12], [11, 23], [12, 24], [23, 24], [11, 13], [13, 15], [12, 14], [14, 16],
        [23, 25], [25, 27], [24, 26], [26, 28], [0, 11], [0, 12]
      ];
      const linePts = [];
      for (const [a, b] of POSE_PAIRS) {
        if (wlm[a] && wlm[b]) {
          linePts.push(new THREE.Vector3(wlm[a].x * 0.16, -wlm[a].y * 0.16, wlm[a].z * 0.16));
          linePts.push(new THREE.Vector3(wlm[b].x * 0.16, -wlm[b].y * 0.16, wlm[b].z * 0.16));
        }
      }
      if (linePts.length) {
        const skel = new THREE.LineSegments(
          new THREE.BufferGeometry().setFromPoints(linePts),
          new THREE.LineBasicMaterial({ color: 0x00f0b4, transparent: true, opacity: 0.95 })
        );
        skel.position.z = 0.05;
        scanModelSpinGroup.add(skel);
      }
    } else if (low.includes('fingerprint')) {
      // 3D Concentric Extruded Fingerprint Ridge Hologram
      for (let r = 0.025; r <= 0.11; r += 0.017) {
        const ring = new THREE.Mesh(
          new THREE.TorusGeometry(r, 0.0022, 6, 32),
          new THREE.MeshBasicMaterial({ color: r < 0.06 ? 0x00f0b4 : 0x38bdf8, transparent: true, opacity: 0.78 })
        );
        ring.position.z = 0.055 + (0.11 - r) * 0.25;
        scanModelSpinGroup.add(ring);
      }
    } else {
      // 3D Bounding Vector Cage + Callout Target Nodes on the Scanned Object
      const cage = new THREE.LineSegments(
        new THREE.EdgesGeometry(new THREE.BoxGeometry(0.26, 0.22, 0.12)),
        new THREE.LineBasicMaterial({ color: 0x00f0b4, transparent: true, opacity: 0.55 })
      );
      scanModelSpinGroup.add(cage);
      for (let k = 0; k < Math.min(4, fbiState.clues.length); k++) {
        const cl = fbiState.clues[k];
        if (!cl.bbox) continue;
        const nx = (cl.bbox[0] + cl.bbox[2] * 0.5 - 0.5) * 0.22;
        const ny = (0.5 - (cl.bbox[1] + cl.bbox[3] * 0.5)) * 0.18;
        const node = new THREE.Mesh(
          new THREE.OctahedronGeometry(0.011, 0),
          new THREE.MeshBasicMaterial({ color: k === fbiState.activeClueIdx ? 0xfbbf24 : 0x00f0b4 })
        );
        node.position.set(nx, ny, 0.065);
        scanModelSpinGroup.add(node);
      }
    }
    redrawScanBackdrop();
  }

  function drawMiniTape(ctx, x1, y1, tx, ty, idStr, textStr, accentCol, isActive) {
    const label = `${idStr} // ${String(textStr || '').toUpperCase().slice(0, 32)}`;
    ctx.save();
    ctx.font = 'bold 14px monospace';
    const tw = Math.min(320, Math.max(130, ctx.measureText(label).width + 24));
    const th = 26;
    const rx = Math.max(8, Math.min(1024 - tw - 8, tx));
    const ry = Math.max(46, Math.min(632 - th - 12, ty));

    // Elbow Leader Line
    ctx.strokeStyle = accentCol;
    ctx.lineWidth = isActive ? 2.5 : 1.5;
    ctx.beginPath();
    ctx.moveTo(x1, y1);
    ctx.lineTo((x1 + rx) * 0.5, ry + th * 0.5);
    ctx.lineTo(rx, ry + th * 0.5);
    ctx.stroke();

    // Leader joint dot
    ctx.fillStyle = accentCol;
    ctx.beginPath(); ctx.arc(x1, y1, isActive ? 5 : 3.5, 0, Math.PI * 2); ctx.fill();

    // Chamfered Mini Tape Strip
    ctx.fillStyle = isActive ? 'rgba(6, 30, 36, 0.94)' : 'rgba(6, 16, 28, 0.86)';
    ctx.beginPath();
    ctx.moveTo(rx, ry); ctx.lineTo(rx + tw - 10, ry); ctx.lineTo(rx + tw, ry + 10);
    ctx.lineTo(rx + tw, ry + th); ctx.lineTo(rx, ry + th); ctx.closePath();
    ctx.fill();
    ctx.strokeStyle = accentCol;
    ctx.lineWidth = isActive ? 2.2 : 1.2;
    ctx.stroke();

    // Accent bar on left of tape
    ctx.fillStyle = accentCol;
    ctx.fillRect(rx, ry, 5, th);

    ctx.fillStyle = isActive ? '#ffffff' : '#bae6fd';
    ctx.textAlign = 'left'; ctx.textBaseline = 'middle';
    ctx.fillText(label, rx + 11, ry + th * 0.5);
    ctx.restore();
  }

      { id: 'face', lbl: 'FACE REVEAL', fn: () => runFaceDisguiseRemoval() },
      { id: 'grapho', lbl: 'GRAPHOLOGY', fn: () => runForensicAnalyse('graphology') },
      { id: 'crime', lbl: 'CRIME SCENE', fn: () => runForensicAnalyse('crime') },
      { id: 'next_clue', lbl: 'NEXT CLUE >', fn: () => cycleClue(1) },
      { id: 'toggle_enh', lbl: fbiState.showEnhanced ? 'VIEW: ENHANCED' : 'VIEW: RAW', active: fbiState.showEnhanced, fn: () => { if (fbiState.enhancedCanvas) { fbiState.showEnhanced = !fbiState.showEnhanced; redrawMainViewport(); redrawCmdDeck(); } } },
      { id: 'edit_kb', lbl: 'EDIT detection.txt', fn: () => openDetectionEditor() },
      { id: 'open_hud', lbl: '2D CONSOLE', fn: () => show2DDrawer(true) },
      { id: 'close_ws', lbl: 'HIDE LAB', warn: true, fn: () => setWorkstationOpen(false) }
    ];

    const cols = 5, pad = 16, gapX = 12, gapY = 12;
    const bw = (W - pad * 2 - gapX * (cols - 1)) / cols;
    const bh = (H - pad * 2 - gapY * 2) / 3;

    defs.forEach((b, idx) => {
      const c = idx % cols, r = Math.floor(idx / cols);
      const bx = pad + c * (bw + gapX), by = pad + r * (bh + gapY);
      const isHov = hoveredCmdId === b.id;
      cmdCtx.fillStyle = b.warn ? '#4c1420' : (b.active ? '#0f5c52' : (isHov ? '#1e3a5f' : '#0d1b2e'));
      cmdCtx.strokeStyle = b.warn ? '#f87171' : (b.active || isHov ? '#00f0b4' : '#2b4d78');
      cmdCtx.lineWidth = isHov ? 3.5 : 2;
      cmdCtx.fillRect(bx, by, bw, bh);
      cmdCtx.strokeRect(bx, by, bw, bh);

      cmdCtx.fillStyle = '#f1f5f9';
      cmdCtx.font = 'bold 17px monospace';
      cmdCtx.textAlign = 'center'; cmdCtx.textBaseline = 'middle';
      cmdCtx.fillText(b.lbl, bx + bw * 0.5, by + bh * 0.5);

      cmdButtons.push({
        id: b.id,
        u0: bx / W, u1: (bx + bw) / W,
        v0: 1 - (by + bh) / H, v1: 1 - by / H,
        action: b.fn
      });
    });
    cmdTex.needsUpdate = true;
  }

  function cycleClue(delta = 1) {
    if (!fbiState.clues.length) return;
    fbiState.activeClueIdx = (fbiState.activeClueIdx + delta + fbiState.clues.length) % fbiState.clues.length;
    redrawMainViewport();
    rebuild3DSubjectScan(fbiState.clues[fbiState.activeClueIdx]?.label || fbiState.evidenceTitle, fbiState.mode);
  }

  async function runForensicAnalyse(forcedDomain, customQuestion) {
    if (!fbiState.open) setWorkstationOpen(true);
    if (fbiState.busy) return { ok: false, error: 'Scan already in progress.' };
    if (!APP.ai?.chat) return { ok: false, error: 'APP.ai not available.' };

    fbiState.busy = true;
    if (forcedDomain) fbiState.mode = forcedDomain;
    fbiState.statusText = 'SCANNING // MEDIAPIPE + SHERLOCK AI';
    refreshAllWindows();

    try {
      await runMediaPipeOnCurrentFrame(false);
      const graphMetrics = computeGraphologyMetrics();
      const kbText = getRelevantKB((forcedDomain || '') + ' ' + (customQuestion || ''));

      const cvContextLines = [];
      if (fbiState.cvDetections.length) {
        cvContextLines.push('DETECTED OBJECTS (MediaPipe): ' + fbiState.cvDetections.map((d) => {
          const cat = d.categories?.[0];
          const b = d.boundingBox || {};
          return `${cat?.categoryName || 'obj'}(${Math.round((cat?.score || 0) * 100)}%)@[${(b.originX / workCanvas.width).toFixed(2)},${(b.originY / workCanvas.height).toFixed(2)},${(b.width / workCanvas.width).toFixed(2)},${(b.height / workCanvas.height).toFixed(2)}]`;
        }).join(', '));
      }
      if (fbiState.cvPoses.length) {
        const p = fbiState.cvPoses[0].landmarks;
        const shDiff = p[11] && p[12] ? Math.abs(p[11].y - p[12].y).toFixed(3) : 'n/a';
        cvContextLines.push(`BODY POSE: ${fbiState.cvPoses.length} person(s), shoulder_asymmetry=${shDiff}`);
      }
      if (fbiState.cvFaces.length) {
        const topB = (fbiState.cvFaces[0].blendshapes || [])
          .filter((b) => b.score > 0.18 && !b.categoryName.startsWith('_'))
          .sort((a, b) => b.score - a.score).slice(0, 6)
          .map((b) => `${b.categoryName}:${(b.score * 100).toFixed(0)}%`).join(', ');
        cvContextLines.push(`FACE FACS BLENDSHAPES: ${topB || 'neutral'}`);
      }
      if (graphMetrics) {
        cvContextLines.push(`GRAPHOLOGY CV METRICS: baseline_slope=${graphMetrics.baselineSlope}%, heavy_pressure_ink=${graphMetrics.pressureRatio}%, edge_sharpness=${graphMetrics.sharpness}`);
      }

      const b64 = getWorkBase64(false);
      const hasVisual = fbiState.evidenceType !== 'none';

      const sysPrompt = [
        'You are the NEXUS-FBI Sherlock Holmes Forensic Deduction Engine.',
        'Use the provided FORENSIC KNOWLEDGE BASE (from detection.txt) and Computer Vision metrics to uncover hidden secrets, behavioral cues, graphology clues, physiognomy/FACS leakage, or crime scene evidence.',
        'IMPORTANT: Respond ONLY with valid JSON matching this schema:',
        '{',
        '  "subject_title": "short name of main object/person/document in picture",',
        '  "domain": "body|graphology|crime|physiognomy|mystery",',
        '  "deduction_summary": "concise Sherlock Holmes deductive conclusion revealing the secrets & connections (max 4 sentences)",',
        '  "clues": [',
        '    { "id": "CLUE-01", "label": "2-5 word tape label", "detail": "specific forensic observation & why it matters", "confidence": 88, "severity": "high|med|low", "bbox": [x, y, w, h] }',
        '  ],',
        '  "links": [',
        '    { "from": "Node A", "to": "Node B", "relation": "implies / contradicts / reveals" }',
        '  ]',
        '}',
        'Note: bbox values MUST be normalized floats 0.0 to 1.0 ([left, top, width, height]). Return 3 to 5 clues.',
        '',
        'FORENSIC KNOWLEDGE BASE (detection.txt):',
        kbText
      ].join('\n');

      const userPrompt = [
        customQuestion ? `INVESTIGATOR QUESTION: ${customQuestion}` : `MODE REQUESTED: ${forcedDomain || 'Full Multi-Domain Sherlock Holmes Forensic Analysis'}`,
        cvContextLines.join('\n') || '(Text/Visual deduction only)'
      ].join('\n\n');

      pushLog('Querying Vision AI + detection.txt rules...');
      const res = await APP.ai.chat({
        systemPrompt: sysPrompt,
        prompt: userPrompt,
        imageB64: hasVisual ? b64 : undefined,
        imageMime: 'image/jpeg'
      });

      if (!res?.ok) throw new Error(res?.error || 'AI analysis failed.');

      let parsed = null;
      try {
        const raw = String(res.text || '');
        const jsonMatch = raw.match(/\{[\s\S]*\}/);
        if (jsonMatch) parsed = JSON.parse(jsonMatch[0]);
      } catch (_) {}

      if (parsed && Array.isArray(parsed.clues)) {
        fbiState.clues = parsed.clues.map((c, idx) => ({
          id: c.id || `CLUE-0${idx + 1}`,
          label: String(c.label || 'Anomaly').slice(0, 32),
          detail: String(c.detail || ''),
          confidence: Number(c.confidence || 85),
          severity: c.severity || 'med',
          bbox: Array.isArray(c.bbox) && c.bbox.length === 4 ? c.bbox.map((v) => Math.max(0, Math.min(1, Number(v) || 0))) : [0.2 + idx * 0.12, 0.25, 0.2, 0.2]
        }));
        fbiState.deductionSummary = parsed.deduction_summary || res.text;
        fbiState.hypothesisLinks = Array.isArray(parsed.links) ? parsed.links : [];
        fbiState.activeClueIdx = 0;
        rebuild3DSubjectScan(parsed.subject_title || fbiState.evidenceTitle, parsed.domain || forcedDomain);
        pushLog(`Identified ${fbiState.clues.length} clues (${parsed.subject_title || 'subject'}).`);
      } else {
        fbiState.deductionSummary = res.text || 'Analysis complete.';
        rebuild3DSubjectScan(fbiState.evidenceTitle, forcedDomain);
      }

      fbiState.statusText = 'ANALYSIS COMPLETE // READY';
      APP.ui?.toast?.('Sherlock Scan complete! ' + fbiState.clues.length + ' clues mapped.', 3200);
      if (APP.bot?.notifySpeech) APP.bot.notifySpeech(fbiState.deductionSummary);
      if (typeof APP.speakTTS === 'function' && fbiState.deductionSummary) APP.speakTTS(fbiState.deductionSummary.slice(0, 260));
      return { ok: true, summary: fbiState.deductionSummary, clues: fbiState.clues };
    } catch (err) {
      fbiState.statusText = 'SCAN ERROR';
      pushLog('Error: ' + err.message);
      APP.ui?.toast?.('Forensic Error: ' + err.message, 5000);
      return { ok: false, error: err.message };
    } finally {
      fbiState.busy = false;
      refreshAllWindows();
    }
  }

  async function runFingerprintMatch() {
    if (!fbiState.open) setWorkstationOpen(true);
    if (fbiState.evidenceType === 'none') return { ok: false, error: 'Load a fingerprint image first.' };

    fbiState.busy = true;
    fbiState.mode = 'fingerprint';
    fbiState.statusText = 'EXTRACTING GALTON MINUTIAE...';
    refreshAllWindows();

    try {
      syncWorkCanvasFromSource();
      const evPrint = analyzeFingerprintImage(workCanvas);
      fbiState.fingerprintReport = evPrint;

      // Compare against secondary uploaded print OR 3 FBI Training DB archetypes
      let bestMatchName = '', bestScore = 0;
      if (fbiState.secondaryPrintImg) {
        const refPrint = analyzeFingerprintImage(fbiState.secondaryPrintImg);
        const endDiff = Math.abs(evPrint.endings - refPrint.endings);
        const bifDiff = Math.abs(evPrint.bifurcations - refPrint.bifurcations);
        const patBonus = evPrint.pattern === refPrint.pattern ? 35 : 0;
        bestScore = Math.max(12, Math.min(99, Math.round(62 + patBonus - (endDiff + bifDiff) * 1.4)));
        bestMatchName = 'Reference Suspect Print';
      } else {
        const db = [
          { name: 'FBI-DB #A104 (Arch Profile)', c: createSyntheticDbPrint(1.2, 'arch') },
          { name: 'FBI-DB #L409 (Loop Profile)', c: createSyntheticDbPrint(2.5, 'loop') },
          { name: 'FBI-DB #W882 (Whorl Profile)', c: createSyntheticDbPrint(4.1, 'whorl') }
        ];
        for (const item of db) {
          const r = analyzeFingerprintImage(item.c);
          const sc = Math.max(20, Math.min(97, Math.round((r.pattern === evPrint.pattern ? 88 : 52) - Math.abs(evPrint.ridgeScore - r.ridgeScore) * 40)));
          if (sc > bestScore) { bestScore = sc; bestMatchName = item.name; }
        }
      }

      fbiState.clues = [
        { id: 'FP-PAT', label: evPrint.pattern, detail: `Level-1 Henry classification: ${evPrint.pattern}`, confidence: 93, severity: 'high', bbox: [0.22, 0.22, 0.56, 0.56] },
        { id: 'FP-MIN', label: `${evPrint.endings} END / ${evPrint.bifurcations} BIF`, detail: `Detected ${evPrint.minutiae.length} Galton ridge minutiae`, confidence: 89, severity: 'med', bbox: [0.30, 0.30, 0.40, 0.40] }
      ];
      fbiState.hypothesisLinks = [
        { from: evPrint.pattern, to: `${evPrint.minutiae.length} Minutiae`, relation: 'classified' },
        { from: `${evPrint.minutiae.length} Minutiae`, to: bestMatchName, relation: `${bestScore}% match` }
      ];
      fbiState.deductionSummary = `FINGERPRINT REPORT: Pattern=${evPrint.pattern} | Ridge Endings=${evPrint.endings} | Bifurcations=${evPrint.bifurcations} | Closest Match: ${bestMatchName} (${bestScore}% similarity).`;
      pushLog(`Fingerprint: ${evPrint.pattern}, ${evPrint.minutiae.length} minutiae -> ${bestScore}%`);
      rebuild3DSubjectScan('3D Fingerprint Ridge Map', 'fingerprint');
      fbiState.statusText = `PRINT MATCHED (${bestScore}%)`;
      return { ok: true, report: evPrint, match: bestMatchName, score: bestScore };
    } finally {
      fbiState.busy = false;
      refreshAllWindows();
    }
  }

  function pickLocalEvidence(acceptStr, asSecondaryPrint = false) {
    const inp = document.createElement('input');
    inp.type = 'file';
    inp.accept = acceptStr || 'image/*,video/*';
    inp.onchange = (e) => {
      const f = e.target.files?.[0];
      if (!f) return;
      const url = URL.createObjectURL(f);
      if (f.type.startsWith('video/')) {
        const v = document.createElement('video');
        v.src = url; v.muted = true; v.loop = true; v.playsInline = true; v.crossOrigin = 'anonymous';
        v.onloadeddata = () => loadEvidenceFromVideo(v, f.name);
      } else {
        const img = new Image();
        img.crossOrigin = 'anonymous';
        img.onload = () => {
          if (asSecondaryPrint) {
            fbiState.secondaryPrintImg = img;
            pushLog(`Loaded Suspect Reference Print: ${f.name}`);
            APP.ui?.toast?.(`Loaded reference print: ${f.name}`);
          } else {
            loadEvidenceFromImage(img, f.name);
          }
        };
        img.src = url;
      }
    };
    inp.click();
  }

  let draggingWs = false;
  const dragPlane = new THREE.Plane(), dragOff = new THREE.Vector3();

  function onPtrDown(e) {
    if (!fbiState.open || e.button !== 0 || !APP.camera) return;
    const el = APP.renderer?.domElement;
    const rect = el ? el.getBoundingClientRect() : { left: 0, top: 0, width: window.innerWidth, height: window.innerHeight };
    dNdc.x = ((e.clientX - rect.left) / Math.max(1, rect.width)) * 2 - 1;
    dNdc.y = -((e.clientY - rect.top) / Math.max(1, rect.height)) * 2 + 1;
    dRay.setFromCamera(dNdc, APP.camera);

    const dh = dRay.intersectObject(dragBar, false);
    if (dh.length) {
      draggingWs = true;
      dragPlane.setFromNormalAndCoplanarPoint(APP.camera.getWorldDirection(new THREE.Vector3()).negate(), dh[0].point);
      dragOff.copy(wsRoot.position).sub(dh[0].point);
      return;
    }

    const mh = dRay.intersectObject(mainScreen, false);
    if (mh.length && mh[0].uv) {
      roiDragStart = { u: mh[0].uv.x, v: 1 - mh[0].uv.y };
    }
  }

  function onPtrMove(e) {
    if (!fbiState.open || !APP.camera) return;
    const el = APP.renderer?.domElement;
    const rect = el ? el.getBoundingClientRect() : { left: 0, top: 0, width: window.innerWidth, height: window.innerHeight };
    dNdc.x = ((e.clientX - rect.left) / Math.max(1, rect.width)) * 2 - 1;
    dNdc.y = -((e.clientY - rect.top) / Math.max(1, rect.height)) * 2 + 1;
    dRay.setFromCamera(dNdc, APP.camera);

    if (draggingWs) {
      const pt = new THREE.Vector3();
      if (dRay.ray.intersectPlane(dragPlane, pt)) wsRoot.position.copy(pt).add(dragOff);
      return;
    }

    if (roiDragStart) {
      const mh = dRay.intersectObject(mainScreen, false);
      if (mh.length && mh[0].uv) {
        const u2 = mh[0].uv.x, v2 = 1 - mh[0].uv.y;
        const x = Math.min(roiDragStart.u, u2), y = Math.min(roiDragStart.v, v2);
        const w = Math.abs(u2 - roiDragStart.u), h = Math.abs(v2 - roiDragStart.v);
        if (w > 0.03 && h > 0.03) { fbiState.roi = { x, y, w, h }; redrawMainViewport(); }
      }
      return;
    }

    const ch = dRay.intersectObject(cmdScreen, false);
    let newHov = null;
    if (ch.length && ch[0].uv) {
      const uv = ch[0].uv;
      for (const b of cmdButtons) if (uv.x >= b.u0 && uv.x <= b.u1 && uv.y >= b.v0 && uv.y <= b.v1) { newHov = b.id; break; }
    }
    if (newHov !== hoveredCmdId) { hoveredCmdId = newHov; redrawCmdDeck(); }
  }

  function onPtrUp() { draggingWs = false; roiDragStart = null; }

  let codeRefreshTick = 0;
  function updateForensicSuite(dt, elapsed) {
    if (!fbiState.open || !wsRoot.visible) return;
    const now = elapsed || (performance.now() * 0.001);
    const delta = Math.min(0.05, dt || 0.016);

    scanOverlayUniforms.uTime.value = now;
    scanOverlayUniforms.uBusy.value = fbiState.busy ? 1.0 : 0.0;

    // Rotate 3D Volumetric Subject Scan Model & Laser Slice Ring
    scanModelSpinGroup.rotation.y = now * 0.65;
    holoRingBase.rotation.z = now * 0.4;
    holoRingInner.rotation.z = -now * 0.8;
    holoSliceRing.position.y = Math.sin(now * 2.2) * 0.13;

    // Drift ambient particles
    const arr = ambGeo.attributes.position.array;
    for (let i = 0; i < ambCount; i++) {
      arr[i * 3 + 1] += 0.06 * delta;
      if (arr[i * 3 + 1] > 0.62) arr[i * 3 + 1] = -0.62;
    }
    ambGeo.attributes.position.needsUpdate = true;

    // Cycle active clue highlight & mini text tape every 3.8s
    if (fbiState.clues.length > 1 && !fbiState.busy) {
      fbiState.clueCycleTimer += delta;
      if (fbiState.clueCycleTimer > 3.8) {
        fbiState.clueCycleTimer = 0;
        fbiState.activeClueIdx = (fbiState.activeClueIdx + 1) % fbiState.clues.length;
        redrawMainViewport();
      }
    }

    // Live Video Frame Rendering + MediaPipe Video Tracking
    if (fbiState.evidenceType === 'video' && fbiState.sourceVideo && !fbiState.sourceVideo.paused) {
      if (fbiState.trackingActive && performance.now() - fbiState.lastVideoDetectTs > 140) {
        fbiState.lastVideoDetectTs = performance.now();
        runMediaPipeOnCurrentFrame(true);
      }
      redrawMainViewport();
    }

    codeRefreshTick += delta;
    if (codeRefreshTick > 0.22) {
      codeRefreshTick = 0;
      redrawCodeWindow(now);
    }
  }
