# ======================================================================
# instruction5.txt — AI Learning Memory Engine (learning.txt Manager)
# ======================================================================

@ability: ai_learning_engine
@autorun
@pin: Learning Memory
@desc: Loads learning.txt on boot, recalls relevant shape/vocab/mistake lessons for AI, records new corrections automatically, and syncs learning.txt to GitHub.
@match: learning memory | show memory | sync learning | ذاكرة التعلم
@code:
(async function () {
  if (typeof APP._inst5Cleanup === 'function') { try { APP._inst5Cleanup(); } catch (_) {} }

  const FILENAME = 'learning.txt';
  const LS_KEY = 'APP_CACHE_' + FILENAME;
  const MAX_CHARS = 28000;
  const DEFAULT_TEMPLATE = [
    '# ======================================================================',
    '# learning.txt — NexusXR Persistent AI Memory & Self-Learning Database',
    '# ======================================================================',
    '',
    '[RECOGNITION_LESSONS]',
    '',
    '[USER_VOCABULARY]',
    '',
    '[MODELING_MISTAKES_AND_FIXES]',
    '',
    '[COMMAND_PREFERENCES]',
    '',
    '[AI_NOTES]',
    '- Memory initialized.'
  ].join('\n');

  let memoryText = '';
  let commitTimer = null;
  let dirtyForGitHub = false;

  const SECTION_MAP = {
    RECOGNITION: '[RECOGNITION_LESSONS]',
    CORRECTION: '[RECOGNITION_LESSONS]',
    VOCAB: '[USER_VOCABULARY]',
    MISTAKE: '[MODELING_MISTAKES_AND_FIXES]',
    FIX: '[MODELING_MISTAKES_AND_FIXES]',
    COMMAND: '[COMMAND_PREFERENCES]',
    NOTE: '[AI_NOTES]'
  };

  function encodeB64(str) {
    const bytes = new TextEncoder().encode(str);
    let bin = '';
    for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
    return btoa(bin);
  }

  async function loadFromSources() {
    let txt = null;
    if (typeof APP.fetchModuleText === 'function') {
      try { txt = await APP.fetchModuleText(FILENAME); } catch (_) {}
    }
    if (!txt) {
      try {
        const res = await fetch(`./${FILENAME}?t=${Date.now()}`, { cache: 'no-store' });
        if (res.ok) txt = await res.text();
      } catch (_) {}
    }
    if (!txt) {
      try { txt = localStorage.getItem(LS_KEY); } catch (_) {}
    }
    memoryText = (txt && txt.trim()) ? txt : DEFAULT_TEMPLATE;
    persistLocal(memoryText);
    return memoryText;
  }

  function persistLocal(txt) {
    memoryText = txt;
    if (APP.modules) APP.modules[FILENAME] = txt;
    try { localStorage.setItem(LS_KEY, txt); } catch (_) {}
  }

  function insertIntoSection(rawTxt, sectionHeader, bulletLine) {
    const cleanLine = '- ' + String(bulletLine || '').replace(/^\s*-\s*/, '').replace(/\r?\n/g, ' ').trim();
    if (!cleanLine || cleanLine === '-') return rawTxt;

    // Deduplicate identical or near-identical lines
    const lines = rawTxt.split('\n');
    const normNew = cleanLine.toLowerCase();
    if (lines.some((l) => l.trim().toLowerCase() === normNew)) return rawTxt;

    const secIdx = lines.findIndex((l) => l.trim().toUpperCase() === sectionHeader.toUpperCase());
    if (secIdx === -1) {
      return rawTxt.trim() + '\n\n' + sectionHeader + '\n' + cleanLine + '\n';
    }

    // Find end of this section (before next [SECTION])
    let insertAt = lines.length;
    for (let i = secIdx + 1; i < lines.length; i++) {
      if (/^\[[A-Z0-9_]+\]$/.test(lines[i].trim())) {
        insertAt = i;
        while (insertAt > secIdx + 1 && lines[insertAt - 1].trim() === '') insertAt--;
        break;
      }
    }
    lines.splice(insertAt, 0, cleanLine);

    let joined = lines.join('\n');
    if (joined.length > MAX_CHARS) {
      // Trim oldest bullet inside largest section if exceeding size cap
      const firstBullet = lines.findIndex((l, idx) => idx > secIdx && l.trim().startsWith('- '));
      if (firstBullet !== -1) {
        lines.splice(firstBullet, 1);
        joined = lines.join('\n');
      }
    }
    return joined;
  }

  function record(type, text) {
    const key = String(type || 'NOTE').toUpperCase();
    const header = SECTION_MAP[key] || '[AI_NOTES]';
    const stamp = new Date().toISOString().slice(0, 10);
    const entry = key === 'CORRECTION' || key === 'MISTAKE' ? `(${stamp}) ${text}` : String(text);
    const updated = insertIntoSection(memoryText || DEFAULT_TEMPLATE, header, entry);
    if (updated === memoryText) return false;
    persistLocal(updated);
    dirtyForGitHub = true;
    scheduleAutoCommit();
    return true;
  }

  function getSectionLines(header) {
    const lines = (memoryText || '').split('\n');
    const idx = lines.findIndex((l) => l.trim().toUpperCase() === header.toUpperCase());
    if (idx === -1) return [];
    const out = [];
    for (let i = idx + 1; i < lines.length; i++) {
      const tr = lines[i].trim();
      if (/^\[[A-Z0-9_]+\]$/.test(tr)) break;
      if (tr && !tr.startsWith('#')) out.push(tr);
    }
    return out;
  }

  function getRelevantContext(opts = {}) {
    if (!memoryText) return '';
    const recLines = getSectionLines('[RECOGNITION_LESSONS]').slice(-10);
    const mistLines = getSectionLines('[MODELING_MISTAKES_AND_FIXES]').slice(-8);
    const vocabLines = getSectionLines('[USER_VOCABULARY]').slice(-8);
    const noteLines = getSectionLines('[AI_NOTES]').slice(-5);

    const chunks = [];
    if (recLines.length) chunks.push('Shape Lessons:\n' + recLines.join('\n'));
    if (mistLines.length) chunks.push('Mistakes To Avoid:\n' + mistLines.join('\n'));
    if (vocabLines.length) chunks.push('User Vocab:\n' + vocabLines.join('\n'));
    if (noteLines.length) chunks.push('Notes:\n' + noteLines.join('\n'));
    return chunks.join('\n\n');
  }

  async function commitToGitHub(silent = false) {
    const cfg = APP.repo || {};
    if (!cfg.owner || !cfg.name || !cfg.pat) {
      if (!silent) APP.ui?.toast?.('Saved locally. Add GitHub Owner/Repo/PAT in GitHub .txt Editor to sync to repo.', 3600);
      return { ok: false, error: 'Missing GitHub Owner/Repo/PAT' };
    }
    const branch = cfg.branch || 'main';
    const headers = {
      'Accept': 'application/vnd.github+json',
      'Authorization': 'Bearer ' + cfg.pat,
      'Content-Type': 'application/json'
    };
    const api = async (path, init) => {
      const res = await fetch(`https://api.github.com/repos/${encodeURIComponent(cfg.owner)}/${encodeURIComponent(cfg.name)}${path}`, Object.assign({}, init || {}, { headers }));
      const txt = await res.text();
      if (!res.ok) throw new Error(`GitHub ${res.status}: ${txt.slice(0, 140)}`);
      return JSON.parse(txt || '{}');
    };
    try {
      const refData = await api(`/git/refs/heads/${encodeURIComponent(branch)}`);
      const headSha = refData.object.sha;
      const headCommit = await api(`/git/commits/${headSha}`);
      const blob = await api('/git/blobs', { method: 'POST', body: JSON.stringify({ content: encodeB64(memoryText), encoding: 'base64' }) });
      const tree = await api('/git/trees', { method: 'POST', body: JSON.stringify({ base_tree: headCommit.tree.sha, tree: [{ path: FILENAME, mode: '100644', type: 'blob', sha: blob.sha }] }) });
      const commit = await api('/git/commits', { method: 'POST', body: JSON.stringify({ message: 'Update learning.txt AI memory', tree: tree.sha, parents: [headSha] }) });
      await api(`/git/refs/heads/${encodeURIComponent(branch)}`, { method: 'PATCH', body: JSON.stringify({ sha: commit.sha, force: false }) });
      dirtyForGitHub = false;
      if (!silent) APP.ui?.toast?.(`Synced learning.txt to GitHub (${commit.sha.slice(0, 7)})`, 2600);
      return { ok: true, sha: commit.sha };
    } catch (e) {
      if (!silent) APP.ui?.toast?.('GitHub sync failed: ' + e.message, 4000);
      return { ok: false, error: e.message };
    }
  }

  function scheduleAutoCommit() {
    const cfg = APP.repo || {};
    if (!cfg.owner || !cfg.name || !cfg.pat) return;
    clearTimeout(commitTimer);
    commitTimer = setTimeout(() => {
      if (dirtyForGitHub) commitToGitHub(true);
    }, 18000);
  }

  function openMemoryViewer() {
    const old = document.getElementById('nxr-learning-modal');
    if (old) old.remove();
    const wrap = document.createElement('div');
    wrap.id = 'nxr-learning-modal';
    wrap.style.cssText = 'position:fixed;inset:0;z-index:99999;background:rgba(0,0,0,0.72);backdrop-filter:blur(6px);display:flex;align-items:center;justify-content:center;padding:16px;';
    wrap.innerHTML = `
      <div class="glass" style="width:min(94vw,720px);max-height:86vh;padding:16px;display:flex;flex-direction:column;gap:10px;background:rgba(10,16,26,0.95);border:1px solid rgba(0,240,180,0.35);border-radius:14px;color:#e2e8f0;">
        <div style="display:flex;justify-content:space-between;align-items:center;">
          <strong style="color:#00f0b4;font-size:15px;">learning.txt — AI Persistent Memory</strong>
          <button id="lrn-close" class="btn">Close</button>
        </div>
        <textarea id="lrn-ta" spellcheck="false" style="width:100%;min-height:340px;font-family:ui-monospace,monospace;font-size:12px;background:rgba(0,0,0,0.45);color:#e2e8f0;border:1px solid rgba(255,255,255,0.2);border-radius:10px;padding:10px;"></textarea>
        <div style="display:flex;gap:8px;justify-content:flex-end;flex-wrap:wrap;">
          <button id="lrn-save-local" class="btn">Save Local</button>
          <button id="lrn-sync-gh" class="btn active">Sync to GitHub</button>
        </div>
      </div>
    `;
    document.body.appendChild(wrap);
    const ta = wrap.querySelector('#lrn-ta');
    ta.value = memoryText;
    wrap.querySelector('#lrn-close').onclick = () => wrap.remove();
    wrap.onclick = (e) => { if (e.target === wrap) wrap.remove(); };
    wrap.querySelector('#lrn-save-local').onclick = () => {
      persistLocal(ta.value);
      APP.ui?.toast?.('Saved learning.txt locally!', 2200);
    };
    wrap.querySelector('#lrn-sync-gh').onclick = async () => {
      persistLocal(ta.value);
      await commitToGitHub(false);
    };
  }

  await loadFromSources();

  APP.learning = {
    read: () => memoryText,
    record,
    getRelevantContext,
    commitToGitHub,
    openViewer: openMemoryViewer,
    reload: loadFromSources
  };

  function registerTools() {
    if (typeof APP.registerBotTool !== 'function') return;
    APP.registerBotTool(
      {
        name: 'save_learning_note',
        description: 'Save a permanent lesson or note into learning.txt so AI remembers shapes, user vocabulary/Darja, preferences, or mistakes to avoid in future sessions.',
        parameters: {
          type: 'object',
          properties: {
            category: { type: 'string', description: 'RECOGNITION | VOCAB | MISTAKE | COMMAND | NOTE' },
            lesson: { type: 'string', description: 'Concise lesson or note to remember permanently.' }
          },
          required: ['category', 'lesson']
        }
      },
      async (args) => {
        const added = record(args?.category || 'NOTE', args?.lesson || '');
        return { ok: true, saved: added, message: added ? 'Lesson saved to learning.txt.' : 'Lesson already in memory.' };
      },
      'LEARNING MEMORY: Whenever the user teaches you a word, corrects a shape/model, or you discover a useful modeling fix, call `save_learning_note` to store it in `learning.txt`.'
    );

    APP.registerBotTool(
      {
        name: 'read_learning_memory',
        description: 'Read notes and past lessons stored in learning.txt.',
        parameters: {
          type: 'object',
          properties: { filter: { type: 'string', description: 'Optional keyword filter' } }
        }
      },
      async (args) => {
        const q = String(args?.filter || '').toLowerCase().trim();
        if (!q) return { ok: true, content: getRelevantContext() || memoryText.slice(0, 2000) };
        const hits = memoryText.split('\n').filter((l) => l.toLowerCase().includes(q));
        return { ok: true, matches: hits.slice(-20) };
      }
    );
  }
  registerTools();
  const unsubMods = APP.on?.('modules:loaded', registerTools);

  APP._inst5Cleanup = () => {
    clearTimeout(commitTimer);
    try { unsubMods && unsubMods(); } catch (_) {}
  };

  APP.registerAbilityHandler?.('ai_learning_engine', async function (text) {
    const low = String(text || '').toLowerCase();
    if (low.includes('sync')) return await commitToGitHub(false);
    openMemoryViewer();
    return { ok: true, message: 'Opened learning.txt viewer.' };
  });
})();
@end
