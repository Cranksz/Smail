  if (APP._inst2ExecWrapped || typeof APP.executeCommand !== 'function') return;

  const origExec = APP.executeCommand;
  APP.executeCommand = async function (rawText, source = 'text') {
    const text = String(rawText || '').trim();
    if (text && APP.instWeight > 0 && APP.instWeight < 100) {
      const ab = APP.findInstructionMatch?.(text);
      if (ab && ab.file === 'instruction2.txt') {
        APP.ui?.setStatus?.(`Running ${ab.id}...`, 'busy');
        const res = await APP.runAbility(ab, text);
        APP.ui?.setStatus?.('Ready', 'ok');
        return res;
      }
    }
    return await origExec.call(this, rawText, source);
  };

})();

(function () {
  const THREE = APP.THREE;
  if (!THREE || !APP.scene) return { ok: false, error: 'Scene not ready.' };

  const existing = (APP.entities || []).find((e) => e && e.name === 'Interactive Water Pool');
  if (existing) APP.removeEntity?.(existing, false);

  const waterGroup = new THREE.Group();
  waterGroup.name = 'Interactive_Water_Pool';

  const POOL_SIZE = 2.05;
  const GRID = 52;
  const TOTAL = GRID * GRID;
  const BASE_Y = 0.13;

  // Sleek dark slate basin walls + bed
  const stoneMat = new THREE.MeshStandardMaterial({ color: 0x1e293b, roughness: 0.78, metalness: 0.18 });
  const bed = new THREE.Mesh(new THREE.BoxGeometry(POOL_SIZE + 0.12, 0.025, POOL_SIZE + 0.12), stoneMat);
  bed.position.y = 0.0125;
  waterGroup.add(bed);

  const wallH = 0.16, wallT = 0.06;
  const half = POOL_SIZE * 0.5 + wallT * 0.5;
  const wN = new THREE.Mesh(new THREE.BoxGeometry(POOL_SIZE + wallT * 2, wallH, wallT), stoneMat);
  wN.position.set(0, wallH * 0.5, -half);
  const wS = wN.clone(); wS.position.z = half;
  const wE = new THREE.Mesh(new THREE.BoxGeometry(wallT, wallH, POOL_SIZE), stoneMat);
  wE.position.set(half, wallH * 0.5, 0);
  const wW = wE.clone(); wW.position.x = -half;
  waterGroup.add(wN, wS, wE, wW);

  // Shimmering underwater caustic bed
  const causticUniforms = { uTime: { value: 0 } };
  const causticBed = new THREE.Mesh(
    new THREE.PlaneGeometry(POOL_SIZE, POOL_SIZE),
    new THREE.ShaderMaterial({
      uniforms: causticUniforms,
      transparent: true,
      depthWrite: false,
      blending: THREE.AdditiveBlending,
      vertexShader: 'varying vec2 vUv; void main() { vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0); }',
      fragmentShader: [
        'uniform float uTime; varying vec2 vUv;',
        'float cPat(vec2 uv, float t) {',
        '  vec2 p = mod(uv * 6.28318, 6.28318) - 250.0;',
        '  vec2 i = p; float c = 1.0; float inten = 0.005;',
        '  for (int n = 0; n < 3; n++) {',
        '    float tn = t * (1.0 - (3.2 / float(n + 1)));',
        '    i = p + vec2(cos(tn - i.x) + sin(tn + i.y), sin(tn - i.y) + cos(tn + i.x));',
        '    c += 1.0 / length(vec2(p.x / (sin(i.x + tn) / inten), p.y / (cos(i.y + tn) / inten)));',
        '  }',
        '  c /= 3.0; c = 1.17 - pow(c, 1.4);',
        '  return pow(abs(c), 7.0);',
        '}',
        'void main() {',
        '  float c = cPat(vUv * 1.8, uTime * 0.85);',
        '  gl_FragColor = vec4(vec3(0.28, 0.82, 1.0) * c * 0.95, clamp(c * 0.85, 0.0, 0.85));',
        '}'
      ].join('\n')
    })
  );
  causticBed.rotation.x = -Math.PI / 2;
  causticBed.position.y = 0.028;
  waterGroup.add(causticBed);

  // 2D Wave Equation Buffers
  let hCur = new Float32Array(TOTAL);
  let hPrev = new Float32Array(TOTAL);
  let hNext = new Float32Array(TOTAL);
  const foamArr = new Float32Array(TOTAL);

  const waterGeo = new THREE.PlaneGeometry(POOL_SIZE, POOL_SIZE, GRID - 1, GRID - 1);
  waterGeo.rotateX(-Math.PI / 2);
  waterGeo.setAttribute('aFoam', new THREE.BufferAttribute(foamArr, 1));

  const waterUniforms = {
    uTime: { value: 0 },
    uDeep: { value: new THREE.Color(0x042b44) },
    uShallow: { value: new THREE.Color(0x0ea5e9) },
    uSky: { value: new THREE.Color(0xbae6fd) }
  };

  const waterMat = new THREE.ShaderMaterial({
    uniforms: waterUniforms,
    transparent: true,
    side: THREE.DoubleSide,
    vertexShader: [
      'attribute float aFoam;',
      'varying vec3 vWPos; varying vec3 vWNorm; varying float vFoam;',
      'void main() {',
      '  vFoam = aFoam;',
      '  vec4 wp = modelMatrix * vec4(position, 1.0);',
      '  vWPos = wp.xyz;',
      '  vWNorm = normalize(mat3(modelMatrix) * normal);',
      '  gl_Position = projectionMatrix * viewMatrix * wp;',
      '}'
    ].join('\n'),
    fragmentShader: [
      'uniform float uTime; uniform vec3 uDeep; uniform vec3 uShallow; uniform vec3 uSky;',
      'varying vec3 vWPos; varying vec3 vWNorm; varying float vFoam;',
      'void main() {',
      '  vec3 N = normalize(vWNorm);',
      '  vec3 V = normalize(cameraPosition - vWPos);',
      '  float fres = pow(1.0 - max(0.0, dot(N, V)), 3.2);',
      '  vec3 L = normalize(vec3(0.45, 0.85, 0.35));',
      '  vec3 H = normalize(L + V);',
      '  float spec = pow(max(0.0, dot(N, H)), 110.0);',
      '  vec3 col = mix(uDeep, uShallow, 0.32 + fres * 0.68);',
      '  col = mix(col, uSky, fres * 0.55);',
      '  col = mix(col, vec3(0.92, 0.98, 1.0), clamp(vFoam * 1.4, 0.0, 0.92));',
      '  col += vec3(1.0, 0.97, 0.90) * spec * 1.65;',
      '  float alpha = clamp(0.64 + fres * 0.30 + vFoam * 0.32 + spec * 0.3, 0.0, 0.95);',
      '  gl_FragColor = vec4(col, alpha);',
      '}'
    ].join('\n')
  });

  const waterMesh = new THREE.Mesh(waterGeo, waterMat);
  waterMesh.position.y = BASE_Y;
  waterGroup.add(waterMesh);

  // Splash droplets pool
  const DROP_COUNT = 60;
  const dropPos = new Float32Array(DROP_COUNT * 3);
  const dropVel = Array.from({ length: DROP_COUNT }, () => new THREE.Vector3());
  const dropLife = new Float32Array(DROP_COUNT);
  const dropGeo = new THREE.BufferGeometry();
  dropGeo.setAttribute('position', new THREE.BufferAttribute(dropPos, 3));
  const dropPts = new THREE.Points(dropGeo, new THREE.PointsMaterial({
    color: 0xe0f2fe, size: 0.016, transparent: true, opacity: 0.85, depthWrite: false
  }));
  dropPts.frustumCulled = false;
  waterGroup.add(dropPts);
  let nextDrop = 0;

  function emitSplash(lx, lz, strength) {
    const count = Math.min(8, Math.ceil(strength * 6));
    for (let i = 0; i < count; i++) {
      const idx = nextDrop;
      nextDrop = (nextDrop + 1) % DROP_COUNT;
      dropPos[idx * 3] = lx + (Math.random() - 0.5) * 0.06;
      dropPos[idx * 3 + 1] = BASE_Y + 0.02;
      dropPos[idx * 3 + 2] = lz + (Math.random() - 0.5) * 0.06;
      dropVel[idx].set((Math.random() - 0.5) * 0.7, 0.8 + Math.random() * 1.1 * strength, (Math.random() - 0.5) * 0.7);
      dropLife[idx] = 0.55 + Math.random() * 0.3;
    }
  }

  function disturb(lx, lz, amp, radiusCells) {
    const u = (lx / POOL_SIZE) + 0.5;
    const v = (lz / POOL_SIZE) + 0.5;
    if (u < 0.04 || u > 0.96 || v < 0.04 || v > 0.96) return;
    const gx = Math.floor(u * (GRID - 1));
    const gz = Math.floor(v * (GRID - 1));
    const r = radiusCells || 2;
    for (let dz = -r; dz <= r; dz++) {
      for (let dx = -r; dx <= r; dx++) {
        const nx = gx + dx, nz = gz + dz;
        if (nx <= 1 || nx >= GRID - 2 || nz <= 1 || nz >= GRID - 2) continue;
        const d = Math.hypot(dx, dz);
        if (d <= r) {
          hCur[nz * GRID + nx] += amp * Math.cos((d / r) * Math.PI * 0.5);
        }
      }
    }
  }

  const spawnPos = APP.getSpawnPosition ? APP.getSpawnPosition(1.35, true) : new THREE.Vector3(0, 0, -1.35);
  waterGroup.position.copy(spawnPos);
  APP.scene.add(waterGroup);

  // Initial central drop ripple
  disturb(0, 0, 0.045, 3);

  const tmpLocal = new THREE.Vector3();
  const prevHands = new Map();
  let ambientTick = 0;

  const ent = APP.registerEntity(waterGroup, {
    name: 'Interactive Water Pool',
    category: 'world_fx',
    grabbable: false,
    holdable: false,
    physics: { enabled: false, gravity: false },
    onUpdate: function (dt) {
      const delta = Math.min(0.033, Math.max(0.004, dt || 0.016));
      const now = performance.now() * 0.001;
      causticUniforms.uTime.value = now;
      waterUniforms.uTime.value = now;

      // Ambient gentle raindrop / breeze ripples
      ambientTick += delta;
      if (ambientTick > 0.55) {
        ambientTick = 0;
        disturb((Math.random() - 0.5) * POOL_SIZE * 0.78, (Math.random() - 0.5) * POOL_SIZE * 0.78, -0.012, 2);
      }

      // Hand & pointer ripples
      for (const [hk, h] of Object.entries(APP.hands || {})) {
        if (!h || !h.active) continue;
        const hp = h.isHand ? h.fingerTips?.index : h.pos;
        if (!hp) continue;
        tmpLocal.copy(hp);
        waterGroup.worldToLocal(tmpLocal);

        const last = prevHands.get(hk);
        const speed = last ? tmpLocal.distanceTo(last) / delta : 0;
        prevHands.set(hk, tmpLocal.clone());

        if (Math.abs(tmpLocal.x) < POOL_SIZE * 0.48 && Math.abs(tmpLocal.z) < POOL_SIZE * 0.48) {
          if (tmpLocal.y >= -0.05 && tmpLocal.y <= BASE_Y + 0.14 && (speed > 0.15 || h.trigger || h.pinching)) {
            disturb(tmpLocal.x, tmpLocal.z, -0.022 * Math.min(2.0, 0.6 + speed * 0.5), 2);
            if (speed > 0.4) emitSplash(tmpLocal.x, tmpLocal.z, Math.min(1.5, speed * 0.5));
          } else if (hk === 'dhand' && h.trigger && h.rayOrigin && h.rayDir) {
            // Desktop click ray intersection on water plane
            const planeY = waterGroup.position.y + BASE_Y;
            if (Math.abs(h.rayDir.y) > 0.01) {
              const tHit = (planeY - h.rayOrigin.y) / h.rayDir.y;
              if (tHit > 0.1 && tHit < 6.0) {
                tmpLocal.copy(h.rayOrigin).addScaledVector(h.rayDir, tHit);
                waterGroup.worldToLocal(tmpLocal);
                if (Math.abs(tmpLocal.x) < POOL_SIZE * 0.48 && Math.abs(tmpLocal.z) < POOL_SIZE * 0.48) {
                  disturb(tmpLocal.x, tmpLocal.z, -0.028, 3);
                  emitSplash(tmpLocal.x, tmpLocal.z, 0.9);
                }
              }
            }
          }
        }
      }

      // Entity splash interaction (any object touching/moving in pool)
      for (const other of APP.entities || []) {
        if (!other || other === ent || !other.group || other.category === 'world_fx' || other.category === 'board') continue;
        tmpLocal.copy(other.group.position);
        waterGroup.worldToLocal(tmpLocal);
        if (Math.abs(tmpLocal.x) < POOL_SIZE * 0.46 && Math.abs(tmpLocal.z) < POOL_SIZE * 0.46 && tmpLocal.y < BASE_Y + 0.18) {
          const spd = other.physics?.vel ? other.physics.vel.length() : 0;
          if (spd > 0.18) {
            disturb(tmpLocal.x, tmpLocal.z, -0.018 * Math.min(2.2, spd), 3);
            emitSplash(tmpLocal.x, tmpLocal.z, Math.min(1.4, spd * 0.4));
          }
        }
      }

      // 2D Wave Equation Solver
      for (let z = 1; z < GRID - 1; z++) {
        const row = z * GRID;
        for (let x = 1; x < GRID - 1; x++) {
          const i = row + x;
          const avg = (hCur[i - 1] + hCur[i + 1] + hCur[i - GRID] + hCur[i + GRID]) * 0.5;
          hNext[i] = THREE.MathUtils.clamp((avg - hPrev[i]) * 0.985, -0.09, 0.09);
        }
      }
      const tmp = hPrev; hPrev = hCur; hCur = hNext; hNext = tmp;

      // Apply heights, normals, and crest foam to geometry
      const posAttr = waterGeo.attributes.position;
      const normAttr = waterGeo.attributes.normal;
      const cell = POOL_SIZE / (GRID - 1);
      for (let z = 0; z < GRID; z++) {
        for (let x = 0; x < GRID; x++) {
          const i = z * GRID + x;
          const h = hCur[i];
          posAttr.setY(i, h);

          const hL = x > 0 ? hCur[i - 1] : h;
          const hR = x < GRID - 1 ? hCur[i + 1] : h;
          const hD = z > 0 ? hCur[i - GRID] : h;
          const hU = z < GRID - 1 ? hCur[i + GRID] : h;
          const nx = hL - hR, ny = 2.0 * cell, nz = hD - hU;
          const len = Math.sqrt(nx * nx + ny * ny + nz * nz) || 1;
          normAttr.setXYZ(i, nx / len, ny / len, nz / len);

          const steep = Math.abs(hR - hL) + Math.abs(hU - hD);
          foamArr[i] = Math.max(0, (h * 12.0 + steep * 18.0) - 0.16);
        }
      }
      posAttr.needsUpdate = true;
      normAttr.needsUpdate = true;
      waterGeo.attributes.aFoam.needsUpdate = true;

      // Update splash droplets
      for (let i = 0; i < DROP_COUNT; i++) {
        if (dropLife[i] <= 0) { dropPos[i * 3 + 1] = -99; continue; }
        dropLife[i] -= delta;
        dropVel[i].y -= 7.5 * delta;
        dropPos[i * 3] += dropVel[i].x * delta;
        dropPos[i * 3 + 1] += dropVel[i].y * delta;
        dropPos[i * 3 + 2] += dropVel[i].z * delta;
        if (dropPos[i * 3 + 1] < BASE_Y) dropLife[i] = 0;
      }
      dropGeo.attributes.position.needsUpdate = true;
    }
  });

  APP.emit?.('entity:created', ent);
  APP.ui?.toast?.('Water Simulation active! Touch or click water to make ripples.');
  APP.speakTTS?.('Interactive water simulation ready.');
  return { ok: true, id: ent.id, message: 'Spawned interactive 2D wave-equation water pool.' };
})();

(function () {
  const THREE = APP.THREE;
  if (!THREE || !APP.scene) return { ok: false, error: 'Scene not ready.' };

  const existing = (APP.entities || []).find((e) => e && e.name === 'Granular Sand Simulation');
  if (existing) APP.removeEntity?.(existing, false);

  const sandGroup = new THREE.Group();
  sandGroup.name = 'Granular_Sand_Sim';

  const SIZE = 1.9;
  const GRID = 44;
  const CELL = SIZE / (GRID - 1);
  const MAX_SLOPE = CELL * 0.68; // ~34 degrees angle of repose
  const pileH = new Float32Array(GRID * GRID);

  // Displaced sand dune surface mesh
  const pileGeo = new THREE.PlaneGeometry(SIZE, SIZE, GRID - 1, GRID - 1);
  pileGeo.rotateX(-Math.PI / 2);

  // Procedural mineral sand canvas texture
  const sCan = document.createElement('canvas');
  sCan.width = 256; sCan.height = 256;
  const sCtx = sCan.getContext('2d');
  sCtx.fillStyle = '#c8a472';
  sCtx.fillRect(0, 0, 256, 256);
  for (let i = 0; i < 6000; i++) {
    const r = Math.random();
    sCtx.fillStyle = r > 0.65 ? '#e4c798' : (r > 0.25 ? '#b38b59' : '#6b4e2e');
    sCtx.fillRect(Math.random() * 256, Math.random() * 256, 2, 2);
  }
  const sandTex = new THREE.CanvasTexture(sCan);
  sandTex.colorSpace = THREE.SRGBColorSpace;
  sandTex.wrapS = sandTex.wrapT = THREE.RepeatWrapping;
  sandTex.repeat.set(4, 4);

  const pileMat = new THREE.MeshStandardMaterial({
    map: sandTex,
    roughness: 0.92,
    metalness: 0.05
  });
  const pileMesh = new THREE.Mesh(pileGeo, pileMat);
  pileMesh.receiveShadow = true;
  pileMesh.castShadow = true;
  sandGroup.add(pileMesh);

  // Subtle brass emitter ring overhead
  const emitterRing = new THREE.Mesh(
    new THREE.TorusGeometry(0.08, 0.014, 10, 24),
    new THREE.MeshStandardMaterial({ color: 0xb45309, metalness: 0.85, roughness: 0.28 })
  );
  emitterRing.rotation.x = Math.PI / 2;
  emitterRing.position.set(0, 1.28, 0);
  sandGroup.add(emitterRing);

  // 1,000 instanced faceted mineral grains
  const GRAIN_COUNT = 1000;
  const grainGeo = new THREE.DodecahedronGeometry(0.0085, 0);
  const grainMat = new THREE.MeshStandardMaterial({ color: 0xffffff, roughness: 0.86, metalness: 0.08 });
  const grainInst = new THREE.InstancedMesh(grainGeo, grainMat, GRAIN_COUNT);
  grainInst.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
  grainInst.castShadow = true;
  sandGroup.add(grainInst);

  const gPos = new Float32Array(GRAIN_COUNT * 3);
  const gVel = new Float32Array(GRAIN_COUNT * 3);
  const gScale = new Float32Array(GRAIN_COUNT);
  const palette = [new THREE.Color(0xdcb884), new THREE.Color(0xc4985e), new THREE.Color(0xe9d2ac), new THREE.Color(0x8c673e)];
  const dummy = new THREE.Object3D();

  function resetGrain(i, ex, ez, burst) {
    const i3 = i * 3;
    gPos[i3] = ex + (Math.random() - 0.5) * (burst ? 0.09 : 0.05);
    gPos[i3 + 1] = burst ? (0.25 + Math.random() * 0.2) : (1.24 - Math.random() * 0.35);
    gPos[i3 + 2] = ez + (Math.random() - 0.5) * (burst ? 0.09 : 0.05);
    gVel[i3] = (Math.random() - 0.5) * (burst ? 1.4 : 0.16);
    gVel[i3 + 1] = burst ? (0.8 + Math.random() * 1.1) : (-0.2 - Math.random() * 0.35);
    gVel[i3 + 2] = (Math.random() - 0.5) * (burst ? 1.4 : 0.16);
    gScale[i] = 0.65 + Math.random() * 0.85;
  }

  for (let i = 0; i < GRAIN_COUNT; i++) {
    resetGrain(i, 0, 0, false);
    gPos[i * 3 + 1] = 0.05 + (i / GRAIN_COUNT) * 1.2;
    grainInst.setColorAt(i, palette[i % palette.length]);
  }
  if (grainInst.instanceColor) grainInst.instanceColor.needsUpdate = true;

  const spawnPos = APP.getSpawnPosition ? APP.getSpawnPosition(1.3, true) : new THREE.Vector3(0, 0, -1.3);
  sandGroup.position.copy(spawnPos);
  APP.scene.add(sandGroup);

  const tmpLocal = new THREE.Vector3();
  const tmpBox = new THREE.Box3();
  const tmpBoxCenter = new THREE.Vector3();

  const ent = APP.registerEntity(sandGroup, {
    name: 'Granular Sand Simulation',
    category: 'world_fx',
    grabbable: false,
    holdable: false,
    physics: { enabled: false, gravity: false },
    onUpdate: function (dt) {
      const delta = Math.min(0.033, Math.max(0.004, dt || 0.016));
      const now = performance.now() * 0.001;

      const emitX = Math.sin(now * 0.7) * 0.18;
      const emitZ = Math.cos(now * 0.55) * 0.18;
      emitterRing.position.set(emitX, 1.28, emitZ);

      // Collect nearby object collision boxes in sandGroup local space
      const obsBoxes = [];
      for (const other of APP.entities || []) {
        if (!other || other === ent || !other.group || other.category === 'world_fx' || other.category === 'board') continue;
        tmpBox.setFromObject(other.group);
        if (!tmpBox.isEmpty()) {
          tmpBox.min.sub(sandGroup.position);
          tmpBox.max.sub(sandGroup.position);
          obsBoxes.push(tmpBox.clone());
        }
      }

      // Hand scoop / crater carving
      for (const h of Object.values(APP.hands || {})) {
        if (!h || !h.active) continue;
        const hp = h.isHand ? h.fingerTips?.index : h.pos;
        if (!hp) continue;
        tmpLocal.copy(hp);
        sandGroup.worldToLocal(tmpLocal);
        const gx = Math.round((tmpLocal.x / SIZE + 0.5) * (GRID - 1));
        const gz = Math.round((tmpLocal.z / SIZE + 0.5) * (GRID - 1));
        if (gx >= 1 && gx < GRID - 1 && gz >= 1 && gz < GRID - 1) {
          const curPile = pileH[gz * GRID + gx];
          if (tmpLocal.y <= curPile + 0.08 && tmpLocal.y >= -0.05) {
            for (let dz = -2; dz <= 2; dz++) {
              for (let dx = -2; dx <= 2; dx++) {
                const nx = gx + dx, nz = gz + dz;
                if (nx > 0 && nx < GRID - 1 && nz > 0 && nz < GRID - 1) {
                  pileH[nz * GRID + nx] = Math.max(0, pileH[nz * GRID + nx] - 0.014);
                }
              }
            }
          }
        }
      }

      // Update falling & colliding grains
      for (let i = 0; i < GRAIN_COUNT; i++) {
        const i3 = i * 3;
        gVel[i3 + 1] -= 7.8 * delta;
        gPos[i3] += gVel[i3] * delta;
        gPos[i3 + 1] += gVel[i3 + 1] * delta;
        gPos[i3 + 2] += gVel[i3 + 2] * delta;

        // Object collision deflection
        for (let b = 0; b < obsBoxes.length; b++) {
          const ob = obsBoxes[b];
          if (gPos[i3] > ob.min.x && gPos[i3] < ob.max.x &&
              gPos[i3 + 2] > ob.min.z && gPos[i3 + 2] < ob.max.z &&

            gPos[i3 + 1] = ob.max.y + 0.01;
            gVel[i3] += (gPos[i3] - tmpBoxCenter.x) * 2.5;
            gVel[i3 + 2] += (gPos[i3 + 2] - tmpBoxCenter.z) * 2.5;
            gVel[i3 + 1] = Math.abs(gVel[i3 + 1]) * 0.25;
          }
        }

        // Pile heightmap deposition
        const gx = Math.floor((gPos[i3] / SIZE + 0.5) * (GRID - 1));
        const gz = Math.floor((gPos[i3 + 2] / SIZE + 0.5) * (GRID - 1));
        const inBounds = gx >= 1 && gx < GRID - 1 && gz >= 1 && gz < GRID - 1;
        const floorY = inBounds ? pileH[gz * GRID + gx] : 0;

        if (gPos[i3 + 1] <= floorY) {

          }

      grainInst.instanceMatrix.needsUpdate = true;

      // Angle-of-repose avalanche relaxation across pile heightmap
      const dirs = [-1, 1, -GRID, GRID];
      for (let pass = 0; pass < 2; pass++) {

      pileGeo.computeVertexNormals();

  APP.speakTTS?.('Granular sand simulation active.');
  return { ok: true, id: ent.id, message: 'Spawned granular sand simulation with angle-of-repose dunes.' };

    physics: { enabled: false, gravity: false },
    onUpdate: function (dt) {
      const safeDt = Math.min(0.025, Math.max(0.004, dt || 0.016));
      const now = performance.now() * 0.001;
      const gravY = -7.8 * safeDt * safeDt;
      const floorLocalY = -clothGroup.position.y + 0.02;

      // Hand grab, push & fast-swipe tear!
      let activeGrab = false;
      for (const [hk, h] of Object.entries(APP.hands || {})) {
        if (!h || !h.active) continue;
        let tip = h.isHand ? h.fingerTips?.index : h.pos;
        if (hk === 'dhand' && h.rayOrigin && h.rayDir) {
          // Project desktop ray onto cloth plane z=0
          const invM = clothGroup.matrixWorld.clone().invert();
          const ro = h.rayOrigin.clone().applyMatrix4(invM);
          const rd = h.rayDir.clone().transformDirection(invM);
          if (Math.abs(rd.z) > 0.01) {
            const t = -ro.z / rd.z;
            if (t > 0.1 && t < 5.0) tip = ro.addScaledVector(rd, t).applyMatrix4(clothGroup.matrixWorld);
          }
        }
        if (!tip) continue;

        tmpLocal.copy(tip);
        clothGroup.worldToLocal(tmpLocal);

        const last = prevHands.get(hk);
        const speed = last ? tmpLocal.distanceTo(last) / safeDt : 0;
        prevHands.set(hk, tmpLocal.clone());

        if ((h.trigger || h.pinching) && grabbedIdx === -1) {
          let bestI = -1, bestD = 0.14;
          for (let i = COLS; i < N_COUNT; i++) {
            const d = cur[i].distanceTo(tmpLocal);
            if (d < bestD) { bestD = d; bestI = i; }
          }
          if (bestI !== -1) { grabbedIdx = bestI; grabbedHand = hk; }
        }

        if (grabbedHand === hk && (h.trigger || h.pinching) && grabbedIdx !== -1) {
          cur[grabbedIdx].lerp(tmpLocal, 0.75);
          prev[grabbedIdx].copy(cur[grabbedIdx]);
          activeGrab = true;
        }

        // Fast swipe across fabric rips springs!
        if (speed > 1.35) {
          tearAtLocalPoint(tmpLocal, 0.085, new THREE.Vector3(0, 0, 0.012));
        } else if (speed > 0.1) {
          // Gentle push
          for (let i = COLS; i < N_COUNT; i++) {
            const d = cur[i].distanceTo(tmpLocal);
            if (d < 0.12 && d > 1e-4) {
              tmpDelta.subVectors(cur[i], tmpLocal).normalize();
              cur[i].addScaledVector(tmpDelta, (0.12 - d) * 0.45);
            }
          }
        }
      }
      if (!activeGrab) { grabbedIdx = -1; grabbedHand = null; }

      // Verlet integration + wind gusts
      for (let i = 0; i < N_COUNT; i++) {
        if (pinned[i] || i === grabbedIdx) continue;
        const p = cur[i], pr = prev[i];
        const vx = (p.x - pr.x) * 0.985;
        const vy = (p.y - pr.y) * 0.985;
        const vz = (p.z - pr.z) * 0.985;
        pr.copy(p);
        const windZ = (Math.sin(now * 2.4 + p.x * 3.1 + p.y * 2.0) * 0.5 + 0.35) * 0.0014;
        const windX = Math.cos(now * 1.7 + p.y * 2.5) * 0.00045;
        p.x += vx + windX;
        p.y += vy + gravY;
        p.z += vz + windZ;
        if (p.y < floorLocalY) p.y = floorLocalY;
      }

      // Distance constraints + stretch tearing
      for (let iter = 0; iter < 5; iter++) {
        for (let i = 0; i < springs.length; i++) {
          const s = springs[i];
          if (s.broken) continue;
          const pa = cur[s.a], pb = cur[s.b];
          tmpDelta.subVectors(pb, pa);
          const d = tmpDelta.length();
          if (d < 1e-6) continue;
          if (iter === 0 && d > s.tearDist) {
            s.broken = true;
            indicesDirty = true;
            emitRipFibers(pa);
            continue;
          }
          const diff = (d - s.rest) / d;
          const pinA = pinned[s.a] || s.a === grabbedIdx;
          const pinB = pinned[s.b] || s.b === grabbedIdx;
          if (pinA && pinB) continue;
          if (pinA) pb.addScaledVector(tmpDelta, -diff);
          else if (pinB) pa.addScaledVector(tmpDelta, diff);
          else { pa.addScaledVector(tmpDelta, diff * 0.5); pb.addScaledVector(tmpDelta, -diff * 0.5); }
        }
      }

      if (indicesDirty) rebuildClothIndices();

      for (let i = 0; i < N_COUNT; i++) {
        posArr[i * 3] = cur[i].x;
        posArr[i * 3 + 1] = cur[i].y;
        posArr[i * 3 + 2] = cur[i].z;
      }
      clothGeo.attributes.position.needsUpdate = true;
      clothGeo.computeVertexNormals();

      // Update torn fiber particles
      for (let i = 0; i < FIBER_MAX; i++) {
        if (fibLife[i] <= 0) { fibPos[i * 3 + 1] = -99; continue; }
        fibLife[i] -= safeDt;
        fibVel[i].y -= 3.5 * safeDt;
        fibPos[i * 3] += fibVel[i].x * safeDt;
        fibPos[i * 3 + 1] += fibVel[i].y * safeDt;
        fibPos[i * 3 + 2] += fibVel[i].z * safeDt;
      }
      fibGeo.attributes.position.needsUpdate = true;
    }
  });

  clothApi.entity = ent;
  APP._activeCloth = clothApi;
  APP._activeClothEnt = ent;
  APP.emit?.('entity:created', ent);
  APP.ui?.toast?.('Cloth Simulation active! Pull hard or swipe fast through it to tear.');
  APP.speakTTS?.('Tearable cloth simulation active.');
  return { ok: true, id: ent.id, message: 'Spawned interactive tearable Verlet cloth.' };
})();
@end

(async function () {

  if (!cloth || !cloth.entity || !APP.entities?.includes(cloth.entity)) {
    const ab = APP.abilities?.get('interactive_cloth_sim');
