@autorun
@desc: Builds composable APP.sim (water, waterfall, granular, cloth, tearing, disintegration) and registers multi-tool AI integration for both REST models and Gemini 3.8 Live.
@match: simulation engine status
@code:
(function () {
  const APP = window.APP || (window.APP = {});
  const THREE = APP.THREE;
  if (!THREE || !APP.scene) return;

  function parseHexColor(col, fallbackHex) {
    if (typeof col === 'number' && isFinite(col)) return new THREE.Color(col);
    if (typeof col === 'string' && col.trim()) {
      try { return new THREE.Color(col.trim()); } catch (_) {}
    }
    return new THREE.Color(fallbackHex);
  }

  function resolvePos(posOpt, distDefault = 1.15) {
    if (Array.isArray(posOpt) && posOpt.length === 3) {
      return new THREE.Vector3(Number(posOpt[0]) || 0, Number(posOpt[1]) || 0, Number(posOpt[2]) || 0);
    }
    return APP.getSpawnPosition ? APP.getSpawnPosition(distDefault, true) : new THREE.Vector3(0, 0, -distDefault);
  }

  function getEntityBounds(ent) {
    if (!ent || !ent.group) return null;
    ent.group.updateMatrixWorld(true);
    const box = new THREE.Box3().setFromObject(ent.group);
    if (box.isEmpty()) return null;
    const size = new THREE.Vector3(), center = new THREE.Vector3();
    box.getSize(size); box.getCenter(center);
    return { box, size, center };
  }

  // ====================================================================
  // 1. COMPOSABLE WATER ENGINE (Attach to ANY object, Waterfall, Pouring)
  // ====================================================================
  function createWaterSimulation(opts = {}) {
    const mode = String(opts.mode || '').toLowerCase();
    const attachEnt = opts.attach_to ? APP.findEntity?.(opts.attach_to) : null;
    const attachInfo = getEntityBounds(attachEnt);

    const container = opts.container
      ? String(opts.container).toLowerCase()
      : (attachInfo ? 'none' : (mode === 'bowl' || mode === 'cup' ? mode : 'basin'));

    const isGlassContainer = container.includes('glass');
    const isCup = container.includes('cup');
    const isBowl = container.includes('bowl') || isCup;
    const isPouring = !!opts.pouring || mode === 'pouring' || !!opts.pour_from;
    const isWaterfall = !!opts.waterfall || mode === 'waterfall';
    const isFountain = !!opts.fountain || mode === 'fountain';

    const waterColor = parseHexColor(opts.color, 0x0ea5e9);
    const deepColor = waterColor.clone().multiplyScalar(0.28);
    const glow = !!opts.glow;

    const waterGroup = new THREE.Group();
    waterGroup.name = attachEnt ? `Water_in_${attachEnt.name}` : isWaterfall ? 'Waterfall_Sim' : isBowl ? 'Water_in_Bowl' : 'Interactive_Water_Sim';

    const radius = Number(opts.radius) || (
      attachInfo ? Math.max(0.10, Math.min(attachInfo.size.x, attachInfo.size.z) * 0.38)
      : isCup ? 0.11 : isBowl ? 0.28 : 0.96
    );
    const vesselH = Number(opts.height) || (
      attachInfo ? Math.max(0.12, attachInfo.size.y * 0.85)
      : isCup ? 0.24 : isBowl ? 0.22 : 0.17
    );
    const holdable = opts.holdable !== undefined ? !!opts.holdable : isBowl;

    // 1A. Optional Container (skipped when container="none" inside custom AI object!)
    let innerProfileFn = () => radius * 0.96;
    if (isBowl && container !== 'none') {
      const pts = [];
      const steps = 14;
      const wallThick = Math.max(0.012, radius * 0.06);
      const baseR = isCup ? radius * 0.65 : radius * 0.42;

      pts.push(new THREE.Vector2(0.001, 0), new THREE.Vector2(baseR, 0));
      for (let i = 1; i <= steps; i++) {
        const t = i / steps;
        pts.push(new THREE.Vector2(THREE.MathUtils.lerp(baseR, radius, isCup ? Math.pow(t, 0.55) : Math.sqrt(t)), t * vesselH));
      }
      pts.push(new THREE.Vector2(radius - wallThick * 0.45, vesselH + wallThick * 0.35), new THREE.Vector2(radius - wallThick, vesselH));
      for (let i = steps - 1; i >= 1; i--) {
        const t = i / steps;
        const r = Math.max(0.01, THREE.MathUtils.lerp(baseR, radius, isCup ? Math.pow(t, 0.55) : Math.sqrt(t)) - wallThick);
        pts.push(new THREE.Vector2(r, Math.max(wallThick, t * vesselH)));
      }
      pts.push(new THREE.Vector2(0.001, wallThick));

      innerProfileFn = (y) => {
        const t = THREE.MathUtils.clamp(y / vesselH, 0.05, 0.98);
        return Math.max(0.02, THREE.MathUtils.lerp(baseR, radius, isCup ? Math.pow(t, 0.55) : Math.sqrt(t)) - wallThick * 1.08);
      };

      const vesselMat = isGlassContainer
        ? new THREE.MeshPhysicalMaterial({ color: 0xe6f7ff, transmission: 0.94, ior: 1.51, thickness: 0.14, roughness: 0.08, clearcoat: 1.0, side: THREE.DoubleSide })
        : new THREE.MeshStandardMaterial({ color: parseHexColor(opts.container_color, 0x334155), roughness: 0.35, metalness: 0.15, side: THREE.DoubleSide });

      const vesselMesh = new THREE.Mesh(new THREE.LatheGeometry(pts, 36), vesselMat);
      vesselMesh.castShadow = true; vesselMesh.receiveShadow = true;
      waterGroup.add(vesselMesh);
    } else if (container === 'basin') {
      const size = radius * 2;
      const stoneMat = new THREE.MeshStandardMaterial({ color: 0x1e293b, roughness: 0.78, metalness: 0.18 });
      const bed = new THREE.Mesh(new THREE.BoxGeometry(size + 0.12, 0.024, size + 0.12), stoneMat);
      bed.position.y = 0.012;
      waterGroup.add(bed);
      const wt = 0.06, half = radius + wt * 0.5;
      const wN = new THREE.Mesh(new THREE.BoxGeometry(size + wt * 2, vesselH, wt), stoneMat);
      wN.position.set(0, vesselH * 0.5, -half);
      const wS = wN.clone(); wS.position.z = half;
      const wE = new THREE.Mesh(new THREE.BoxGeometry(wt, vesselH, size), stoneMat);
      wE.position.set(half, vesselH * 0.5, 0);
      const wW = wE.clone(); wW.position.x = -half;
      waterGroup.add(wN, wS, wE, wW);
    }

    // 1B. Polar Wave-Equation Water Surface
    const RINGS = isBowl || attachInfo ? 14 : 24;
    const SEGS = isBowl || attachInfo ? 32 : 44;
    const V_COUNT = 1 + RINGS * SEGS;
    const wPos = new Float32Array(V_COUNT * 3);
    const wNorm = new Float32Array(V_COUNT * 3);
    const wFoam = new Float32Array(V_COUNT);
    const wIdx = [];

    wNorm[1] = 1;
    for (let r = 1; r <= RINGS; r++) {
      const rn = r / RINGS;
      for (let s = 0; s < SEGS; s++) {
        const ang = (s / SEGS) * Math.PI * 2;
        const vi = 1 + (r - 1) * SEGS + s;
        wPos[vi * 3] = Math.cos(ang) * rn;
        wPos[vi * 3 + 2] = Math.sin(ang) * rn;
        wNorm[vi * 3 + 1] = 1;
      }
    }
    for (let s = 0; s < SEGS; s++) wIdx.push(0, 1 + ((s + 1) % SEGS), 1 + s);
    for (let r = 1; r < RINGS; r++) {
      for (let s = 0; s < SEGS; s++) {
        const sn = (s + 1) % SEGS;
        const a = 1 + (r - 1) * SEGS + s, b = 1 + (r - 1) * SEGS + sn;
        const c = 1 + r * SEGS + s, d = 1 + r * SEGS + sn;
        wIdx.push(a, b, d, a, d, c);
      }
    }

    const waterGeo = new THREE.BufferGeometry();
    waterGeo.setAttribute('position', new THREE.BufferAttribute(wPos, 3));
    waterGeo.setAttribute('normal', new THREE.BufferAttribute(wNorm, 3));
    waterGeo.setAttribute('aFoam', new THREE.BufferAttribute(wFoam, 1));
    waterGeo.setIndex(wIdx);

    const waterUniforms = { uTime: { value: 0 }, uDeep: { value: deepColor }, uShallow: { value: waterColor }, uGlow: { value: glow ? 0.65 : 0.0 } };
    const waterMesh = new THREE.Mesh(waterGeo, new THREE.ShaderMaterial({
      uniforms: waterUniforms,
      transparent: true,
      side: THREE.DoubleSide,
      vertexShader: [
        'attribute float aFoam; varying vec3 vWPos; varying vec3 vWNorm; varying float vFoam;',
        'void main() { vFoam = aFoam; vec4 wp = modelMatrix * vec4(position, 1.0); vWPos = wp.xyz; vWNorm = normalize(mat3(modelMatrix) * normal); gl_Position = projectionMatrix * viewMatrix * wp; }'
      ].join('\n'),
      fragmentShader: [
        'uniform float uTime; uniform vec3 uDeep; uniform vec3 uShallow; uniform float uGlow;',
        'varying vec3 vWPos; varying vec3 vWNorm; varying float vFoam;',
        'void main() {',
        '  vec3 N = normalize(vWNorm); vec3 V = normalize(cameraPosition - vWPos);',
        '  float fres = pow(1.0 - max(0.0, dot(N, V)), 3.0);',
        '  vec3 H = normalize(normalize(vec3(0.45, 0.85, 0.35)) + V);',
        '  float spec = pow(max(0.0, dot(N, H)), 96.0);',
        '  vec3 col = mix(uDeep, uShallow, 0.36 + fres * 0.64) + uShallow * uGlow;',
        '  col = mix(col, vec3(0.92, 0.98, 1.0), clamp(vFoam * 1.5, 0.0, 0.9));',
        '  col += vec3(1.0, 0.97, 0.92) * spec * 1.55;',
        '  gl_FragColor = vec4(col, clamp(0.68 + fres * 0.28 + vFoam * 0.30 + spec * 0.25, 0.0, 0.96));',
        '}'
      ].join('\n')
    }));

    // Position waterGroup (attached to custom AI object OR standalone)
    if (attachInfo) {
      waterGroup.position.set(attachInfo.center.x, attachInfo.box.min.y, attachInfo.center.z);
    } else {
      waterGroup.position.copy(resolvePos(opts.position, isBowl ? 0.95 : 1.35));
    }
    APP.scene.add(waterGroup);

    let fillLevel = isPouring && !attachInfo ? 0.22 : (Number(opts.fill) || 0.65);
    const maxFill = 0.88;
    const minY = isBowl ? vesselH * 0.15 : (attachInfo ? attachInfo.size.y * 0.25 : 0.05);
    const maxY = isBowl ? vesselH * 0.88 : (attachInfo ? attachInfo.size.y * 0.72 : vesselH * 0.82);
    let curWaterY = THREE.MathUtils.lerp(minY, maxY, fillLevel);
    let curR = innerProfileFn(curWaterY);
    waterMesh.position.y = curWaterY;
    waterMesh.scale.set(curR, 1, curR);
    waterGroup.add(waterMesh);

    // 1C. Pouring Stream / Waterfall
    const streamUniforms = { uTime: { value: 0 }, uColor: { value: waterColor }, uGlow: { value: glow ? 0.65 : 0.0 } };
    const streamMat = new THREE.ShaderMaterial({
      uniforms: streamUniforms, transparent: true, side: THREE.DoubleSide,
      vertexShader: 'varying vec2 vUv; void main() { vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0); }',
      fragmentShader: [
        'uniform float uTime; uniform vec3 uColor; uniform float uGlow; varying vec2 vUv;',
        'void main() {',
        '  float flow = sin(vUv.x * 28.0 - uTime * 18.0) * 0.5 + 0.5;',
        '  float foam = smoothstep(0.60, 1.0, flow) * 0.5;',
        '  gl_FragColor = vec4(mix(uColor * 0.85, vec3(0.92, 0.98, 1.0), foam) + uColor * uGlow, 0.80 + foam * 0.18);',
        '}'
      ].join('\n')
    });

    let impactLocal = new THREE.Vector3(0, curWaterY, 0);
    if (isPouring) {
      let spoutLocal = new THREE.Vector3(radius * 0.85, vesselH + 0.52, 0);
      let customSpout = false;
      if (Array.isArray(opts.pour_from) && opts.pour_from.length === 3) {
        spoutLocal.set(opts.pour_from[0], opts.pour_from[1], opts.pour_from[2]).sub(waterGroup.position);
        customSpout = true;
      } else if (typeof opts.pour_from === 'string') {
        const srcInfo = getEntityBounds(APP.findEntity?.(opts.pour_from));
        if (srcInfo) {
          spoutLocal.copy(srcInfo.center).sub(waterGroup.position);
          spoutLocal.y = srcInfo.box.min.y - waterGroup.position.y + srcInfo.size.y * 0.35;
          customSpout = true;
        }
      }

      if (!customSpout && !attachInfo) {
        const pitcher = new THREE.Group();
        const body = new THREE.Mesh(new THREE.CylinderGeometry(0.045, 0.065, 0.16, 16), new THREE.MeshStandardMaterial({ color: 0xb45309, metalness: 0.82, roughness: 0.26 }));
        const spout = new THREE.Mesh(new THREE.ConeGeometry(0.025, 0.08, 12), new THREE.MeshStandardMaterial({ color: 0xb45309, metalness: 0.82, roughness: 0.26 }));
        spout.rotation.z = Math.PI * 0.35; spout.position.set(-0.065, 0.045, 0);
        pitcher.add(body, spout);
        pitcher.position.copy(spoutLocal).add(new THREE.Vector3(0.09, 0.04, 0));
        pitcher.rotation.z = 0.55;
        waterGroup.add(pitcher);
      }

      const mid = new THREE.Vector3(spoutLocal.x * 0.35, Math.max(spoutLocal.y, curWaterY + 0.18), spoutLocal.z * 0.35);
      const curve = new THREE.QuadraticBezierCurve3(spoutLocal, mid, new THREE.Vector3(0.02, curWaterY * 0.6, 0));
      waterGroup.add(new THREE.Mesh(new THREE.TubeGeometry(curve, 24, Math.min(0.026, radius * 0.09), 12, false), streamMat));
      impactLocal.set(0.02, curWaterY, 0);
    } else if (isWaterfall) {
      const wfW = radius * 0.95, wfH = 1.05;
      const wfGeo = new THREE.PlaneGeometry(wfW, wfH, 12, 16);
      const pAttr = wfGeo.attributes.position;
      for (let i = 0; i < pAttr.count; i++) {
        const yn = (pAttr.getY(i) + wfH * 0.5) / wfH;
        pAttr.setZ(i, Math.pow(1 - yn, 1.5) * 0.35);
      }
      wfGeo.computeVertexNormals();
      const wfMesh = new THREE.Mesh(wfGeo, new THREE.ShaderMaterial({
        uniforms: streamUniforms, transparent: true, side: THREE.DoubleSide,
        vertexShader: 'varying vec2 vUv; void main() { vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0); }',
        fragmentShader: [
          'uniform float uTime; uniform vec3 uColor; uniform float uGlow; varying vec2 vUv;',
          'void main() {',
          '  float s1 = sin(vUv.y * 32.0 + uTime * 14.0) * 0.5 + 0.5;',
          '  float s2 = cos(vUv.x * 26.0 + vUv.y * 18.0 + uTime * 11.0) * 0.5 + 0.5;',
          '  float foam = smoothstep(0.55, 1.0, s1 * s2) + (1.0 - smoothstep(0.0, 0.22, vUv.y)) * 0.65;',
          '  gl_FragColor = vec4(mix(uColor, vec3(0.95, 0.99, 1.0), clamp(foam, 0.0, 1.0)) + uColor * uGlow, 0.82);',
          '}'
        ].join('\n')
      }));
      wfMesh.position.set(0, curWaterY + wfH * 0.45, -radius * 0.55);
      waterGroup.add(wfMesh);
      impactLocal.set(0, curWaterY, -radius * 0.2);
    }

    const DROP_COUNT = 56;
    const dropPos = new Float32Array(DROP_COUNT * 3);
    const dropVel = Array.from({ length: DROP_COUNT }, () => new THREE.Vector3());
    const dropLife = new Float32Array(DROP_COUNT);
    const dropGeo = new THREE.BufferGeometry();
    dropGeo.setAttribute('position', new THREE.BufferAttribute(dropPos, 3));
    const dropPts = new THREE.Points(dropGeo, new THREE.PointsMaterial({
      color: waterColor.clone().lerp(new THREE.Color(0xffffff), 0.5), size: isCup ? 0.008 : 0.015, transparent: true, opacity: 0.88, depthWrite: false
    }));
    dropPts.frustumCulled = false;
    waterGroup.add(dropPts);
    let nextDrop = 0;

    function emitSplash(lx, ly, lz, strength) {
      const cnt = Math.min(6, Math.ceil(strength * 4));
      for (let i = 0; i < cnt; i++) {
        const idx = nextDrop; nextDrop = (nextDrop + 1) % DROP_COUNT;
        dropPos[idx * 3] = lx + (Math.random() - 0.5) * 0.04;
        dropPos[idx * 3 + 1] = ly + 0.015;
        dropPos[idx * 3 + 2] = lz + (Math.random() - 0.5) * 0.04;
        const spread = isCup ? 0.25 : 0.55;
        dropVel[idx].set((Math.random() - 0.5) * spread, (0.45 + Math.random() * 0.75) * strength, (Math.random() - 0.5) * spread);
        dropLife[idx] = 0.45 + Math.random() * 0.25;
      }
    }

    let hCur = new Float32Array(V_COUNT), hPrev = new Float32Array(V_COUNT), hNext = new Float32Array(V_COUNT);
    function disturbLocal(lx, lz, amp) {
      const dist = Math.hypot(lx, lz);
      if (dist > curR * 0.94) return;
      const rIdx = Math.max(0, Math.min(RINGS - 1, Math.round((dist / Math.max(0.01, curR)) * RINGS)));
      if (rIdx === 0) { hCur[0] += amp; return; }
      let ang = Math.atan2(lz, lx); if (ang < 0) ang += Math.PI * 2;
      const sIdx = Math.round((ang / (Math.PI * 2)) * SEGS) % SEGS;
      hCur[1 + (rIdx - 1) * SEGS + sIdx] += amp;
    }

    const tmpLocal = new THREE.Vector3(), prevPos = waterGroup.position.clone(), prevHands = new Map();
    let dropTick = 0;

    const ent = APP.registerEntity(waterGroup, {
      name: opts.name || waterGroup.name,
      category: 'world_fx',
      grabbable: holdable,
      holdable: holdable,
      physics: { enabled: holdable, gravity: holdable },
      onUpdate: function (dt) {
        const delta = Math.min(0.033, Math.max(0.004, dt || 0.016));
        const now = performance.now() * 0.001;
        waterUniforms.uTime.value = now; streamUniforms.uTime.value = now;

        if (isPouring || isWaterfall || isFountain) {
          if (isPouring && fillLevel < maxFill && !attachInfo) {
            fillLevel = Math.min(maxFill, fillLevel + delta * 0.065);
            curWaterY = THREE.MathUtils.lerp(minY, maxY, fillLevel);
            curR = innerProfileFn(curWaterY);
            waterMesh.position.y = curWaterY;
            waterMesh.scale.set(curR, 1, curR);
          }
          dropTick += delta;
          if (dropTick > 0.055) {
            dropTick = 0;
            disturbLocal(impactLocal.x, impactLocal.z, isFountain ? 0.022 : -0.014);
            emitSplash(impactLocal.x, curWaterY, impactLocal.z, isFountain ? 1.45 : (isCup ? 0.45 : 0.8));
          }
        }

        if (waterGroup.position.distanceTo(prevPos) > 0.002) {
          disturbLocal(-Math.sign(waterGroup.position.x - prevPos.x) * curR * 0.5, -Math.sign(waterGroup.position.z - prevPos.z) * curR * 0.5, -0.008);
          prevPos.copy(waterGroup.position);
        }

        for (const [hk, h] of Object.entries(APP.hands || {})) {
          if (!h || !h.active) continue;
          const hp = h.isHand ? h.fingerTips?.index : h.pos;
          if (!hp) continue;
          tmpLocal.copy(hp); waterGroup.worldToLocal(tmpLocal);
          const last = prevHands.get(hk), spd = last ? tmpLocal.distanceTo(last) / delta : 0;
          prevHands.set(hk, tmpLocal.clone());
          if (Math.hypot(tmpLocal.x, tmpLocal.z) < curR && Math.abs(tmpLocal.y - curWaterY) < 0.14 && spd > 0.15) {
            disturbLocal(tmpLocal.x, tmpLocal.z, -0.016);
            emitSplash(tmpLocal.x, curWaterY, tmpLocal.z, 0.65);
          }
        }

        const ampMax = isBowl ? 0.024 : 0.065;
        let ring1Sum = 0;
        for (let s = 0; s < SEGS; s++) ring1Sum += hCur[1 + s];
        hNext[0] = THREE.MathUtils.clamp(((ring1Sum / SEGS) * 2.0 - hPrev[0]) * 0.982, -ampMax, ampMax);
        for (let r = 1; r < RINGS; r++) {
          for (let s = 0; s < SEGS; s++) {
            const vi = 1 + (r - 1) * SEGS + s, sn = (s + 1) % SEGS, sp = (s - 1 + SEGS) % SEGS;
            const vIn = r === 1 ? 0 : (1 + (r - 2) * SEGS + s), vOut = 1 + r * SEGS + s;
            const avg = (hCur[vIn] + hCur[vOut] + hCur[1 + (r - 1) * SEGS + sp] + hCur[1 + (r - 1) * SEGS + sn]) * 0.5;
            hNext[vi] = THREE.MathUtils.clamp((avg - hPrev[vi]) * 0.982, -ampMax, ampMax);
          }
        }
        const tmp = hPrev; hPrev = hCur; hCur = hNext; hNext = tmp;

        const posA = waterGeo.attributes.position, normA = waterGeo.attributes.normal;
        for (let i = 0; i < V_COUNT; i++) {
          posA.setY(i, hCur[i]);
          wFoam[i] = Math.max(0, Math.abs(hCur[i]) * (isBowl ? 22.0 : 12.0) - 0.12);
        }
        for (let r = 1; r < RINGS; r++) {
          for (let s = 0; s < SEGS; s++) {
            const vi = 1 + (r - 1) * SEGS + s, vIn = r === 1 ? 0 : (1 + (r - 2) * SEGS + s), vOut = 1 + r * SEGS + s;
            const dh = (hCur[vIn] - hCur[vOut]) * 6.0, ang = (s / SEGS) * Math.PI * 2;
            const nx = Math.cos(ang) * dh, nz = Math.sin(ang) * dh, len = Math.hypot(nx, 1.0, nz) || 1;
            normA.setXYZ(vi, nx / len, 1.0 / len, nz / len);
          }
        }
        posA.needsUpdate = true; normA.needsUpdate = true; waterGeo.attributes.aFoam.needsUpdate = true;

        for (let i = 0; i < DROP_COUNT; i++) {
          if (dropLife[i] <= 0) { dropPos[i * 3 + 1] = -99; continue; }
          dropLife[i] -= delta; dropVel[i].y -= 7.8 * delta;
          dropPos[i * 3] += dropVel[i].x * delta; dropPos[i * 3 + 1] += dropVel[i].y * delta; dropPos[i * 3 + 2] += dropVel[i].z * delta;
          if (dropPos[i * 3 + 1] < curWaterY) dropLife[i] = 0;
        }
        dropGeo.attributes.position.needsUpdate = true;
      }
    });

    APP.lastEntityId = ent.id;
    APP.emit?.('entity:created', ent);
    return { ok: true, id: ent.id, name: ent.name, message: `Created ${ent.name}.` };
  }

  // ====================================================================
  // 2. COMPOSABLE GRANULAR ENGINE (Pour from/onto ANY object!)
  // ====================================================================
  function createGranularSimulation(opts = {}) {
    const matType = String(opts.material || 'sand').toLowerCase();
    const defaultCol = matType === 'gold' ? 0xfbbf24 : matType === 'snow' ? 0xe2e8f0 : matType === 'ash' ? 0x475569 : 0xc8a472;
    const baseCol = parseHexColor(opts.color, defaultCol);
    const targetEnt = opts.pour_on_target ? APP.findEntity?.(opts.pour_on_target) : null;
    const fromEnt = opts.pour_from_entity ? APP.findEntity?.(opts.pour_from_entity) : null;
    const fromInfo = getEntityBounds(fromEnt);

    const sandGroup = new THREE.Group();
    sandGroup.name = `Granular_${matType}_Sim`;

    const SIZE = Number(opts.size) || 1.75;
    const GRID = 40;
    const CELL = SIZE / (GRID - 1);
    const MAX_SLOPE = CELL * (matType === 'gravel' ? 0.85 : matType === 'snow' ? 0.92 : 0.68);
    const pileH = new Float32Array(GRID * GRID);

    const pileGeo = new THREE.PlaneGeometry(SIZE, SIZE, GRID - 1, GRID - 1);
    pileGeo.rotateX(-Math.PI / 2);
    const pileMesh = new THREE.Mesh(pileGeo, new THREE.MeshStandardMaterial({
      color: baseCol, roughness: matType === 'gold' ? 0.28 : 0.92, metalness: matType === 'gold' ? 0.88 : 0.04
    }));
    pileMesh.receiveShadow = true; pileMesh.castShadow = true;
    sandGroup.add(pileMesh);

    const GRAIN_COUNT = Math.min(1200, Math.max(400, Number(opts.count) || 950));
    const grainGeo = new THREE.DodecahedronGeometry(matType === 'gravel' ? 0.015 : 0.0085, 0);
    const grainInst = new THREE.InstancedMesh(grainGeo, new THREE.MeshStandardMaterial({
      color: 0xffffff, roughness: matType === 'gold' ? 0.25 : 0.86, metalness: matType === 'gold' ? 0.9 : 0.06
    }), GRAIN_COUNT);
    grainInst.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
    sandGroup.add(grainInst);

    const gPos = new Float32Array(GRAIN_COUNT * 3), gVel = new Float32Array(GRAIN_COUNT * 3), gScale = new Float32Array(GRAIN_COUNT);
    const dummy = new THREE.Object3D();

    const basePos = fromInfo ? new THREE.Vector3(fromInfo.center.x, 0, fromInfo.center.z)
      : targetEnt?.group ? new THREE.Vector3(targetEnt.group.position.x, 0, targetEnt.group.position.z)
      : resolvePos(opts.position, 1.25);
    sandGroup.position.copy(basePos);
    APP.scene.add(sandGroup);

    const emitH = fromInfo ? Math.max(0.45, fromInfo.box.min.y + fromInfo.size.y * 0.25) : 1.25;

    for (let i = 0; i < GRAIN_COUNT; i++) {
      gPos[i * 3] = (Math.random() - 0.5) * 0.06;
      gPos[i * 3 + 1] = 0.08 + Math.random() * emitH;
      gPos[i * 3 + 2] = (Math.random() - 0.5) * 0.06;
      gVel[i * 3 + 1] = -0.3 - Math.random() * 0.4;
      gScale[i] = 0.65 + Math.random() * 0.85;
      grainInst.setColorAt(i, baseCol.clone().offsetHSL(0, 0, (Math.random() - 0.5) * 0.14));
    }
    if (grainInst.instanceColor) grainInst.instanceColor.needsUpdate = true;

    const tmpLocal = new THREE.Vector3(), tmpBox = new THREE.Box3(), tmpCenter = new THREE.Vector3();

    const ent = APP.registerEntity(sandGroup, {
      name: `Granular ${matType}`,
      category: 'world_fx',
      grabbable: false,
      physics: { enabled: false },
      onUpdate: function (dt) {
        const delta = Math.min(0.033, Math.max(0.004, dt || 0.016));
        const now = performance.now() * 0.001;
        const emitX = (targetEnt || fromEnt) ? 0 : Math.sin(now * 0.65) * 0.16;
        const emitZ = (targetEnt || fromEnt) ? 0 : Math.cos(now * 0.55) * 0.16;

        const obsBoxes = [];
        for (const other of APP.entities || []) {
          if (!other || other === ent || other === fromEnt || !other.group || other.category === 'world_fx' || other.category === 'board') continue;
          tmpBox.setFromObject(other.group);
          if (!tmpBox.isEmpty()) {
            tmpBox.min.sub(sandGroup.position); tmpBox.max.sub(sandGroup.position);
            obsBoxes.push(tmpBox.clone());
          }
        }

        for (const h of Object.values(APP.hands || {})) {
          if (!h || !h.active) continue;
          const hp = h.isHand ? h.fingerTips?.index : h.pos;
          if (!hp) continue;
          tmpLocal.copy(hp); sandGroup.worldToLocal(tmpLocal);
          const gx = Math.round((tmpLocal.x / SIZE + 0.5) * (GRID - 1)), gz = Math.round((tmpLocal.z / SIZE + 0.5) * (GRID - 1));
          if (gx >= 1 && gx < GRID - 1 && gz >= 1 && gz < GRID - 1 && tmpLocal.y <= pileH[gz * GRID + gx] + 0.08) {
            for (let dz = -2; dz <= 2; dz++) for (let dx = -2; dx <= 2; dx++) {
              const nx = gx + dx, nz = gz + dz;
              if (nx > 0 && nx < GRID - 1 && nz > 0 && nz < GRID - 1) pileH[nz * GRID + nx] = Math.max(0, pileH[nz * GRID + nx] - 0.014);
            }
          }
        }

        for (let i = 0; i < GRAIN_COUNT; i++) {
          const i3 = i * 3;
          gVel[i3 + 1] -= 7.8 * delta;
          gPos[i3] += gVel[i3] * delta; gPos[i3 + 1] += gVel[i3 + 1] * delta; gPos[i3 + 2] += gVel[i3 + 2] * delta;

          for (let b = 0; b < obsBoxes.length; b++) {
            const ob = obsBoxes[b];
            if (gPos[i3] > ob.min.x && gPos[i3] < ob.max.x && gPos[i3 + 2] > ob.min.z && gPos[i3 + 2] < ob.max.z && gPos[i3 + 1] > ob.min.y && gPos[i3 + 1] < ob.max.y) {
              ob.getCenter(tmpCenter);
              gPos[i3 + 1] = ob.max.y + 0.01;
              gVel[i3] += (gPos[i3] - tmpCenter.x) * 2.8; gVel[i3 + 2] += (gPos[i3 + 2] - tmpCenter.z) * 2.8;
              gVel[i3 + 1] = Math.abs(gVel[i3 + 1]) * 0.25;
            }
          }

          const gx = Math.floor((gPos[i3] / SIZE + 0.5) * (GRID - 1)), gz = Math.floor((gPos[i3 + 2] / SIZE + 0.5) * (GRID - 1));
          const inB = gx >= 1 && gx < GRID - 1 && gz >= 1 && gz < GRID - 1;
          const fY = inB ? pileH[gz * GRID + gx] : 0;
          if (gPos[i3 + 1] <= fY) {
            if (inB && pileH[gz * GRID + gx] < 0.42) pileH[gz * GRID + gx] += 0.0028;
            gPos[i3] = emitX + (Math.random() - 0.5) * 0.06;
            gPos[i3 + 1] = emitH - Math.random() * 0.12;
            gPos[i3 + 2] = emitZ + (Math.random() - 0.5) * 0.06;
            gVel[i3] = (Math.random() - 0.5) * 0.16; gVel[i3 + 1] = -0.25 - Math.random() * 0.3; gVel[i3 + 2] = (Math.random() - 0.5) * 0.16;
          }

          dummy.position.set(gPos[i3], gPos[i3 + 1], gPos[i3 + 2]);
          dummy.rotation.set(i + now * 2, i * 0.7, 0); dummy.scale.setScalar(gScale[i]); dummy.updateMatrix();
          grainInst.setMatrixAt(i, dummy.matrix);
        }
        grainInst.instanceMatrix.needsUpdate = true;

        const dirs = [-1, 1, -GRID, GRID];
        for (let z = 1; z < GRID - 1; z++) for (let x = 1; x < GRID - 1; x++) {
          const idx = z * GRID + x, h = pileH[idx];
          if (h <= 0.001) continue;
          for (let d = 0; d < 4; d++) {
            const ni = idx + dirs[d], diff = h - pileH[ni];
            if (diff > MAX_SLOPE) { const mv = (diff - MAX_SLOPE) * 0.28; pileH[idx] -= mv; pileH[ni] += mv; }
          }
        }
        const posA = pileGeo.attributes.position;
        for (let i = 0; i < pileH.length; i++) posA.setY(i, pileH[i]);
        posA.needsUpdate = true; pileGeo.computeVertexNormals();
      }
    });

    APP.lastEntityId = ent.id;
    APP.emit?.('entity:created', ent);
    return { ok: true, id: ent.id, name: ent.name, message: `Spawned granular ${matType} simulation.` };
  }

  // ====================================================================
  // 3. COMPOSABLE CLOTH & TEARING ENGINE (Banner, Drape, Between Pillars)
  // ====================================================================
  function createClothSimulation(opts = {}) {
    const mode = String(opts.mode || 'banner').toLowerCase();
    const isDrape = mode === 'drape' || mode === 'tablecloth';
    const targetEnt = opts.target ? APP.findEntity?.(opts.target) : null;
    const betEnts = Array.isArray(opts.between_entities) && opts.between_entities.length === 2
      ? [APP.findEntity?.(opts.between_entities[0]), APP.findEntity?.(opts.between_entities[1])]
      : null;

    if (APP._activeClothEnt && APP.entities?.includes(APP._activeClothEnt) && opts.replace !== false) {
      APP.removeEntity?.(APP._activeClothEnt, false);
    }

    const clothGroup = new THREE.Group();
    clothGroup.name = isDrape ? 'Draped_Cloth_Sim' : 'Interactive_Cloth_Sim';

    const COLS = 22, ROWS = 16;
    let W = Number(opts.width) || (isDrape ? 1.15 : 1.08);
    const H = Number(opts.height) || (isDrape ? 1.15 : 0.80);

    const bInfoA = betEnts?.[0] ? getEntityBounds(betEnts[0]) : null;
    const bInfoB = betEnts?.[1] ? getEntityBounds(betEnts[1]) : null;
    if (bInfoA && bInfoB) {
      W = Math.max(0.45, Math.min(2.8, bInfoA.center.distanceTo(bInfoB.center)));
    }

    const DX = W / (COLS - 1), DY = H / (ROWS - 1), DIAG = Math.hypot(DX, DY), N_COUNT = COLS * ROWS;

    if (!isDrape && !(bInfoA && bInfoB)) {
      const rod = new THREE.Mesh(
        new THREE.CylinderGeometry(0.013, 0.013, W + 0.14, 14),
        new THREE.MeshStandardMaterial({ color: 0x64748b, metalness: 0.88, roughness: 0.25 })
      );
      rod.rotation.z = Math.PI / 2; rod.position.set(0, H * 0.5 + 0.015, 0);
      clothGroup.add(rod);
    }

    const baseCol = parseHexColor(opts.color, 0x991b1b);
    const fCan = document.createElement('canvas'); fCan.width = 512; fCan.height = 512;
    const fCtx = fCan.getContext('2d');
    fCtx.fillStyle = '#' + baseCol.getHexString(); fCtx.fillRect(0, 0, 512, 512);
    fCtx.strokeStyle = 'rgba(255,255,255,0.14)'; fCtx.lineWidth = 2;
    for (let i = 0; i < 512; i += 8) {
      fCtx.beginPath(); fCtx.moveTo(i, 0); fCtx.lineTo(i, 512); fCtx.stroke();
      fCtx.beginPath(); fCtx.moveTo(0, i); fCtx.lineTo(512, i); fCtx.stroke();
    }
    fCtx.strokeStyle = '#fbbf24'; fCtx.lineWidth = 12; fCtx.strokeRect(16, 16, 480, 480);
    const fabricTex = new THREE.CanvasTexture(fCan); fabricTex.colorSpace = THREE.SRGBColorSpace;

    const cur = [], prev = [], pinned = [];
    for (let r = 0; r < ROWS; r++) {
      for (let c = 0; c < COLS; c++) {
        const x = (c / (COLS - 1) - 0.5) * W;
        const y = isDrape ? 0.0 : (H * 0.5 - r * DY);
        const z = isDrape ? (r / (ROWS - 1) - 0.5) * H : 0.0;
        cur.push(new THREE.Vector3(x, y, z));
        prev.push(new THREE.Vector3(x, y, z));
        pinned.push(bInfoA && bInfoB ? (r === 0 && (c === 0 || c === COLS - 1)) : (!isDrape && r === 0));
      }
    }

    const springs = [], hGrid = Array.from({ length: ROWS }, () => new Array(COLS - 1)), vGrid = Array.from({ length: ROWS - 1 }, () => new Array(COLS));
    const tearable = opts.tearable !== false, tearMult = tearable ? 2.15 : 99.0;
    function addSpring(a, b, rest, mult) { const s = { a, b, rest, tearDist: rest * mult, broken: false }; springs.push(s); return s; }
    for (let r = 0; r < ROWS; r++) for (let c = 0; c < COLS; c++) {
      const i = r * COLS + c;
      if (c < COLS - 1) hGrid[r][c] = addSpring(i, i + 1, DX, tearMult);
      if (r < ROWS - 1) vGrid[r][c] = addSpring(i, i + COLS, DY, tearMult);
      if (r < ROWS - 1 && c < COLS - 1) { addSpring(i, i + COLS + 1, DIAG, tearMult * 1.15); addSpring(i + 1, i + COLS, DIAG, tearMult * 1.15); }
    }

    const clothGeo = new THREE.BufferGeometry();
    const posArr = new Float32Array(N_COUNT * 3), uvArr = new Float32Array(N_COUNT * 2);
    for (let r = 0; r < ROWS; r++) for (let c = 0; c < COLS; c++) {
      const idx = r * COLS + c; uvArr[idx * 2] = c / (COLS - 1); uvArr[idx * 2 + 1] = 1.0 - r / (ROWS - 1);
    }
    const idxArr = new Uint16Array((ROWS - 1) * (COLS - 1) * 6);
    clothGeo.setAttribute('position', new THREE.BufferAttribute(posArr, 3));
    clothGeo.setAttribute('uv', new THREE.BufferAttribute(uvArr, 2));
    clothGeo.setIndex(new THREE.BufferAttribute(idxArr, 1));

    const clothMesh = new THREE.Mesh(clothGeo, new THREE.MeshStandardMaterial({ map: fabricTex, roughness: 0.78, metalness: 0.04, side: THREE.DoubleSide }));
    clothMesh.castShadow = true; clothMesh.receiveShadow = true;
    clothGroup.add(clothMesh);

    const FIBER_MAX = 80, fibPos = new Float32Array(FIBER_MAX * 3), fibVel = Array.from({ length: FIBER_MAX }, () => new THREE.Vector3()), fibLife = new Float32Array(FIBER_MAX);
    const fibGeo = new THREE.BufferGeometry(); fibGeo.setAttribute('position', new THREE.BufferAttribute(fibPos, 3));
    clothGroup.add(new THREE.Points(fibGeo, new THREE.PointsMaterial({ color: 0xfef08a, size: 0.015, transparent: true, opacity: 0.9 })));
    let nextFib = 0;
    function emitRipFibers(pt) {
      for (let k = 0; k < 3; k++) {
        const fi = nextFib; nextFib = (nextFib + 1) % FIBER_MAX;
        fibPos[fi * 3] = pt.x; fibPos[fi * 3 + 1] = pt.y; fibPos[fi * 3 + 2] = pt.z;
        fibVel[fi].set((Math.random() - 0.5) * 1.4, Math.random() * 0.8, (Math.random() - 0.5) * 1.4); fibLife[fi] = 0.55;
      }
    }

    let indicesDirty = true;
    function rebuildClothIndices() {
      let ptr = 0;
      for (let r = 0; r < ROWS - 1; r++) for (let c = 0; c < COLS - 1; c++) {
        if ((!hGrid[r][c].broken ? 1 : 0) + (!hGrid[r + 1][c].broken ? 1 : 0) + (!vGrid[r][c].broken ? 1 : 0) + (!vGrid[r][c + 1].broken ? 1 : 0) < 3) continue;
        const a = r * COLS + c, b = a + 1, d = a + COLS, e = d + 1;
        idxArr[ptr++] = a; idxArr[ptr++] = d; idxArr[ptr++] = b; idxArr[ptr++] = b; idxArr[ptr++] = d; idxArr[ptr++] = e;
      }
      clothGeo.index.needsUpdate = true; clothGeo.setDrawRange(0, ptr); indicesDirty = false;
    }
    rebuildClothIndices();

    if (bInfoA && bInfoB) {
      const mid = new THREE.Vector3().addVectors(bInfoA.center, bInfoB.center).multiplyScalar(0.5);
      mid.y = Math.max(bInfoA.box.max.y, bInfoB.box.max.y) - H * 0.45;
      clothGroup.position.copy(mid);
      const dir = new THREE.Vector3().subVectors(bInfoB.center, bInfoA.center).setY(0).normalize();
      if (dir.lengthSq() > 0.01) clothGroup.rotation.y = Math.atan2(-dir.z, dir.x);
    } else if (isDrape && targetEnt?.group) {
      const tInfo = getEntityBounds(targetEnt);
      clothGroup.position.set(tInfo.center.x, tInfo.box.max.y + 0.25, tInfo.center.z);
    } else {
      const sp = resolvePos(opts.position, 1.05);
      clothGroup.position.set(sp.x, isDrape ? 1.0 : 1.15, sp.z);
      const hp = APP.headPos || new THREE.Vector3(0, 1.5, 0);
      if (!isDrape) clothGroup.lookAt(hp.x, clothGroup.position.y, hp.z);
    }
    APP.scene.add(clothGroup);

    const tmpDelta = new THREE.Vector3(), tmpLocal = new THREE.Vector3(), tmpSphereC = new THREE.Vector3(), tmpBox = new THREE.Box3(), prevHands = new Map();
    let grabbedIdx = -1, grabbedHand = null;

    function tearAtLocalPoint(pt, rad, imp) {
      let tore = false;
      for (let i = 0; i < springs.length; i++) {
        const s = springs[i]; if (s.broken) continue;
        const mx = (cur[s.a].x + cur[s.b].x) * 0.5, my = (cur[s.a].y + cur[s.b].y) * 0.5, mz = (cur[s.a].z + cur[s.b].z) * 0.5;
        if (Math.hypot(mx - pt.x, my - pt.y, mz - pt.z) < rad) {
          s.broken = true; tore = true; indicesDirty = true; emitRipFibers(cur[s.a]);
          if (imp) { prev[s.a].sub(imp); prev[s.b].add(imp); }
        }
      }
      return tore;
    }

    const clothApi = {
      group: clothGroup,
      tearSlice: (yRatio = 0.45) => {
        const targetRow = Math.max(1, Math.min(ROWS - 2, Math.floor(yRatio * ROWS)));
        for (let c = 0; c < COLS; c++) tearAtLocalPoint(cur[THREE.MathUtils.clamp(targetRow + Math.round(Math.sin(c * 0.9)), 1, ROWS - 2) * COLS + c], DX * 0.95, new THREE.Vector3(0, 0, 0.018));
        rebuildClothIndices();
      },
      shred: () => {
        for (let c = 3; c < COLS - 2; c += 4) for (let r = 1; r < ROWS - 1; r++) tearAtLocalPoint(cur[r * COLS + c], DX * 0.85, new THREE.Vector3(0, 0, 0.02));
        rebuildClothIndices();
      }
    };

    const ent = APP.registerEntity(clothGroup, {
      name: opts.name || (isDrape ? 'Draped Cloth' : 'Interactive Cloth'),
      category: 'cloth',
      grabbable: false,
      physics: { enabled: false },
      onUpdate: function (dt) {
        const safeDt = Math.min(0.025, Math.max(0.004, dt || 0.016)), now = performance.now() * 0.001;
        const gravY = -7.8 * safeDt * safeDt, floorLocalY = -clothGroup.position.y + 0.02;

        const colliders = [];
        for (const other of APP.entities || []) {
          if (!other || other === ent || !other.group || other.category === 'world_fx' || other.category === 'board') continue;
          tmpBox.setFromObject(other.group);
          if (!tmpBox.isEmpty()) {
            const sz = new THREE.Vector3(); tmpBox.getSize(sz); tmpBox.getCenter(tmpSphereC); clothGroup.worldToLocal(tmpSphereC);
            colliders.push({ c: tmpSphereC.clone(), r: Math.max(sz.x, sz.y, sz.z) * 0.52 + 0.015 });
          }
        }

        let activeGrab = false;
        for (const [hk, h] of Object.entries(APP.hands || {})) {
          if (!h || !h.active) continue;
          const tip = h.isHand ? h.fingerTips?.index : h.pos;
          if (!tip) continue;
          tmpLocal.copy(tip); clothGroup.worldToLocal(tmpLocal);
          const last = prevHands.get(hk), speed = last ? tmpLocal.distanceTo(last) / safeDt : 0;
          prevHands.set(hk, tmpLocal.clone());

          if ((h.trigger || h.pinching) && grabbedIdx === -1) {
            let bestI = -1, bestD = 0.16;
            for (let i = (isDrape ? 0 : COLS); i < N_COUNT; i++) { const d = cur[i].distanceTo(tmpLocal); if (d < bestD) { bestD = d; bestI = i; } }
            if (bestI !== -1) { grabbedIdx = bestI; grabbedHand = hk; }
          }
          if (grabbedHand === hk && (h.trigger || h.pinching) && grabbedIdx !== -1) {
            cur[grabbedIdx].lerp(tmpLocal, 0.72); prev[grabbedIdx].copy(cur[grabbedIdx]); activeGrab = true;
          }
          if (tearable && speed > 1.35) tearAtLocalPoint(tmpLocal, 0.09, new THREE.Vector3(0, 0, 0.012));
        }
        if (!activeGrab) { grabbedIdx = -1; grabbedHand = null; }

        const windStrength = Number(opts.wind ?? (isDrape ? 0.35 : 1.0));
        for (let i = 0; i < N_COUNT; i++) {
          if (pinned[i] || i === grabbedIdx) continue;
          const p = cur[i], pr = prev[i];
          const vx = (p.x - pr.x) * 0.982, vy = (p.y - pr.y) * 0.982, vz = (p.z - pr.z) * 0.982;
          pr.copy(p);
          p.x += vx + Math.cos(now * 1.7 + p.y * 2.5) * 0.0004 * windStrength;
          p.y += vy + gravY;
          p.z += vz + (Math.sin(now * 2.4 + p.x * 3.1 + p.y * 2.0) * 0.5 + 0.35) * 0.0013 * windStrength;
          if (p.y < floorLocalY) p.y = floorLocalY;
        }

        for (let iter = 0; iter < 5; iter++) {
          for (let i = 0; i < springs.length; i++) {
            const s = springs[i]; if (s.broken) continue;
            const pa = cur[s.a], pb = cur[s.b]; tmpDelta.subVectors(pb, pa); const d = tmpDelta.length();
            if (d < 1e-6) continue;
            if (tearable && iter === 0 && d > s.tearDist) { s.broken = true; indicesDirty = true; emitRipFibers(pa); continue; }
            const diff = (d - s.rest) / d, pinA = pinned[s.a] || s.a === grabbedIdx, pinB = pinned[s.b] || s.b === grabbedIdx;
            if (pinA && pinB) continue;
            if (pinA) pb.addScaledVector(tmpDelta, -diff);
            else if (pinB) pa.addScaledVector(tmpDelta, diff);
            else { pa.addScaledVector(tmpDelta, diff * 0.5); pb.addScaledVector(tmpDelta, -diff * 0.5); }
          }
          for (let ci = 0; ci < colliders.length; ci++) {
            const col = colliders[ci], rSq = col.r * col.r;
            for (let i = 0; i < N_COUNT; i++) {
              if (pinned[i]) continue;
              tmpDelta.subVectors(cur[i], col.c); const dSq = tmpDelta.lengthSq();
              if (dSq < rSq && dSq > 1e-6) cur[i].copy(col.c).addScaledVector(tmpDelta, col.r / Math.sqrt(dSq));
            }
          }
        }

        if (indicesDirty) rebuildClothIndices();
        for (let i = 0; i < N_COUNT; i++) { posArr[i * 3] = cur[i].x; posArr[i * 3 + 1] = cur[i].y; posArr[i * 3 + 2] = cur[i].z; }
        clothGeo.attributes.position.needsUpdate = true; clothGeo.computeVertexNormals();
        for (let i = 0; i < FIBER_MAX; i++) {
          if (fibLife[i] <= 0) { fibPos[i * 3 + 1] = -99; continue; }
          fibLife[i] -= safeDt; fibVel[i].y -= 3.5 * safeDt;
          fibPos[i * 3] += fibVel[i].x * safeDt; fibPos[i * 3 + 1] += fibVel[i].y * safeDt; fibPos[i * 3 + 2] += fibVel[i].z * safeDt;
        }
        fibGeo.attributes.position.needsUpdate = true;
      }
    });

    clothApi.entity = ent; APP._activeCloth = clothApi; APP._activeClothEnt = ent;
    APP.lastEntityId = ent.id; APP.emit?.('entity:created', ent);
    return { ok: true, id: ent.id, name: ent.name, message: `Spawned ${ent.name}.` };
  }

  // ====================================================================
  // 4. DISINTEGRATION ENGINE
  // ====================================================================
  function createDisintegration(opts = {}) {
    const q = String(opts.target || '').trim();
    let target = q ? APP.findEntity?.(q) : null;
    if (!target || target.category === 'world_fx' || target.category === 'board') {
      const candidates = (APP.entities || []).filter((e) => e && e.group?.parent && e.category !== 'world_fx' && e.category !== 'board' && !e.isPen);
      target = candidates[candidates.length - 1] || null;
    }
    if (!target) {
      const artGroup = new THREE.Group();
      const ped = new THREE.Mesh(new THREE.CylinderGeometry(0.18, 0.22, 0.28, 14), new THREE.MeshStandardMaterial({ color: 0x334155, roughness: 0.55, metalness: 0.45 }));
      ped.position.y = 0.14;
      const core = new THREE.Mesh(new THREE.OctahedronGeometry(0.22, 2), new THREE.MeshStandardMaterial({ color: 0x38bdf8, emissive: 0x0ea5e9, emissiveIntensity: 0.5, roughness: 0.25, metalness: 0.65 }));
      core.position.y = 0.50;
      artGroup.add(ped, core); artGroup.position.copy(resolvePos(null, 1.15));
      target = APP.registerEntity(artGroup, { name: 'Artifact Monolith', category: 'artifact' });
    }

    const edgeCol = parseHexColor(opts.color, 0xff7b14);
    const dur = Math.max(1.2, Math.min(6.0, Number(opts.duration) || 2.6));
    target.group.updateMatrixWorld(true);
    const box = new THREE.Box3().setFromObject(target.group);
    const minY = isFinite(box.min.y) ? box.min.y : 0, maxY = isFinite(box.max.y) ? Math.max(minY + 0.1, box.max.y) : 1.0;

    const samplePos = [], sampleCol = [], vTmp = new THREE.Vector3();
    target.group.traverse((c) => {
      if (!c.isMesh || !c.geometry?.attributes?.position) return;
      const posA = c.geometry.attributes.position, colA = c.geometry.attributes.color;
      const mat = Array.isArray(c.material) ? c.material[0] : c.material;
      const baseCol = mat?.color ? mat.color : new THREE.Color(0x94a3b8);
      const step = Math.max(1, Math.floor(posA.count / 450));
      for (let i = 0; i < posA.count; i += step) {
        vTmp.fromBufferAttribute(posA, i).applyMatrix4(c.matrixWorld);
        samplePos.push(vTmp.x, vTmp.y, vTmp.z);
        if (colA) sampleCol.push(colA.getX(i), colA.getY(i), colA.getZ(i));
        else sampleCol.push(baseCol.r, baseCol.g, baseCol.b);
      }
    });

    const P_COUNT = Math.min(1800, Math.max(600, samplePos.length / 3));
    const pPos = new Float32Array(P_COUNT * 3), pCol = new Float32Array(P_COUNT * 3), pThresh = new Float32Array(P_COUNT);
    const pVel = new Float32Array(P_COUNT * 3), pDetached = new Uint8Array(P_COUNT), pLife = new Float32Array(P_COUNT), pActive = new Float32Array(P_COUNT);
    const srcLen = Math.max(1, samplePos.length / 3);
    for (let i = 0; i < P_COUNT; i++) {
      const si = (i % srcLen) * 3;
      pPos[i * 3] = samplePos[si] + (Math.random() - 0.5) * 0.015;
      pPos[i * 3 + 1] = samplePos[si + 1] + (Math.random() - 0.5) * 0.015;
      pPos[i * 3 + 2] = samplePos[si + 2] + (Math.random() - 0.5) * 0.015;
      pCol[i * 3] = sampleCol[si]; pCol[i * 3 + 1] = sampleCol[si + 1]; pCol[i * 3 + 2] = sampleCol[si + 2];
      const normY = THREE.MathUtils.clamp((pPos[i * 3 + 1] - minY) / (maxY - minY), 0, 1);
      pThresh[i] = THREE.MathUtils.clamp((opts.direction === 'top_down' ? 1 - normY : normY) * 0.72 + Math.random() * 0.28, 0.01, 0.99);
      pLife[i] = 1.0; pVel[i * 3] = (Math.random() - 0.3) * 0.35 + 0.18; pVel[i * 3 + 1] = 0.24 + Math.random() * 0.45; pVel[i * 3 + 2] = (Math.random() - 0.5) * 0.35;
    }

    const uProg = { value: 0.0 }, topDownFlag = { value: opts.direction === 'top_down' ? 1.0 : 0.0 };
    target.group.traverse((c) => {
      if (!c.isMesh || !c.material) return;
      const mats = Array.isArray(c.material) ? c.material : [c.material];
      c.material = mats.map((m) => {
        const cl = m.clone(); cl.side = THREE.DoubleSide; cl.customProgramCacheKey = () => 'nxr_dis_v2';
        cl.onBeforeCompile = function (shader) {
          shader.uniforms.uDisProg = uProg; shader.uniforms.uMinY = { value: minY }; shader.uniforms.uSpanY = { value: maxY - minY };
          shader.uniforms.uTopDown = topDownFlag; shader.uniforms.uEdgeCol = { value: edgeCol };
          shader.vertexShader = shader.vertexShader.replace('#include <common>', '#include <common>\nvarying vec3 vDisWPos;').replace('#include <project_vertex>', '#include <project_vertex>\nvDisWPos = (modelMatrix * vec4(transformed, 1.0)).xyz;');
          shader.fragmentShader = shader.fragmentShader.replace('#include <common>', '#include <common>\nuniform float uDisProg; uniform float uMinY; uniform float uSpanY; uniform float uTopDown; uniform vec3 uEdgeCol;\nvarying vec3 vDisWPos;\nfloat disHash(vec3 p){p=fract(p*0.3183+0.1);p*=17.0;return fract(p.x*p.y*p.z*(p.x+p.y+p.z));}').replace('#include <opaque_fragment>', 'float disN=disHash(floor(vDisWPos*46.0));float disH=clamp((vDisWPos.y-uMinY)/max(0.05,uSpanY),0.0,1.0);if(uTopDown>0.5)disH=1.0-disH;float disVal=disH*0.72+disN*0.28;if(disVal<uDisProg)discard;float rim=1.0-smoothstep(uDisProg,uDisProg+0.08,disVal);outgoingLight=mix(outgoingLight,uEdgeCol*3.4,rim);\n#include <opaque_fragment>');
        };
        cl.needsUpdate = true; return cl;
      });
      if (!Array.isArray(c.material)) c.material = c.material[0];

    pGeo.setAttribute('position', new THREE.BufferAttribute(pPos, 3)); pGeo.setAttribute('aCol', new THREE.BufferAttribute(pCol, 3));
    pGeo.setAttribute('aLife', new THREE.BufferAttribute(pLife, 1)); pGeo.setAttribute('aActive', new THREE.BufferAttribute(pActive, 1));
    const dustPts = new THREE.Points(pGeo, new THREE.ShaderMaterial({
      transparent: true, depthWrite: false, blending: THREE.AdditiveBlending,

      fragmentShader: 'varying vec3 vCol; varying float vAlpha; void main(){ if(vAlpha<0.01)discard; float d=length(gl_PointCoord-0.5)*2.0; if(d>1.0)discard; gl_FragColor=vec4(vCol,pow(1.0-d,1.8)*vAlpha); }'
    }));
    dustPts.frustumCulled = false; dustGroup.add(dustPts); APP.scene.add(dustGroup);

    let meshRemoved = false, elapsed = 0;
    const dustEnt = APP.registerEntity(dustGroup, {
      name: 'Disintegration Dust', category: 'world_fx', grabbable: false, physics: { enabled: false },
      onUpdate: function (dt) {
        const delta = Math.min(0.04, dt || 0.016); elapsed += delta; uProg.value = Math.min(1.08, elapsed / dur);
        let anyAlive = false;
        for (let i = 0; i < P_COUNT; i++) {
          if (!pDetached[i]) { if (uProg.value >= pThresh[i]) { pDetached[i] = 1; pActive[i] = 1.0; } anyAlive = true; }
          else if (pLife[i] > 0) {
            anyAlive = true; const i3 = i * 3;
            pVel[i3] += Math.sin(pPos[i3 + 1] * 5.0 + elapsed * 3.0) * 0.55 * delta; pVel[i3 + 2] += Math.cos(pPos[i3] * 5.0 + elapsed * 2.5) * 0.55 * delta;
            pPos[i3] += pVel[i3] * delta; pPos[i3 + 1] += pVel[i3 + 1] * delta; pPos[i3 + 2] += pVel[i3 + 2] * delta; pLife[i] -= delta * 0.48;
          }
        }
        if (!meshRemoved && uProg.value >= 1.05) { meshRemoved = true; APP.removeEntity?.(target, false); }
        pGeo.attributes.position.needsUpdate = true; pGeo.attributes.aLife.needsUpdate = true; pGeo.attributes.aActive.needsUpdate = true;
        if (meshRemoved && !anyAlive) APP.removeEntity?.(dustEnt, false);
      }
    });
    return { ok: true, message: `Disintegrating ${target.name}.` };
  }
