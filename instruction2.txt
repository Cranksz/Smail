  function parseHexColor(col, fallbackHex) {
    if (typeof col === 'number' && isFinite(col)) return new THREE.Color(col);
    if (typeof col === 'string' && col.trim()) {
      try { return new THREE.Color(col.trim()); } catch (_) {}
    }
    return new THREE.Color(fallbackHex);
  }

  function resolvePos(posOpt, distDefault = 1.15) {
    if (Array.isArray(posOpt) && posOpt.length === 3) {
      return new THREE.Vector3(Number(posOpt[0]) || 0, Number(posOpt[1]) || 0, Number(posOpt[2]) || 0);
    }
    return APP.getSpawnPosition ? APP.getSpawnPosition(distDefault, true) : new THREE.Vector3(0, 0, -distDefault);
  }

  function getEntityBounds(ent) {
    if (!ent || !ent.group) return null;
    ent.group.updateMatrixWorld(true);
    const box = new THREE.Box3().setFromObject(ent.group);
    if (box.isEmpty()) return null;
    const size = new THREE.Vector3(), center = new THREE.Vector3();
    box.getSize(size); box.getCenter(center);
    return { box, size, center };
  }

  async function resolveAssetImageUrl(query) {
    const q = String(query || '').trim();
    if (!q) return null;
    if (q.startsWith('http') || q.startsWith('blob:') || q.startsWith('./') || q.startsWith('data:')) return q;
    const clean = q.replace(/\.[^.]+$/, '').toLowerCase();
    if (APP.importedImages) {
      const exact = APP.importedImages.get(clean);
      if (exact) return exact.url;
      for (const rec of APP.importedImages.values()) {
        if (rec.cleanName.includes(clean)) return rec.url;
      }
    }
    const probes = [
      `./assets/${clean}.png`, `./assets/${clean}.jpg`, `./assets/${clean}.jpeg`, `./assets/${clean}.webp`,
      `./${clean}.png`, `./${clean}.jpg`, `./${clean}.jpeg`
    ];
    for (const p of probes) {
      try { const h = await fetch(p, { method: 'HEAD' }); if (h.ok) return p; } catch (_) {}
    }
    return null;
  }

      }

  function createWaterSimulation(opts = {}) {
    const mode = String(opts.mode || '').toLowerCase();
    const fluidType = String(opts.fluid || opts.type || (mode === 'honey' ? 'honey' : mode === 'mud' ? 'mud' : mode === 'jelly' ? 'jelly' : mode === 'elastic' || mode === 'slime' ? 'elastic' : 'water')).toLowerCase();
    const attachEnt = opts.attach_to ? APP.findEntity?.(opts.attach_to) : null;
    const attachInfo = getEntityBounds(attachEnt);

    let thickness = opts.thickness !== undefined ? THREE.MathUtils.clamp(Number(opts.thickness), 0, 1)
      : (fluidType === 'honey' ? 0.28 : fluidType === 'mud' ? 0.56 : fluidType === 'jelly' ? 0.80 : (fluidType === 'elastic' || fluidType === 'slime') ? 0.96 : 0.02);

    const isSpreading = mode === 'spreading' || mode === 'spill' || mode === 'puddle' || mode === 'flood' || !!opts.spreading;
    const isPouring = !!opts.pouring || mode === 'pouring' || !!opts.pour_from;
    const isWaterfall = !!opts.waterfall || mode === 'waterfall';
    const isFountain = !!opts.fountain || mode === 'fountain' || mode === 'splashing' || !!opts.splashing;

    const container = opts.container
      ? String(opts.container).toLowerCase()
      : (attachInfo || isSpreading ? 'none' : (mode === 'bowl' || mode === 'cup' ? mode : 'basin'));

    const isGlassContainer = container.includes('glass');
    const isCup = container.includes('cup');
    const isBowl = container.includes('bowl') || isCup;
    const customColor = opts.color ? parseHexColor(opts.color, 0x38bdf8) : null;

    const waterGroup = new THREE.Group();
    waterGroup.name = attachEnt ? `Fluid_in_${attachEnt.name}` : isSpreading ? 'Fluid_Spill_Sim' : isPouring ? 'Fluid_Pour_Sim' : 'Fluid_Pool_Sim';

    const radius = Number(opts.radius) || (
      attachInfo ? Math.max(0.12, Math.min(attachInfo.size.x, attachInfo.size.z) * 0.40)
      : isSpreading ? 1.25 : isCup ? 0.12 : isBowl ? 0.30 : 0.96
    );
    const vesselH = Number(opts.height) || (
      attachInfo ? Math.max(0.12, attachInfo.size.y * 0.85)
      : isSpreading ? 0.06 : isCup ? 0.25 : isBowl ? 0.23 : 0.24
    );
    const holdable = opts.holdable !== undefined ? !!opts.holdable : (isBowl && !isSpreading);

    let innerProfileFn = () => radius * 0.96;
    if (isBowl && container !== 'none') {
      const pts = [], steps = 16, wallThick = Math.max(0.012, radius * 0.06), baseR = isCup ? radius * 0.65 : radius * 0.42;
      pts.push(new THREE.Vector2(0.001, 0), new THREE.Vector2(baseR, 0));
      for (let i = 1; i <= steps; i++) {
        const t = i / steps;
        pts.push(new THREE.Vector2(THREE.MathUtils.lerp(baseR, radius, isCup ? Math.pow(t, 0.55) : Math.sqrt(t)), t * vesselH));
      }
      pts.push(new THREE.Vector2(radius - wallThick * 0.45, vesselH + wallThick * 0.35), new THREE.Vector2(radius - wallThick, vesselH));
      for (let i = steps - 1; i >= 1; i--) {
        const t = i / steps;
        pts.push(new THREE.Vector2(Math.max(0.01, THREE.MathUtils.lerp(baseR, radius, isCup ? Math.pow(t, 0.55) : Math.sqrt(t)) - wallThick), Math.max(wallThick, t * vesselH)));
      }
      pts.push(new THREE.Vector2(0.001, wallThick));
      innerProfileFn = (y) => Math.max(0.02, THREE.MathUtils.lerp(baseR, radius, isCup ? Math.pow(THREE.MathUtils.clamp(y / vesselH, 0.05, 0.98), 0.55) : Math.sqrt(THREE.MathUtils.clamp(y / vesselH, 0.05, 0.98))) - wallThick * 1.08);
      const vesselMat = isGlassContainer
        ? new THREE.MeshPhysicalMaterial({ color: 0xeaf6ff, transmission: 0.92, ior: 1.51, thickness: 0.14, roughness: 0.06, clearcoat: 1.0, side: THREE.DoubleSide })
        : new THREE.MeshStandardMaterial({ color: parseHexColor(opts.container_color, 0x2a2f38), roughness: 0.38, metalness: 0.14, side: THREE.DoubleSide });
      const vesselMesh = new THREE.Mesh(new THREE.LatheGeometry(pts, 44), vesselMat);
      vesselMesh.castShadow = true; vesselMesh.receiveShadow = true;
      waterGroup.add(vesselMesh);
    } else if (container === 'basin') {
      const size = radius * 2;
      const slateMat = new THREE.MeshStandardMaterial({ color: 0x262d36, roughness: 0.72, metalness: 0.14 });
      const bed = new THREE.Mesh(new THREE.BoxGeometry(size + 0.14, 0.024, size + 0.14), slateMat);
      bed.position.y = 0.012; bed.receiveShadow = true; waterGroup.add(bed);
      const wt = 0.065, half = radius + wt * 0.5;
      const wN = new THREE.Mesh(new THREE.BoxGeometry(size + wt * 2, vesselH, wt), slateMat); wN.position.set(0, vesselH * 0.5, -half);
      const wS = wN.clone(); wS.position.z = half;
      const wE = new THREE.Mesh(new THREE.BoxGeometry(wt, vesselH, size), slateMat); wE.position.set(half, vesselH * 0.5, 0);
      const wW = wE.clone(); wW.position.x = -half;
      waterGroup.add(wN, wS, wE, wW);
    }

    if (attachInfo) waterGroup.position.set(attachInfo.center.x, attachInfo.box.min.y, attachInfo.center.z);
    else waterGroup.position.copy(resolvePos(opts.position, isBowl ? 0.95 : 1.35));
    APP.scene.add(waterGroup);

    // 84x84 High-Res Shallow-Water / Viscoelastic Solver
    const GRID = isBowl || isCup ? 52 : 84;
    const SIZE = radius * 2.0, N = GRID * GRID;
    const sH = new Float32Array(N), sVel = new Float32Array(N), sNext = new Float32Array(N);
    const sFoam = new Float32Array(N), sDepth = new Float32Array(N), sObs = new Uint8Array(N);

    let fillLevel = isPouring && !attachInfo ? 0.42 : (Number(opts.fill) || 0.72);
    const maxFill = 0.88;
    const minY = isBowl ? vesselH * 0.16 : (attachInfo ? attachInfo.size.y * 0.22 : 0.05);
    const maxY = isBowl ? vesselH * 0.88 : (attachInfo ? attachInfo.size.y * 0.74 : vesselH * 0.80);
    let curWaterY = isSpreading ? 0.008 : THREE.MathUtils.lerp(minY, maxY, fillLevel);
    let curR = innerProfileFn(curWaterY);
    const isCircular = isBowl || !!attachInfo;

    const waterGeo = new THREE.PlaneGeometry(SIZE, SIZE, GRID - 1, GRID - 1);
    waterGeo.rotateX(-Math.PI / 2);
    waterGeo.setAttribute('aFoam', new THREE.BufferAttribute(sFoam, 1));
    waterGeo.setAttribute('aDepth', new THREE.BufferAttribute(sDepth, 1));

    if (isSpreading && !isPouring) {
      const cx = Math.floor(GRID / 2), cz = Math.floor(GRID / 2);
      for (let z = -6; z <= 6; z++) for (let x = -6; x <= 6; x++) {
        const d = Math.hypot(x, z);
        if (d <= 6.2) sH[(cz + z) * GRID + (cx + x)] = (1.0 - d / 6.5) * 0.04;
      }
    }

    // Dynamic Rheology Shader Parameters (Morphed continuously by `thickness`)
    const fluidUniforms = {
      uTime: { value: 0 },
      uThick: { value: thickness },
      uTint: { value: customColor || new THREE.Color(0x38bdf8) },
      uHasCustomCol: { value: customColor ? 1.0 : 0.0 },
      uIsSpreading: { value: isSpreading ? 1.0 : 0.0 },
      uIsCircular: { value: isCircular ? 1.0 : 0.0 },
      uClipR: { value: curR * 0.99 }
    };

    const fluidMat = new THREE.ShaderMaterial({
      uniforms: fluidUniforms,
      transparent: true,
      side: THREE.DoubleSide,
      vertexShader: [
        'attribute float aFoam; attribute float aDepth;',
        'varying vec3 vWPos; varying vec3 vLocalPos; varying vec3 vWNorm; varying float vFoam; varying float vDepth;',
        'void main() {',
        '  vFoam = aFoam; vDepth = aDepth; vLocalPos = position;',
        '  vec4 wp = modelMatrix * vec4(position, 1.0); vWPos = wp.xyz;',
        '  vWNorm = normalize(mat3(modelMatrix) * normal);',
        '  gl_Position = projectionMatrix * viewMatrix * wp;',
        '}'
      ].join('\n'),
      fragmentShader: [
        'uniform float uTime; uniform float uThick; uniform vec3 uTint; uniform float uHasCustomCol;',
        'uniform float uIsSpreading; uniform float uIsCircular; uniform float uClipR;',
        'varying vec3 vWPos; varying vec3 vLocalPos; varying vec3 vWNorm; varying float vFoam; varying float vDepth;',
        'float h21(vec2 p){ p=fract(p*vec2(127.1,311.7)); p+=dot(p,p+34.5); return fract(p.x*p.y); }',
        'float n2(vec2 p){ vec2 i=floor(p); vec2 f=fract(p); f=f*f*(3.0-2.0*f); return mix(mix(h21(i),h21(i+vec2(1,0)),f.x),mix(h21(i+vec2(0,1)),h21(i+vec2(1,1)),f.x),f.y); }',
        'float fbm(vec2 p){ return 0.55*n2(p)+0.30*n2(p*2.1)+0.15*n2(p*4.3); }',
        'float caust(vec2 uv, float t){ vec2 p=uv*5.5; float a=sin(p.x*1.7+t*1.5+cos(p.y*2.1-t*1.1)); float b=cos(p.y*2.3-t*1.3+sin(p.x*1.9+t*0.9)); return pow(clamp(1.0-abs(a+b)*0.5,0.0,1.0),6.0); }',
        'vec3 envSample(vec3 R){ float u=clamp(R.y,-1.0,1.0); vec3 sky=mix(vec3(0.42,0.50,0.60),vec3(0.14,0.28,0.50),smoothstep(0.0,0.85,max(0.0,u))); vec3 gnd=vec3(0.08,0.09,0.11); vec3 env=u>=0.0?sky:mix(vec3(0.25,0.29,0.34),gnd,smoothstep(0.0,-0.5,u)); float sun=smoothstep(0.96,0.996,dot(R,normalize(vec3(0.55,0.82,0.38)))); return env+vec3(1.0,0.96,0.88)*(sun*2.2); }',
        'void main() {',
        '  if (uIsCircular > 0.5 && length(vLocalPos.xz) > uClipR) discard;',
        '  if (uIsSpreading > 0.5 && vDepth < 0.0008) discard;',
        '  vec3 N = normalize(vWNorm);',
        '  float flowT = uTime * mix(1.4, 0.25, uThick);',
        '  float nx = (fbm(vWPos.xz * 10.0 + vec2(flowT*0.4,-flowT*0.3)) - 0.5);',
        '  float nz = (fbm(vWPos.xz * 10.0 + vec2(-flowT*0.35,flowT*0.45)+7.3) - 0.5);',
        '  N = normalize(N + vec3(nx, 0.0, nz) * mix(0.07, 0.012, uThick) * (0.35 + clamp(vFoam*1.5,0.0,1.0)));',
        '  vec3 I = normalize(vWPos - cameraPosition); vec3 V = -I;',
        '  vec3 R_refl = reflect(I, N); vec3 R_refr = refract(I, N, 1.0 / mix(1.333, 1.52, uThick));',
        '  float depth = max(0.008, vDepth);',
        '  vec2 floorXZ = vWPos.xz + R_refr.xz * clamp(depth / max(0.15, -R_refr.y), 0.01, 0.65);',
        '  vec3 bedCol = vec3(0.22, 0.25, 0.28) * (0.7 + 0.35 * fbm(floorXZ * 16.0));',
        '  bedCol += vec3(0.9, 0.97, 1.0) * caust(floorXZ, uTime * 1.2) * (1.0 - smoothstep(0.05, 0.40, uThick));',
        '  // Continuous Rheology Color Palette: Water -> Honey -> Mud -> Jelly -> Elastic',
        '  vec3 cWater = vec3(0.18, 0.72, 0.92);',
        '  vec3 cHoney = vec3(0.88, 0.52, 0.04);',
        '  vec3 cMud   = vec3(0.28, 0.19, 0.12);',
        '  vec3 cJelly = vec3(0.82, 0.12, 0.28);',
        '  vec3 cElast = vec3(0.14, 0.85, 0.52);',
        '  vec3 matCol = cWater;',
        '  matCol = mix(matCol, cHoney, smoothstep(0.05, 0.28, uThick));',
        '  matCol = mix(matCol, cMud,   smoothstep(0.32, 0.55, uThick));',
        '  matCol = mix(matCol, cJelly, smoothstep(0.60, 0.80, uThick));',
        '  matCol = mix(matCol, cElast, smoothstep(0.84, 0.98, uThick));',
        '  if (uHasCustomCol > 0.5) matCol = uTint;',
        '  float isOpaqueMud = smoothstep(0.38, 0.55, uThick) * (1.0 - smoothstep(0.62, 0.76, uThick));',
        '  vec3 trans = exp(-mix(vec3(2.8,0.72,0.35), vec3(0.4,2.6,7.0), smoothstep(0.0,0.3,uThick)) * depth * 1.6);',
        '  vec3 refrCol = mix(matCol * 0.45, bedCol * trans, 0.85 * (1.0 - smoothstep(0.15, 0.45, uThick)));',
        '  refrCol = mix(refrCol, matCol * (0.75 + 0.35 * fbm(vWPos.xz * 22.0)), isOpaqueMud);',
        '  float fres = 0.025 + 0.975 * pow(1.0 - max(0.0, dot(N, V)), 5.0);',
        '  vec3 L = normalize(vec3(0.55, 0.82, 0.38)); vec3 H = normalize(L + V);',
        '  float spec = pow(max(0.0, dot(N, H)), mix(340.0, 48.0, isOpaqueMud));',
        '  float sss = pow(clamp(dot(V, -L + N * 0.55), 0.0, 1.0), 3.0) * (1.0 - isOpaqueMud) * 0.55;',
        '  vec3 col = mix(refrCol, envSample(R_refl), clamp(fres * mix(1.0, 0.35, isOpaqueMud), 0.0, 0.96)) + matCol * sss;',
        '  float foamMask = clamp((vFoam - 0.10) * (0.6 + fbm(vWPos.xz * 28.0 + uTime * 0.6) * 0.8), 0.0, 0.95) * (1.0 - smoothstep(0.05, 0.35, uThick));',
        '  col = mix(col, vec3(0.92, 0.97, 1.0), foamMask) + vec3(1.0, 0.97, 0.92) * spec * mix(2.2, 0.85, isOpaqueMud);',
        '  float alpha = uIsSpreading > 0.5 ? smoothstep(0.0008, 0.006, vDepth) * mix(0.75, 0.99, fres + isOpaqueMud) : mix(0.84, 0.99, fres + isOpaqueMud + foamMask);',
        '  gl_FragColor = vec4(col, clamp(alpha, 0.0, 0.99));',
        '}'
      ].join('\n')
    });

    const waterMesh = new THREE.Mesh(waterGeo, fluidMat);
    waterMesh.position.y = isSpreading ? 0.002 : curWaterY;
    waterGroup.add(waterMesh);

    // 1B. TRUE VOLUMETRIC FLUID PARTICLE STREAM (1,100 Shaded Sphere-Impostors, NO FAKE LINES!)
    const P_COUNT = 1100;
    const pPos = new Float32Array(P_COUNT * 3);
    const pVel = new Float32Array(P_COUNT * 3);
    const pSize = new Float32Array(P_COUNT);
    const pActive = new Float32Array(P_COUNT); // 0=pool-splash/inactive, 1=active volumetric jet/splash

    for (let i = 0; i < P_COUNT; i++) {
      pPos[i * 3 + 1] = -99;
      pSize[i] = 16.0 + Math.random() * 14.0;
    }

    const pGeo = new THREE.BufferGeometry();
    pGeo.setAttribute('position', new THREE.BufferAttribute(pPos, 3));
    pGeo.setAttribute('aSize', new THREE.BufferAttribute(pSize, 1));
    pGeo.setAttribute('aActive', new THREE.BufferAttribute(pActive, 1));

    const pUniforms = {
      uPx: { value: window.devicePixelRatio || 1 },
      uThick: { value: thickness },
      uTint: { value: customColor || new THREE.Color(0x38bdf8) },
      uHasCustomCol: { value: customColor ? 1.0 : 0.0 }
    };

    const fluidPts = new THREE.Points(pGeo, new THREE.ShaderMaterial({
      uniforms: pUniforms,
      transparent: true,
      depthWrite: true,
      vertexShader: [
        'attribute float aSize; attribute float aActive; uniform float uPx; uniform float uThick; varying float vActive; varying vec3 vWPos;',
        'void main(){',
        '  vActive = aActive;',
        '  vec4 wp = modelMatrix * vec4(position, 1.0); vWPos = wp.xyz;',
        '  vec4 mv = viewMatrix * wp;',
        '  float sz = aSize * mix(0.85, 1.45, uThick);',
        '  gl_PointSize = aActive > 0.5 ? clamp((sz * uPx) / max(0.25, -mv.z), 2.0, 26.0) : 0.0;',
        '  gl_Position = projectionMatrix * mv;',
        '}'
      ].join('\n'),
      fragmentShader: [
        'uniform float uThick; uniform vec3 uTint; uniform float uHasCustomCol; varying float vActive; varying vec3 vWPos;',
        'void main(){',
        '  if (vActive < 0.5) discard;',
        '  vec2 c = (gl_PointCoord - 0.5) * 2.0;',
        '  float r2 = dot(c, c); if (r2 > 1.0) discard;',
        '  // Real 3D Sphere Impostor Normal!',
        '  vec3 N = normalize(vec3(c.x, -c.y, sqrt(max(0.0, 1.0 - r2))));',
        '  vec3 V = vec3(0.0, 0.0, 1.0);',
        '  vec3 L = normalize(vec3(0.5, 0.8, 0.5));',
        '  float ndl = max(0.0, dot(N, L));',
        '  float fres = 0.04 + 0.96 * pow(1.0 - max(0.0, dot(N, V)), 3.2);',
        '  float spec = pow(max(0.0, dot(N, normalize(L + V))), mix(96.0, 24.0, uThick));',
        '  vec3 cWater = vec3(0.45, 0.84, 0.98); vec3 cHoney = vec3(0.92, 0.56, 0.06);',
        '  vec3 cMud = vec3(0.32, 0.22, 0.14); vec3 cJelly = vec3(0.88, 0.16, 0.34); vec3 cElast = vec3(0.18, 0.90, 0.55);',
        '  vec3 base = mix(cWater, cHoney, smoothstep(0.05, 0.28, uThick));',
        '  base = mix(base, cMud, smoothstep(0.32, 0.55, uThick));',
        '  base = mix(base, cJelly, smoothstep(0.60, 0.80, uThick));',
        '  base = mix(base, cElast, smoothstep(0.84, 0.98, uThick));',
        '  if (uHasCustomCol > 0.5) base = uTint;',
        '  float isMud = smoothstep(0.38, 0.55, uThick) * (1.0 - smoothstep(0.62, 0.76, uThick));',
        '  vec3 col = base * (0.45 + 0.65 * ndl) + vec3(0.85, 0.96, 1.0) * fres * (1.0 - isMud * 0.7) * 0.65 + vec3(1.0) * spec * 1.4;',
        '  float alpha = (1.0 - r2 * 0.4) * mix(0.62 + fres * 0.35, 0.96, smoothstep(0.18, 0.6, uThick));',
        '  gl_FragColor = vec4(col, clamp(alpha, 0.0, 0.98));',
        '}'
      ].join('\n')
    }));
    fluidPts.frustumCulled = false;
    waterGroup.add(fluidPts);

    let pourSourceEnt = null, pourCustomEnt = null;
    if (isPouring) {
      let initSpoutWorld = waterGroup.position.clone().add(new THREE.Vector3(isSpreading ? 0.12 : radius * 0.58, isSpreading ? 0.68 : vesselH + 0.54, 0));
      if (typeof opts.pour_from === 'string') {
        pourCustomEnt = APP.findEntity?.(opts.pour_from);
        const info = getEntityBounds(pourCustomEnt);
        if (info) { initSpoutWorld.copy(info.center); initSpoutWorld.y = info.box.min.y + info.size.y * 0.35; }
      } else if (Array.isArray(opts.pour_from) && opts.pour_from.length === 3) {
        initSpoutWorld.set(opts.pour_from[0], opts.pour_from[1], opts.pour_from[2]);
      } else if (!attachInfo && opts.movable_source !== false) {
        const pitcherG = new THREE.Group();
        pitcherG.name = 'Fluid_Pitcher';
        const pMat = new THREE.MeshPhysicalMaterial({ color: 0xe6f4ff, transmission: 0.82, roughness: 0.08, ior: 1.5 });
        const body = new THREE.Mesh(new THREE.CylinderGeometry(0.052, 0.072, 0.17, 20), pMat);
        const spout = new THREE.Mesh(new THREE.ConeGeometry(0.022, 0.085, 16), pMat);
        spout.rotation.z = Math.PI * 0.38; spout.position.set(-0.074, 0.05, 0);
        const handle = new THREE.Mesh(new THREE.TorusGeometry(0.048, 0.008, 10, 20, Math.PI), pMat);
        handle.rotation.z = -Math.PI / 2; handle.position.set(0.068, 0.0, 0);
        pitcherG.add(body, spout, handle);
        pitcherG.position.copy(initSpoutWorld).add(new THREE.Vector3(0.09, 0.03, 0));
        pitcherG.rotation.z = 0.52;
        APP.scene.add(pitcherG);
        pourSourceEnt = APP.registerEntity(pitcherG, {
          name: 'Fluid Pitcher', category: 'tool', grabbable: true, holdable: true, physics: { enabled: false },
          onUpdate: () => { if (!waterGroup.parent) APP.removeEntity?.(pourSourceEnt, false); }
        });
      }
    }

    function disturbGridDirectional(lx, lz, amp, vx, vz, radCells = 2) {
      const gx = Math.round((lx / SIZE + 0.5) * (GRID - 1));
      const gz = Math.round((lz / SIZE + 0.5) * (GRID - 1));
      if (gx < 2 || gx >= GRID - 2 || gz < 2 || gz >= GRID - 2) return;
      if (isCircular && Math.hypot(lx, lz) > curR * 0.96) return;
      for (let dz = -radCells; dz <= radCells; dz++) {
        for (let dx = -radCells; dx <= radCells; dx++) {
          const nx = gx + dx, nz = gz + dz, d = Math.hypot(dx, dz);
          if (d <= radCells + 0.2) {
            const w = Math.exp(-(d * d) / (radCells * 0.8 + 0.3));
            const bow = (dx * vx + dz * vz) * 0.004;
            const idx = nz * GRID + nx;
            sVel[idx] += (amp + bow) * w;
            sFoam[idx] = Math.min(1.0, sFoam[idx] + (Math.abs(amp) + Math.abs(bow)) * 28.0 * w);
          }
        }
      }
    }

    let nextP = 0;
    function emitFluidParticle(lx, ly, lz, vx, vy, vz, sz) {
      const idx = nextP; nextP = (nextP + 1) % P_COUNT;
      pActive[idx] = 1.0;
      pPos[idx * 3] = lx; pPos[idx * 3 + 1] = ly; pPos[idx * 3 + 2] = lz;
      pVel[idx * 3] = vx; pVel[idx * 3 + 1] = vy; pVel[idx * 3 + 2] = vz;
      pSize[idx] = sz || (14.0 + Math.random() * 12.0);
    }

    const tmpLocal = new THREE.Vector3(), tmpStart = new THREE.Vector3(), tmpDir = new THREE.Vector3();
    const prevPos = waterGroup.position.clone(), prevHands = new Map(), prevEntPos = new Map();
    const tmpBox = new THREE.Box3(), tmpC = new THREE.Vector3(), tmpSz = new THREE.Vector3();
    const handColliders = [];

    interactives.push({
      type: 'water', group: waterGroup, mesh: waterMesh,
      onPointerHit: (worldPt, worldVel, isClick) => {
        tmpLocal.copy(worldPt); waterGroup.worldToLocal(tmpLocal);
        const spd = worldVel.length();
        if (isClick || spd > 0.002) {
          const amp = isClick ? -0.018 : -Math.min(0.015, spd * 0.38);
          disturbGridDirectional(tmpLocal.x, tmpLocal.z, amp, worldVel.x * 20, worldVel.z * 20, isClick ? 3 : 2);
          if ((isClick || spd > 0.02) && thickness < 0.7) {
            const cnt = isClick ? 12 : 4;
            for (let k = 0; k < cnt; k++) emitFluidParticle(tmpLocal.x, curWaterY + 0.015, tmpLocal.z, (Math.random() - 0.5) * 0.7, 0.55 + Math.random() * 0.85, (Math.random() - 0.5) * 0.7, 13.0);
          }
        }
      }
    });

    const ent = APP.registerEntity(waterGroup, {
      name: opts.name || waterGroup.name,
      category: 'world_fx',
      grabbable: holdable,
      holdable: holdable,
      physics: { enabled: holdable, gravity: holdable },
      onUpdate: function (dt) {
        const delta = Math.min(0.033, Math.max(0.004, dt || 0.016));
        const now = performance.now() * 0.001;
        fluidUniforms.uTime.value = now;
        fluidUniforms.uThick.value = thickness;
        pUniforms.uThick.value = thickness;

        // Collect VR Hand Colliders (with velocity!)
        handColliders.length = 0;
        for (const [hk, h] of Object.entries(APP.hands || {})) {
          if (!h || !h.active) continue;
          const pts = [];
          if (h.isHand && h.fingerTips?.index) pts.push([hk + '_idx', h.fingerTips.index, 0.045]);
          if (h.pos) pts.push([hk + '_palm', h.pos, 0.075]);
          for (const [key, hp, radH] of pts) {
            tmpLocal.copy(hp); waterGroup.worldToLocal(tmpLocal);
            const last = prevHands.get(key);
            const vx = last ? (tmpLocal.x - last.x) / delta : 0;
            const vy = last ? (tmpLocal.y - last.y) / delta : 0;
            const vz = last ? (tmpLocal.z - last.z) / delta : 0;
            prevHands.set(key, tmpLocal.clone());
            handColliders.push({ pos: tmpLocal.clone(), vel: new THREE.Vector3(vx, vy, vz), r: radH });

            const spd = Math.hypot(vx, vy, vz);
            if (tmpLocal.y < curWaterY + 0.12 && tmpLocal.y > -0.10 && spd > 0.05) {
              disturbGridDirectional(tmpLocal.x, tmpLocal.z, -Math.min(0.020, spd * 0.012), vx, vz, 3);
              if (spd > 0.35 && thickness < 0.72) {
                for (let k = 0; k < 4; k++) emitFluidParticle(tmpLocal.x, curWaterY + 0.015, tmpLocal.z, vx * 0.4 + (Math.random() - 0.5) * 0.5, 0.45 + Math.random() * 0.6, vz * 0.4 + (Math.random() - 0.5) * 0.5, 13.0);
              }
            }
          }
        }

        // Emit Volumetric Fluid Particles from Pouring Spout / Waterfall / Fountain
        if (isPouring) {
          let spoutVelX = 0, spoutVelZ = 0;
          if (pourSourceEnt?.group?.parent) {
            tmpStart.set(-0.088, 0.046, 0);
            pourSourceEnt.group.localToWorld(tmpStart);
            tmpDir.set(-1.0, -0.35, 0).applyQuaternion(pourSourceEnt.group.quaternion).normalize();
            waterGroup.worldToLocal(tmpStart);
            spoutVelX = tmpDir.x * 0.46; spoutVelZ = tmpDir.z * 0.46;
          } else if (pourCustomEnt?.group?.parent) {
            const info = getEntityBounds(pourCustomEnt);
            if (info) { tmpStart.copy(info.center); tmpStart.y = info.box.min.y + info.size.y * 0.35; waterGroup.worldToLocal(tmpStart); spoutVelX = -tmpStart.x * 0.4; spoutVelZ = -tmpStart.z * 0.4; }
          } else {
            tmpStart.set(isSpreading ? 0.12 : radius * 0.58, isSpreading ? 0.68 : vesselH + 0.50, 0);
            spoutVelX = -tmpStart.x * 0.4; spoutVelZ = -tmpStart.z * 0.4;
          }

          const emitCount = thickness > 0.4 ? 10 : 14;
          const jitter = THREE.MathUtils.lerp(0.018, 0.008, thickness);
          const coil = thickness > 0.18 && thickness < 0.75 ? 0.04 : 0.0;
          for (let k = 0; k < emitCount; k++) {
            emitFluidParticle(
              tmpStart.x + (Math.random() - 0.5) * jitter,
              tmpStart.y + (Math.random() - 0.5) * jitter,
              tmpStart.z + (Math.random() - 0.5) * jitter,
              spoutVelX + Math.sin(now * 8.0) * coil + (Math.random() - 0.5) * 0.06,
              -0.35 - Math.random() * 0.25,
              spoutVelZ + Math.cos(now * 8.0) * coil + (Math.random() - 0.5) * 0.06,
              THREE.MathUtils.lerp(13.0, 22.0, thickness)
            );
          }
        } else if (isFountain) {
          for (let k = 0; k < 12; k++) {
            emitFluidParticle((Math.random() - 0.5) * 0.05, curWaterY + 0.02, (Math.random() - 0.5) * 0.05, (Math.random() - 0.5) * 0.7, 1.55 + Math.random() * 0.85, (Math.random() - 0.5) * 0.7, 14.0);
          }
        } else if (isWaterfall) {
          for (let k = 0; k < 14; k++) {
            emitFluidParticle((Math.random() - 0.5) * radius * 0.85, curWaterY + 1.05, -radius * 0.55, (Math.random() - 0.5) * 0.08, -0.4, 0.48 + Math.random() * 0.15, 18.0);
          }
        }

        // Collect Scene Entity Obstacles
        sObs.fill(0);
        const obsBoxes = [], waterTopWorldY = waterGroup.position.y + curWaterY;
        for (const other of APP.entities || []) {
          if (!other || other === ent || other === pourSourceEnt || other === pourCustomEnt || !other.group || other.category === 'world_fx' || other.category === 'board') continue;
          tmpBox.setFromObject(other.group);
          if (tmpBox.isEmpty()) continue;
          tmpBox.getCenter(tmpC); tmpBox.getSize(tmpSz);
          const lBox = tmpBox.clone(); lBox.min.sub(waterGroup.position); lBox.max.sub(waterGroup.position);
          obsBoxes.push(lBox);

          if (tmpBox.min.y < waterTopWorldY + 0.04 && tmpBox.max.y > waterGroup.position.y - 0.05) {
            if (isSpreading) {
              const x0 = Math.max(1, Math.floor((lBox.min.x / SIZE + 0.5) * (GRID - 1))), x1 = Math.min(GRID - 2, Math.ceil((lBox.max.x / SIZE + 0.5) * (GRID - 1)));
              const z0 = Math.max(1, Math.floor((lBox.min.z / SIZE + 0.5) * (GRID - 1))), z1 = Math.min(GRID - 2, Math.ceil((lBox.max.z / SIZE + 0.5) * (GRID - 1)));
              for (let gz = z0; gz <= z1; gz++) for (let gx = x0; gx <= x1; gx++) sObs[gz * GRID + gx] = 1;
            }
            tmpLocal.copy(tmpC); waterGroup.worldToLocal(tmpLocal);
            const prevE = prevEntPos.get(other.id), eSpd = prevE ? tmpLocal.distanceTo(prevE) / delta : 0;
            prevEntPos.set(other.id, tmpLocal.clone());
            if (eSpd > 0.06 && Math.hypot(tmpLocal.x, tmpLocal.z) < radius * 1.05) {
              disturbGridDirectional(tmpLocal.x, tmpLocal.z, -Math.min(0.018, eSpd * 0.015), 0, 0, 3);
            }
          }
        }

        // Step Volumetric Fluid Particles (Collision with Hands, Scene Objects & Pool!)
        const grav = 9.2;
        const bounceFactor = thickness > 0.68 ? THREE.MathUtils.lerp(0.28, 0.65, (thickness - 0.68) / 0.32) : 0.18;
        for (let i = 0; i < P_COUNT; i++) {
          if (pActive[i] < 0.5) { pPos[i * 3 + 1] = -99; continue; }
          const i3 = i * 3;
          pVel[i3 + 1] -= grav * delta;
          pPos[i3] += pVel[i3] * delta; pPos[i3 + 1] += pVel[i3 + 1] * delta; pPos[i3 + 2] += pVel[i3 + 2] * delta;

          // Collide with VR Hands/Controllers
          for (let hc = 0; hc < handColliders.length; hc++) {
            const h = handColliders[hc];
            const dx = pPos[i3] - h.pos.x, dy = pPos[i3 + 1] - h.pos.y, dz = pPos[i3 + 2] - h.pos.z;
            const dSq = dx * dx + dy * dy + dz * dz;
            if (dSq < h.r * h.r && dSq > 1e-6) {
              const d = Math.sqrt(dSq), nx = dx / d, ny = dy / d, nz = dz / d;
              pPos[i3] = h.pos.x + nx * h.r; pPos[i3 + 1] = h.pos.y + ny * h.r; pPos[i3 + 2] = h.pos.z + nz * h.r;
              pVel[i3] = h.vel.x * 0.55 + nx * 0.85; pVel[i3 + 1] = Math.max(0.2, h.vel.y * 0.55 + ny * 0.85); pVel[i3 + 2] = h.vel.z * 0.55 + nz * 0.85;
            }
          }

          // Collide with Scene Objects
          for (let b = 0; b < obsBoxes.length; b++) {
            const ob = obsBoxes[b];
            if (pPos[i3] > ob.min.x && pPos[i3] < ob.max.x && pPos[i3 + 2] > ob.min.z && pPos[i3 + 2] < ob.max.z && pPos[i3 + 1] > ob.min.y && pPos[i3 + 1] < ob.max.y) {
              pPos[i3 + 1] = ob.max.y + 0.008;
              pVel[i3 + 1] = Math.abs(pVel[i3 + 1]) * bounceFactor;
              pVel[i3] += (Math.random() - 0.5) * 0.35; pVel[i3 + 2] += (Math.random() - 0.5) * 0.35;
            }
          }

          const floorCheckY = isSpreading ? 0.012 : curWaterY;
          if (pPos[i3 + 1] <= floorCheckY) {
            if (thickness > 0.72 && Math.abs(pVel[i3 + 1]) > 0.6) {
              // Elastic jelly/slime bounce!
              pPos[i3 + 1] = floorCheckY + 0.01;
              pVel[i3 + 1] = Math.abs(pVel[i3 + 1]) * bounceFactor;
              disturbGridDirectional(pPos[i3], pPos[i3 + 2], -0.008, 0, 0, 2);
            } else {
              if (isSpreading) {
                const gx = Math.round((pPos[i3] / SIZE + 0.5) * (GRID - 1)), gz = Math.round((pPos[i3 + 2] / SIZE + 0.5) * (GRID - 1));
                if (gx >= 2 && gx < GRID - 2 && gz >= 2 && gz < GRID - 2) {
                  sH[gz * GRID + gx] = Math.min(0.14, sH[gz * GRID + gx] + 0.0022);
                  curWaterY = sH[gz * GRID + gx];
                }
              } else if (fillLevel < maxFill && !attachInfo && (!isCircular || Math.hypot(pPos[i3], pPos[i3 + 2]) <= curR)) {
                fillLevel = Math.min(maxFill, fillLevel + 0.00012);
                curWaterY = THREE.MathUtils.lerp(minY, maxY, fillLevel);
                curR = innerProfileFn(curWaterY);
                waterMesh.position.y = curWaterY;
                fluidUniforms.uClipR.value = curR * 0.99;
              }
              disturbGridDirectional(pPos[i3], pPos[i3 + 2], -0.0028 * (1.0 - thickness * 0.5), 0, 0, 1);
              pActive[i] = 0.0;
            }
          }
        }
        pGeo.attributes.position.needsUpdate = true;
        pGeo.attributes.aActive.needsUpdate = true;
        pGeo.attributes.aSize.needsUpdate = true;

        // Step 2D Surface Grid (Morphed by Continuous Rheology `thickness`!)
        if (isSpreading) {
          const spreadRate = THREE.MathUtils.lerp(0.22, 0.018, thickness);
          const minSurf = thickness * 0.0045;
          const damp = THREE.MathUtils.lerp(0.965, 0.72, thickness);
          sNext.set(sH);
          const dirs = [-1, 1, -GRID, GRID];
          for (let z = 1; z < GRID - 1; z++) for (let x = 1; x < GRID - 1; x++) {
            const idx = z * GRID + x, h0 = sH[idx];
            if (h0 <= 0.0003) { sFoam[idx] *= 0.88; continue; }
            let outSum = 0;
            for (let d = 0; d < 4; d++) {
              const ni = idx + dirs[d]; if (sObs[ni]) continue;
              const diff = h0 - sH[ni];
              if (diff > minSurf) {
                const flux = Math.min(h0 * 0.22, (diff - minSurf) * spreadRate);
                sNext[idx] -= flux; sNext[ni] += flux; outSum += flux;
              }
            }
            sFoam[idx] = THREE.MathUtils.lerp(sFoam[idx], Math.min(1.0, outSum * 120.0), 0.22);
          }
          for (let i = 0; i < N; i++) {
            sVel[i] = (sVel[i] + (sNext[i] - sH[i])) * damp;
            sH[i] = Math.max(0, sNext[i] + sVel[i] * (thickness > 0.7 ? 0.35 : 0.16));
            sDepth[i] = sH[i];
          }
        } else {
          // Water (fast) -> Honey/Mud (viscous damped) -> Jelly/Elastic (springy wobble!)
          const c2 = thickness < 0.55 ? THREE.MathUtils.lerp(0.42, 0.06, thickness / 0.55) : THREE.MathUtils.lerp(0.06, 0.36, (thickness - 0.55) / 0.45);
          const damp = thickness < 0.55 ? THREE.MathUtils.lerp(0.992, 0.80, thickness / 0.55) : THREE.MathUtils.lerp(0.80, 0.96, (thickness - 0.55) / 0.45);
          const maxH = isBowl ? 0.025 : 0.068, rLimit = curR * 0.98;
          for (let z = 1; z < GRID - 1; z++) {
            const lz = (z / (GRID - 1) - 0.5) * SIZE;
            for (let x = 1; x < GRID - 1; x++) {
              const idx = z * GRID + x;
              if (isCircular && Math.hypot((x / (GRID - 1) - 0.5) * SIZE, lz) > rLimit) { sH[idx] = 0; sVel[idx] = 0; continue; }
              const avg = (sH[idx - 1] + sH[idx + 1] + sH[idx - GRID] + sH[idx + GRID]) * 0.25;
              sVel[idx] = (sVel[idx] + (avg - sH[idx]) * c2) * damp;
              sNext[idx] = THREE.MathUtils.clamp(sH[idx] + sVel[idx], -maxH, maxH);
              const slope = Math.abs(sH[idx + 1] - sH[idx - 1]) + Math.abs(sH[idx + GRID] - sH[idx - GRID]);
              sFoam[idx] = THREE.MathUtils.lerp(sFoam[idx], Math.min(1.0, slope * 14.0 + Math.abs(sVel[idx]) * 18.0), 0.18);
            }
          }
          for (let i = 0; i < GRID; i++) {
            sNext[i] = sNext[GRID + i]; sNext[(GRID - 1) * GRID + i] = sNext[(GRID - 2) * GRID + i];
            sNext[i * GRID] = sNext[i * GRID + 1]; sNext[i * GRID + GRID - 1] = sNext[i * GRID + GRID - 2];
          }
          for (let i = 0; i < N; i++) { sH[i] = sNext[i]; sDepth[i] = curWaterY + sH[i]; }
        }

        const posA = waterGeo.attributes.position;
        for (let i = 0; i < N; i++) posA.setY(i, sH[i]);
        posA.needsUpdate = true; waterGeo.computeVertexNormals();
        waterGeo.attributes.aFoam.needsUpdate = true; waterGeo.attributes.aDepth.needsUpdate = true;
      }
    });

    APP._activeFluid = {
      entity: ent,
      getThickness: () => thickness,
      setThickness: (val) => { thickness = THREE.MathUtils.clamp(Number(val), 0, 1); syncSimUI(); }
    };
    syncSimUI();
    APP.lastEntityId = ent.id;
    APP.emit?.('entity:created', ent);
    return { ok: true, id: ent.id, name: ent.name, message: `Created volumetric fluid simulation (${getThicknessLabel(thickness)}).` };
  }

  // ====================================================================
  // 2. REALISTIC GRANULAR ENGINE (104x104 Oren-Nayar Sand + 7,500 Fine Grains
  //    + Random Emitter + Full Hand & Object Collision & Volume Displacement)
  // ====================================================================
  function createGranularSimulation(opts = {}) {
    const matType = String(opts.material || 'sand').toLowerCase();
    const isSnow = matType === 'snow';
    const isWetSand = matType === 'wet_sand' || matType === 'mud' || matType === 'clay';
    const isGold = matType === 'gold';
    const isGravel = matType === 'gravel';
    const isAsh = matType === 'ash';

    const defaultCol = isGold ? 0xd4a017 : isSnow ? 0xeef4fa : isWetSand ? 0x6f4e32 : isAsh ? 0x475569 : isGravel ? 0x64748b : 0xc6a574;
    const baseCol = parseHexColor(opts.color, defaultCol);
    const targetEnt = opts.pour_on_target ? APP.findEntity?.(opts.pour_on_target) : null;
    const fromEnt = opts.pour_from_entity ? APP.findEntity?.(opts.pour_from_entity) : null;
    const fromInfo = getEntityBounds(fromEnt);

    const sandGroup = new THREE.Group();
    sandGroup.name = `Granular_${matType}_Sim`;

    const SIZE = Number(opts.size) || 2.20;
    const GRID = 104;
    const CELL = SIZE / (GRID - 1);
    const slopeFactor = isSnow ? 1.25 : isWetSand ? 1.05 : isGravel ? 0.85 : isAsh ? 0.52 : 0.66;
    const MAX_SLOPE = CELL * slopeFactor;
    const pileH = new Float32Array(GRID * GRID);
    const pileAO = new Float32Array(GRID * GRID);

    const pileGeo = new THREE.PlaneGeometry(SIZE, SIZE, GRID - 1, GRID - 1);
    pileGeo.rotateX(-Math.PI / 2);
    pileGeo.setAttribute('aAO', new THREE.BufferAttribute(pileAO, 1));

    const pileMat = new THREE.MeshStandardMaterial({
      color: baseCol,
      roughness: isGold ? 0.28 : isWetSand ? 0.45 : isSnow ? 0.88 : 1.0,
      metalness: isGold ? 0.85 : 0.0
    });
    pileMat.customProgramCacheKey = () => `nxr_gran_v5_${matType}`;
    pileMat.onBeforeCompile = (shader) => {
      shader.uniforms.uIsSnow = { value: isSnow ? 1.0 : 0.0 };
      shader.vertexShader = shader.vertexShader
        .replace('#include <common>', '#include <common>\nattribute float aAO; varying vec3 vGranWPos; varying float vGranAO;')
        .replace('#include <begin_vertex>', '#include <begin_vertex>\nvGranWPos = (modelMatrix * vec4(position, 1.0)).xyz; vGranAO = aAO;');
      shader.fragmentShader = shader.fragmentShader
        .replace('#include <common>', '#include <common>\nuniform float uIsSnow; varying vec3 vGranWPos; varying float vGranAO;\nfloat gHash(vec2 p){ p=fract(p*vec2(127.1,311.7)); p+=dot(p,p+34.5); return fract(p.x*p.y); }')
        .replace('#include <map_fragment>', [
          '#include <map_fragment>',
          'float fine = gHash(floor(vGranWPos.xz * 950.0 + vGranWPos.y * 160.0));',
          'float med = gHash(floor(vGranWPos.xz * 320.0));',
          'float ao = clamp(1.0 - vGranAO * 2.1, 0.52, 1.05);',
          'if (uIsSnow > 0.5) { diffuseColor.rgb = mix(diffuseColor.rgb * vec3(0.76, 0.86, 0.98), vec3(0.98, 0.99, 1.0), ao * 0.78 + fine * 0.08); }',
          'else { diffuseColor.rgb *= ao * (0.84 + fine * 0.18 + med * 0.10); }'
        ].join('\n'))
        .replace('#include <emissivemap_fragment>', [
          '#include <emissivemap_fragment>',
          'vec3 Vg = normalize(cameraPosition - vGranWPos);',
          'float glint = step(0.995, gHash(floor(vGranWPos.xz * 560.0) + floor(Vg.xz * 18.0)));',
          'totalEmissiveRadiance += glint * (uIsSnow > 0.5 ? vec3(0.7, 0.85, 1.0) : diffuseColor.rgb * 0.55) * 0.45;'
        ].join('\n'));
    };

    const pileMesh = new THREE.Mesh(pileGeo, pileMat);
    pileMesh.receiveShadow = true; pileMesh.castShadow = true;
    sandGroup.add(pileMesh);

    // 7,500 Tiny Fine Mineral Grains (1.0 - 2.6px max!)
    const GRAIN_COUNT = Math.min(9000, Math.max(4000, Number(opts.count) || 7500));
    const gPos = new Float32Array(GRAIN_COUNT * 3), gVel = new Float32Array(GRAIN_COUNT * 3);
    const gCol = new Float32Array(GRAIN_COUNT * 3), gSize = new Float32Array(GRAIN_COUNT);
    const gState = new Uint8Array(GRAIN_COUNT), gRollTime = new Float32Array(GRAIN_COUNT), gSeed = new Float32Array(GRAIN_COUNT);

    const basePos = fromInfo ? new THREE.Vector3(fromInfo.center.x, 0, fromInfo.center.z)
      : targetEnt?.group ? new THREE.Vector3(targetEnt.group.position.x, 0, targetEnt.group.position.z)
      : resolvePos(opts.position, 1.25);
    sandGroup.position.copy(basePos);
    APP.scene.add(sandGroup);

    const tmpC = new THREE.Color();
    for (let i = 0; i < GRAIN_COUNT; i++) {
      gPos[i * 3] = (Math.random() - 0.5) * 0.04;
      gPos[i * 3 + 1] = 0.04 + Math.random() * 1.15;
      gPos[i * 3 + 2] = (Math.random() - 0.5) * 0.04;
      gVel[i * 3 + 1] = -0.35 - Math.random() * 0.65;
      gSize[i] = isGravel ? (5.5 + Math.random() * 3.0) : isSnow ? (3.5 + Math.random() * 2.5) : (1.6 + Math.random() * 1.4);
      gSeed[i] = Math.random() * 100.0;
      const isMica = !isSnow && Math.random() < 0.08;
      tmpC.copy(baseCol).offsetHSL((Math.random() - 0.5) * 0.03, (Math.random() - 0.5) * 0.10, isMica ? -0.28 : (Math.random() - 0.5) * 0.18);
      gCol[i * 3] = tmpC.r; gCol[i * 3 + 1] = tmpC.g; gCol[i * 3 + 2] = tmpC.b;
    }

    const grainGeo = new THREE.BufferGeometry();
    grainGeo.setAttribute('position', new THREE.BufferAttribute(gPos, 3));
    grainGeo.setAttribute('aCol', new THREE.BufferAttribute(gCol, 3));
    grainGeo.setAttribute('aSize', new THREE.BufferAttribute(gSize, 1));
    const grainPts = new THREE.Points(grainGeo, new THREE.ShaderMaterial({
      transparent: true, depthWrite: true,
      uniforms: { uPx: { value: window.devicePixelRatio || 1 }, uIsSnow: { value: isSnow ? 1.0 : 0.0 } },
      vertexShader: 'attribute vec3 aCol; attribute float aSize; uniform float uPx; varying vec3 vCol; void main(){ vCol=aCol; vec4 mv=modelViewMatrix*vec4(position,1.0); gl_PointSize=clamp((aSize*uPx)/max(0.25,-mv.z), 1.0, 2.8); gl_Position=projectionMatrix*mv; }',
      fragmentShader: 'uniform float uIsSnow; varying vec3 vCol; void main(){ vec2 c=gl_PointCoord-0.5; float r=dot(c,c)*4.0; if(r>1.0)discard; float shade=mix(0.72, 1.15, clamp(-c.y+0.5, 0.0, 1.0)); gl_FragColor=vec4(vCol*shade, uIsSnow>0.5 ? (1.0-r)*0.9 : 1.0); }'
    }));
    grainPts.frustumCulled = false;
    sandGroup.add(grainPts);

    let emitterEnt = null;
    if (!fromEnt && opts.emitter !== false) {
      const emG = new THREE.Group(); emG.name = `${matType}_Emitter`;
      const funnel = new THREE.Mesh(new THREE.CylinderGeometry(0.082, 0.020, 0.16, 24, 1, true), new THREE.MeshPhysicalMaterial({ color: 0xe2e8f0, transmission: 0.78, roughness: 0.12, ior: 1.48, side: THREE.DoubleSide }));
      const collar = new THREE.Mesh(new THREE.TorusGeometry(0.082, 0.008, 10, 24), new THREE.MeshStandardMaterial({ color: 0x64748b, metalness: 0.88, roughness: 0.22 }));
      collar.rotation.x = Math.PI / 2; collar.position.y = 0.08;
      const nozzle = new THREE.Mesh(new THREE.CylinderGeometry(0.022, 0.016, 0.04, 18), new THREE.MeshStandardMaterial({ color: 0x334155, metalness: 0.9, roughness: 0.25 }));
      nozzle.position.y = -0.09;
      const fillCore = new THREE.Mesh(new THREE.ConeGeometry(0.070, 0.11, 18), new THREE.MeshStandardMaterial({ color: baseCol, roughness: 0.94 }));
      fillCore.rotation.x = Math.PI; fillCore.position.y = 0.01;
      emG.add(funnel, collar, nozzle, fillCore);
      emG.position.copy(basePos).add(new THREE.Vector3(0, 1.08, 0));
      APP.scene.add(emG);
      emitterEnt = APP.registerEntity(emG, {
        name: `${matType.charAt(0).toUpperCase() + matType.slice(1)} Emitter`, category: 'tool', grabbable: true, holdable: true, physics: { enabled: false },
        onUpdate: () => { if (!sandGroup.parent) APP.removeEntity?.(emitterEnt, false); }
      });
    }

    const tmpLocal = new THREE.Vector3(), tmpEmPos = new THREE.Vector3(), tmpBox = new THREE.Box3(), tmpCenter = new THREE.Vector3();
    const prevEmPos = new THREE.Vector3(0, 1.0, 0), emVel = new THREE.Vector3(), prevHands = new Map();
    const handCols = [];
    let splashPtr = 0;

    function carveAndThrow(lx, lz, vx, vz, rCells = 2, depthTake = 0.016) {
      const gx = Math.round((lx / SIZE + 0.5) * (GRID - 1)), gz = Math.round((lz / SIZE + 0.5) * (GRID - 1));
      if (gx < 2 || gx >= GRID - 2 || gz < 2 || gz >= GRID - 2) return;
      const centerH = pileH[gz * GRID + gx];
      if (centerH <= 0.003) return;
      for (let dz = -rCells; dz <= rCells; dz++) for (let dx = -rCells; dx <= rCells; dx++) {
        const nx = gx + dx, nz = gz + dz, d = Math.hypot(dx, dz);
        if (nx > 1 && nx < GRID - 2 && nz > 1 && nz < GRID - 2 && d <= rCells + 0.2) {
          const take = Math.min(pileH[nz * GRID + nx], depthTake * (1.0 - d / (rCells + 0.5)));
          pileH[nz * GRID + nx] -= take;
          // Volume conservation: push displaced sand outward into realistic berm ridges!
          const rx = THREE.MathUtils.clamp(nx + Math.sign(dx) * (rCells + 1), 1, GRID - 2);
          const rz = THREE.MathUtils.clamp(nz + Math.sign(dz) * (rCells + 1), 1, GRID - 2);
          pileH[rz * GRID + rx] += take * 0.85;
        }
      }
      for (let k = 0; k < 28; k++) {
        const idx = splashPtr; splashPtr = (splashPtr + 1) % GRAIN_COUNT;
        const i3 = idx * 3; gState[idx] = 2;
        gPos[i3] = lx + (Math.random() - 0.5) * 0.06; gPos[i3 + 1] = centerH + 0.012; gPos[i3 + 2] = lz + (Math.random() - 0.5) * 0.06;
        gVel[i3] = vx * 0.45 + (Math.random() - 0.5) * 0.65; gVel[i3 + 1] = 0.42 + Math.random() * 0.65; gVel[i3 + 2] = vz * 0.45 + (Math.random() - 0.5) * 0.65;
      }
    }

    interactives.push({
      type: 'granular', group: sandGroup, mesh: pileMesh,
      onPointerHit: (worldPt, worldVel, isClick, isDown) => {
        if (!isClick && !isDown && worldVel.length() < 0.006) return;
        tmpLocal.copy(worldPt); sandGroup.worldToLocal(tmpLocal);
        carveAndThrow(tmpLocal.x, tmpLocal.z, worldVel.x * 16.0, worldVel.z * 16.0);
      }
    });

    function depositSmooth(gx, gz, amt) {
      const w0 = amt * 0.44, w1 = amt * 0.14;
      pileH[gz * GRID + gx] = Math.min(0.62, pileH[gz * GRID + gx] + w0);
      pileH[gz * GRID + gx + 1] = Math.min(0.62, pileH[gz * GRID + gx + 1] + w1);
      pileH[gz * GRID + gx - 1] = Math.min(0.62, pileH[gz * GRID + gx - 1] + w1);
      pileH[(gz + 1) * GRID + gx] = Math.min(0.62, pileH[(gz + 1) * GRID + gx] + w1);
      pileH[(gz - 1) * GRID + gx] = Math.min(0.62, pileH[(gz - 1) * GRID + gx] + w1);
    }

    const ent = APP.registerEntity(sandGroup, {
      name: `Granular ${matType}`, category: 'world_fx', grabbable: false, physics: { enabled: false },
      onUpdate: function (dt) {
        const delta = Math.min(0.033, Math.max(0.004, dt || 0.016)), now = performance.now() * 0.001;

        if (emitterEnt?.group?.parent) {
          tmpEmPos.set(0, -0.11, 0); emitterEnt.group.localToWorld(tmpEmPos); sandGroup.worldToLocal(tmpEmPos);
        } else if (fromEnt?.group?.parent) {
          const fi = getEntityBounds(fromEnt);
          if (fi) { tmpEmPos.copy(fi.center); tmpEmPos.y = fi.box.min.y + fi.size.y * 0.25; sandGroup.worldToLocal(tmpEmPos); }
        } else {
          // Organic random emitter drift if no custom source
          tmpEmPos.set(targetEnt ? 0 : Math.sin(now * 0.72) * 0.22 + Math.sin(now * 2.1) * 0.04, 1.12, targetEnt ? 0 : Math.cos(now * 0.58) * 0.22);
        }
        emVel.subVectors(tmpEmPos, prevEmPos).divideScalar(delta).clampLength(0, 2.5);
        prevEmPos.copy(tmpEmPos);

        // Scene Object Colliders (both Falling Grains AND Sand Pile displacement!)
        const obsBoxes = [];
        for (const other of APP.entities || []) {
          if (!other || other === ent || other === fromEnt || other === emitterEnt || !other.group || other.category === 'world_fx' || other.category === 'board') continue;
          tmpBox.setFromObject(other.group);
          if (!tmpBox.isEmpty()) {
            tmpBox.min.sub(sandGroup.position); tmpBox.max.sub(sandGroup.position);
            obsBoxes.push(tmpBox.clone());
            // Displace sand pile when an object rests on or pushes through it!
            if (tmpBox.min.y < 0.45 && tmpBox.max.y > -0.02) {
              tmpBox.getCenter(tmpCenter);
              const ox = Math.round((tmpCenter.x / SIZE + 0.5) * (GRID - 1)), oz = Math.round((tmpCenter.z / SIZE + 0.5) * (GRID - 1));
              if (ox >= 2 && ox < GRID - 2 && oz >= 2 && oz < GRID - 2 && tmpBox.min.y <= pileH[oz * GRID + ox] + 0.01) {
                carveAndThrow(tmpCenter.x, tmpCenter.z, 0, 0, 2, 0.008);
              }
            }
          }
        }

        // VR Hands / Controllers Colliders (Deflect Falling Sand + Carve Pile!)
        handCols.length = 0;
        for (const [hk, h] of Object.entries(APP.hands || {})) {
          if (!h || !h.active) continue;
          const pts = [];
          if (h.isHand && h.fingerTips?.index) pts.push([hk + '_idx', h.fingerTips.index, 0.042]);
          if (h.pos) pts.push([hk + '_palm', h.pos, 0.072]);
          for (const [key, hp, rH] of pts) {
            tmpLocal.copy(hp); sandGroup.worldToLocal(tmpLocal);
            const last = prevHands.get(key);
            const hVx = last ? (tmpLocal.x - last.x) / delta : 0, hVy = last ? (tmpLocal.y - last.y) / delta : 0, hVz = last ? (tmpLocal.z - last.z) / delta : 0;
            prevHands.set(key, tmpLocal.clone());
            handCols.push({ pos: tmpLocal.clone(), vel: new THREE.Vector3(hVx, hVy, hVz), r: rH });
            const gx = Math.round((tmpLocal.x / SIZE + 0.5) * (GRID - 1)), gz = Math.round((tmpLocal.z / SIZE + 0.5) * (GRID - 1));
            if (gx >= 2 && gx < GRID - 2 && gz >= 2 && gz < GRID - 2 && tmpLocal.y <= pileH[gz * GRID + gx] + 0.08) {
              carveAndThrow(tmpLocal.x, tmpLocal.z, hVx, hVz, 2, 0.018);
            }
          }
        }

        const grav = isSnow ? 1.85 : isAsh ? 1.2 : 8.8;
        const depositPerGrain = isSnow ? 0.0005 : 0.00042;
        const nozzleSpread = isSnow ? 0.065 : 0.028;

        for (let i = 0; i < GRAIN_COUNT; i++) {
          const i3 = i * 3, st = gState[i];
          if (st === 1) {
            gRollTime[i] -= delta;
            const gx = Math.floor((gPos[i3] / SIZE + 0.5) * (GRID - 1)), gz = Math.floor((gPos[i3 + 2] / SIZE + 0.5) * (GRID - 1));
            if (gx >= 2 && gx < GRID - 2 && gz >= 2 && gz < GRID - 2 && gRollTime[i] > 0) {
              const idx = gz * GRID + gx;
              const dhx = (pileH[idx + 1] - pileH[idx - 1]) / (2 * CELL), dhz = (pileH[idx + GRID] - pileH[idx - GRID]) / (2 * CELL);
              gVel[i3] = gVel[i3] * 0.88 - dhx * 1.45 * delta * 60.0;
              gVel[i3 + 2] = gVel[i3 + 2] * 0.88 - dhz * 1.45 * delta * 60.0;
              gPos[i3] += gVel[i3] * delta; gPos[i3 + 2] += gVel[i3 + 2] * delta; gPos[i3 + 1] = pileH[idx] + 0.002;
              if (Math.hypot(dhx, dhz) < slopeFactor * 0.35 && gRollTime[i] < 0.2) gRollTime[i] = 0;
              continue;
            }
            if (gx >= 2 && gx < GRID - 2 && gz >= 2 && gz < GRID - 2) depositSmooth(gx, gz, depositPerGrain);
            gState[i] = 0;
            gPos[i3] = tmpEmPos.x + (Math.random() - 0.5) * nozzleSpread; gPos[i3 + 1] = Math.max(0.08, tmpEmPos.y) - Math.random() * 0.04; gPos[i3 + 2] = tmpEmPos.z + (Math.random() - 0.5) * nozzleSpread;
            gVel[i3] = emVel.x * 0.55 + (Math.random() - 0.5) * 0.16; gVel[i3 + 1] = emVel.y * 0.35 - 0.18 - Math.random() * 0.32; gVel[i3 + 2] = emVel.z * 0.55 + (Math.random() - 0.5) * 0.16;
            continue;
          }

          if (isSnow || isAsh) {
            gVel[i3] += Math.sin(now * 2.4 + gSeed[i]) * 0.42 * delta; gVel[i3 + 2] += Math.cos(now * 2.0 + gSeed[i] * 1.3) * 0.42 * delta;
            gVel[i3] *= 0.97; gVel[i3 + 2] *= 0.97; gVel[i3 + 1] = Math.max(-0.85, gVel[i3 + 1] - grav * delta);
          } else gVel[i3 + 1] -= grav * delta;

          gPos[i3] += gVel[i3] * delta; gPos[i3 + 1] += gVel[i3 + 1] * delta; gPos[i3 + 2] += gVel[i3 + 2] * delta;

          // Collide Poured Grains with VR Hands / Controllers!
          for (let hc = 0; hc < handCols.length; hc++) {
            const h = handCols[hc];
            const dx = gPos[i3] - h.pos.x, dy = gPos[i3 + 1] - h.pos.y, dz = gPos[i3 + 2] - h.pos.z;
            const dSq = dx * dx + dy * dy + dz * dz;
            if (dSq < h.r * h.r && dSq > 1e-6) {
              const d = Math.sqrt(dSq), nx = dx / d, ny = dy / d, nz = dz / d;
              gPos[i3] = h.pos.x + nx * h.r; gPos[i3 + 1] = h.pos.y + ny * h.r; gPos[i3 + 2] = h.pos.z + nz * h.r;
              gVel[i3] = h.vel.x * 0.65 + nx * 0.55 + (Math.random() - 0.5) * 0.25;
              gVel[i3 + 1] = Math.max(0.1, h.vel.y * 0.65 + ny * 0.45);
              gVel[i3 + 2] = h.vel.z * 0.65 + nz * 0.55 + (Math.random() - 0.5) * 0.25;
            }
          }

          // Collide Poured Grains with Scene Objects
          for (let b = 0; b < obsBoxes.length; b++) {
            const ob = obsBoxes[b];
            if (gPos[i3] > ob.min.x && gPos[i3] < ob.max.x && gPos[i3 + 2] > ob.min.z && gPos[i3 + 2] < ob.max.z && gPos[i3 + 1] > ob.min.y && gPos[i3 + 1] < ob.max.y) {
              ob.getCenter(tmpCenter);
              gPos[i3 + 1] = ob.max.y + 0.005;
              gVel[i3] += (gPos[i3] - tmpCenter.x) * 3.2; gVel[i3 + 2] += (gPos[i3 + 2] - tmpCenter.z) * 3.2;
              gVel[i3 + 1] = Math.abs(gVel[i3 + 1]) * (isSnow ? 0.05 : 0.22);
            }
          }

          const gx = Math.floor((gPos[i3] / SIZE + 0.5) * (GRID - 1)), gz = Math.floor((gPos[i3 + 2] / SIZE + 0.5) * (GRID - 1));
          const inB = gx >= 2 && gx < GRID - 2 && gz >= 2 && gz < GRID - 2;
          const fY = inB ? pileH[gz * GRID + gx] : 0;

          if (gPos[i3 + 1] <= fY) {
            if (inB && !isSnow && Math.random() < 0.42) {
              gState[i] = 1; gRollTime[i] = 0.22 + Math.random() * 0.35; gPos[i3 + 1] = fY + 0.002;
              gVel[i3] *= 0.4; gVel[i3 + 2] *= 0.4; gVel[i3 + 1] = 0;
            } else {
              if (inB) depositSmooth(gx, gz, depositPerGrain);
              gState[i] = 0;
              gPos[i3] = tmpEmPos.x + (Math.random() - 0.5) * nozzleSpread; gPos[i3 + 1] = Math.max(0.08, tmpEmPos.y) - Math.random() * 0.04; gPos[i3 + 2] = tmpEmPos.z + (Math.random() - 0.5) * nozzleSpread;
              gVel[i3] = emVel.x * 0.55 + (Math.random() - 0.5) * 0.16; gVel[i3 + 1] = emVel.y * 0.35 - 0.18 - Math.random() * 0.32; gVel[i3 + 2] = emVel.z * 0.55 + (Math.random() - 0.5) * 0.16;
            }
          }
        }
        grainGeo.attributes.position.needsUpdate = true;

        const dirs = [-1, 1, -GRID, GRID, -GRID - 1, -GRID + 1, GRID - 1, GRID + 1];
        const diagScale = [1, 1, 1, 1, 1.414, 1.414, 1.414, 1.414];
        const slumpRate = isWetSand ? 0.14 : isSnow ? 0.18 : 0.32;
        for (let pass = 0; pass < 2; pass++) {
          for (let z = 1; z < GRID - 1; z++) for (let x = 1; x < GRID - 1; x++) {
            const idx = z * GRID + x, h = pileH[idx];
            if (h <= 0.0004) { pileAO[idx] = 0; continue; }
            let aoSum = 0;
            for (let d = 0; d < 8; d++) {
              const ni = idx + dirs[d], lim = MAX_SLOPE * diagScale[d], diff = h - pileH[ni];
              if (diff < 0) aoSum += -diff;
              if (diff > lim) { const mv = (diff - lim) * slumpRate * (d >= 4 ? 0.5 : 1.0); pileH[idx] -= mv; pileH[ni] += mv; }
            }
            pileAO[idx] = aoSum * 12.0;
          }
        }
        const posA = pileGeo.attributes.position;
        for (let i = 0; i < pileH.length; i++) posA.setY(i, pileH[i]);
        posA.needsUpdate = true; pileGeo.attributes.aAO.needsUpdate = true; pileGeo.computeVertexNormals();
      }
    });

    APP.lastEntityId = ent.id; APP.emit?.('entity:created', ent);
    return { ok: true, id: ent.id, emitterId: emitterEnt?.id, name: ent.name, message: `Spawned realistic ${matType} simulation.` };
  }

  async function buildClothTextures(texName, baseCol) {
    const type = String(texName || 'linen').toLowerCase();
    const b = document.createElement('canvas'); b.width = 512; b.height = 512;
    const btx = b.getContext('2d');
    btx.fillStyle = '#808080'; btx.fillRect(0, 0, 512, 512);
    for (let i = 0; i < 512; i += 4) {
      btx.fillStyle = (i % 8 === 0) ? '#626262' : '#a8a8a8';
      btx.fillRect(i, 0, 2, 512); btx.fillRect(0, i, 512, 2);
    }
    const bumpTex = new THREE.CanvasTexture(b);

    // Check if texName matches a photo in /assets/ or importedImages!
    const builtins = ['linen', 'silk', 'denim', 'velvet', 'lace', 'chainmail', 'royal'];
    if (!builtins.includes(type)) {
      const url = await resolveAssetImageUrl(texName);
      if (url) {
        try {
          const mapTex = await new THREE.TextureLoader().loadAsync(url);
          mapTex.colorSpace = THREE.SRGBColorSpace;
          mapTex.anisotropy = 4;
          return { map: mapTex, bumpMap: bumpTex, alphaMap: null, roughness: 0.78, metalness: 0.04, sheen: 0.45, sheenRoughness: 0.6, bumpScale: 0.018, alphaTest: 0, label: texName };
        } catch (_) {}
      }
    }

    const c = document.createElement('canvas'); c.width = 512; c.height = 512;
    const ctx = c.getContext('2d');
    const hex = '#' + baseCol.getHexString();
    ctx.fillStyle = hex; ctx.fillRect(0, 0, 512, 512);

    let alphaTex = null;
    let matProps = { roughness: 0.85, metalness: 0.0, sheen: 0.45, sheenRoughness: 0.65, bumpScale: 0.018, alphaTest: 0 };

    if (type === 'silk') {
      const grad = ctx.createLinearGradient(0, 0, 512, 512);
      grad.addColorStop(0, 'rgba(255,255,255,0.12)'); grad.addColorStop(0.5, 'rgba(0,0,0,0.05)'); grad.addColorStop(1, 'rgba(255,255,255,0.15)');

      matProps = { roughness: 0.30, metalness: 0.08, sheen: 1.0, sheenRoughness: 0.24, bumpScale: 0.007, alphaTest: 0 };
    } else if (type === 'denim') {
      ctx.fillStyle = baseCol.getHex() === 0x7f1d1d ? '#1e3557' : hex; ctx.fillRect(0, 0, 512, 512);
      for (let i = -512; i < 1024; i += 4) { ctx.strokeStyle = (i % 8 === 0) ? 'rgba(210,225,245,0.13)' : 'rgba(10,18,35,0.16)'; ctx.lineWidth = 1.5; ctx.beginPath(); ctx.moveTo(i, 0); ctx.lineTo(i + 512, 512); ctx.stroke(); }
      matProps = { roughness: 0.90, metalness: 0.0, sheen: 0.18, sheenRoughness: 0.85, bumpScale: 0.025, alphaTest: 0 };
    } else if (type === 'velvet') {
      ctx.fillStyle = baseCol.clone().multiplyScalar(0.68).getStyle(); ctx.fillRect(0, 0, 512, 512);
      for (let i = 0; i < 12000; i++) { ctx.fillStyle = Math.random() > 0.5 ? 'rgba(255,255,255,0.045)' : 'rgba(0,0,0,0.14)'; ctx.fillRect(Math.random() * 512, Math.random() * 512, 2, 2); }
      matProps = { roughness: 0.94, metalness: 0.0, sheen: 1.0, sheenRoughness: 0.32, bumpScale: 0.022, alphaTest: 0 };
    } else if (type === 'chainmail') {
      ctx.fillStyle = '#2d3748'; ctx.fillRect(0, 0, 512, 512); ctx.strokeStyle = '#cbd5e1'; ctx.lineWidth = 2.5;
      for (let y = 8; y < 512; y += 14) for (let x = 8; x < 512; x += 14) { const ox = (Math.floor(y / 14) % 2) * 7; ctx.beginPath(); ctx.arc(x + ox, y, 6, 0, Math.PI * 2); ctx.stroke(); }
      matProps = { roughness: 0.30, metalness: 0.92, sheen: 0.0, sheenRoughness: 0.5, bumpScale: 0.036, alphaTest: 0 };
    } else if (type === 'lace') {
      const a = document.createElement('canvas'); a.width = 512; a.height = 512; const atx = a.getContext('2d');
      atx.fillStyle = '#000000'; atx.fillRect(0, 0, 512, 512); atx.strokeStyle = '#ffffff'; atx.lineWidth = 3.5;
      ctx.fillStyle = baseCol.getHex() === 0x7f1d1d ? '#f1f5f9' : hex; ctx.fillRect(0, 0, 512, 512);
      for (let y = 0; y <= 512; y += 28) for (let x = 0; x <= 512; x += 28) { atx.beginPath(); atx.arc(x, y, 12, 0, Math.PI * 2); atx.stroke(); }
      alphaTex = new THREE.CanvasTexture(a);
      matProps = { roughness: 0.72, metalness: 0.02, sheen: 0.55, sheenRoughness: 0.5, bumpScale: 0.015, alphaTest: 0.45 };
    } else {

      for (let i = 0; i < 512; i += 4) {

    const mapTex = new THREE.CanvasTexture(c); mapTex.colorSpace = THREE.SRGBColorSpace;
    return { map: mapTex, bumpMap: bumpTex, alphaMap: alphaTex, ...matProps, label: type };
  }

  function createClothFlameMaterial() {
    const uniforms = { uTime: { value: 0 } };
    const mat = new THREE.ShaderMaterial({
      uniforms,
      transparent: true, depthWrite: false, side: THREE.DoubleSide, blending: THREE.AdditiveBlending,
      vertexShader: 'uniform float uTime; varying vec2 vUv; varying float vH; void main(){ vUv=uv; vH=clamp(uv.y,0.0,1.0); vec3 p=position; p.x+=sin(uTime*12.0+p.y*16.0)*0.02*vH*vH; p.z+=cos(uTime*9.0+p.y*14.0)*0.02*vH*vH; gl_Position=projectionMatrix*modelViewMatrix*vec4(p,1.0); }',
      fragmentShader: [
        'uniform float uTime; varying vec2 vUv; varying float vH;',
        'float fH(vec2 p){ return fract(sin(dot(p,vec2(127.1,311.7)))*43758.5453); }',
        'float fN(vec2 p){ vec2 i=floor(p); vec2 f=fract(p); f=f*f*(3.0-2.0*f); return mix(mix(fH(i),fH(i+vec2(1,0)),f.x),mix(fH(i+vec2(0,1)),fH(i+vec2(1,1)),f.x),f.y); }',
        'float fbm(vec2 p){ return 0.55*fN(p)+0.30*fN(p*2.1)+0.15*fN(p*4.2); }',
        'void main(){',
        '  float n = fbm(vec2(vUv.x * 4.5, vUv.y * 3.8 - uTime * 4.2));',
        '  float dens = smoothstep(0.28, 0.88, (1.0 - vH) * 1.12 + (n - 0.42) * 0.72);',
        '  float rad = 1.0 - smoothstep(0.14, 0.50, abs(vUv.x - 0.5));',
        '  float a = dens * rad * smoothstep(0.0, 0.14, vH) * (1.0 - smoothstep(0.75, 1.0, vH));',
        '  if (a < 0.01) discard;',
        '  vec3 col = mix(vec3(0.92, 0.16, 0.02), vec3(1.0, 0.65, 0.10), smoothstep(0.25, 0.68, dens));',
        '  col = mix(col, vec3(1.0, 0.96, 0.82), smoothstep(0.68, 0.95, dens) * (1.0 - vH * 0.6));',
        '  gl_FragColor = vec4(col * 1.45, a * 0.95);',
        '}'
      ].join('\n')
    });
    return { mat, uniforms };
  }

  async function createClothSimulation(opts = {}) {

    }

    }

    const baseCol = parseHexColor(opts.color, 0x7f1d1d);

    const texData = await buildClothTextures(curTexName, baseCol);

    const cur = [], prev = [], pinned = [];
    for (let r = 0; r < ROWS; r++) for (let c = 0; c < COLS; c++) {
      const x = (c / (COLS - 1) - 0.5) * W, y = isDrape ? 0.0 : (H * 0.5 - r * DY), z = isDrape ? (r / (ROWS - 1) - 0.5) * H : 0.0;
      cur.push(new THREE.Vector3(x, y, z)); prev.push(new THREE.Vector3(x, y, z));

    function addSpring(a, b, rest, mult, stiff = 1.0, isBend = false) {
      const s = { a, b, rest, tearDist: rest * mult, stiff, isBend, broken: false }; springs.push(s); return s;
    }
    for (let r = 0; r < ROWS; r++) for (let c = 0; c < COLS; c++) {
      const i = r * COLS + c;
      if (c < COLS - 1) hGrid[r][c] = addSpring(i, i + 1, DX, 2.35, 1.0, false);
      if (r < ROWS - 1) vGrid[r][c] = addSpring(i, i + COLS, DY, 2.35, 1.0, false);
      if (r < ROWS - 1 && c < COLS - 1) { addSpring(i, i + COLS + 1, DIAG, 2.65, 0.88, false); addSpring(i + 1, i + COLS, DIAG, 2.65, 0.88, false); }
      if (c < COLS - 2) addSpring(i, i + 2, DX * 2, 99.0, 0.36, true);

    }

    const clothMat = new THREE.MeshPhysicalMaterial({
      map: texData.map, bumpMap: texData.bumpMap, bumpScale: texData.bumpScale,

      clothMat.userData.shader = shader;
      shader.vertexShader = shader.vertexShader
        .replace('#include <common>', '#include <common>\nattribute float aHeat; attribute float aChar; attribute float aTear;\nvarying float vHeat; varying float vChar; varying float vTear; varying vec2 vClothUv; varying vec3 vClothWPos;')
        .replace('#include <begin_vertex>', '#include <begin_vertex>\nvHeat=aHeat; vChar=aChar; vTear=aTear; vClothUv=uv; vClothWPos=(modelMatrix*vec4(position,1.0)).xyz;');
      shader.fragmentShader = shader.fragmentShader
        .replace('#include <common>', [
          '#include <common>',
          'uniform float uBurnTime; varying float vHeat; varying float vChar; varying float vTear; varying vec2 vClothUv; varying vec3 vClothWPos;',
          'float cH(vec2 p){ p=fract(p*vec2(127.1,311.7)); p+=dot(p,p+34.5); return fract(p.x*p.y); }',
          'float cN(vec2 p){ vec2 i=floor(p); vec2 f=fract(p); f=f*f*(3.0-2.0*f); return mix(mix(cH(i),cH(i+vec2(1,0)),f.x),mix(cH(i+vec2(0,1)),cH(i+vec2(1,1)),f.x),f.y); }',
          'float cFbm(vec2 p){ return 0.55*cN(p)+0.30*cN(p*2.1)+0.15*cN(p*4.3); }'
        ].join('\n'))
        .replace('#include <map_fragment>', [
          '#include <map_fragment>',
          // Smooth organic curve holes + frayed woven threads (NOT blocky Minecraft quads!)
          'float fbmEdge = cFbm(vClothUv * 38.0);',
          'float frayThread = max(sin(vClothUv.x * 240.0), sin(vClothUv.y * 240.0));',
          'if (vTear > 0.15 && (vTear * 0.7 + fbmEdge * 0.5) > 0.88 && frayThread < 0.35) discard;',
          'float burnProg = vChar + (fbmEdge - 0.5) * 0.24;',
          'if (vChar > 0.02 && burnProg > 0.92) discard;',
          'float scorch = smoothstep(0.14, 0.48, vHeat) * (1.0 - smoothstep(0.55, 0.88, burnProg));',
          'diffuseColor.rgb = mix(diffuseColor.rgb, diffuseColor.rgb * vec3(0.40, 0.25, 0.12), scorch);',
          'diffuseColor.rgb = mix(diffuseColor.rgb, vec3(0.02, 0.018, 0.016), smoothstep(0.50, 0.82, burnProg));'
        ].join('\n'))
        .replace('#include <emissivemap_fragment>', [
          '#include <emissivemap_fragment>',
          'float fbmEm = cFbm(vClothUv * 38.0);',
          'float burnP = vChar + (fbmEm - 0.5) * 0.24;',
          'float flick = 0.72 + 0.35 * sin(uBurnTime * 24.0 + vClothWPos.x * 20.0 + vClothWPos.y * 16.0);',
          'float emberLine = smoothstep(0.74, 0.88, burnP) * (1.0 - smoothstep(0.88, 0.94, burnP)) * step(0.14, vHeat);',
          'vec3 emberCol = mix(vec3(1.0, 0.16, 0.01), vec3(1.0, 0.82, 0.25), smoothstep(0.82, 0.91, burnP));',
          'totalEmissiveRadiance += emberCol * emberLine * flick * 4.5;'
        ].join('\n'));
    };

    const clothMesh = new THREE.Mesh(clothGeo, clothMat);
    clothMesh.castShadow = true; clothMesh.receiveShadow = true;
    clothGroup.add(clothMesh);

    // REAL 3D VOLUMETRIC FIRE FLAMES ON CLOTH (24 Dynamic FBM Flame Clusters!)
    const FLAME_POOL = 24;
    const flameShader = createClothFlameMaterial();
    const flameGeo = new THREE.PlaneGeometry(0.11, 0.20, 6, 10);
    flameGeo.translate(0, 0.10, 0);
    const clothFlames = [];
    for (let i = 0; i < FLAME_POOL; i++) {
      const fg = new THREE.Group();
      const m1 = new THREE.Mesh(flameGeo, flameShader.mat);
      const m2 = new THREE.Mesh(flameGeo, flameShader.mat); m2.rotation.y = Math.PI * 0.5;
      fg.add(m1, m2);
      fg.visible = false;
      clothGroup.add(fg);
      clothFlames.push(fg);
    }
    const burnLight = new THREE.PointLight(0xff6b1a, 0.0, 3.8);
    clothGroup.add(burnLight);

    // Natural Alpha-Blended Rising Smoke Plumes + Frayed Fibers
    const SMOKE_MAX = 150;
    const smPos = new Float32Array(SMOKE_MAX * 3), smVel = Array.from({ length: SMOKE_MAX }, () => new THREE.Vector3());
    const smLife = new Float32Array(SMOKE_MAX), smType = new Float32Array(SMOKE_MAX);
    const smGeo = new THREE.BufferGeometry();
    smGeo.setAttribute('position', new THREE.BufferAttribute(smPos, 3));
    smGeo.setAttribute('aLife', new THREE.BufferAttribute(smLife, 1));
    smGeo.setAttribute('aType', new THREE.BufferAttribute(smType, 1));
    clothGroup.add(new THREE.Points(smGeo, new THREE.ShaderMaterial({
      transparent: true, depthWrite: false,
      uniforms: { uPx: { value: window.devicePixelRatio || 1 } },
      vertexShader: 'attribute float aLife; attribute float aType; uniform float uPx; varying float vLife; varying float vType; void main(){ vLife=aLife; vType=aType; vec4 mv=modelViewMatrix*vec4(position,1.0); float sz = aType < 0.5 ? mix(42.0, 18.0, aLife) : 6.0; gl_PointSize=aLife>0.01?clamp((sz*uPx)/max(0.25,-mv.z),1.5,24.0):0.0; gl_Position=projectionMatrix*mv; }',

    let nextSm = 0;
    function emitParticle(pt, type, vx, vy, vz, life) {
      const fi = nextSm; nextSm = (nextSm + 1) % SMOKE_MAX;
      smPos[fi * 3] = pt.x + (Math.random() - 0.5) * 0.02; smPos[fi * 3 + 1] = pt.y + (Math.random() - 0.5) * 0.02; smPos[fi * 3 + 2] = pt.z + (Math.random() - 0.5) * 0.02;
      smVel[fi].set(vx, vy, vz); smLife[fi] = life; smType[fi] = type;
    }

    let indicesDirty = true;
    function rebuildClothIndices() {
      let ptr = 0;
      for (let r = 0; r < ROWS - 1; r++) for (let c = 0; c < COLS - 1; c++) {
        const a = r * COLS + c, b = a + 1, d = a + COLS, e = d + 1;
        if (burned[a] + burned[b] + burned[d] + burned[e] >= 3) continue;
        if ((!hGrid[r][c].broken ? 1 : 0) + (!hGrid[r + 1][c].broken ? 1 : 0) + (!vGrid[r][c].broken ? 1 : 0) + (!vGrid[r][c + 1].broken ? 1 : 0) < 3) continue;
        idxArr[ptr++] = a; idxArr[ptr++] = d; idxArr[ptr++] = b; idxArr[ptr++] = b; idxArr[ptr++] = d; idxArr[ptr++] = e;
      }
      clothGeo.index.needsUpdate = true; clothGeo.setDrawRange(0, ptr); indicesDirty = false;
    }
    rebuildClothIndices();

    if (bInfoA && bInfoB) {
      const mid = new THREE.Vector3().addVectors(bInfoA.center, bInfoB.center).multiplyScalar(0.5);
      mid.y = Math.max(bInfoA.box.max.y, bInfoB.box.max.y) - H * 0.45;
      clothGroup.position.copy(mid);
      const dir = new THREE.Vector3().subVectors(bInfoB.center, bInfoA.center).setY(0).normalize();
      if (dir.lengthSq() > 0.01) clothGroup.rotation.y = Math.atan2(-dir.z, dir.x);
    } else if (isDrape && targetEnt?.group) {
      const tInfo = getEntityBounds(targetEnt);
      clothGroup.position.set(tInfo.center.x, tInfo.box.max.y + 0.25, tInfo.center.z);
    } else {
      const sp = resolvePos(opts.position, 1.05);
      clothGroup.position.set(sp.x, isDrape ? 1.0 : 1.15, sp.z);
      const hp = APP.headPos || new THREE.Vector3(0, 1.5, 0);
      if (!isDrape) clothGroup.lookAt(hp.x, clothGroup.position.y, hp.z);
    }
    APP.scene.add(clothGroup);

    const tmpDelta = new THREE.Vector3(), tmpLocal = new THREE.Vector3(), tmpSphereC = new THREE.Vector3(), tmpBox = new THREE.Box3(), prevHands = new Map();
    const tmpWind = new THREE.Vector3(), tmpNorm = new THREE.Vector3();
    let grabbedIdx = -1, grabbedHand = null, ptrGrabIdx = -1, ptrDragTarget = null;
    let isBurning = !!opts.burning, burnSpeed = Number(opts.burn_speed) || 1.0;

    function tearAtLocalPoint(pt, rad, imp) {
      let tore = false;
      for (let i = 0; i < springs.length; i++) {
        const s = springs[i]; if (s.broken || s.isBend) continue;
        const mx = (cur[s.a].x + cur[s.b].x) * 0.5, my = (cur[s.a].y + cur[s.b].y) * 0.5, mz = (cur[s.a].z + cur[s.b].z) * 0.5;
        if (Math.hypot(mx - pt.x, my - pt.y, mz - pt.z) < rad) {
          s.broken = true; tore = true; indicesDirty = true;
          tearArr[s.a] = 1.0; tearArr[s.b] = 1.0;
          emitParticle(cur[s.a], 2, (Math.random() - 0.5) * 0.8, Math.random() * 0.5, (Math.random() - 0.5) * 0.8, 0.5);
          if (imp) { prev[s.a].sub(imp); prev[s.b].add(imp); }
        }
      }
      if (tore) clothGeo.attributes.aTear.needsUpdate = true;
      return tore;
    }

    interactives.push({
      type: 'cloth', group: clothGroup, mesh: clothMesh,
      onPointerHit: (worldPt, worldVel, isClick) => {
        tmpLocal.copy(worldPt); clothGroup.worldToLocal(tmpLocal);
        if (isClick) {
          let bestI = -1, bestD = 0.16;
          for (let i = (isDrape ? 0 : COLS); i < N_COUNT; i++) { if (burned[i]) continue; const d = cur[i].distanceTo(tmpLocal); if (d < bestD) { bestD = d; bestI = i; } }
          if (bestI !== -1) { ptrGrabIdx = bestI; ptrDragTarget = tmpLocal.clone(); }
        } else if (worldVel.length() > 0.002) {
          for (let i = (isDrape ? 0 : COLS); i < N_COUNT; i++) if (cur[i].distanceTo(tmpLocal) < 0.14) prev[i].addScaledVector(worldVel, -0.45);
        }
      },
      onPointerDrag: (worldPt) => { if (ptrGrabIdx === -1) return; tmpLocal.copy(worldPt); clothGroup.worldToLocal(tmpLocal); ptrDragTarget = tmpLocal.clone(); },
      onPointerRelease: () => { ptrGrabIdx = -1; ptrDragTarget = null; }
    });

    function igniteAt(origin = 'bottom') {
      isBurning = true;
      const o = String(origin).toLowerCase();
      if (o === 'center') { const ci = Math.floor(ROWS * 0.55) * COLS + Math.floor(COLS * 0.5); heatArr[ci] = 1.0; if (ci + 1 < N_COUNT) heatArr[ci + 1] = 0.9; }
      else if (o === 'corners') { heatArr[(ROWS - 1) * COLS] = 1.0; heatArr[N_COUNT - 1] = 1.0; }
      else if (o === 'random') { for (let k = 0; k < 3; k++) heatArr[Math.floor(Math.random() * N_COUNT)] = 1.0; }
      else { const r = ROWS - 1; for (let c = Math.floor(COLS * 0.35); c <= Math.floor(COLS * 0.65); c++) heatArr[r * COLS + c] = 0.95 + Math.random() * 0.05; }
    }
    if (isBurning) igniteAt(opts.burn_origin || 'bottom');

    const clothApi = {
      group: clothGroup,
      isTearable: () => tearable,
      setTearable: (val) => { tearable = !!val; syncSimUI(); },
      tearSlice: (yRatio = 0.45) => {
        const targetRow = Math.max(1, Math.min(ROWS - 2, Math.floor(yRatio * ROWS)));
        for (let c = 0; c < COLS; c++) tearAtLocalPoint(cur[THREE.MathUtils.clamp(targetRow + Math.round(Math.sin(c * 0.9)), 1, ROWS - 2) * COLS + c], DX * 1.1, new THREE.Vector3(0, 0, 0.018));
        rebuildClothIndices();
      },
      shred: () => {
        for (let c = 3; c < COLS - 2; c += 5) for (let r = 1; r < ROWS - 1; r++) tearAtLocalPoint(cur[r * COLS + c], DX * 0.95, new THREE.Vector3(0, 0, 0.02));
        rebuildClothIndices();
      },
      burn: (o = {}) => {
        if (o.action === 'extinguish') { isBurning = false; heatArr.fill(0); clothGeo.attributes.aHeat.needsUpdate = true; clothFlames.forEach((f) => { f.visible = false; }); burnLight.intensity = 0; return; }
        if (o.speed) burnSpeed = Number(o.speed) || 1.0;
        igniteAt(o.origin || 'bottom');
      },
      setTexture: async (tex, newCol) => {
        curTexName = String(tex || curTexName);
        const cObj = newCol ? parseHexColor(newCol, 0x7f1d1d) : baseCol;
        const td = await buildClothTextures(curTexName, cObj);
        clothMat.map = td.map; clothMat.bumpMap = td.bumpMap; clothMat.bumpScale = td.bumpScale;
        clothMat.alphaMap = td.alphaMap; clothMat.alphaTest = td.alphaTest;
        clothMat.roughness = td.roughness; clothMat.metalness = td.metalness;
        clothMat.sheen = td.sheen; clothMat.sheenRoughness = td.sheenRoughness;
        clothMat.sheenColor = cObj.clone().lerp(new THREE.Color(0xffffff), 0.45);
        clothMat.needsUpdate = true;
        return td.label || curTexName;
      }
    };

    const ent = APP.registerEntity(clothGroup, {
      name: opts.name || (isDrape ? 'Draped Cloth' : 'Interactive Cloth'), category: 'cloth', grabbable: false, physics: { enabled: false },
      onUpdate: function (dt) {
        const safeDt = Math.min(0.025, Math.max(0.004, dt || 0.016)), now = performance.now() * 0.001;
        if (clothMat.userData.shader) clothMat.userData.shader.uniforms.uBurnTime.value = now;
        flameShader.uniforms.uTime.value = now;
        const floorLocalY = -clothGroup.position.y + 0.02;

        const colliders = [];
        for (const other of APP.entities || []) {
          if (!other || other === ent || !other.group || other.category === 'world_fx' || other.category === 'board') continue;
          tmpBox.setFromObject(other.group);
          if (!tmpBox.isEmpty()) {
            const sz = new THREE.Vector3(); tmpBox.getSize(sz); tmpBox.getCenter(tmpSphereC); clothGroup.worldToLocal(tmpSphereC);
            colliders.push({ c: tmpSphereC.clone(), r: Math.max(sz.x, sz.y, sz.z) * 0.52 + 0.015 });
          }
        }

        let activeGrab = false;
        for (const [hk, h] of Object.entries(APP.hands || {})) {
          if (!h || !h.active) continue;
          const tip = h.isHand ? h.fingerTips?.index : h.pos;
          if (!tip) continue;
          tmpLocal.copy(tip); clothGroup.worldToLocal(tmpLocal);
          const last = prevHands.get(hk), speed = last ? tmpLocal.distanceTo(last) / safeDt : 0;
          prevHands.set(hk, tmpLocal.clone());

          if ((h.trigger || h.pinching) && grabbedIdx === -1) {
            let bestI = -1, bestD = 0.16;
            for (let i = (isDrape ? 0 : COLS); i < N_COUNT; i++) { if (burned[i]) continue; const d = cur[i].distanceTo(tmpLocal); if (d < bestD) { bestD = d; bestI = i; } }
            if (bestI !== -1) { grabbedIdx = bestI; grabbedHand = hk; }
          }
          if (grabbedHand === hk && (h.trigger || h.pinching) && grabbedIdx !== -1) {
            cur[grabbedIdx].lerp(tmpLocal, 0.72); prev[grabbedIdx].copy(cur[grabbedIdx]); activeGrab = true;
          }
          // Only tear on swipe if Tearable option is ON!
          if (tearable && speed > 1.35) tearAtLocalPoint(tmpLocal, 0.085, new THREE.Vector3(0, 0, 0.012));
        }
        if (!activeGrab) { grabbedIdx = -1; grabbedHand = null; }
        if (ptrGrabIdx !== -1 && ptrDragTarget) { cur[ptrGrabIdx].lerp(ptrDragTarget, 0.65); prev[ptrGrabIdx].copy(cur[ptrGrabIdx]); }

        // Burning + Live Volumetric Flames Riding Active Combustion Front!
        let activeFlamePtr = 0;
        const lightCenter = new THREE.Vector3();
        if (isBurning && curTexName !== 'chainmail') {
          for (let i = 0; i < springs.length; i++) {
            const s = springs[i]; if (s.broken || s.isBend) continue;
            const ha = heatArr[s.a], hb = heatArr[s.b];
            if (ha > 0.25 && !burned[s.b]) { const up = cur[s.b].y > cur[s.a].y ? 2.4 : 0.95; heatArr[s.b] = Math.min(1.0, hb + ha * safeDt * 0.82 * burnSpeed * up); }
            if (hb > 0.25 && !burned[s.a]) { const up = cur[s.a].y > cur[s.b].y ? 2.4 : 0.95; heatArr[s.a] = Math.min(1.0, ha + hb * safeDt * 0.82 * burnSpeed * up); }
          }
          const flameNodes = [];
          for (let i = 0; i < N_COUNT; i++) {
            if (burned[i]) continue;
            if (heatArr[i] > 0.34) {
              charArr[i] = Math.min(1.22, charArr[i] + safeDt * 0.40 * burnSpeed * (0.7 + Math.sin(i * 1.7) * 0.3));
              if (charArr[i] > 0.25 && charArr[i] < 0.98) {
                flameNodes.push(i);
                if (Math.random() < 0.06) emitParticle(cur[i], 0, (Math.random() - 0.5) * 0.14, 0.32 + Math.random() * 0.35, (Math.random() - 0.5) * 0.14, 1.15);
                if (Math.random() < 0.03) emitParticle(cur[i], 1, (Math.random() - 0.5) * 0.35, 0.55 + Math.random() * 0.6, (Math.random() - 0.5) * 0.35, 0.55);
              }
              if (charArr[i] >= 1.18) {
                burned[i] = 1; heatArr[i] = 0; indicesDirty = true;
                for (let k = 0; k < springs.length; k++) if (!springs[k].broken && (springs[k].a === i || springs[k].b === i)) springs[k].broken = true;
              }
            }
          }
          if (flameNodes.length > 0) {
            const stride = Math.max(1, Math.floor(flameNodes.length / FLAME_POOL));
            for (let k = 0; k < flameNodes.length && activeFlamePtr < FLAME_POOL; k += stride) {
              const ni = flameNodes[k];
              const fg = clothFlames[activeFlamePtr++];
              fg.visible = true;
              fg.position.copy(cur[ni]).add(new THREE.Vector3(0, -0.02, 0));
              const sc = 0.75 + Math.sin(now * 9.0 + k) * 0.25;
              fg.scale.set(sc, sc * 1.15, sc);
              lightCenter.add(cur[ni]);
            }
            lightCenter.divideScalar(activeFlamePtr);
            burnLight.position.copy(lightCenter).add(new THREE.Vector3(0, 0.1, 0.15));
            burnLight.intensity = 2.2 + Math.sin(now * 18.0) * 0.45;
          } else {
            burnLight.intensity = 0;
          }
          clothGeo.attributes.aHeat.needsUpdate = true;
          clothGeo.attributes.aChar.needsUpdate = true;
        }
        for (let i = activeFlamePtr; i < FLAME_POOL; i++) clothFlames[i].visible = false;

        const SUB = 2, subDt = safeDt / SUB, gravY = -8.2 * subDt * subDt;
        const windStrength = Number(opts.wind ?? (isDrape ? 0.28 : 0.85)), nAttr = clothGeo.attributes.normal;

        for (let step = 0; step < SUB; step++) {
          for (let i = 0; i < N_COUNT; i++) {
            if (pinned[i] || i === grabbedIdx || i === ptrGrabIdx || burned[i]) continue;
            const p = cur[i], pr = prev[i], vx = (p.x - pr.x) * 0.990, vy = (p.y - pr.y) * 0.990, vz = (p.z - pr.z) * 0.990;
            pr.copy(p);
            tmpWind.set(Math.cos(now * 1.5 + p.y * 2.2) * 0.00014, isBurning && heatArr[i] > 0.3 ? 0.00015 : 0.0, (Math.sin(now * 2.1 + p.x * 2.8 + p.y * 1.6) * 0.55 + 0.38) * 0.00038).multiplyScalar(windStrength);
            if (nAttr) { tmpNorm.fromBufferAttribute(nAttr, i); const aero = Math.abs(tmpNorm.dot(tmpWind)); p.x += vx + tmpWind.x * (0.4 + aero * 450.0); p.y += vy + gravY + tmpWind.y; p.z += vz + tmpWind.z * (0.4 + aero * 450.0); }
            else { p.x += vx + tmpWind.x; p.y += vy + gravY + tmpWind.y; p.z += vz + tmpWind.z; }
            if (p.y < floorLocalY) p.y = floorLocalY;
          }
          for (let iter = 0; iter < 5; iter++) {
            for (let i = 0; i < springs.length; i++) {
              const s = springs[i]; if (s.broken) continue;
              const pa = cur[s.a], pb = cur[s.b]; tmpDelta.subVectors(pb, pa); const d = tmpDelta.length();
              if (d < 1e-6) continue;
              if (tearable && !s.isBend && iter === 0 && step === 0 && d > s.tearDist) {
                s.broken = true; indicesDirty = true; tearArr[s.a] = 1.0; tearArr[s.b] = 1.0; clothGeo.attributes.aTear.needsUpdate = true;
                emitParticle(pa, 2, (Math.random() - 0.5) * 0.6, 0.3, (Math.random() - 0.5) * 0.6, 0.45); continue;
              }
              const diff = ((d - s.rest) / d) * s.stiff;
              const pinA = pinned[s.a] || s.a === grabbedIdx || s.a === ptrGrabIdx, pinB = pinned[s.b] || s.b === grabbedIdx || s.b === ptrGrabIdx;
              if (pinA && pinB) continue;
              if (pinA) pb.addScaledVector(tmpDelta, -diff); else if (pinB) pa.addScaledVector(tmpDelta, diff);
              else { pa.addScaledVector(tmpDelta, diff * 0.5); pb.addScaledVector(tmpDelta, -diff * 0.5); }
            }
            for (let ci = 0; ci < colliders.length; ci++) {
              const col = colliders[ci], rSq = col.r * col.r;
              for (let i = 0; i < N_COUNT; i++) {
                if (pinned[i] || burned[i]) continue;
                tmpDelta.subVectors(cur[i], col.c); const dSq = tmpDelta.lengthSq();
                if (dSq < rSq && dSq > 1e-6) cur[i].copy(col.c).addScaledVector(tmpDelta, col.r / Math.sqrt(dSq));
              }
            }
          }
        }

        if (indicesDirty) rebuildClothIndices();
        for (let i = 0; i < N_COUNT; i++) { posArr[i * 3] = cur[i].x; posArr[i * 3 + 1] = cur[i].y; posArr[i * 3 + 2] = cur[i].z; }
        clothGeo.attributes.position.needsUpdate = true; clothGeo.computeVertexNormals();

        for (let i = 0; i < SMOKE_MAX; i++) {
          if (smLife[i] <= 0) { smPos[i * 3 + 1] = -99; continue; }
          smLife[i] -= safeDt;
          if (smType[i] < 0.5) { smVel[i].x += Math.sin(now * 3.0 + i) * 0.15 * safeDt; smVel[i].y += 0.24 * safeDt; }
          else if (smType[i] < 1.5) smVel[i].y += 0.9 * safeDt;
          else smVel[i].y -= 3.5 * safeDt;
          smPos[i * 3] += smVel[i].x * safeDt; smPos[i * 3 + 1] += smVel[i].y * safeDt; smPos[i * 3 + 2] += smVel[i].z * safeDt;
        }
        smGeo.attributes.position.needsUpdate = true; smGeo.attributes.aLife.needsUpdate = true; smGeo.attributes.aType.needsUpdate = true;
      }
    });

    clothApi.entity = ent; APP._activeCloth = clothApi; APP._activeClothEnt = ent;
    syncSimUI();
    APP.lastEntityId = ent.id; APP.emit?.('entity:created', ent);
    return { ok: true, id: ent.id, name: ent.name, message: `Spawned ${texData.label || curTexName} cloth (Tearable: ${tearable ? 'ON' : 'OFF'}).` };
  }

  // ====================================================================
  // 4. DISINTEGRATION ENGINE
  // ====================================================================
  function createDisintegration(opts = {}) {
    const q = String(opts.target || '').trim();
    let target = q ? APP.findEntity?.(q) : null;
    if (!target || target.category === 'world_fx' || target.category === 'board') {
      const candidates = (APP.entities || []).filter((e) => e && e.group?.parent && e.category !== 'world_fx' && e.category !== 'board' && !e.isPen);
      target = candidates[candidates.length - 1] || null;
    }
    if (!target) {
      const artGroup = new THREE.Group();
      const ped = new THREE.Mesh(new THREE.CylinderGeometry(0.18, 0.22, 0.28, 16), new THREE.MeshStandardMaterial({ color: 0x334155, roughness: 0.55, metalness: 0.45 }));
      ped.position.y = 0.14;
      const core = new THREE.Mesh(new THREE.OctahedronGeometry(0.22, 2), new THREE.MeshStandardMaterial({ color: 0x38bdf8, emissive: 0x0ea5e9, emissiveIntensity: 0.4, roughness: 0.25, metalness: 0.65 }));
      core.position.y = 0.50;
      artGroup.add(ped, core); artGroup.position.copy(resolvePos(null, 1.15));
      target = APP.registerEntity(artGroup, { name: 'Artifact Monolith', category: 'artifact' });
    }

    const edgeCol = parseHexColor(opts.color, 0xff7b14), dur = Math.max(1.2, Math.min(6.0, Number(opts.duration) || 2.6));
    target.group.updateMatrixWorld(true);
    const box = new THREE.Box3().setFromObject(target.group), minY = isFinite(box.min.y) ? box.min.y : 0, maxY = isFinite(box.max.y) ? Math.max(minY + 0.1, box.max.y) : 1.0;

    const samplePos = [], sampleCol = [], vTmp = new THREE.Vector3();
    target.group.traverse((c) => {
      if (!c.isMesh || !c.geometry?.attributes?.position) return;
      const posA = c.geometry.attributes.position, colA = c.geometry.attributes.color, mat = Array.isArray(c.material) ? c.material[0] : c.material, baseCol = mat?.color ? mat.color : new THREE.Color(0x94a3b8);
      const step = Math.max(1, Math.floor(posA.count / 450));
      for (let i = 0; i < posA.count; i += step) {
        vTmp.fromBufferAttribute(posA, i).applyMatrix4(c.matrixWorld);
        samplePos.push(vTmp.x, vTmp.y, vTmp.z);
        if (colA) sampleCol.push(colA.getX(i), colA.getY(i), colA.getZ(i)); else sampleCol.push(baseCol.r, baseCol.g, baseCol.b);
      }
    });

    const P_COUNT = Math.min(1800, Math.max(600, samplePos.length / 3)), pPos = new Float32Array(P_COUNT * 3), pCol = new Float32Array(P_COUNT * 3), pThresh = new Float32Array(P_COUNT);
    const pVel = new Float32Array(P_COUNT * 3), pDetached = new Uint8Array(P_COUNT), pLife = new Float32Array(P_COUNT), pActive = new Float32Array(P_COUNT), srcLen = Math.max(1, samplePos.length / 3);
    for (let i = 0; i < P_COUNT; i++) {
      const si = (i % srcLen) * 3;
      pPos[i * 3] = samplePos[si] + (Math.random() - 0.5) * 0.015; pPos[i * 3 + 1] = samplePos[si + 1] + (Math.random() - 0.5) * 0.015; pPos[i * 3 + 2] = samplePos[si + 2] + (Math.random() - 0.5) * 0.015;
      pCol[i * 3] = sampleCol[si]; pCol[i * 3 + 1] = sampleCol[si + 1]; pCol[i * 3 + 2] = sampleCol[si + 2];
      const normY = THREE.MathUtils.clamp((pPos[i * 3 + 1] - minY) / (maxY - minY), 0, 1);
      pThresh[i] = THREE.MathUtils.clamp((opts.direction === 'top_down' ? 1 - normY : normY) * 0.72 + Math.random() * 0.28, 0.01, 0.99);
      pLife[i] = 1.0; pVel[i * 3] = (Math.random() - 0.3) * 0.35 + 0.18; pVel[i * 3 + 1] = 0.24 + Math.random() * 0.45; pVel[i * 3 + 2] = (Math.random() - 0.5) * 0.35;
    }

    const uProg = { value: 0.0 }, topDownFlag = { value: opts.direction === 'top_down' ? 1.0 : 0.0 };
    target.group.traverse((c) => {
      if (!c.isMesh || !c.material) return;
      const mats = Array.isArray(c.material) ? c.material : [c.material];
      c.material = mats.map((m) => {
        const cl = m.clone(); cl.side = THREE.DoubleSide; cl.customProgramCacheKey = () => 'nxr_dis_v5';
        cl.onBeforeCompile = function (shader) {
          shader.uniforms.uDisProg = uProg; shader.uniforms.uMinY = { value: minY }; shader.uniforms.uSpanY = { value: maxY - minY }; shader.uniforms.uTopDown = topDownFlag; shader.uniforms.uEdgeCol = { value: edgeCol };
          shader.vertexShader = shader.vertexShader.replace('#include <common>', '#include <common>\nvarying vec3 vDisWPos;').replace('#include <project_vertex>', '#include <project_vertex>\nvDisWPos = (modelMatrix * vec4(transformed, 1.0)).xyz;');
          const outAnchor = shader.fragmentShader.includes('#include <opaque_fragment>') ? '#include <opaque_fragment>' : '#include <output_fragment>';
          shader.fragmentShader = shader.fragmentShader.replace('#include <common>', '#include <common>\nuniform float uDisProg; uniform float uMinY; uniform float uSpanY; uniform float uTopDown; uniform vec3 uEdgeCol;\nvarying vec3 vDisWPos;\nfloat disHash(vec3 p){p=fract(p*0.3183+0.1);p*=17.0;return fract(p.x*p.y*p.z*(p.x+p.y+p.z));}').replace(outAnchor, `float disN=disHash(floor(vDisWPos*46.0));float disH=clamp((vDisWPos.y-uMinY)/max(0.05,uSpanY),0.0,1.0);if(uTopDown>0.5)disH=1.0-disH;float disVal=disH*0.72+disN*0.28;if(disVal<uDisProg)discard;float rim=1.0-smoothstep(uDisProg,uDisProg+0.08,disVal);outgoingLight=mix(outgoingLight,uEdgeCol*3.4,rim);\n${outAnchor}`);
        };
        cl.needsUpdate = true; return cl;
      });
      if (!Array.isArray(c.material)) c.material = c.material[0];
    });

    const dustGroup = new THREE.Group(), pGeo = new THREE.BufferGeometry();
    pGeo.setAttribute('position', new THREE.BufferAttribute(pPos, 3)); pGeo.setAttribute('aCol', new THREE.BufferAttribute(pCol, 3));
    pGeo.setAttribute('aLife', new THREE.BufferAttribute(pLife, 1)); pGeo.setAttribute('aActive', new THREE.BufferAttribute(pActive, 1));
    const dustPts = new THREE.Points(pGeo, new THREE.ShaderMaterial({
      transparent: true, depthWrite: false, blending: THREE.AdditiveBlending,
      uniforms: { uSize: { value: 20.0 * (window.devicePixelRatio || 1) }, uEdgeCol: { value: edgeCol } },
      vertexShader: 'attribute vec3 aCol; attribute float aLife; attribute float aActive; uniform float uSize; uniform vec3 uEdgeCol; varying vec3 vCol; varying float vAlpha; void main(){ vCol=mix(uEdgeCol,aCol,smoothstep(1.0,0.5,aLife)); vAlpha=aActive*clamp(aLife,0.0,1.0); vec4 mv=modelViewMatrix*vec4(position,1.0); gl_PointSize=aActive>0.5?clamp((uSize/max(0.3,-mv.z))*aLife,1.5,9.0):0.0; gl_Position=projectionMatrix*mv; }',
      fragmentShader: 'varying vec3 vCol; varying float vAlpha; void main(){ if(vAlpha<0.01)discard; float d=length(gl_PointCoord-0.5)*2.0; if(d>1.0)discard; gl_FragColor=vec4(vCol,pow(1.0-d,1.8)*vAlpha); }'
    }));
    dustPts.frustumCulled = false; dustGroup.add(dustPts); APP.scene.add(dustGroup);

    let meshRemoved = false, elapsed = 0;
    const dustEnt = APP.registerEntity(dustGroup, {
      name: 'Disintegration Dust', category: 'world_fx', grabbable: false, physics: { enabled: false },
      onUpdate: function (dt) {
        const delta = Math.min(0.04, dt || 0.016); elapsed += delta; uProg.value = Math.min(1.08, elapsed / dur);
        let anyAlive = false;
        for (let i = 0; i < P_COUNT; i++) {
          if (!pDetached[i]) { if (uProg.value >= pThresh[i]) { pDetached[i] = 1; pActive[i] = 1.0; } anyAlive = true; }
          else if (pLife[i] > 0) {
            anyAlive = true; const i3 = i * 3;
            pVel[i3] += Math.sin(pPos[i3 + 1] * 5.0 + elapsed * 3.0) * 0.55 * delta; pVel[i3 + 2] += Math.cos(pPos[i3] * 5.0 + elapsed * 2.5) * 0.55 * delta;
            pPos[i3] += pVel[i3] * delta; pPos[i3 + 1] += pVel[i3 + 1] * delta; pPos[i3 + 2] += pVel[i3 + 2] * delta; pLife[i] -= delta * 0.48;
          }
        }
        if (!meshRemoved && uProg.value >= 1.05) { meshRemoved = true; APP.removeEntity?.(target, false); }
