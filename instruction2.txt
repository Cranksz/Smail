mat.color : new THREE.Color(0xffffff); const hsl = {}; col.getHSL(hsl);

    (mat.transparent && mat.opacity < 0.88 && mat.opacity > 0.08) ||
    (mat.transmission && mat.transmission > 0.2)) {
  return 'glass';
}
if (/\b(metal|steel|iron|gold|silver|chrome|brass|copper|bronze|armor|blade|sword|gun|robot|chassis|mech|coin)\b/i.test(label) ||
    (mat.metalness !== undefined && mat.metalness >= 0.58)) {
  return 'metal';

  return 'sss';

  return 'rough_organic';

// ==================================================================== // 5.
GEOMETRY OCCLUSION (CORNERS, GAPS, CONTACTS) & EDGE WEAR BAKER //
==================================================================== function
bakeCreviceAndEdgeAttributes(group) { group.updateMatrixWorld(true); const
invGroup = group.matrixWorld.clone().invert(); const meshes = []; const
meshBoxes = [];

group.traverse((c) => {
  if (c.isMesh && c.geometry && !c.userData?.nxrDecal) {
    if (!c.geometry.attributes.normal) c.geometry.computeVertexNormals();
    c.geometry.computeBoundingBox();
    const boxG = c.geometry.boundingBox.clone();
    const relM = invGroup.clone().multiply(c.matrixWorld);
    boxG.applyMatrix4(relM);
    meshes.push({ mesh: c, relM, boxG });
    meshBoxes.push(boxG);
  }
});

const vPos = new THREE.Vector3();
const vNorm = new THREE.Vector3();
const closest = new THREE.Vector3();

for (let mi = 0; mi < meshes.length; mi++) {
  const { mesh, relM, boxG } = meshes[mi];
  const geo = mesh.geometry;
  const posAttr = geo.attributes.position;
  const normAttr = geo.attributes.normal;
  if (!posAttr) continue;

  const count = posAttr.count;
  const crevArr = new Float32Array(count);
  const edgeArr = new Float32Array(count);
  const boxCenter = new THREE.Vector3();
  boxG.getCenter(boxCenter);
  const boxSize = new THREE.Vector3();
  boxG.getSize(boxSize);
  const minYGroup = meshes.reduce((m, x) => Math.min(m, x.boxG.min.y), Infinity);

  const stride = count > 9000 ? 3 : 1;

  for (let i = 0; i < count; i++) {
    if (stride > 1 && i % stride !== 0) {
      crevArr[i] = crevArr[i - 1] || 0;
      edgeArr[i] = edgeArr[i - 1] || 0;
      continue;
    }
    vPos.fromBufferAttribute(posAttr, i).applyMatrix4(relM);
    if (normAttr) vNorm.fromBufferAttribute(normAttr, i).transformDirection(relM).normalize();
    else vNorm.set(0, 1, 0);

    // 1. Contact occlusion near sibling meshes (seams, joints, gaps)
    let occ = 0;
    for (let oi = 0; oi < meshBoxes.length; oi++) {
      if (oi === mi) continue;
      const ob = meshBoxes[oi];
      ob.clampPoint(vPos, closest);
      const d = vPos.distanceTo(closest);
      if (d < 0.085) {
        occ = Math.max(occ, (1.0 - d / 0.085) * 0.92);
      }
    }

    // 2. Ground contact occlusion near bottom of entity
    const floorDist = vPos.y - minYGroup;
    if (floorDist < 0.065) {
      occ = Math.max(occ, (1.0 - Math.max(0, floorDist) / 0.065) * 0.55);
    }

    // 3. Local concavity vs outer exposed edge wear
    const outDir = vPos.clone().sub(boxCenter);
    const radLen = outDir.length() || 1;
    outDir.divideScalar(radLen);
    const alignment = vNorm.dot(outDir);
    if (alignment < 0.15) occ = Math.max(occ, (0.15 - alignment) * 0.55);

    // Exposed corners & high-normal divergence => Edge Wear
    const nx = Math.abs(vPos.x - boxCenter.x) / Math.max(0.01, boxSize.x * 0.5);
    const ny = Math.abs(vPos.y - boxCenter.y) / Math.max(0.01, boxSize.y * 0.5);
    const nz = Math.abs(vPos.z - boxCenter.z) / Math.max(0.01, boxSize.z * 0.5);
    const cornerScore = THREE.MathUtils.clamp((nx + ny + nz - 1.45) * 0.72, 0, 1);

    crevArr[i] = THREE.MathUtils.clamp(occ, 0, 1);
    edgeArr[i] = cornerScore;
  }

  geo.setAttribute('aNxrCrevice', new THREE.BufferAttribute(crevArr, 1));
  geo.setAttribute('aNxrEdge', new THREE.BufferAttribute(edgeArr, 1));
}

}

  const t = i / segs;
  const r = THREE.MathUtils.lerp(rStart, rEnd, t) * (1.0 + Math.sin(t * Math.PI) * 0.14);
  const p = pts[i], N = frames.normals[i], B = frames.binormals[i];

    const nx = cos * N.x + sin * B.x;
    const ny = cos * N.y + sin * B.y;
    const nz = cos * N.z + sin * B.z;
    positions.push(p.x + r * nx, p.y + r * ny, p.z + r * nz);
    normals.push(nx, ny, nz);
    uvs.push(j / radSegs, t);
  }
}

  for (let j = 0; j < radSegs; j++) {
    const idxA = i * (radSegs + 1) + j;
    const idxB = (i + 1) * (radSegs + 1) + j;
    const idxC = (i + 1) * (radSegs + 1) + (j + 1);
    const idxD = i * (radSegs + 1) + (j + 1);
    indices.push(idxA, idxB, idxD, idxB, idxC, idxD);
  }
}

geo.setAttribute('normal', new THREE.Float32BufferAttribute(normals, 3));
geo.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
