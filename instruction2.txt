(function () {
  const APP = window.APP || (window.APP = {});
  const THREE = APP.THREE;
  if (!THREE || !APP.scene) return;

  function parseHexColor(col, fallbackHex) {
    if (typeof col === 'number' && isFinite(col)) return new THREE.Color(col);
    if (typeof col === 'string' && col.trim()) {
      try { return new THREE.Color(col.trim()); } catch (_) {}
    }
    return new THREE.Color(fallbackHex);
  }

  function resolvePos(posOpt, distDefault = 1.15) {
    if (Array.isArray(posOpt) && posOpt.length === 3) {
      return new THREE.Vector3(Number(posOpt[0]) || 0, Number(posOpt[1]) || 0, Number(posOpt[2]) || 0);
    }
    return APP.getSpawnPosition ? APP.getSpawnPosition(distDefault, true) : new THREE.Vector3(0, 0, -distDefault);
  }

  function getEntityBounds(ent) {
    if (!ent || !ent.group) return null;
    ent.group.updateMatrixWorld(true);
    const box = new THREE.Box3().setFromObject(ent.group);
    if (box.isEmpty()) return null;
    const size = new THREE.Vector3(), center = new THREE.Vector3();
    box.getSize(size); box.getCenter(center);
    return { box, size, center };
  }

  // ====================================================================
  // 1. HOUDINI-GRADE FLUID & VISCOUS ENGINE
  //    (Water, Honey, Lava, Slime | Pool, Pouring, Splashing, Spreading)
  // ====================================================================
  function createWaterSimulation(opts = {}) {
    const mode = String(opts.mode || '').toLowerCase();
    const fluidType = String(opts.fluid || opts.type || (mode === 'honey' ? 'honey' : 'water')).toLowerCase();
    const attachEnt = opts.attach_to ? APP.findEntity?.(opts.attach_to) : null;
    const attachInfo = getEntityBounds(attachEnt);

    const isHoney = fluidType === 'honey' || fluidType === 'syrup' || fluidType === 'amber';
    const isLava = fluidType === 'lava' || fluidType === 'magma';
    const isSlime = fluidType === 'slime';
    const viscosity = Number(opts.viscosity ?? (isHoney ? 0.88 : isLava ? 0.92 : isSlime ? 0.72 : 0.05));

    const isSpreading = mode === 'spreading' || mode === 'spill' || mode === 'puddle' || mode === 'flood' || !!opts.spreading;
    const isPouring = !!opts.pouring || mode === 'pouring' || !!opts.pour_from;
    const isWaterfall = !!opts.waterfall || mode === 'waterfall';
    const isFountain = !!opts.fountain || mode === 'fountain' || mode === 'splashing' || !!opts.splashing;

    const container = opts.container
      ? String(opts.container).toLowerCase()
      : (attachInfo || isSpreading ? 'none' : (mode === 'bowl' || mode === 'cup' ? mode : 'basin'));

    const isGlassContainer = container.includes('glass');
    const isCup = container.includes('cup');
    const isBowl = container.includes('bowl') || isCup;

    const defaultFluidCol = isHoney ? 0xd97706 : isLava ? 0xff4500 : isSlime ? 0x22c55e : 0x0ea5e9;
    const fluidColor = parseHexColor(opts.color, defaultFluidCol);
    const deepColor = isHoney ? new THREE.Color(0x78350f)
      : isLava ? new THREE.Color(0x450a0a)
      : isSlime ? new THREE.Color(0x14532d)
      : fluidColor.clone().multiplyScalar(0.18);
    const glow = opts.glow !== undefined ? !!opts.glow : isLava;

    const waterGroup = new THREE.Group();
    const labelFluid = fluidType.charAt(0).toUpperCase() + fluidType.slice(1);
    waterGroup.name = attachEnt ? `${labelFluid}_in_${attachEnt.name}`
      : isSpreading ? `${labelFluid}_Spill_Sim`
      : isWaterfall ? `${labelFluid}_Waterfall_Sim`
      : isFountain ? `${labelFluid}_Splash_Sim`
      : isPouring ? `${labelFluid}_Pour_Sim`
      : `${labelFluid}_Pool_Sim`;

    const radius = Number(opts.radius) || (
      attachInfo ? Math.max(0.10, Math.min(attachInfo.size.x, attachInfo.size.z) * 0.40)
      : isSpreading ? 1.25
      : isCup ? 0.11 : isBowl ? 0.28 : 0.98
    );
    const vesselH = Number(opts.height) || (
      attachInfo ? Math.max(0.12, attachInfo.size.y * 0.85)
      : isSpreading ? 0.08
      : isCup ? 0.24 : isBowl ? 0.22 : 0.20
    );
    const holdable = opts.holdable !== undefined ? !!opts.holdable : (isBowl && !isSpreading);

    let innerProfileFn = () => radius * 0.96;
    if (isBowl && container !== 'none') {
      const pts = [], steps = 16, wallThick = Math.max(0.012, radius * 0.06), baseR = isCup ? radius * 0.65 : radius * 0.42;
      pts.push(new THREE.Vector2(0.001, 0), new THREE.Vector2(baseR, 0));
      for (let i = 1; i <= steps; i++) {
        const t = i / steps;
        pts.push(new THREE.Vector2(THREE.MathUtils.lerp(baseR, radius, isCup ? Math.pow(t, 0.55) : Math.sqrt(t)), t * vesselH));
      }
      pts.push(new THREE.Vector2(radius - wallThick * 0.45, vesselH + wallThick * 0.35), new THREE.Vector2(radius - wallThick, vesselH));
      for (let i = steps - 1; i >= 1; i--) {
        const t = i / steps;
        const r = Math.max(0.01, THREE.MathUtils.lerp(baseR, radius, isCup ? Math.pow(t, 0.55) : Math.sqrt(t)) - wallThick);
        pts.push(new THREE.Vector2(r, Math.max(wallThick, t * vesselH)));
      }
      pts.push(new THREE.Vector2(0.001, wallThick));

      innerProfileFn = (y) => {
        const t = THREE.MathUtils.clamp(y / vesselH, 0.05, 0.98);
        return Math.max(0.02, THREE.MathUtils.lerp(baseR, radius, isCup ? Math.pow(t, 0.55) : Math.sqrt(t)) - wallThick * 1.08);
      };

      const vesselMat = isGlassContainer
        ? new THREE.MeshPhysicalMaterial({ color: 0xe6f7ff, transmission: 0.92, ior: 1.51, thickness: 0.14, roughness: 0.06, clearcoat: 1.0, side: THREE.DoubleSide })
        : new THREE.MeshStandardMaterial({ color: parseHexColor(opts.container_color, 0x334155), roughness: 0.32, metalness: 0.18, side: THREE.DoubleSide });

      const vesselMesh = new THREE.Mesh(new THREE.LatheGeometry(pts, 40), vesselMat);
      vesselMesh.castShadow = true; vesselMesh.receiveShadow = true;
      waterGroup.add(vesselMesh);
    } else if (container === 'basin') {
      const size = radius * 2;
      const stoneMat = new THREE.MeshStandardMaterial({ color: 0x1e293b, roughness: 0.78, metalness: 0.15 });
      const bed = new THREE.Mesh(new THREE.BoxGeometry(size + 0.14, 0.024, size + 0.14), stoneMat);
      bed.position.y = 0.012; bed.receiveShadow = true;
      waterGroup.add(bed);
      const wt = 0.065, half = radius + wt * 0.5;
      const wN = new THREE.Mesh(new THREE.BoxGeometry(size + wt * 2, vesselH, wt), stoneMat);
      wN.position.set(0, vesselH * 0.5, -half);
      const wS = wN.clone(); wS.position.z = half;
      const wE = new THREE.Mesh(new THREE.BoxGeometry(wt, vesselH, size), stoneMat);
      wE.position.set(half, vesselH * 0.5, 0);
      const wW = wE.clone(); wW.position.x = -half;
      waterGroup.add(wN, wS, wE, wW);

      if (!isHoney && !isLava) {
        const causticsMesh = new THREE.Mesh(
          new THREE.PlaneGeometry(size * 0.98, size * 0.98),
          new THREE.ShaderMaterial({
            transparent: true, depthWrite: false, blending: THREE.AdditiveBlending,
            uniforms: { uTime: { value: 0 }, uCol: { value: fluidColor } },
            vertexShader: 'varying vec2 vUv; void main(){ vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }',
            fragmentShader: [
              'uniform float uTime; uniform vec3 uCol; varying vec2 vUv;',
              'float caustic(vec2 uv, float t){',
              '  vec2 p = mod(uv * 6.28318, 6.28318) - 250.0;',
              '  vec2 i = p; float c = 1.0; float inten = 0.005;',
              '  for(int n=0; n<4; n++){',
              '    float nt = t * (1.0 - (3.5 / float(n+1)));',
              '    i = p + vec2(cos(nt - i.x) + sin(nt + i.y), sin(nt - i.y) + cos(nt + i.x));',
              '    c += 1.0 / length(vec2(p.x / (sin(i.x + nt)/inten), p.y / (cos(i.y + nt)/inten)));',
              '  }',
              '  c /= 4.0; c = 1.17 - pow(c, 1.4); return pow(abs(c), 8.0);',
              '}',
              'void main(){ float c = caustic(vUv * 1.4, uTime * 0.55); gl_FragColor = vec4((uCol * 0.6 + 0.5) * c, clamp(c * 0.65, 0.0, 0.7)); }'
            ].join('\n')
          })
        );
        causticsMesh.rotation.x = -Math.PI / 2;
        causticsMesh.position.y = 0.026;
        causticsMesh.userData.isCaustics = true;
        waterGroup.add(causticsMesh);
      }
    }

    if (attachInfo) {
      waterGroup.position.set(attachInfo.center.x, attachInfo.box.min.y, attachInfo.center.z);
    } else {
      waterGroup.position.copy(resolvePos(opts.position, isBowl ? 0.95 : 1.35));
    }
    APP.scene.add(waterGroup);

    // 1A. Fluid Shader (Houdini-inspired Fresnel + SSS + Caustic Shimmer + Foam FBM)
    const fluidUniforms = {
      uTime: { value: 0 },
      uDeep: { value: deepColor },
      uShallow: { value: fluidColor },
      uGlow: { value: glow ? 0.85 : (isHoney ? 0.22 : 0.0) },
      uViscosity: { value: viscosity },
      uIsSpreading: { value: isSpreading ? 1.0 : 0.0 }
    };

    const fluidMat = new THREE.ShaderMaterial({
      uniforms: fluidUniforms,
      transparent: true,
      side: THREE.DoubleSide,
      vertexShader: [
        'attribute float aFoam; attribute float aDepth;',
        'varying vec3 vWPos; varying vec3 vWNorm; varying float vFoam; varying float vDepth;',
        'void main() {',
        '  vFoam = aFoam; vDepth = aDepth;',
        '  vec4 wp = modelMatrix * vec4(position, 1.0);',
        '  vWPos = wp.xyz;',
        '  vWNorm = normalize(mat3(modelMatrix) * normal);',
        '  gl_Position = projectionMatrix * viewMatrix * wp;',
        '}'
      ].join('\n'),
      fragmentShader: [
        'uniform float uTime; uniform vec3 uDeep; uniform vec3 uShallow; uniform float uGlow; uniform float uViscosity; uniform float uIsSpreading;',
        'varying vec3 vWPos; varying vec3 vWNorm; varying float vFoam; varying float vDepth;',
        'float hash21(vec2 p){ p = fract(p * vec2(123.34, 456.21)); p += dot(p, p + 45.32); return fract(p.x * p.y); }',
        'float noise2(vec2 p){ vec2 i = floor(p); vec2 f = fract(p); f = f*f*(3.0-2.0*f); return mix(mix(hash21(i), hash21(i+vec2(1,0)), f.x), mix(hash21(i+vec2(0,1)), hash21(i+vec2(1,1)), f.x), f.y); }',
        'void main() {',
        '  if (uIsSpreading > 0.5 && vDepth < 0.0012) discard;',
        '  vec3 N = normalize(vWNorm);',
        '  float microT = uTime * mix(2.2, 0.35, uViscosity);',
        '  float n1 = noise2(vWPos.xz * 18.0 + vec2(microT, -microT * 0.7));',
        '  float n2 = noise2(vWPos.xz * 34.0 - vec2(microT * 1.1, microT * 0.5));',
        '  N = normalize(N + vec3((n1 - 0.5), 0.0, (n2 - 0.5)) * mix(0.14, 0.025, uViscosity));',
        '  vec3 V = normalize(cameraPosition - vWPos);',
        '  vec3 L = normalize(vec3(0.45, 0.88, 0.35));',
        '  float fres = pow(1.0 - max(0.0, dot(N, V)), mix(3.4, 2.2, uViscosity));',
        '  vec3 H = normalize(L + V);',
        '  float specPow = mix(180.0, 48.0, uViscosity * 0.5);',
        '  float spec = pow(max(0.0, dot(N, H)), specPow);',
        '  float sss = pow(clamp(dot(V, -L + N * 0.55), 0.0, 1.0), 3.0) * mix(0.55, 1.25, uViscosity);',
        '  vec3 skyRefl = mix(vec3(0.15, 0.25, 0.38), vec3(0.85, 0.93, 1.0), clamp(reflect(-V, N).y * 0.5 + 0.5, 0.0, 1.0));',
        '  float absorb = 1.0 - exp(-max(0.02, vDepth) * mix(6.5, 14.0, uViscosity));',
        '  vec3 bodyCol = mix(uShallow, uDeep, clamp(absorb * 0.82, 0.0, 1.0));',
        '  vec3 col = mix(bodyCol, skyRefl, fres * 0.78) + uShallow * (sss * 0.55 + uGlow);',
        '  float foamNoise = noise2(vWPos.xz * 24.0 + uTime * 0.8);',
        '  float foamMask = clamp((vFoam - 0.15) * (0.65 + foamNoise * 0.7), 0.0, 0.92) * (1.0 - uViscosity * 0.85);',
        '  col = mix(col, vec3(0.95, 0.99, 1.0), foamMask);',
        '  col += vec3(1.0, 0.98, 0.92) * spec * mix(1.85, 1.15, uViscosity);',
        '  float alpha = uIsSpreading > 0.5 ? smoothstep(0.0012, 0.012, vDepth) * mix(0.76, 0.95, absorb + fres * 0.3) : clamp(0.68 + fres * 0.28 + foamMask * 0.3 + spec * 0.25, 0.0, 0.97);',
        '  gl_FragColor = vec4(col, alpha);',
        '}'
      ].join('\n')
    });

    // 1B. Surface Geometry: Either Spreading 2D Flood Grid OR Polar Wave Surface
    let fillLevel = isPouring && !attachInfo ? 0.20 : (Number(opts.fill) || 0.68);
    const maxFill = 0.88;
    const minY = isBowl ? vesselH * 0.15 : (attachInfo ? attachInfo.size.y * 0.22 : 0.05);
    const maxY = isBowl ? vesselH * 0.88 : (attachInfo ? attachInfo.size.y * 0.74 : vesselH * 0.82);
    let curWaterY = isSpreading ? 0.012 : THREE.MathUtils.lerp(minY, maxY, fillLevel);
    let curR = innerProfileFn(curWaterY);

    let waterGeo, waterMesh, S_GRID = 52, S_SIZE = radius * 2.0, sH, sNextH, sVelH, sFoam, sDepth;
    let RINGS = isBowl || attachInfo ? 16 : 26, SEGS = isBowl || attachInfo ? 34 : 46, V_COUNT = 1 + RINGS * SEGS;
    let hCur, hPrev, hNext, wFoam, wDepth;

    if (isSpreading) {
      const N = S_GRID * S_GRID;
      sH = new Float32Array(N); sNextH = new Float32Array(N); sVelH = new Float32Array(N);
      sFoam = new Float32Array(N); sDepth = new Float32Array(N);
      waterGeo = new THREE.PlaneGeometry(S_SIZE, S_SIZE, S_GRID - 1, S_GRID - 1);
      waterGeo.rotateX(-Math.PI / 2);
      waterGeo.setAttribute('aFoam', new THREE.BufferAttribute(sFoam, 1));
      waterGeo.setAttribute('aDepth', new THREE.BufferAttribute(sDepth, 1));
      waterMesh = new THREE.Mesh(waterGeo, fluidMat);
      waterMesh.position.y = 0.002;
      waterGroup.add(waterMesh);
      const cx = Math.floor(S_GRID / 2), cz = Math.floor(S_GRID / 2);
      if (!isPouring) {
        for (let z = -4; z <= 4; z++) for (let x = -4; x <= 4; x++) {
          const d = Math.hypot(x, z);
          if (d <= 4.2) sH[(cz + z) * S_GRID + (cx + x)] = (1.0 - d / 4.5) * (isHoney ? 0.055 : 0.032);
        }
      }
    } else {
      const wPos = new Float32Array(V_COUNT * 3);
      const wNorm = new Float32Array(V_COUNT * 3);
      wFoam = new Float32Array(V_COUNT);
      wDepth = new Float32Array(V_COUNT);
      wDepth.fill(isBowl ? 0.14 : 0.25);
      const wIdx = [];
      wNorm[1] = 1;
      for (let r = 1; r <= RINGS; r++) {
        const rn = r / RINGS;
        for (let s = 0; s < SEGS; s++) {
          const ang = (s / SEGS) * Math.PI * 2;
          const vi = 1 + (r - 1) * SEGS + s;
          wPos[vi * 3] = Math.cos(ang) * rn;
          wPos[vi * 3 + 2] = Math.sin(ang) * rn;
          wNorm[vi * 3 + 1] = 1;
        }
      }
      for (let s = 0; s < SEGS; s++) wIdx.push(0, 1 + ((s + 1) % SEGS), 1 + s);
      for (let r = 1; r < RINGS; r++) {
        for (let s = 0; s < SEGS; s++) {
          const sn = (s + 1) % SEGS;
          const a = 1 + (r - 1) * SEGS + s, b = 1 + (r - 1) * SEGS + sn;
          const c = 1 + r * SEGS + s, d = 1 + r * SEGS + sn;
          wIdx.push(a, b, d, a, d, c);
        }
      }
      waterGeo = new THREE.BufferGeometry();
      waterGeo.setAttribute('position', new THREE.BufferAttribute(wPos, 3));
      waterGeo.setAttribute('normal', new THREE.BufferAttribute(wNorm, 3));
      waterGeo.setAttribute('aFoam', new THREE.BufferAttribute(wFoam, 1));
      waterGeo.setAttribute('aDepth', new THREE.BufferAttribute(wDepth, 1));
      waterGeo.setIndex(wIdx);
      waterMesh = new THREE.Mesh(waterGeo, fluidMat);
      waterMesh.position.y = curWaterY;
      waterMesh.scale.set(curR, 1, curR);
      waterGroup.add(waterMesh);
      hCur = new Float32Array(V_COUNT); hPrev = new Float32Array(V_COUNT); hNext = new Float32Array(V_COUNT);
    }

    // 1C. Dynamic Parabolic Pouring Stream (Zero-GC, Coiling for Honey, Movable Spout)
    const streamUniforms = {
      uTime: { value: 0 },
      uColor: { value: fluidColor },
      uDeep: { value: deepColor },
      uGlow: { value: glow ? 0.75 : 0.0 },
      uVisc: { value: viscosity }
    };
    const streamMat = new THREE.ShaderMaterial({
      uniforms: streamUniforms, transparent: true, side: THREE.DoubleSide,
      vertexShader: 'varying vec2 vUv; varying vec3 vN; void main(){ vUv = uv; vN = normalize(normalMatrix * normal); gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }',
      fragmentShader: [
        'uniform float uTime; uniform vec3 uColor; uniform vec3 uDeep; uniform float uGlow; uniform float uVisc;',
        'varying vec2 vUv; varying vec3 vN;',
        'void main(){',
        '  float speed = mix(16.0, 4.5, uVisc);',
        '  float streak = sin(vUv.x * 26.0 - uTime * speed) * 0.5 + 0.5;',
        '  float ripple = sin(vUv.y * 18.0 + vUv.x * 12.0 - uTime * speed * 0.8) * 0.5 + 0.5;',
        '  float fres = pow(1.0 - abs(vN.z), 2.2);',
        '  vec3 col = mix(uDeep, uColor, 0.45 + fres * 0.55) + uColor * uGlow;',
        '  float foam = smoothstep(0.62, 1.0, streak * ripple) * (1.0 - uVisc * 0.85);',
        '  col = mix(col, vec3(0.95, 0.99, 1.0), foam * 0.55) + vec3(1.0, 0.96, 0.88) * fres * 0.45;',
        '  gl_FragColor = vec4(col, mix(0.84, 0.95, uVisc) + foam * 0.1);',
        '}'
      ].join('\n')
    });

    let pourSourceEnt = null, pourCustomEnt = null, streamMesh = null;
    const STREAM_SEGS = 24, RAD_SEGS = 10;
    const streamGeo = new THREE.CylinderGeometry(1, 1, 1, RAD_SEGS, STREAM_SEGS, true);
    const streamBasePos = streamGeo.attributes.position.array.slice();
    let impactLocal = new THREE.Vector3(0, curWaterY, 0);

    if (isPouring) {
      let initSpoutWorld = waterGroup.position.clone().add(new THREE.Vector3(isSpreading ? 0.12 : radius * 0.72, isSpreading ? 0.65 : vesselH + 0.54, 0));
      if (typeof opts.pour_from === 'string') {
        pourCustomEnt = APP.findEntity?.(opts.pour_from);
        const info = getEntityBounds(pourCustomEnt);
        if (info) {
          initSpoutWorld.copy(info.center);
          initSpoutWorld.y = info.box.min.y + info.size.y * 0.35;
        }
      } else if (Array.isArray(opts.pour_from) && opts.pour_from.length === 3) {
        initSpoutWorld.set(opts.pour_from[0], opts.pour_from[1], opts.pour_from[2]);
      } else if (!attachInfo && opts.movable_source !== false) {
        const pitcherG = new THREE.Group();
        pitcherG.name = `${labelFluid}_Pitcher`;
        const pMat = isHoney
          ? new THREE.MeshPhysicalMaterial({ color: 0xfef3c7, transmission: 0.72, roughness: 0.14, ior: 1.5 })
          : new THREE.MeshStandardMaterial({ color: 0xb45309, metalness: 0.82, roughness: 0.24 });
        const body = new THREE.Mesh(new THREE.CylinderGeometry(0.05, 0.072, 0.17, 18), pMat);
        const spout = new THREE.Mesh(new THREE.ConeGeometry(0.024, 0.085, 14), pMat);
        spout.rotation.z = Math.PI * 0.38; spout.position.set(-0.072, 0.048, 0);
        const rim = new THREE.Mesh(new THREE.TorusGeometry(0.052, 0.007, 10, 20), new THREE.MeshStandardMaterial({ color: 0x38bdf8, emissive: fluidColor, emissiveIntensity: 0.45 }));
        rim.rotation.x = Math.PI / 2; rim.position.y = 0.082;
        pitcherG.add(body, spout, rim);
        pitcherG.position.copy(initSpoutWorld).add(new THREE.Vector3(0.09, 0.03, 0));
        pitcherG.rotation.z = 0.55;
        APP.scene.add(pitcherG);
        pourSourceEnt = APP.registerEntity(pitcherG, {
          name: `${labelFluid} Pitcher`,
          category: 'tool',
          grabbable: true,
          holdable: true,
          physics: { enabled: false },
          onUpdate: function () {
            if (!waterGroup.parent) APP.removeEntity?.(pourSourceEnt, false);
          }
        });
      }
      streamMesh = new THREE.Mesh(streamGeo, streamMat);
      streamMesh.frustumCulled = false;
      waterGroup.add(streamMesh);
    } else if (isWaterfall) {
      const wfW = radius * 1.05, wfH = 1.15;
      const wfGeo = new THREE.PlaneGeometry(wfW, wfH, 16, 20);
      const pAttr = wfGeo.attributes.position;
      for (let i = 0; i < pAttr.count; i++) {
        const yn = (pAttr.getY(i) + wfH * 0.5) / wfH;
        pAttr.setZ(i, Math.pow(1 - yn, 1.5) * 0.38);
      }
      wfGeo.computeVertexNormals();
      const wfMesh = new THREE.Mesh(wfGeo, new THREE.ShaderMaterial({
        uniforms: streamUniforms, transparent: true, side: THREE.DoubleSide,
        vertexShader: 'varying vec2 vUv; void main() { vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0); }',
        fragmentShader: [
          'uniform float uTime; uniform vec3 uColor; uniform vec3 uDeep; uniform float uGlow; uniform float uVisc; varying vec2 vUv;',
          'void main() {',
          '  float sp = mix(14.0, 4.0, uVisc);',
          '  float s1 = sin(vUv.y * 34.0 + uTime * sp) * 0.5 + 0.5;',
          '  float s2 = cos(vUv.x * 26.0 + vUv.y * 18.0 + uTime * sp * 0.8) * 0.5 + 0.5;',
          '  float foam = (smoothstep(0.54, 1.0, s1 * s2) + (1.0 - smoothstep(0.0, 0.24, vUv.y)) * 0.72) * (1.0 - uVisc * 0.8);',
          '  vec3 col = mix(uDeep, uColor, 0.55) + uColor * uGlow;',
          '  gl_FragColor = vec4(mix(col, vec3(0.95, 0.99, 1.0), clamp(foam, 0.0, 1.0)), 0.84);',
          '}'
        ].join('\n')
      }));
      wfMesh.position.set(0, curWaterY + wfH * 0.45, -radius * 0.55);
      waterGroup.add(wfMesh);
      impactLocal.set(0, curWaterY, -radius * 0.2);
    }

    // 1D. Multi-Tier Splash Droplets + Mist Spray
    const DROP_COUNT = isHoney ? 32 : 160;
    const dropPos = new Float32Array(DROP_COUNT * 3);
    const dropVel = Array.from({ length: DROP_COUNT }, () => new THREE.Vector3());
    const dropLife = new Float32Array(DROP_COUNT);
    const dropSize = new Float32Array(DROP_COUNT);
    const dropGeo = new THREE.BufferGeometry();
    dropGeo.setAttribute('position', new THREE.BufferAttribute(dropPos, 3));
    dropGeo.setAttribute('aSize', new THREE.BufferAttribute(dropSize, 1));
    dropGeo.setAttribute('aLife', new THREE.BufferAttribute(dropLife, 1));
    const dropPts = new THREE.Points(dropGeo, new THREE.ShaderMaterial({
      transparent: true, depthWrite: false,
      uniforms: { uCol: { value: fluidColor.clone().lerp(new THREE.Color(0xffffff), isHoney ? 0.15 : 0.65) }, uPx: { value: window.devicePixelRatio || 1 } },
      vertexShader: 'attribute float aSize; attribute float aLife; uniform float uPx; varying float vLife; void main(){ vLife=aLife; vec4 mv=modelViewMatrix*vec4(position,1.0); gl_PointSize=aLife>0.01?clamp((aSize*uPx)/max(0.25,-mv.z),1.5,14.0):0.0; gl_Position=projectionMatrix*mv; }',
      fragmentShader: 'uniform vec3 uCol; varying float vLife; void main(){ float d=length(gl_PointCoord-0.5)*2.0; if(d>1.0)discard; gl_FragColor=vec4(uCol, (1.0-d*d)*clamp(vLife*1.5,0.0,0.9)); }'
    }));
    dropPts.frustumCulled = false;
    waterGroup.add(dropPts);
    let nextDrop = 0;

    function emitSplash(lx, ly, lz, strength) {
      if (isHoney && strength < 0.8) return;
      const cnt = isHoney ? 2 : Math.min(12, Math.ceil(strength * 7));
      for (let i = 0; i < cnt; i++) {
        const idx = nextDrop; nextDrop = (nextDrop + 1) % DROP_COUNT;
        dropPos[idx * 3] = lx + (Math.random() - 0.5) * 0.05;
        dropPos[idx * 3 + 1] = ly + 0.015;
        dropPos[idx * 3 + 2] = lz + (Math.random() - 0.5) * 0.05;
        const isMist = !isHoney && Math.random() < 0.38;
        const spread = (isCup ? 0.28 : 0.72) * (isMist ? 1.4 : 1.0);
        dropVel[idx].set(
          (Math.random() - 0.5) * spread,
          (isMist ? 0.25 + Math.random() * 0.45 : 0.55 + Math.random() * 0.95) * strength * (isHoney ? 0.45 : 1.0),
          (Math.random() - 0.5) * spread
        );
        dropLife[idx] = isMist ? 0.65 + Math.random() * 0.45 : 0.42 + Math.random() * 0.28;
        dropSize[idx] = isMist ? 28.0 : (isCup ? 12.0 : 18.0);
      }
    }

    function disturbLocal(lx, lz, amp) {
      if (isSpreading) {
        const gx = Math.round((lx / S_SIZE + 0.5) * (S_GRID - 1));
        const gz = Math.round((lz / S_SIZE + 0.5) * (S_GRID - 1));
        if (gx >= 1 && gx < S_GRID - 1 && gz >= 1 && gz < S_GRID - 1) {
          sVelH[gz * S_GRID + gx] += amp * 1.4;
          sFoam[gz * S_GRID + gx] = Math.min(1.0, sFoam[gz * S_GRID + gx] + Math.abs(amp) * 18.0);
        }
        return;
      }
      const dist = Math.hypot(lx, lz);
      if (dist > curR * 0.94) return;
      const rIdx = Math.max(0, Math.min(RINGS - 1, Math.round((dist / Math.max(0.01, curR)) * RINGS)));
      if (rIdx === 0) { hCur[0] += amp; return; }
      let ang = Math.atan2(lz, lx); if (ang < 0) ang += Math.PI * 2;
      const sIdx = Math.round((ang / (Math.PI * 2)) * SEGS) % SEGS;
      hCur[1 + (rIdx - 1) * SEGS + sIdx] += amp;
    }

    const tmpLocal = new THREE.Vector3(), tmpStart = new THREE.Vector3(), tmpEnd = new THREE.Vector3();
    const tmpP0 = new THREE.Vector3(), tmpP1 = new THREE.Vector3(), tmpT = new THREE.Vector3(), tmpN = new THREE.Vector3(), tmpB = new THREE.Vector3();
    const prevPos = waterGroup.position.clone(), prevHands = new Map(), tmpBox = new THREE.Box3();
    let dropTick = 0;

    function updateParabolicStream(now) {
      if (!streamMesh) return;
      if (pourSourceEnt?.group?.parent) {
        tmpStart.set(-0.085, 0.045, 0);
        pourSourceEnt.group.localToWorld(tmpStart);
        waterGroup.worldToLocal(tmpStart);
      } else if (pourCustomEnt?.group?.parent) {
        const info = getEntityBounds(pourCustomEnt);
        if (info) {
          tmpStart.copy(info.center);
          tmpStart.y = info.box.min.y + info.size.y * 0.35;
          waterGroup.worldToLocal(tmpStart);
        }
      } else {
        tmpStart.set(isSpreading ? 0.12 : radius * 0.68, isSpreading ? 0.65 : vesselH + 0.50, 0);
      }

      const fallH = Math.max(0.08, tmpStart.y - curWaterY);
      const vx = isSpreading ? -0.08 : -tmpStart.x * 0.45;
      const vz = isSpreading ? 0.0 : -tmpStart.z * 0.45;
      const endX = tmpStart.x + vx * 0.32;
      const endZ = tmpStart.z + vz * 0.32;
      impactLocal.set(endX, curWaterY, endZ);

      const posAttr = streamGeo.attributes.position;
      const baseR = Math.min(0.024, radius * 0.09) * (isHoney ? 0.72 : 1.0);
      const coilAmp = isHoney ? 0.022 : 0.002;
      const coilFreq = isHoney ? 9.5 : 14.0;

      for (let iy = 0; iy <= STREAM_SEGS; iy++) {
        const t = iy / STREAM_SEGS;
        const invT = 1.0 - t;
        const cx = THREE.MathUtils.lerp(tmpStart.x, endX, Math.pow(t, 0.85)) + Math.sin(now * coilFreq - t * 6.0) * coilAmp * Math.pow(t, 2.2);
        const cz = THREE.MathUtils.lerp(tmpStart.z, endZ, Math.pow(t, 0.85)) + Math.cos(now * coilFreq - t * 6.0) * coilAmp * Math.pow(t, 2.2);
        const cy = tmpStart.y - fallH * Math.pow(t, 1.45);
        tmpP0.set(cx, cy, cz);

        const t2 = Math.min(1.0, t + 0.02);
        tmpP1.set(
          THREE.MathUtils.lerp(tmpStart.x, endX, Math.pow(t2, 0.85)),
          tmpStart.y - fallH * Math.pow(t2, 1.45),
          THREE.MathUtils.lerp(tmpStart.z, endZ, Math.pow(t2, 0.85))
        );
        tmpT.subVectors(tmpP1, tmpP0).normalize();
        if (tmpT.lengthSq() < 0.001) tmpT.set(0, -1, 0);
        tmpN.set(0, 0, 1).cross(tmpT);
        if (tmpN.lengthSq() < 0.001) tmpN.set(1, 0, 0).cross(tmpT);
        tmpN.normalize();
        tmpB.crossVectors(tmpT, tmpN).normalize();

        const neckScale = isHoney
          ? (0.95 - t * 0.35 + Math.pow(t, 4.0) * 0.55)
          : (1.0 - t * 0.32 + Math.sin(t * 22.0 - now * 16.0) * 0.08 * t);
        const r = Math.max(0.004, baseR * neckScale);

        for (let ir = 0; ir <= RAD_SEGS; ir++) {
          const idx = (iy * (RAD_SEGS + 1) + ir) * 3;
          const bx = streamBasePos[idx], bz = streamBasePos[idx + 2];
          posAttr.array[idx] = tmpP0.x + (tmpN.x * bx + tmpB.x * bz) * r;
          posAttr.array[idx + 1] = tmpP0.y + (tmpN.y * bx + tmpB.y * bz) * r;
          posAttr.array[idx + 2] = tmpP0.z + (tmpN.z * bx + tmpB.z * bz) * r;
        }
      }
      posAttr.needsUpdate = true;
      streamGeo.computeVertexNormals();
    }

    const ent = APP.registerEntity(waterGroup, {
      name: opts.name || waterGroup.name,
      category: 'world_fx',
      grabbable: holdable,
      holdable: holdable,
      physics: { enabled: holdable, gravity: holdable },
      onUpdate: function (dt) {
        const delta = Math.min(0.033, Math.max(0.004, dt || 0.016));
        const now = performance.now() * 0.001;
        fluidUniforms.uTime.value = now;
        streamUniforms.uTime.value = now;
        for (const c of waterGroup.children) {
          if (c.userData?.isCaustics && c.material?.uniforms?.uTime) c.material.uniforms.uTime.value = now;
        }

        if (isPouring) updateParabolicStream(now);

        if (isPouring || isWaterfall || isFountain) {
          if (isPouring && !isSpreading && fillLevel < maxFill && !attachInfo) {
            fillLevel = Math.min(maxFill, fillLevel + delta * (isHoney ? 0.032 : 0.065));
            curWaterY = THREE.MathUtils.lerp(minY, maxY, fillLevel);
            curR = innerProfileFn(curWaterY);
            waterMesh.position.y = curWaterY;
            waterMesh.scale.set(curR, 1, curR);
          }
          dropTick += delta;
          if (dropTick > (isHoney ? 0.09 : 0.045)) {
            dropTick = 0;
            if (isSpreading) {
              const gx = Math.round((impactLocal.x / S_SIZE + 0.5) * (S_GRID - 1));
              const gz = Math.round((impactLocal.z / S_SIZE + 0.5) * (S_GRID - 1));
              if (gx >= 2 && gx < S_GRID - 2 && gz >= 2 && gz < S_GRID - 2) {
                const addV = isHoney ? 0.0095 : 0.014;
                sH[gz * S_GRID + gx] = Math.min(0.16, sH[gz * S_GRID + gx] + addV);
                curWaterY = sH[gz * S_GRID + gx];
              }
            }
            disturbLocal(impactLocal.x, impactLocal.z, isFountain ? 0.026 : (isHoney ? 0.004 : -0.016));
            emitSplash(impactLocal.x, curWaterY, impactLocal.z, isFountain ? 1.55 : (isCup ? 0.45 : 0.85));
          }
        }

        if (!isSpreading && waterGroup.position.distanceTo(prevPos) > 0.002) {
          disturbLocal(-Math.sign(waterGroup.position.x - prevPos.x) * curR * 0.5, -Math.sign(waterGroup.position.z - prevPos.z) * curR * 0.5, isHoney ? -0.003 : -0.01);
          prevPos.copy(waterGroup.position);
        }

        for (const [hk, h] of Object.entries(APP.hands || {})) {
          if (!h || !h.active) continue;
          const hp = h.isHand ? h.fingerTips?.index : h.pos;
          if (!hp) continue;
          tmpLocal.copy(hp); waterGroup.worldToLocal(tmpLocal);
          const last = prevHands.get(hk), spd = last ? tmpLocal.distanceTo(last) / delta : 0;
          prevHands.set(hk, tmpLocal.clone());
          if (tmpLocal.y < curWaterY + 0.14 && tmpLocal.y > -0.05 && spd > 0.14) {
            if (isSpreading || Math.hypot(tmpLocal.x, tmpLocal.z) < curR) {
              disturbLocal(tmpLocal.x, tmpLocal.z, isHoney ? -0.006 : -0.018);
              emitSplash(tmpLocal.x, curWaterY, tmpLocal.z, 0.72);
            }
          }
        }

        if (isSpreading) {
          // 2D Shallow-Water / Viscous Flood Solver
          const spreadRate = isHoney ? 0.042 : (isLava ? 0.028 : 0.21);
          const minSurfTension = isHoney ? 0.0035 : 0.0001;
          const damp = isHoney ? 0.78 : 0.965;

          // Collect obstacles for puddle wrapping
          const obs = [];
          for (const other of APP.entities || []) {
            if (!other || other === ent || other === pourSourceEnt || other === pourCustomEnt || !other.group || other.category === 'world_fx' || other.category === 'board') continue;
            tmpBox.setFromObject(other.group);
            if (!tmpBox.isEmpty() && tmpBox.min.y < waterGroup.position.y + 0.15) {
              tmpBox.min.sub(waterGroup.position); tmpBox.max.sub(waterGroup.position);
              obs.push(tmpBox.clone());
            }
          }

          sNextH.set(sH);
          const dirs = [-1, 1, -S_GRID, S_GRID];
          for (let z = 1; z < S_GRID - 1; z++) {
            for (let x = 1; x < S_GRID - 1; x++) {
              const idx = z * S_GRID + x;
              const h0 = sH[idx];
              if (h0 <= 0.0004) { sFoam[idx] *= 0.9; continue; }
              let totalOut = 0;
              for (let d = 0; d < 4; d++) {
                const ni = idx + dirs[d];
                const diff = h0 - sH[ni];
                if (diff > minSurfTension) {
                  const nx = (x + (d === 1 ? 1 : d === 0 ? -1 : 0)) / (S_GRID - 1) - 0.5;
                  const nz = (z + (d === 3 ? 1 : d === 2 ? -1 : 0)) / (S_GRID - 1) - 0.5;
                  const wx = nx * S_SIZE, wz = nz * S_SIZE;
                  let blocked = false;
                  for (let b = 0; b < obs.length; b++) {
                    if (wx > obs[b].min.x && wx < obs[b].max.x && wz > obs[b].min.z && wz < obs[b].max.z) { blocked = true; break; }
                  }
                  if (!blocked) {
                    const flux = Math.min(h0 * 0.24, (diff - minSurfTension) * spreadRate);
                    sNextH[idx] -= flux;
                    sNextH[ni] += flux;
                    totalOut += flux;
                  }
                }
              }
              sFoam[idx] = THREE.MathUtils.lerp(sFoam[idx], Math.min(1.0, totalOut * 140.0), 0.25);
            }
          }
          for (let i = 0; i < sH.length; i++) {
            sVelH[i] = (sVelH[i] + (sNextH[i] - sH[i])) * damp;
            sH[i] = Math.max(0, sNextH[i] + sVelH[i] * 0.18);
            sDepth[i] = sH[i];
          }
          const posA = waterGeo.attributes.position;
          for (let i = 0; i < sH.length; i++) posA.setY(i, sH[i]);
          posA.needsUpdate = true;
          waterGeo.computeVertexNormals();
          waterGeo.attributes.aFoam.needsUpdate = true;
          waterGeo.attributes.aDepth.needsUpdate = true;
        } else {
          // Polar Wave Equation Solver (Damped heavily for Honey/Lava)
          const damp = isHoney ? 0.905 : isLava ? 0.86 : 0.985;
          const ampMax = isHoney ? 0.012 : (isBowl ? 0.025 : 0.07);
          let ring1Sum = 0;
          for (let s = 0; s < SEGS; s++) ring1Sum += hCur[1 + s];
          hNext[0] = THREE.MathUtils.clamp(((ring1Sum / SEGS) * 2.0 - hPrev[0]) * damp, -ampMax, ampMax);
          for (let r = 1; r < RINGS; r++) {
            for (let s = 0; s < SEGS; s++) {
              const vi = 1 + (r - 1) * SEGS + s, sn = (s + 1) % SEGS, sp = (s - 1 + SEGS) % SEGS;
              const vIn = r === 1 ? 0 : (1 + (r - 2) * SEGS + s), vOut = 1 + r * SEGS + s;
              const avg = (hCur[vIn] + hCur[vOut] + hCur[1 + (r - 1) * SEGS + sp] + hCur[1 + (r - 1) * SEGS + sn]) * 0.5;
              hNext[vi] = THREE.MathUtils.clamp((avg - hPrev[vi]) * damp, -ampMax, ampMax);
            }
          }
          const tmp = hPrev; hPrev = hCur; hCur = hNext; hNext = tmp;

          const posA = waterGeo.attributes.position, normA = waterGeo.attributes.normal;
          for (let i = 0; i < V_COUNT; i++) {
            const bulge = (isHoney && isPouring) ? Math.exp(-Math.pow(i / V_COUNT, 1.5) * 3.5) * 0.025 : 0.0;
            posA.setY(i, hCur[i] + bulge);
            wFoam[i] = Math.max(0, Math.abs(hCur[i]) * (isBowl ? 22.0 : 13.0) - 0.10);
            wDepth[i] = curWaterY + hCur[i];
          }
          for (let r = 1; r < RINGS; r++) {
            for (let s = 0; s < SEGS; s++) {
              const vi = 1 + (r - 1) * SEGS + s, vIn = r === 1 ? 0 : (1 + (r - 2) * SEGS + s), vOut = 1 + r * SEGS + s;
              const dh = (hCur[vIn] - hCur[vOut]) * 6.5, ang = (s / SEGS) * Math.PI * 2;
              const nx = Math.cos(ang) * dh, nz = Math.sin(ang) * dh, len = Math.hypot(nx, 1.0, nz) || 1;
              normA.setXYZ(vi, nx / len, 1.0 / len, nz / len);
            }
          }
          posA.needsUpdate = true; normA.needsUpdate = true;
          waterGeo.attributes.aFoam.needsUpdate = true;
          waterGeo.attributes.aDepth.needsUpdate = true;
        }

        for (let i = 0; i < DROP_COUNT; i++) {
          if (dropLife[i] <= 0) { dropPos[i * 3 + 1] = -99; continue; }
          dropLife[i] -= delta; dropVel[i].y -= 7.8 * delta;
          dropPos[i * 3] += dropVel[i].x * delta; dropPos[i * 3 + 1] += dropVel[i].y * delta; dropPos[i * 3 + 2] += dropVel[i].z * delta;
          if (dropPos[i * 3 + 1] < curWaterY) dropLife[i] = 0;
        }
        dropGeo.attributes.position.needsUpdate = true;
        dropGeo.attributes.aLife.needsUpdate = true;
      }
    });

    APP.lastEntityId = ent.id;
    APP.emit?.('entity:created', ent);
    return { ok: true, id: ent.id, name: ent.name, message: `Created ${ent.name}.` };
  }

  // ====================================================================
  // 2. ULTRA-FINE GRANULAR & IN-BETWEEN ENGINE (Sand, Snow, Wet Sand)
  //    + Grabbable VR Controller Emitter + Avalanche Heightfield
  // ====================================================================
  function createGranularSimulation(opts = {}) {
    const matType = String(opts.material || 'sand').toLowerCase();
    const isSnow = matType === 'snow';
    const isWetSand = matType === 'wet_sand' || matType === 'mud' || matType === 'clay';
    const isGold = matType === 'gold';
    const isGravel = matType === 'gravel';
    const isAsh = matType === 'ash';

    const defaultCol = isGold ? 0xfbbf24 : isSnow ? 0xf1f5f9 : isWetSand ? 0x855e3c : isAsh ? 0x475569 : isGravel ? 0x64748b : 0xd6b47c;
    const baseCol = parseHexColor(opts.color, defaultCol);
    const targetEnt = opts.pour_on_target ? APP.findEntity?.(opts.pour_on_target) : null;
    const fromEnt = opts.pour_from_entity ? APP.findEntity?.(opts.pour_from_entity) : null;
    const fromInfo = getEntityBounds(fromEnt);

    const sandGroup = new THREE.Group();
    sandGroup.name = `Granular_${matType}_Sim`;

    const SIZE = Number(opts.size) || 2.35;
    const GRID = 68;
    const CELL = SIZE / (GRID - 1);
    const slopeFactor = isSnow ? 1.28 : isWetSand ? 1.05 : isGravel ? 0.85 : isAsh ? 0.52 : 0.66;
    const MAX_SLOPE = CELL * slopeFactor;
    const pileH = new Float32Array(GRID * GRID);

    const pileGeo = new THREE.PlaneGeometry(SIZE, SIZE, GRID - 1, GRID - 1);
    pileGeo.rotateX(-Math.PI / 2);

    const pileMat = new THREE.MeshStandardMaterial({
      color: baseCol,
      roughness: isGold ? 0.25 : isWetSand ? 0.42 : isSnow ? 0.85 : 0.94,
      metalness: isGold ? 0.88 : 0.04
    });
    pileMat.customProgramCacheKey = () => `nxr_gran_${matType}`;
    pileMat.onBeforeCompile = (shader) => {
      shader.uniforms.uTime = { value: 0 };
      shader.uniforms.uIsSnow = { value: isSnow ? 1.0 : 0.0 };
      shader.uniforms.uIsWet = { value: isWetSand ? 1.0 : 0.0 };
      pileMat.userData.shader = shader;
      shader.vertexShader = shader.vertexShader
        .replace('#include <common>', '#include <common>\nvarying vec3 vGranWPos;')
        .replace('#include <begin_vertex>', '#include <begin_vertex>\nvGranWPos = (modelMatrix * vec4(position, 1.0)).xyz;');
      shader.fragmentShader = shader.fragmentShader
        .replace('#include <common>', [
          '#include <common>',
          'uniform float uTime; uniform float uIsSnow; uniform float uIsWet;',
          'varying vec3 vGranWPos;',
          'float gHash(vec2 p){ p = fract(p * vec2(234.34, 435.345)); p += dot(p, p + 34.23); return fract(p.x * p.y); }'
        ].join('\n'))
        .replace('#include <map_fragment>', [
          '#include <map_fragment>',
          'float micro = gHash(floor(vGranWPos.xz * 260.0));',
          'float ripple = sin(vGranWPos.x * 48.0 + vGranWPos.y * 22.0) * 0.03;',
          'if (uIsSnow > 0.5) {',
          '  diffuseColor.rgb = mix(diffuseColor.rgb * vec3(0.86, 0.92, 1.0), vec3(1.0), smoothstep(0.0, 0.08, vGranWPos.y) * 0.5 + micro * 0.12);',
          '} else {',
          '  diffuseColor.rgb *= (0.88 + micro * 0.24 + ripple);',
          '}'
        ].join('\n'))
        .replace('#include <emissivemap_fragment>', [
          '#include <emissivemap_fragment>',
          'vec3 Vg = normalize(cameraPosition - vGranWPos);',
          'float glint = step(0.988, gHash(floor(vGranWPos.xz * 320.0) + floor(Vg.xz * 12.0)));',
          'totalEmissiveRadiance += glint * (uIsSnow > 0.5 ? vec3(0.85, 0.95, 1.0) : diffuseColor.rgb * 0.9) * 0.75;'
        ].join('\n'));
    };

    const pileMesh = new THREE.Mesh(pileGeo, pileMat);
    pileMesh.receiveShadow = true; pileMesh.castShadow = true;
    sandGroup.add(pileMesh);

    // Ultra-fine Micro-Grain Particle Cloud (4,000 fine grains via single Points draw call)
    const GRAIN_COUNT = Math.min(6000, Math.max(1800, Number(opts.count) || 4000));
    const gPos = new Float32Array(GRAIN_COUNT * 3);
    const gVel = new Float32Array(GRAIN_COUNT * 3);
    const gCol = new Float32Array(GRAIN_COUNT * 3);
    const gSize = new Float32Array(GRAIN_COUNT);
    const gSeed = new Float32Array(GRAIN_COUNT);

    const basePos = fromInfo ? new THREE.Vector3(fromInfo.center.x, 0, fromInfo.center.z)
      : targetEnt?.group ? new THREE.Vector3(targetEnt.group.position.x, 0, targetEnt.group.position.z)
      : resolvePos(opts.position, 1.25);
    sandGroup.position.copy(basePos);
    APP.scene.add(sandGroup);

    const tmpC = new THREE.Color();
    for (let i = 0; i < GRAIN_COUNT; i++) {
      gPos[i * 3] = (Math.random() - 0.5) * 0.03;
      gPos[i * 3 + 1] = 0.05 + Math.random() * 0.95;
      gPos[i * 3 + 2] = (Math.random() - 0.5) * 0.03;
      gVel[i * 3 + 1] = -0.25 - Math.random() * 0.4;
      gSize[i] = isGravel ? (18.0 + Math.random() * 14.0) : isSnow ? (12.0 + Math.random() * 12.0) : (6.0 + Math.random() * 6.5);
      gSeed[i] = Math.random() * 100.0;
      tmpC.copy(baseCol).offsetHSL((Math.random() - 0.5) * 0.02, (Math.random() - 0.5) * 0.08, (Math.random() - 0.5) * 0.16);
      gCol[i * 3] = tmpC.r; gCol[i * 3 + 1] = tmpC.g; gCol[i * 3 + 2] = tmpC.b;
    }

    const grainGeo = new THREE.BufferGeometry();
    grainGeo.setAttribute('position', new THREE.BufferAttribute(gPos, 3));
    grainGeo.setAttribute('aCol', new THREE.BufferAttribute(gCol, 3));
    grainGeo.setAttribute('aSize', new THREE.BufferAttribute(gSize, 1));
    const grainPts = new THREE.Points(grainGeo, new THREE.ShaderMaterial({
      transparent: true, depthWrite: true,
      uniforms: { uPx: { value: window.devicePixelRatio || 1 }, uIsSnow: { value: isSnow ? 1.0 : 0.0 } },
      vertexShader: 'attribute vec3 aCol; attribute float aSize; uniform float uPx; varying vec3 vCol; void main(){ vCol=aCol; vec4 mv=modelViewMatrix*vec4(position,1.0); gl_PointSize=clamp((aSize*uPx)/max(0.25,-mv.z), 1.5, 11.0); gl_Position=projectionMatrix*mv; }',
      fragmentShader: 'uniform float uIsSnow; varying vec3 vCol; void main(){ vec2 c=gl_PointCoord-0.5; float r=dot(c,c)*4.0; if(r>1.0)discard; float shade=mix(0.75, 1.15, clamp(-c.y+0.5, 0.0, 1.0)); gl_FragColor=vec4(vCol*shade, uIsSnow>0.5 ? (1.0-r)*0.92 : 1.0); }'
    }));
    grainPts.frustumCulled = false;
    sandGroup.add(grainPts);

    // Movable VR Controller Emitter Hopper / Nozzle
    let emitterEnt = null;
    if (!fromEnt && opts.emitter !== false) {
      const emG = new THREE.Group();
      emG.name = `${matType}_Emitter`;
      const funnel = new THREE.Mesh(
        new THREE.CylinderGeometry(0.085, 0.022, 0.16, 20, 1, true),
        new THREE.MeshPhysicalMaterial({ color: 0xe2e8f0, transmission: 0.78, roughness: 0.12, ior: 1.48, side: THREE.DoubleSide })
      );
      const collar = new THREE.Mesh(
        new THREE.TorusGeometry(0.085, 0.008, 10, 24),
        new THREE.MeshStandardMaterial({ color: 0x94a3b8, metalness: 0.85, roughness: 0.25 })
      );
      collar.rotation.x = Math.PI / 2; collar.position.y = 0.08;
      const nozzle = new THREE.Mesh(
        new THREE.CylinderGeometry(0.024, 0.018, 0.04, 16),
        new THREE.MeshStandardMaterial({ color: 0x334155, metalness: 0.9, roughness: 0.28, emissive: baseCol, emissiveIntensity: 0.25 })
      );
      nozzle.position.y = -0.09;
      const fillCore = new THREE.Mesh(
        new THREE.ConeGeometry(0.072, 0.11, 16),
        new THREE.MeshStandardMaterial({ color: baseCol, roughness: 0.9 })
      );
      fillCore.rotation.x = Math.PI; fillCore.position.y = 0.01;
      emG.add(funnel, collar, nozzle, fillCore);

      const startEmPos = basePos.clone().add(new THREE.Vector3(0, 1.12, 0));
      emG.position.copy(startEmPos);
      APP.scene.add(emG);

      emitterEnt = APP.registerEntity(emG, {
        name: `${matType.charAt(0).toUpperCase() + matType.slice(1)} Emitter`,
        category: 'tool',
        grabbable: true,
        holdable: true,
        physics: { enabled: false },
        onUpdate: function () {
          if (!sandGroup.parent) APP.removeEntity?.(emitterEnt, false);
        }
      });
    }

    const tmpLocal = new THREE.Vector3(), tmpEmPos = new THREE.Vector3(), tmpBox = new THREE.Box3(), tmpCenter = new THREE.Vector3();
    const prevEmPos = new THREE.Vector3(0, 1.0, 0), emVel = new THREE.Vector3();

    const ent = APP.registerEntity(sandGroup, {
      name: `Granular ${matType}`,
      category: 'world_fx',
      grabbable: false,
      physics: { enabled: false },
      onUpdate: function (dt) {
        const delta = Math.min(0.033, Math.max(0.004, dt || 0.016));
        const now = performance.now() * 0.001;
        if (pileMat.userData.shader) pileMat.userData.shader.uniforms.uTime.value = now;

        // Track movable emitter or source entity in sandGroup local space
        if (emitterEnt?.group?.parent) {
          tmpEmPos.set(0, -0.11, 0);
          emitterEnt.group.localToWorld(tmpEmPos);
          sandGroup.worldToLocal(tmpEmPos);
        } else if (fromEnt?.group?.parent) {
          const fi = getEntityBounds(fromEnt);
          if (fi) {
            tmpEmPos.copy(fi.center);
            tmpEmPos.y = fi.box.min.y + fi.size.y * 0.25;
            sandGroup.worldToLocal(tmpEmPos);
          }
        } else {
          tmpEmPos.set(
            targetEnt ? 0 : Math.sin(now * 0.65) * 0.16,
            1.15,
            targetEnt ? 0 : Math.cos(now * 0.55) * 0.16
          );
        }
        emVel.subVectors(tmpEmPos, prevEmPos).divideScalar(delta).clampLength(0, 2.5);
        prevEmPos.copy(tmpEmPos);

        const obsBoxes = [];
        for (const other of APP.entities || []) {
          if (!other || other === ent || other === fromEnt || other === emitterEnt || !other.group || other.category === 'world_fx' || other.category === 'board') continue;
          tmpBox.setFromObject(other.group);
          if (!tmpBox.isEmpty()) {
            tmpBox.min.sub(sandGroup.position); tmpBox.max.sub(sandGroup.position);
            obsBoxes.push(tmpBox.clone());
          }
        }

        // Hand/Finger carving through pile
        for (const h of Object.values(APP.hands || {})) {
          if (!h || !h.active) continue;
          const hp = h.isHand ? h.fingerTips?.index : h.pos;
          if (!hp) continue;
          tmpLocal.copy(hp); sandGroup.worldToLocal(tmpLocal);
          const gx = Math.round((tmpLocal.x / SIZE + 0.5) * (GRID - 1));
          const gz = Math.round((tmpLocal.z / SIZE + 0.5) * (GRID - 1));
          if (gx >= 2 && gx < GRID - 2 && gz >= 2 && gz < GRID - 2 && tmpLocal.y <= pileH[gz * GRID + gx] + 0.08) {
            for (let dz = -2; dz <= 2; dz++) for (let dx = -2; dx <= 2; dx++) {
              const nx = gx + dx, nz = gz + dz;
              const d = Math.hypot(dx, dz);
              if (d <= 2.2) {
                const take = Math.min(pileH[nz * GRID + nx], 0.012 * (1 - d / 2.6));
                pileH[nz * GRID + nx] -= take;
                const rx = nx + Math.sign(dx) * 2, rz = nz + Math.sign(dz) * 2;
                if (rx > 1 && rx < GRID - 2 && rz > 1 && rz < GRID - 2) pileH[rz * GRID + rx] += take * 0.85;
              }
            }
          }
        }

        const grav = isSnow ? 1.85 : isAsh ? 1.2 : 8.2;
        const depositPerGrain = isSnow ? 0.00055 : 0.00042;
        const nozzleSpread = isSnow ? 0.065 : 0.022;

        for (let i = 0; i < GRAIN_COUNT; i++) {
          const i3 = i * 3;
          if (isSnow || isAsh) {
            gVel[i3] += Math.sin(now * 2.5 + gSeed[i]) * 0.45 * delta;
            gVel[i3 + 2] += Math.cos(now * 2.1 + gSeed[i] * 1.3) * 0.45 * delta;
            gVel[i3] *= 0.97; gVel[i3 + 2] *= 0.97;
            gVel[i3 + 1] = Math.max(-0.85, gVel[i3 + 1] - grav * delta);
          } else {
            gVel[i3 + 1] -= grav * delta;
          }

          gPos[i3] += gVel[i3] * delta;
          gPos[i3 + 1] += gVel[i3 + 1] * delta;
          gPos[i3 + 2] += gVel[i3 + 2] * delta;

          for (let b = 0; b < obsBoxes.length; b++) {
            const ob = obsBoxes[b];
            if (gPos[i3] > ob.min.x && gPos[i3] < ob.max.x && gPos[i3 + 2] > ob.min.z && gPos[i3 + 2] < ob.max.z && gPos[i3 + 1] > ob.min.y && gPos[i3 + 1] < ob.max.y) {
              ob.getCenter(tmpCenter);
              gPos[i3 + 1] = ob.max.y + 0.008;
              if (isSnow) {
                gVel[i3] *= 0.4; gVel[i3 + 2] *= 0.4; gVel[i3 + 1] = 0.02;
              } else {
                gVel[i3] += (gPos[i3] - tmpCenter.x) * 3.2;
                gVel[i3 + 2] += (gPos[i3 + 2] - tmpCenter.z) * 3.2;
                gVel[i3 + 1] = Math.abs(gVel[i3 + 1]) * 0.22;
              }
            }
          }

          const gx = Math.floor((gPos[i3] / SIZE + 0.5) * (GRID - 1));
          const gz = Math.floor((gPos[i3 + 2] / SIZE + 0.5) * (GRID - 1));
          const inB = gx >= 2 && gx < GRID - 2 && gz >= 2 && gz < GRID - 2;
          const fY = inB ? pileH[gz * GRID + gx] : 0;

          if (gPos[i3 + 1] <= fY) {
            if (inB && pileH[gz * GRID + gx] < 0.55) {
              pileH[gz * GRID + gx] += depositPerGrain;
            }
            gPos[i3] = tmpEmPos.x + (Math.random() - 0.5) * nozzleSpread;
            gPos[i3 + 1] = Math.max(0.08, tmpEmPos.y) - Math.random() * 0.04;
            gPos[i3 + 2] = tmpEmPos.z + (Math.random() - 0.5) * nozzleSpread;
            gVel[i3] = emVel.x * 0.55 + (Math.random() - 0.5) * (isSnow ? 0.24 : 0.09);
            gVel[i3 + 1] = emVel.y * 0.35 - 0.18 - Math.random() * 0.28;
            gVel[i3 + 2] = emVel.z * 0.55 + (Math.random() - 0.5) * (isSnow ? 0.24 : 0.09);
          }
        }
        grainGeo.attributes.position.needsUpdate = true;

        // Dual-direction Thermal Erosion (Avalanche Repose)
        const dirs = [-1, 1, -GRID, GRID, -GRID - 1, -GRID + 1, GRID - 1, GRID + 1];
        const diagScale = [1, 1, 1, 1, 1.414, 1.414, 1.414, 1.414];
        const slumpRate = isWetSand ? 0.14 : isSnow ? 0.18 : 0.30;
        for (let pass = 0; pass < 2; pass++) {
          for (let z = 1; z < GRID - 1; z++) {
            for (let x = 1; x < GRID - 1; x++) {
              const idx = z * GRID + x, h = pileH[idx];
              if (h <= 0.0005) continue;
              for (let d = 0; d < 8; d++) {
                const ni = idx + dirs[d];
                const lim = MAX_SLOPE * diagScale[d];
                const diff = h - pileH[ni];
                if (diff > lim) {
                  const mv = (diff - lim) * slumpRate * (d >= 4 ? 0.5 : 1.0);
                  pileH[idx] -= mv;
                  pileH[ni] += mv;
                }
              }
            }
          }
        }

        const posA = pileGeo.attributes.position;
        for (let i = 0; i < pileH.length; i++) posA.setY(i, pileH[i]);
        posA.needsUpdate = true;
        pileGeo.computeVertexNormals();
      }
    });

    APP.lastEntityId = ent.id;
    APP.emit?.('entity:created', ent);
    return { ok: true, id: ent.id, emitterId: emitterEnt?.id, name: ent.name, message: `Spawned ${matType} simulation with movable controller emitter.` };
  }

  function buildClothTextures(texName, baseCol) {
    const type = String(texName || 'royal').toLowerCase();
    const c = document.createElement('canvas'); c.width = 512; c.height = 512;
    const ctx = c.getContext('2d');
    const b = document.createElement('canvas'); b.width = 512; b.height = 512;
    const btx = b.getContext('2d');

    const hex = '#' + baseCol.getHexString();
    ctx.fillStyle = hex; ctx.fillRect(0, 0, 512, 512);
    btx.fillStyle = '#808080'; btx.fillRect(0, 0, 512, 512);

    let alphaTex = null;
    let matProps = { roughness: 0.78, metalness: 0.04, sheen: 0.35, sheenRoughness: 0.6, bumpScale: 0.015, alphaTest: 0 };

    if (type === 'silk') {
      const grad = ctx.createLinearGradient(0, 0, 512, 512);
      grad.addColorStop(0, 'rgba(255,255,255,0.16)');
      grad.addColorStop(0.5, 'rgba(0,0,0,0.06)');
      grad.addColorStop(1, 'rgba(255,255,255,0.22)');
      ctx.fillStyle = grad; ctx.fillRect(0, 0, 512, 512);
      ctx.strokeStyle = 'rgba(255,255,255,0.08)'; ctx.lineWidth = 1;
      for (let i = -512; i < 1024; i += 4) {
        ctx.beginPath(); ctx.moveTo(i, 0); ctx.lineTo(i + 512, 512); ctx.stroke();
        btx.strokeStyle = 'rgba(255,255,255,0.18)';
        btx.beginPath(); btx.moveTo(i, 0); btx.lineTo(i + 512, 512); btx.stroke();
      }
      matProps = { roughness: 0.28, metalness: 0.14, sheen: 1.0, sheenRoughness: 0.22, bumpScale: 0.006, alphaTest: 0 };
    } else if (type === 'denim') {
      ctx.fillStyle = baseCol.getHex() === 0x991b1b ? '#1e3a5f' : hex;
      ctx.fillRect(0, 0, 512, 512);
      ctx.strokeStyle = 'rgba(255,255,255,0.13)'; ctx.lineWidth = 2;
      btx.strokeStyle = 'rgba(255,255,255,0.4)'; btx.lineWidth = 2;
      for (let i = -512; i < 1024; i += 6) {
        ctx.beginPath(); ctx.moveTo(i, 0); ctx.lineTo(i + 512, 512); ctx.stroke();
        btx.beginPath(); btx.moveTo(i, 0); btx.lineTo(i + 512, 512); btx.stroke();
      }
      ctx.setLineDash([10, 6]); ctx.strokeStyle = '#f59e0b'; ctx.lineWidth = 5;
      ctx.strokeRect(14, 14, 484, 484); ctx.setLineDash([]);
      matProps = { roughness: 0.88, metalness: 0.02, sheen: 0.15, sheenRoughness: 0.85, bumpScale: 0.024, alphaTest: 0 };
    } else if (type === 'velvet') {
      ctx.fillStyle = baseCol.clone().multiplyScalar(0.75).getStyle();
      ctx.fillRect(0, 0, 512, 512);
      for (let i = 0; i < 7000; i++) {
        const x = Math.random() * 512, y = Math.random() * 512;
        ctx.fillStyle = Math.random() > 0.5 ? 'rgba(255,255,255,0.06)' : 'rgba(0,0,0,0.14)';
        ctx.fillRect(x, y, 2, 2);
        btx.fillStyle = Math.random() > 0.5 ? '#a0a0a0' : '#606060';
        btx.fillRect(x, y, 2, 2);
      }
      matProps = { roughness: 0.92, metalness: 0.0, sheen: 1.0, sheenRoughness: 0.35, bumpScale: 0.02, alphaTest: 0 };
    } else if (type === 'linen') {
      ctx.fillStyle = baseCol.getHex() === 0x991b1b ? '#d6c7b2' : hex;
      ctx.fillRect(0, 0, 512, 512);
      for (let i = 0; i < 512; i += 5) {
        ctx.fillStyle = (i % 10 === 0) ? 'rgba(0,0,0,0.08)' : 'rgba(255,255,255,0.08)';
        ctx.fillRect(i, 0, 2, 512); ctx.fillRect(0, i, 512, 2);
        btx.fillStyle = (i % 10 === 0) ? '#606060' : '#b0b0b0';
        btx.fillRect(i, 0, 2, 512); btx.fillRect(0, i, 512, 2);
      }
      matProps = { roughness: 0.94, metalness: 0.0, sheen: 0.1, sheenRoughness: 0.9, bumpScale: 0.028, alphaTest: 0 };
    } else if (type === 'chainmail') {
      ctx.fillStyle = '#334155'; ctx.fillRect(0, 0, 512, 512);
      ctx.strokeStyle = '#cbd5e1'; ctx.lineWidth = 3;
      btx.fillStyle = '#202020'; btx.fillRect(0, 0, 512, 512);
      btx.strokeStyle = '#ffffff'; btx.lineWidth = 3;
      for (let y = 8; y < 512; y += 16) {
        for (let x = 8; x < 512; x += 16) {
          const ox = (Math.floor(y / 16) % 2) * 8;
          ctx.beginPath(); ctx.arc(x + ox, y, 7, 0, Math.PI * 2); ctx.stroke();
          btx.beginPath(); btx.arc(x + ox, y, 7, 0, Math.PI * 2); btx.stroke();
        }
      }
      matProps = { roughness: 0.32, metalness: 0.90, sheen: 0.0, sheenRoughness: 0.5, bumpScale: 0.035, alphaTest: 0 };
    } else if (type === 'lace') {
      const a = document.createElement('canvas'); a.width = 512; a.height = 512;
      const atx = a.getContext('2d');
      atx.fillStyle = '#000000'; atx.fillRect(0, 0, 512, 512);
      atx.strokeStyle = '#ffffff'; atx.lineWidth = 4;
      ctx.fillStyle = baseCol.getHex() === 0x991b1b ? '#f8fafc' : hex; ctx.fillRect(0, 0, 512, 512);
      for (let y = 0; y <= 512; y += 32) {
        for (let x = 0; x <= 512; x += 32) {
          atx.beginPath(); atx.arc(x, y, 14, 0, Math.PI * 2); atx.stroke();
          atx.beginPath(); atx.arc(x + 16, y + 16, 10, 0, Math.PI * 2); atx.stroke();
        }
      }
      atx.lineWidth = 18; atx.strokeRect(9, 9, 494, 494);
      alphaTex = new THREE.CanvasTexture(a);
      matProps = { roughness: 0.68, metalness: 0.05, sheen: 0.5, sheenRoughness: 0.5, bumpScale: 0.015, alphaTest: 0.45 };
    } else {
      // Royal tapestry default
      ctx.strokeStyle = 'rgba(255,255,255,0.12)'; ctx.lineWidth = 2;
      btx.strokeStyle = 'rgba(255,255,255,0.3)'; btx.lineWidth = 2;
      for (let i = 0; i < 512; i += 6) {
        ctx.beginPath(); ctx.moveTo(i, 0); ctx.lineTo(i, 512); ctx.stroke();
        ctx.beginPath(); ctx.moveTo(0, i); ctx.lineTo(512, i); ctx.stroke();
        btx.beginPath(); btx.moveTo(i, 0); btx.lineTo(i, 512); btx.stroke();
        btx.beginPath(); btx.moveTo(0, i); btx.lineTo(512, i); btx.stroke();
      }
      ctx.strokeStyle = '#fbbf24'; ctx.lineWidth = 14; ctx.strokeRect(18, 18, 476, 476);
      ctx.lineWidth = 4; ctx.strokeRect(32, 32, 448, 448);
    }

    const mapTex = new THREE.CanvasTexture(c); mapTex.colorSpace = THREE.SRGBColorSpace;
    const bumpTex = new THREE.CanvasTexture(b);
    return { map: mapTex, bumpMap: bumpTex, alphaMap: alphaTex, ...matProps };
  }

  function createClothSimulation(opts = {}) {
    const mode = String(opts.mode || 'banner').toLowerCase();
    const isDrape = mode === 'drape' || mode === 'tablecloth';
    const targetEnt = opts.target ? APP.findEntity?.(opts.target) : null;
    const betEnts = Array.isArray(opts.between_entities) && opts.between_entities.length === 2
      ? [APP.findEntity?.(opts.between_entities[0]), APP.findEntity?.(opts.between_entities[1])]
      : null;

    if (APP._activeClothEnt && APP.entities?.includes(APP._activeClothEnt) && opts.replace !== false) {
      APP.removeEntity?.(APP._activeClothEnt, false);
    }

    const clothGroup = new THREE.Group();
    clothGroup.name = isDrape ? 'Draped_Cloth_Sim' : 'Interactive_Cloth_Sim';

    const COLS = 30, ROWS = 22;
    let W = Number(opts.width) || (isDrape ? 1.22 : 1.14);
    const H = Number(opts.height) || (isDrape ? 1.22 : 0.86);

    const bInfoA = betEnts?.[0] ? getEntityBounds(betEnts[0]) : null;
    const bInfoB = betEnts?.[1] ? getEntityBounds(betEnts[1]) : null;
    if (bInfoA && bInfoB) {
      W = Math.max(0.45, Math.min(2.8, bInfoA.center.distanceTo(bInfoB.center)));
    }

    const DX = W / (COLS - 1), DY = H / (ROWS - 1), DIAG = Math.hypot(DX, DY), N_COUNT = COLS * ROWS;

    if (!isDrape && !(bInfoA && bInfoB)) {
      const rod = new THREE.Mesh(
        new THREE.CylinderGeometry(0.013, 0.013, W + 0.16, 16),
        new THREE.MeshStandardMaterial({ color: 0x64748b, metalness: 0.88, roughness: 0.25 })
      );
      rod.rotation.z = Math.PI / 2; rod.position.set(0, H * 0.5 + 0.015, 0);
      clothGroup.add(rod);
    }

    const baseCol = parseHexColor(opts.color, 0x991b1b);
    let curTexName = String(opts.texture || 'royal').toLowerCase();
    const texData = buildClothTextures(curTexName, baseCol);

    const cur = [], prev = [], pinned = [];
    for (let r = 0; r < ROWS; r++) {
      for (let c = 0; c < COLS; c++) {
        const x = (c / (COLS - 1) - 0.5) * W;
        const y = isDrape ? 0.0 : (H * 0.5 - r * DY);
        const z = isDrape ? (r / (ROWS - 1) - 0.5) * H : 0.0;
        cur.push(new THREE.Vector3(x, y, z));
        prev.push(new THREE.Vector3(x, y, z));
        pinned.push(bInfoA && bInfoB ? (r === 0 && (c === 0 || c === COLS - 1)) : (!isDrape && r === 0));
      }
    }

    const springs = [], hGrid = Array.from({ length: ROWS }, () => new Array(COLS - 1)), vGrid = Array.from({ length: ROWS - 1 }, () => new Array(COLS));
    const tearable = opts.tearable !== false, tearMult = tearable ? 2.15 : 99.0;
    function addSpring(a, b, rest, mult, stiff = 1.0, isBend = false) {
      const s = { a, b, rest, tearDist: rest * mult, stiff, isBend, broken: false };
      springs.push(s); return s;
    }
    for (let r = 0; r < ROWS; r++) for (let c = 0; c < COLS; c++) {
      const i = r * COLS + c;
      if (c < COLS - 1) hGrid[r][c] = addSpring(i, i + 1, DX, tearMult, 1.0, false);
      if (r < ROWS - 1) vGrid[r][c] = addSpring(i, i + COLS, DY, tearMult, 1.0, false);
      if (r < ROWS - 1 && c < COLS - 1) {
        addSpring(i, i + COLS + 1, DIAG, tearMult * 1.15, 0.85, false);
        addSpring(i + 1, i + COLS, DIAG, tearMult * 1.15, 0.85, false);
      }
      // Bend springs for authentic fabric stiffness
      if (c < COLS - 2) addSpring(i, i + 2, DX * 2, 99.0, 0.34, true);
      if (r < ROWS - 2) addSpring(i, i + COLS * 2, DY * 2, 99.0, 0.34, true);
    }

    const clothGeo = new THREE.BufferGeometry();
    const posArr = new Float32Array(N_COUNT * 3), uvArr = new Float32Array(N_COUNT * 2);
    const heatArr = new Float32Array(N_COUNT), charArr = new Float32Array(N_COUNT), burned = new Uint8Array(N_COUNT);
    for (let r = 0; r < ROWS; r++) for (let c = 0; c < COLS; c++) {
      const idx = r * COLS + c; uvArr[idx * 2] = c / (COLS - 1); uvArr[idx * 2 + 1] = 1.0 - r / (ROWS - 1);
    }
    const idxArr = new Uint16Array((ROWS - 1) * (COLS - 1) * 6);
    clothGeo.setAttribute('position', new THREE.BufferAttribute(posArr, 3));
    clothGeo.setAttribute('uv', new THREE.BufferAttribute(uvArr, 2));
    clothGeo.setAttribute('aHeat', new THREE.BufferAttribute(heatArr, 1));
    clothGeo.setAttribute('aChar', new THREE.BufferAttribute(charArr, 1));
    clothGeo.setIndex(new THREE.BufferAttribute(idxArr, 1));

    const clothMat = new THREE.MeshPhysicalMaterial({
      map: texData.map,
      bumpMap: texData.bumpMap,
      bumpScale: texData.bumpScale,
      alphaMap: texData.alphaMap,
      alphaTest: texData.alphaTest,
      roughness: texData.roughness,
      metalness: texData.metalness,
      sheen: texData.sheen,
      sheenRoughness: texData.sheenRoughness,
      sheenColor: baseCol.clone().lerp(new THREE.Color(0xffffff), 0.5),
      side: THREE.DoubleSide
    });

        .replace('#include <map_fragment>', [
          '#include <map_fragment>',
          'float soot = smoothstep(0.05, 0.72, vChar);',

          'float front = smoothstep(0.22, 0.55, vHeat) * (1.0 - smoothstep(0.70, 0.99, vChar));',
          'vec3 emberCol = mix(vec3(1.0, 0.18, 0.02), vec3(1.0, 0.82, 0.25), clamp((vHeat - 0.3) * 1.6, 0.0, 1.0));',
          'totalEmissiveRadiance += emberCol * front * flicker * 3.8;'

    };

    const clothMesh = new THREE.Mesh(clothGeo, clothMat);
    clothMesh.castShadow = true; clothMesh.receiveShadow = true;
    clothGroup.add(clothMesh);

    // Tear fibers + Fire/Ember/Smoke Particle Pool
    const FIBER_MAX = 220;
    const fibPos = new Float32Array(FIBER_MAX * 3), fibVel = Array.from({ length: FIBER_MAX }, () => new THREE.Vector3());
    const fibLife = new Float32Array(FIBER_MAX), fibCol = new Float32Array(FIBER_MAX * 3);
    const fibGeo = new THREE.BufferGeometry();
    fibGeo.setAttribute('position', new THREE.BufferAttribute(fibPos, 3));
    fibGeo.setAttribute('aCol', new THREE.BufferAttribute(fibCol, 3));
    fibGeo.setAttribute('aLife', new THREE.BufferAttribute(fibLife, 1));
    clothGroup.add(new THREE.Points(fibGeo, new THREE.ShaderMaterial({
      transparent: true, depthWrite: false, blending: THREE.AdditiveBlending,
      uniforms: { uPx: { value: window.devicePixelRatio || 1 } },
      vertexShader: 'attribute vec3 aCol; attribute float aLife; uniform float uPx; varying vec3 vCol; varying float vLife; void main(){ vCol=aCol; vLife=aLife; vec4 mv=modelViewMatrix*vec4(position,1.0); gl_PointSize=aLife>0.01?clamp((26.0*uPx*aLife)/max(0.25,-mv.z),2.0,16.0):0.0; gl_Position=projectionMatrix*mv; }',
      fragmentShader: 'varying vec3 vCol; varying float vLife; void main(){ float d=length(gl_PointCoord-0.5)*2.0; if(d>1.0)discard; gl_FragColor=vec4(vCol, (1.0-d)*clamp(vLife*1.4,0.0,0.95)); }'
    })));
    let nextFib = 0;

    function emitRipFibers(pt) {
      for (let k = 0; k < 3; k++) {
        const fi = nextFib; nextFib = (nextFib + 1) % FIBER_MAX;
        fibPos[fi * 3] = pt.x; fibPos[fi * 3 + 1] = pt.y; fibPos[fi * 3 + 2] = pt.z;
        fibVel[fi].set((Math.random() - 0.5) * 1.3, Math.random() * 0.7, (Math.random() - 0.5) * 1.3);
        fibLife[fi] = 0.55;
        fibCol[fi * 3] = 0.99; fibCol[fi * 3 + 1] = 0.94; fibCol[fi * 3 + 2] = 0.55;
      }
    }

    function emitEmber(pt) {
      const fi = nextFib; nextFib = (nextFib + 1) % FIBER_MAX;
      fibPos[fi * 3] = pt.x + (Math.random() - 0.5) * 0.02;
      fibPos[fi * 3 + 1] = pt.y + (Math.random() - 0.5) * 0.02;
      fibPos[fi * 3 + 2] = pt.z + (Math.random() - 0.5) * 0.02;
      fibVel[fi].set((Math.random() - 0.5) * 0.35, 0.55 + Math.random() * 0.85, (Math.random() - 0.5) * 0.35);
      fibLife[fi] = 0.45 + Math.random() * 0.55;
      const hot = Math.random();
      fibCol[fi * 3] = 1.0; fibCol[fi * 3 + 1] = 0.25 + hot * 0.6; fibCol[fi * 3 + 2] = hot * 0.18;
    }

    let indicesDirty = true;
    function rebuildClothIndices() {
      let ptr = 0;
      for (let r = 0; r < ROWS - 1; r++) for (let c = 0; c < COLS - 1; c++) {
        const a = r * COLS + c, b = a + 1, d = a + COLS, e = d + 1;
        if (burned[a] || burned[b] || burned[d] || burned[e]) {
          if (burned[a] + burned[b] + burned[d] + burned[e] >= 2) continue;
        }
        if ((!hGrid[r][c].broken ? 1 : 0) + (!hGrid[r + 1][c].broken ? 1 : 0) + (!vGrid[r][c].broken ? 1 : 0) + (!vGrid[r][c + 1].broken ? 1 : 0) < 3) continue;
        idxArr[ptr++] = a; idxArr[ptr++] = d; idxArr[ptr++] = b; idxArr[ptr++] = b; idxArr[ptr++] = d; idxArr[ptr++] = e;
      }
      clothGeo.index.needsUpdate = true; clothGeo.setDrawRange(0, ptr); indicesDirty = false;
    }
    rebuildClothIndices();

    if (bInfoA && bInfoB) {
      const mid = new THREE.Vector3().addVectors(bInfoA.center, bInfoB.center).multiplyScalar(0.5);
      mid.y = Math.max(bInfoA.box.max.y, bInfoB.box.max.y) - H * 0.45;
      clothGroup.position.copy(mid);
      const dir = new THREE.Vector3().subVectors(bInfoB.center, bInfoA.center).setY(0).normalize();
      if (dir.lengthSq() > 0.01) clothGroup.rotation.y = Math.atan2(-dir.z, dir.x);
    } else if (isDrape && targetEnt?.group) {
      const tInfo = getEntityBounds(targetEnt);
      clothGroup.position.set(tInfo.center.x, tInfo.box.max.y + 0.25, tInfo.center.z);
    } else {
      const sp = resolvePos(opts.position, 1.05);
      clothGroup.position.set(sp.x, isDrape ? 1.0 : 1.15, sp.z);
      const hp = APP.headPos || new THREE.Vector3(0, 1.5, 0);
      if (!isDrape) clothGroup.lookAt(hp.x, clothGroup.position.y, hp.z);
    }
    APP.scene.add(clothGroup);

    const tmpDelta = new THREE.Vector3(), tmpLocal = new THREE.Vector3(), tmpSphereC = new THREE.Vector3(), tmpBox = new THREE.Box3(), prevHands = new Map();
    const tmpWind = new THREE.Vector3(), tmpNorm = new THREE.Vector3();
    let grabbedIdx = -1, grabbedHand = null, isBurning = !!opts.burning, burnSpeed = Number(opts.burn_speed) || 1.0;

    function tearAtLocalPoint(pt, rad, imp) {
      let tore = false;
      for (let i = 0; i < springs.length; i++) {
        const s = springs[i]; if (s.broken || s.isBend) continue;
        const mx = (cur[s.a].x + cur[s.b].x) * 0.5, my = (cur[s.a].y + cur[s.b].y) * 0.5, mz = (cur[s.a].z + cur[s.b].z) * 0.5;
        if (Math.hypot(mx - pt.x, my - pt.y, mz - pt.z) < rad) {
          s.broken = true; tore = true; indicesDirty = true; emitRipFibers(cur[s.a]);
          if (imp) { prev[s.a].sub(imp); prev[s.b].add(imp); }
        }
      }
      return tore;
    }

    function igniteAt(origin = 'bottom') {
      isBurning = true;
      const o = String(origin).toLowerCase();
      if (o === 'center') {
        const ci = Math.floor(ROWS * 0.55) * COLS + Math.floor(COLS * 0.5);
        heatArr[ci] = 1.0;
        if (ci + 1 < N_COUNT) heatArr[ci + 1] = 0.9;
      } else if (o === 'corners') {
        heatArr[(ROWS - 1) * COLS] = 1.0;
        heatArr[N_COUNT - 1] = 1.0;
      } else if (o === 'random') {
        for (let k = 0; k < 3; k++) heatArr[Math.floor(Math.random() * N_COUNT)] = 1.0;
      } else {
        // Bottom edge
        const r = ROWS - 1;
        for (let c = Math.floor(COLS * 0.35); c <= Math.floor(COLS * 0.65); c++) {
          heatArr[r * COLS + c] = 0.95 + Math.random() * 0.05;
        }
      }
    }

    if (isBurning) igniteAt(opts.burn_origin || 'bottom');

    const clothApi = {
      group: clothGroup,
      tearSlice: (yRatio = 0.45) => {
        const targetRow = Math.max(1, Math.min(ROWS - 2, Math.floor(yRatio * ROWS)));
        for (let c = 0; c < COLS; c++) tearAtLocalPoint(cur[THREE.MathUtils.clamp(targetRow + Math.round(Math.sin(c * 0.9)), 1, ROWS - 2) * COLS + c], DX * 0.98, new THREE.Vector3(0, 0, 0.018));
        rebuildClothIndices();
      },
      shred: () => {
        for (let c = 3; c < COLS - 2; c += 4) for (let r = 1; r < ROWS - 1; r++) tearAtLocalPoint(cur[r * COLS + c], DX * 0.88, new THREE.Vector3(0, 0, 0.02));
        rebuildClothIndices();
      },
      burn: (o = {}) => {
        if (o.action === 'extinguish') {
          isBurning = false;
          heatArr.fill(0);
          clothGeo.attributes.aHeat.needsUpdate = true;
          return;
        }
        if (o.speed) burnSpeed = Number(o.speed) || 1.0;
        igniteAt(o.origin || 'bottom');
      },
      setTexture: (tex, newCol) => {
        curTexName = String(tex || curTexName).toLowerCase();
        const cObj = newCol ? parseHexColor(newCol, 0x991b1b) : baseCol;
        const td = buildClothTextures(curTexName, cObj);
        clothMat.map = td.map;
        clothMat.bumpMap = td.bumpMap;
        clothMat.bumpScale = td.bumpScale;
        clothMat.alphaMap = td.alphaMap;
        clothMat.alphaTest = td.alphaTest;
        clothMat.roughness = td.roughness;
        clothMat.metalness = td.metalness;
        clothMat.sheen = td.sheen;
        clothMat.sheenRoughness = td.sheenRoughness;
        clothMat.sheenColor = cObj.clone().lerp(new THREE.Color(0xffffff), 0.5);
        clothMat.needsUpdate = true;
      }
    };

    const ent = APP.registerEntity(clothGroup, {
      name: opts.name || (isDrape ? 'Draped Cloth' : 'Interactive Cloth'),
      category: 'cloth',
      grabbable: false,
      physics: { enabled: false },
      onUpdate: function (dt) {
        const safeDt = Math.min(0.025, Math.max(0.004, dt || 0.016)), now = performance.now() * 0.001;
        if (clothMat.userData.shader) clothMat.userData.shader.uniforms.uBurnTime.value = now;
        const gravY = -7.8 * safeDt * safeDt, floorLocalY = -clothGroup.position.y + 0.02;

        const colliders = [];
        for (const other of APP.entities || []) {
          if (!other || other === ent || !other.group || other.category === 'world_fx' || other.category === 'board') continue;
          tmpBox.setFromObject(other.group);
          if (!tmpBox.isEmpty()) {
            const sz = new THREE.Vector3(); tmpBox.getSize(sz); tmpBox.getCenter(tmpSphereC); clothGroup.worldToLocal(tmpSphereC);
            colliders.push({ c: tmpSphereC.clone(), r: Math.max(sz.x, sz.y, sz.z) * 0.52 + 0.015 });
          }
        }

        let activeGrab = false;
        for (const [hk, h] of Object.entries(APP.hands || {})) {
          if (!h || !h.active) continue;
          const tip = h.isHand ? h.fingerTips?.index : h.pos;
          if (!tip) continue;
          tmpLocal.copy(tip); clothGroup.worldToLocal(tmpLocal);
          const last = prevHands.get(hk), speed = last ? tmpLocal.distanceTo(last) / safeDt : 0;
          prevHands.set(hk, tmpLocal.clone());

          if ((h.trigger || h.pinching) && grabbedIdx === -1) {
            let bestI = -1, bestD = 0.16;
            for (let i = (isDrape ? 0 : COLS); i < N_COUNT; i++) {
              if (burned[i]) continue;
              const d = cur[i].distanceTo(tmpLocal);
              if (d < bestD) { bestD = d; bestI = i; }
            }
            if (bestI !== -1) { grabbedIdx = bestI; grabbedHand = hk; }
          }
          if (grabbedHand === hk && (h.trigger || h.pinching) && grabbedIdx !== -1) {
            cur[grabbedIdx].lerp(tmpLocal, 0.72); prev[grabbedIdx].copy(cur[grabbedIdx]); activeGrab = true;
          }
          if (tearable && speed > 1.35) tearAtLocalPoint(tmpLocal, 0.085, new THREE.Vector3(0, 0, 0.012));
        }
        if (!activeGrab) { grabbedIdx = -1; grabbedHand = null; }

        // Burning & Thermal Convection Propagation
        if (isBurning && curTexName !== 'chainmail') {
          for (let i = 0; i < springs.length; i++) {
            const s = springs[i]; if (s.broken || s.isBend) continue;
            const ha = heatArr[s.a], hb = heatArr[s.b];
            if (ha > 0.25 && !burned[s.b]) {
              const up = cur[s.b].y > cur[s.a].y ? 2.4 : 0.95;
              heatArr[s.b] = Math.min(1.0, hb + ha * safeDt * 0.85 * burnSpeed * up);
            }
            if (hb > 0.25 && !burned[s.a]) {
              const up = cur[s.a].y > cur[s.b].y ? 2.4 : 0.95;
              heatArr[s.a] = Math.min(1.0, ha + hb * safeDt * 0.85 * burnSpeed * up);
            }
          }
          for (let i = 0; i < N_COUNT; i++) {
            if (burned[i]) continue;
            if (heatArr[i] > 0.34) {
              charArr[i] = Math.min(1.0, charArr[i] + safeDt * 0.42 * burnSpeed * (0.7 + Math.sin(i * 1.7) * 0.3));
              if (charArr[i] > 0.2 && charArr[i] < 0.95 && Math.random() < 0.14) emitEmber(cur[i]);
              if (charArr[i] >= 1.0) {
                burned[i] = 1;
                heatArr[i] = 0;
                indicesDirty = true;
                for (let k = 0; k < springs.length; k++) {
                  if (!springs[k].broken && (springs[k].a === i || springs[k].b === i)) springs[k].broken = true;
                }
              }
            }
          }
          clothGeo.attributes.aHeat.needsUpdate = true;
          clothGeo.attributes.aChar.needsUpdate = true;
        }

        // Aerodynamic Wind + Verlet Integration
        const windStrength = Number(opts.wind ?? (isDrape ? 0.32 : 1.0));
        const nAttr = clothGeo.attributes.normal;
        for (let i = 0; i < N_COUNT; i++) {
          if (pinned[i] || i === grabbedIdx || burned[i]) continue;
          const p = cur[i], pr = prev[i];
          const vx = (p.x - pr.x) * 0.984, vy = (p.y - pr.y) * 0.984, vz = (p.z - pr.z) * 0.984;
          pr.copy(p);

          tmpWind.set(
            Math.cos(now * 1.6 + p.y * 2.2) * 0.0005,
            isBurning && heatArr[i] > 0.3 ? 0.0004 : 0.0,
            (Math.sin(now * 2.3 + p.x * 3.0 + p.y * 1.8) * 0.55 + 0.42) * 0.0014
          ).multiplyScalar(windStrength);

          if (nAttr) {
            tmpNorm.fromBufferAttribute(nAttr, i);
            const aero = Math.abs(tmpNorm.dot(tmpWind));
            p.x += vx + tmpWind.x * (0.5 + aero * 400.0);
            p.y += vy + gravY + tmpWind.y;
            p.z += vz + tmpWind.z * (0.5 + aero * 400.0);
          } else {
            p.x += vx + tmpWind.x; p.y += vy + gravY + tmpWind.y; p.z += vz + tmpWind.z;
          }
          if (p.y < floorLocalY) p.y = floorLocalY;
        }

        for (let iter = 0; iter < 6; iter++) {
          for (let i = 0; i < springs.length; i++) {
            const s = springs[i]; if (s.broken) continue;
            const pa = cur[s.a], pb = cur[s.b]; tmpDelta.subVectors(pb, pa); const d = tmpDelta.length();
            if (d < 1e-6) continue;
            if (tearable && !s.isBend && iter === 0 && d > s.tearDist) { s.broken = true; indicesDirty = true; emitRipFibers(pa); continue; }
            const diff = ((d - s.rest) / d) * s.stiff;
            const pinA = pinned[s.a] || s.a === grabbedIdx, pinB = pinned[s.b] || s.b === grabbedIdx;
            if (pinA && pinB) continue;
            if (pinA) pb.addScaledVector(tmpDelta, -diff);
            else if (pinB) pa.addScaledVector(tmpDelta, diff);
            else { pa.addScaledVector(tmpDelta, diff * 0.5); pb.addScaledVector(tmpDelta, -diff * 0.5); }
          }
          for (let ci = 0; ci < colliders.length; ci++) {
            const col = colliders[ci], rSq = col.r * col.r;
            for (let i = 0; i < N_COUNT; i++) {
              if (pinned[i] || burned[i]) continue;
              tmpDelta.subVectors(cur[i], col.c); const dSq = tmpDelta.lengthSq();
              if (dSq < rSq && dSq > 1e-6) cur[i].copy(col.c).addScaledVector(tmpDelta, col.r / Math.sqrt(dSq));
            }
          }
        }

        if (indicesDirty) rebuildClothIndices();
        for (let i = 0; i < N_COUNT; i++) { posArr[i * 3] = cur[i].x; posArr[i * 3 + 1] = cur[i].y; posArr[i * 3 + 2] = cur[i].z; }
        clothGeo.attributes.position.needsUpdate = true;
        clothGeo.computeVertexNormals();

        for (let i = 0; i < FIBER_MAX; i++) {
          if (fibLife[i] <= 0) { fibPos[i * 3 + 1] = -99; continue; }
          fibLife[i] -= safeDt;
          fibVel[i].y += (fibCol[i * 3] > 0.99 && fibCol[i * 3 + 2] < 0.3 ? 1.8 : -3.5) * safeDt;
          fibPos[i * 3] += fibVel[i].x * safeDt; fibPos[i * 3 + 1] += fibVel[i].y * safeDt; fibPos[i * 3 + 2] += fibVel[i].z * safeDt;
        }
        fibGeo.attributes.position.needsUpdate = true;
        fibGeo.attributes.aLife.needsUpdate = true;
        fibGeo.attributes.aCol.needsUpdate = true;
      }
    });

    clothApi.entity = ent; APP._activeCloth = clothApi; APP._activeClothEnt = ent;
    APP.lastEntityId = ent.id; APP.emit?.('entity:created', ent);
    return { ok: true, id: ent.id, name: ent.name, message: `Spawned ${curTexName} ${ent.name}.` };
  }

  function createDisintegration(opts = {}) {
    const q = String(opts.target || '').trim();
    let target = q ? APP.findEntity?.(q) : null;
    if (!target || target.category === 'world_fx' || target.category === 'board') {
      const candidates = (APP.entities || []).filter((e) => e && e.group?.parent && e.category !== 'world_fx' && e.category !== 'board' && !e.isPen);
      target = candidates[candidates.length - 1] || null;
    }
    if (!target) {
      const artGroup = new THREE.Group();
      const ped = new THREE.Mesh(new THREE.CylinderGeometry(0.18, 0.22, 0.28, 14), new THREE.MeshStandardMaterial({ color: 0x334155, roughness: 0.55, metalness: 0.45 }));
      ped.position.y = 0.14;
      const core = new THREE.Mesh(new THREE.OctahedronGeometry(0.22, 2), new THREE.MeshStandardMaterial({ color: 0x38bdf8, emissive: 0x0ea5e9, emissiveIntensity: 0.5, roughness: 0.25, metalness: 0.65 }));
      core.position.y = 0.50;
      artGroup.add(ped, core); artGroup.position.copy(resolvePos(null, 1.15));
      target = APP.registerEntity(artGroup, { name: 'Artifact Monolith', category: 'artifact' });
    }

    const edgeCol = parseHexColor(opts.color, 0xff7b14);
    const dur = Math.max(1.2, Math.min(6.0, Number(opts.duration) || 2.6));
    target.group.updateMatrixWorld(true);
    const box = new THREE.Box3().setFromObject(target.group);
    const minY = isFinite(box.min.y) ? box.min.y : 0, maxY = isFinite(box.max.y) ? Math.max(minY + 0.1, box.max.y) : 1.0;

    const samplePos = [], sampleCol = [], vTmp = new THREE.Vector3();
    target.group.traverse((c) => {
      if (!c.isMesh || !c.geometry?.attributes?.position) return;
      const posA = c.geometry.attributes.position, colA = c.geometry.attributes.color;
      const mat = Array.isArray(c.material) ? c.material[0] : c.material;
      const baseCol = mat?.color ? mat.color : new THREE.Color(0x94a3b8);
      const step = Math.max(1, Math.floor(posA.count / 450));
      for (let i = 0; i < posA.count; i += step) {
        vTmp.fromBufferAttribute(posA, i).applyMatrix4(c.matrixWorld);
        samplePos.push(vTmp.x, vTmp.y, vTmp.z);
        if (colA) sampleCol.push(colA.getX(i), colA.getY(i), colA.getZ(i));
        else sampleCol.push(baseCol.r, baseCol.g, baseCol.b);
      }
    });

    const P_COUNT = Math.min(1800, Math.max(600, samplePos.length / 3));
    const pPos = new Float32Array(P_COUNT * 3), pCol = new Float32Array(P_COUNT * 3), pThresh = new Float32Array(P_COUNT);
    const pVel = new Float32Array(P_COUNT * 3), pDetached = new Uint8Array(P_COUNT), pLife = new Float32Array(P_COUNT), pActive = new Float32Array(P_COUNT);
    const srcLen = Math.max(1, samplePos.length / 3);
    for (let i = 0; i < P_COUNT; i++) {
      const si = (i % srcLen) * 3;
      pPos[i * 3] = samplePos[si] + (Math.random() - 0.5) * 0.015;
      pPos[i * 3 + 1] = samplePos[si + 1] + (Math.random() - 0.5) * 0.015;
      pPos[i * 3 + 2] = samplePos[si + 2] + (Math.random() - 0.5) * 0.015;
      pCol[i * 3] = sampleCol[si]; pCol[i * 3 + 1] = sampleCol[si + 1]; pCol[i * 3 + 2] = sampleCol[si + 2];
      const normY = THREE.MathUtils.clamp((pPos[i * 3 + 1] - minY) / (maxY - minY), 0, 1);
      pThresh[i] = THREE.MathUtils.clamp((opts.direction === 'top_down' ? 1 - normY : normY) * 0.72 + Math.random() * 0.28, 0.01, 0.99);
      pLife[i] = 1.0; pVel[i * 3] = (Math.random() - 0.3) * 0.35 + 0.18; pVel[i * 3 + 1] = 0.24 + Math.random() * 0.45; pVel[i * 3 + 2] = (Math.random() - 0.5) * 0.35;
    }

    const uProg = { value: 0.0 }, topDownFlag = { value: opts.direction === 'top_down' ? 1.0 : 0.0 };
    target.group.traverse((c) => {
      if (!c.isMesh || !c.material) return;
      const mats = Array.isArray(c.material) ? c.material : [c.material];
      c.material = mats.map((m) => {
        const cl = m.clone(); cl.side = THREE.DoubleSide; cl.customProgramCacheKey = () => 'nxr_dis_v3';
