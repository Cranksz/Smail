  function lsGet(k, fb) {
    try { const v = localStorage.getItem(k); return v !== null ? v : fb; } catch (_) { return fb; }
  }
  function lsSet(k, v) {
    try {
      if (v === null || v === undefined || v === '') localStorage.removeItem(k);
      else localStorage.setItem(k, String(v));
    } catch (_) {}
  }

  // Auto-guess GitHub owner/repo if hosted on *.github.io/<repo>/
  function detectDefaultRepo() {
    let defOwner = lsGet(GH_STORAGE.owner, '');
    let defRepo = lsGet(GH_STORAGE.repo, '');
    const defBranch = lsGet(GH_STORAGE.branch, 'main');
    if (!defOwner && window.location.hostname.endsWith('.github.io')) {
      defOwner = window.location.hostname.split('.')[0] || '';
      const segs = window.location.pathname.split('/').filter(Boolean);
      if (!defRepo && segs.length > 0) defRepo = segs[0];
    }
    return { owner: defOwner, repo: defRepo, branch: defBranch || 'main', pat: lsGet(GH_STORAGE.pat, '') };
  }

      return;

      const chip = document.createElement('button');
      chip.type = 'button';
      chip.className = 'hud-chip';
      chip.textContent = ab.label || ab.id;

    function drawRoundedRect(x, y, w, h, r, fill, stroke) {
      ctx.beginPath();
      ctx.moveTo(x + r, y);
      ctx.arcTo(x + w, y, x + w, y + h, r);
      ctx.arcTo(x + w, y + h, x, y + h, r);
      ctx.arcTo(x, y + h, x, y, r);
      ctx.arcTo(x, y, x + w, y, r);
      ctx.closePath();
      if (fill) { ctx.fillStyle = fill; ctx.fill(); }
      if (stroke) { ctx.strokeStyle = stroke; ctx.lineWidth = 3; ctx.stroke(); }
    }

      ctx.textAlign = 'center';

      hitZones.push({
        id: 'prev',
        u0: prevRect.x / canvas.width,
        u1: (prevRect.x + prevRect.w) / canvas.width,
        v0: 1 - (prevRect.y + prevRect.h) / canvas.height,
        v1: 1 - prevRect.y / canvas.height,
        action: () => { currentPage = (currentPage - 1 + pages.length) % pages.length; }

    function placeInFrontOfUser() {
      const hp = APP.headPos || new THREE.Vector3(0, 1.5, 0);
      const hf = APP.headFwd || new THREE.Vector3(0, 0, -1);
      const flatFwd = new THREE.Vector3(hf.x, 0, hf.z).normalize();
      if (flatFwd.lengthSq() < 0.01) flatFwd.set(0, 0, -1);
      const right = new THREE.Vector3().crossVectors(flatFwd, new THREE.Vector3(0, 1, 0)).normalize();
      root.position.copy(hp).addScaledVector(flatFwd, 0.85).addScaledVector(right, -0.32);
      root.position.y = Math.max(0.85, hp.y - 0.12);
      root.lookAt(hp.x, root.position.y, hp.z);
    }

      else show(true);

    function pulseGamepad(hand) {
      try {
        const gp = hand && hand.gamepad;
        if (gp && gp.hapticActuators && gp.hapticActuators[0]) {
          gp.hapticActuators[0].pulse(0.35, 30);
        }
      } catch (_) {}
    }

    function triggerZone(zone, hand) {
      const now = performance.now();
      if (now - lastClickTime < 280) return;
      lastClickTime = now;
      pulseGamepad(hand);
      try { zone.action(); } catch (e) { console.warn('[vrMenu] action err:', e); }
      syncHUDState();
      redraw();
    }

    function updateInteraction(dt, elapsed) {
      if (!root.visible) return;
      if (minimized) {
        orbRing.rotation.z = elapsed * 1.4;
        orbRing.rotation.x = Math.sin(elapsed * 0.9) * 0.4;
      }

      const hands = APP.hands || {};
      const handEntries = Object.entries(hands);
      let newHover = null;

      // Active dragging
      if (draggingHandKey && hands[draggingHandKey]) {
        const dh = hands[draggingHandKey];
        const stillDown = !!(dh.selecting || dh.pinching || dh.squeezing);
        if (stillDown) {
          const anchor = dh.pinching && dh.indexTip ? dh.indexTip : null;
          if (anchor) {
            root.position.copy(anchor).add(dragOffset);
          } else if (dh.rayOrigin && dh.rayDir) {
            root.position.copy(dh.rayOrigin).addScaledVector(dh.rayDir, 0.85).add(dragOffset);
          }
          if (APP.headPos) root.lookAt(APP.headPos.x, root.position.y, APP.headPos.z);
          return;
        } else {
          draggingHandKey = null;
        }
      }

      for (const [key, hand] of handEntries) {
        if (!hand) continue;
        const pressed = !!(hand.justSelected || hand.justPinched);

        // 1. Hand-poke on orb or screen (fingertip proximity)
        if (hand.indexTip) {
          if (minimized) {
            orbMesh.getWorldPosition(tmpVec);
            if (hand.indexTip.distanceTo(tmpVec) < 0.095) {
              if (pressed || hand.indexTip.distanceTo(tmpVec) < 0.065) {
                const now = performance.now();
                if (now - lastClickTime > 320) {
                  lastClickTime = now;
                  setMinimized(false);
                  return;
                }
              }
            }
          } else {
            // Check dragbar poke+pinch
            dragBar.getWorldPosition(tmpVec);
            if (hand.pinching && hand.indexTip.distanceTo(tmpVec) < 0.12) {
              draggingHandKey = key;
              dragOffset.copy(root.position).sub(hand.indexTip);
              return;
            }

            // Convert fingertip into screenMesh local space
            tmpLocal.copy(hand.indexTip);
            screenMesh.worldToLocal(tmpLocal);
            if (Math.abs(tmpLocal.z) < 0.045) {
              const u = tmpLocal.x / PANEL_W + 0.5;
              const v = tmpLocal.y / PANEL_H + 0.5;
              if (u >= 0 && u <= 1 && v >= 0 && v <= 1) {
                for (const z of hitZones) {
                  if (u >= z.u0 && u <= z.u1 && v >= z.v0 && v <= z.v1) {
                    newHover = z.id;
                    if (pressed || tmpLocal.z < 0.015) {
                      triggerZone(z, hand);
                    }
                    break;
                  }
                }
              }
            }
          }
        }

        // 2. Raycast laser interaction (VR controller OR desktop/mobile pointer)
        if (hand.rayOrigin && hand.rayDir) {
          raycaster.set(hand.rayOrigin, hand.rayDir);
          if (minimized) {
            const hits = raycaster.intersectObject(orbMesh, false);
            if (hits.length > 0 && (pressed || hand.justSelected)) {
              if (hand.pinching || hand.squeezing) {
                draggingHandKey = key;
                dragOffset.set(0, 0, 0);
              } else {
                setMinimized(false);
              }
              return;
            }
          } else {
            const dragHits = raycaster.intersectObject(dragBar, false);
            if (dragHits.length > 0 && (hand.selecting || hand.pinching)) {
              draggingHandKey = key;
              dragOffset.copy(root.position).sub(dragHits[0].point);
              return;
            }

            const hits = raycaster.intersectObject(screenMesh, false);
            if (hits.length > 0 && hits[0].uv) {
              const uv = hits[0].uv;
              for (const z of hitZones) {
                if (uv.x >= z.u0 && uv.x <= z.u1 && uv.y >= z.v0 && uv.y <= z.v1) {
                  newHover = z.id;
                  if (pressed) {
                    triggerZone(z, hand);
                  }
                  break;
                }
              }
            }
          }
        }
      }

      if (newHover !== hoveredId) {
        hoveredId = newHover;
        redraw();
      }
    }

    return {
      root,
      show,
      hide,
      toggle,
      minimize: () => setMinimized(true),
      expand: () => setMinimized(false),
      isVisible: () => root.visible,
      redraw,
      update: updateInteraction
    };
  })();

  function buildLineDiff(oldText, newText) {
    const a = String(oldText || '').split('\n');
    const b = String(newText || '').split('\n');
    const out = [];
    const maxLen = Math.max(a.length, b.length);
    let shown = 0;
    for (let i = 0; i < maxLen; i++) {
      if (a[i] !== b[i]) {
        if (a[i] !== undefined) out.push('- ' + (i + 1) + ': ' + a[i]);
        if (b[i] !== undefined) out.push('+ ' + (i + 1) + ': ' + b[i]);
        shown++;
        if (shown > 140) { out.push('... (diff truncated)'); break; }
      }
    }
    return out.length ? out.join('\n') : '(No text differences)';
  }

  function encodeBase64Utf8(str) {
    const bytes = new TextEncoder().encode(str);
    let bin = '';
    for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
    return btoa(bin);
  }

  function decodeBase64Utf8(b64) {
    const bin = atob(String(b64 || '').replace(/\s+/g, ''));
    const bytes = new Uint8Array(bin.length);
    for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
    return new TextDecoder().decode(bytes);
  }

  async function ghApiFetch(path, opts) {
    const cfg = getGhConfig();
    if (!cfg.owner || !cfg.repo || !cfg.pat) {
      throw new Error('Set GitHub Owner, Repo, and Personal Access Token (PAT) first.');
    }
    const url = 'https://api.github.com/repos/' + encodeURIComponent(cfg.owner) + '/' + encodeURIComponent(cfg.repo) + path;
    const res = await fetch(url, Object.assign({}, opts || {}, {
      headers: Object.assign({
        'Accept': 'application/vnd.github+json',
        'Authorization': 'Bearer ' + cfg.pat,
        'Content-Type': 'application/json'
      }, (opts && opts.headers) || {})
    }));
    const txt = await res.text();
    let json = {};
    try { json = JSON.parse(txt || '{}'); } catch (_) {}
    if (!res.ok) {
      throw new Error('GitHub API ' + res.status + ': ' + (json.message || txt));
    }
    return json;
  }

  // Commit directly to GitHub using Git Blobs + Trees + Commits + Refs API
  async function commitToGitHub() {
    saveGhConfig();
    const cfg = getGhConfig();
    const filename = ghModal.querySelector('#gh-file-select').value;
    const content = ghModal.querySelector('#gh-code-editor').value;
    const commitMsg = ghModal.querySelector('#gh-commit-msg').value.trim() || ('Update ' + filename + ' via WebXR Studio');

    setGhLog('Committing ' + filename + ' to ' + cfg.owner + '/' + cfg.repo + '@' + cfg.branch + '...');
    try {
      // 1. Get branch ref
      const refData = await ghApiFetch('/git/refs/heads/' + encodeURIComponent(cfg.branch));
      const headSha = refData.object.sha;

      // 2. Get head commit to obtain base tree SHA
      const headCommit = await ghApiFetch('/git/commits/' + headSha);
      const baseTreeSha = headCommit.tree.sha;

      // 3. Create new blob for file content
      const blob = await ghApiFetch('/git/blobs', {
        method: 'POST',
        body: JSON.stringify({ content: encodeBase64Utf8(content), encoding: 'base64' })
      });

      // 4. Create new tree referencing the blob
      const newTree = await ghApiFetch('/git/trees', {
        method: 'POST',
        body: JSON.stringify({
          base_tree: baseTreeSha,
          tree: [{ path: filename, mode: '100644', type: 'blob', sha: blob.sha }]
        })
      });

      // 5. Create new commit
      const newCommit = await ghApiFetch('/git/commits', {
        method: 'POST',
        body: JSON.stringify({
          message: commitMsg,
          tree: newTree.sha,
          parents: [headSha]
        })
      });

      // 6. Update branch ref
      await ghApiFetch('/git/refs/heads/' + encodeURIComponent(cfg.branch), {
        method: 'PATCH',
        body: JSON.stringify({ sha: newCommit.sha, force: false })
      });

      sandboxSnapshot = null;
      if (typeof APP.reloadModuleText === 'function') {
        await APP.reloadModuleText(filename, content);
      }
      setGhLog('Committed ' + filename + ' (' + newCommit.sha.slice(0, 7) + ').');
      loadRecentCommits();
    } catch (err) {
      setGhLog('Commit error: ' + (err.message || err));
    }
  }

    } catch (err) {

    }
  }

        setGhLog('Fetched latest ' + filename + ' from GitHub.');
      } catch (e) {
        setGhLog('Fetch error: ' + (e.message || e));
      }
    });
  }

  function openGitHubEditor(nlPrompt) {
    ensureGitHubModal();
    populateFileSelect();
    loadSelectedFileIntoEditor();
    if (nlPrompt) ghModal.querySelector('#gh-nl-prompt').value = String(nlPrompt);
    ghModal.classList.add('open');
  }

  // ====================================================================
  // 4. HOOKS & EXPORTS
  // ====================================================================
  buildHUD();

  if (typeof APP.onUpdate === 'function') {
    APP.onUpdate((dt, elapsed) => vrMenu.update(dt, elapsed));
  }

  if (typeof APP.on === 'function') {
    APP.on('modules:loaded', () => { renderPinChips(); syncHUDState(); });
    APP.on('ability:registered', () => { renderPinChips(); syncHUDState(); });
    APP.on('mode:changed', syncHUDState);
    APP.on('mic:changed', syncHUDState);
    APP.on('drawing:gravity_changed', syncHUDState);
    APP.on('bot:state_changed', syncHUDState);
    APP.on('visual:env_changed', syncHUDState);
    APP.on('ai:changed', syncHUDState);
    APP.on('status', setStatus);
    APP.on('xr:start', () => { vrMenu.show(true); });
  }

  APP.toolbar = {
    vrMenu,
    openGitHubEditor,
    renderPinChips,
    syncHUDState,
    setStatus
  };

  if (typeof APP.registerAbilityHandler === 'function') {
    APP.registerAbilityHandler('toolbar_hud_and_github', function (text) {
      const low = String(text || '').toLowerCase();
      if (low.includes('github') || low.includes('edit') || low.includes('جيتهب')) {
        openGitHubEditor();
        return { ok: true, message: 'Opened GitHub .txt Editor.' };
      }
      vrMenu.toggle();
      return { ok: true, message: 'Toggled 3D Spatial VR Menu.' };
    });
  }
})();
