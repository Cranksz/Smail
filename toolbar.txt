  function getPinnedList() {
    if (typeof APP.getPinnedAbilities === 'function') {
      return APP.getPinnedAbilities();
    }
    if (APP.abilities instanceof Map) {
      return Array.from(APP.abilities.values()).filter((a) => a && a.pin);
    }
    return [];
  }

  function renderPinChips() {
    if (!pinStrip) return;
    const pins = getPinnedList();
    pinStrip.innerHTML = '';
    if (!pins.length) {
      const span = document.createElement('span');
      span.style.cssText = 'font-size:11px;color:#94a3b8;';
      span.textContent = 'Pinned @pin abilities appear here automatically.';
      pinStrip.appendChild(span);
      return;
    }
    for (const ab of pins) {
      const chip = document.createElement('button');
      chip.type = 'button';
      chip.className = 'tb-chip';
      chip.textContent = ab.pinLabel || ab.label || ab.id;
      chip.title = `${ab.file || ''} — ${ab.matchRaw || ab.id}`;
      chip.addEventListener('click', () => {
        APP.runAbility?.(ab, ab.samplePhrase || ab.pinLabel || ab.id);
      });
      pinStrip.appendChild(chip);
    }
  }

  const vrMenu = (function buildVRMenu() {
    const root = new THREE.Group();
    root.name = 'APP_SPATIAL_VR_TOOLBAR';
    root.position.set(-0.42, 1.32, -0.92);
    root.visible = false;
    APP.scene.add(root);

    const PANEL_W = 0.72;
    const PANEL_H = 0.50;

    // Minimized glowing orb
    const orbGroup = new THREE.Group();
    orbGroup.visible = false;
    const orbMesh = new THREE.Mesh(
      new THREE.SphereGeometry(0.052, 24, 24),
      new THREE.MeshStandardMaterial({
        color: 0x00f0b4,
        emissive: 0x0ea5e9,
        emissiveIntensity: 0.95,
        roughness: 0.2,
        metalness: 0.4
      })
    );
    const orbRing = new THREE.Mesh(
      new THREE.TorusGeometry(0.072, 0.0045, 10, 36),
      new THREE.MeshBasicMaterial({ color: 0x6ef3d6, transparent: true, opacity: 0.85 })
    );
    orbGroup.add(orbMesh, orbRing);
    root.add(orbGroup);

    // Expanded panel
    const panelGroup = new THREE.Group();
    root.add(panelGroup);

    const backMesh = new THREE.Mesh(
      new THREE.BoxGeometry(PANEL_W + 0.024, PANEL_H + 0.024, 0.014),
      new THREE.MeshStandardMaterial({ color: 0x09111e, roughness: 0.45, metalness: 0.35 })
    );
    backMesh.position.z = -0.009;
    panelGroup.add(backMesh);

    const canvas = document.createElement('canvas');
    canvas.width = 1024;
    canvas.height = 712;
    const ctx = canvas.getContext('2d');
    const panelTex = new THREE.CanvasTexture(canvas);
    panelTex.colorSpace = THREE.SRGBColorSpace;
    panelTex.needsUpdate = true;

    const screenMesh = new THREE.Mesh(
      new THREE.PlaneGeometry(PANEL_W, PANEL_H),
      new THREE.MeshBasicMaterial({ map: panelTex, transparent: true })
    );
    panelGroup.add(screenMesh);

    const dragBar = new THREE.Mesh(
      new THREE.BoxGeometry(0.32, 0.028, 0.018),
      new THREE.MeshStandardMaterial({
        color: 0x00f0b4,
        emissive: 0x0ea5e9,
        emissiveIntensity: 0.65,
        roughness: 0.3
      })
    );
    dragBar.position.set(0, PANEL_H * 0.5 + 0.024, 0);
    panelGroup.add(dragBar);

    let minimized = false;
    let currentPage = 0;
    let hitZones = [];
    let hoveredId = null;
    let lastClickTime = 0;
    let draggingHandKey = null;
    const dragOffset = new THREE.Vector3();
    const tmpVec = new THREE.Vector3();
    const tmpLocal = new THREE.Vector3();

    function getPages() {
      const mode = APP.execMode || 'ai_instructions';
      const env = APP.visual ? APP.visual.getEnvironment() : 'default';
      const botOn = APP.bot ? !!APP.bot.isEnabled?.() : true;
      const pages = [{
        title: 'CORE CONTROLS (1)',
        items: [
          { id: 'm_ai', label: 'Mode: AI Only', active: mode === 'ai', action: () => APP.setExecMode?.('ai') },
          { id: 'm_in', label: 'Mode: Instructions', active: mode === 'instructions', action: () => APP.setExecMode?.('instructions') },
          { id: 'm_mx', label: 'Mode: AI + Inst', active: mode === 'ai_instructions', action: () => APP.setExecMode?.('ai_instructions') },
          { id: 'mic', label: 'Mic: ' + (APP.micActive ? 'ON' : 'OFF'), active: !!APP.micActive, action: () => APP.toggleMic?.() },
          { id: 'bot', label: 'Bot: ' + (botOn ? 'ON' : 'OFF'), active: botOn, action: () => APP.bot?.toggle?.() },
          { id: 'grav', label: 'Draw Gravity: ' + (APP.drawingGravity ? 'ON' : 'OFF'), active: !!APP.drawingGravity, action: () => APP.setDrawingGravity?.(!APP.drawingGravity) },
          { id: 'env', label: 'Env: ' + env, active: env !== 'default', action: () => APP.visual?.cycleEnvironment?.() },
          { id: 'clr', label: 'Clear Scene', warn: true, action: () => APP.clearAllEntities?.() }
        ]
      }];

      const pins = getPinnedList();
      const perPage = 6;
      const pCount = Math.max(1, Math.ceil(pins.length / perPage));
      for (let p = 0; p < pCount; p++) {
        const slice = pins.slice(p * perPage, (p + 1) * perPage);

          title: 'PINNED ABILITIES (' + (p + 2) + ')',

            label: ab.pinLabel || ab.label || ab.id,

          }))
        });
      }

    }

    function drawRoundedRect(x, y, w, h, r, fill, stroke) {
      ctx.beginPath();
      ctx.moveTo(x + r, y);
      ctx.arcTo(x + w, y, x + w, y + h, r);
      ctx.arcTo(x + w, y + h, x, y + h, r);
      ctx.arcTo(x, y + h, x, y, r);
      ctx.arcTo(x, y, x + w, y, r);
      ctx.closePath();
      if (fill) { ctx.fillStyle = fill; ctx.fill(); }
      if (stroke) { ctx.strokeStyle = stroke; ctx.lineWidth = 3; ctx.stroke(); }
    }

    function redraw() {
      const pages = getPages();
      if (currentPage >= pages.length) currentPage = 0;
      const page = pages[currentPage];
      hitZones = [];

      ctx.clearRect(0, 0, canvas.width, canvas.height);
      drawRoundedRect(6, 6, canvas.width - 12, canvas.height - 12, 28, '#0b1424', '#00f0b4');

      ctx.fillStyle = '#6ef3d6';
      ctx.font = 'bold 30px system-ui, sans-serif';
      ctx.textAlign = 'left';
      ctx.textBaseline = 'middle';
      ctx.fillText(page.title, 34, 54);

      const minRect = { x: canvas.width - 190, y: 24, w: 158, h: 54 };
      drawRoundedRect(minRect.x, minRect.y, minRect.w, minRect.h, 14, hoveredId === 'minimize' ? '#254a78' : '#162c49', '#5eead4');
      ctx.fillStyle = '#e0f2fe';
      ctx.font = 'bold 22px system-ui, sans-serif';
      ctx.textAlign = 'center';
      ctx.fillText('Minimize', minRect.x + minRect.w / 2, minRect.y + minRect.h / 2);
      hitZones.push({
        id: 'minimize',
        u0: minRect.x / canvas.width,
        u1: (minRect.x + minRect.w) / canvas.width,
        v0: 1 - (minRect.y + minRect.h) / canvas.height,
        v1: 1 - minRect.y / canvas.height,
        action: () => setMinimized(true)
      });

      const cols = 2, padX = 34, startY = 104, gapX = 22, gapY = 20;
      const btnW = (canvas.width - padX * 2 - gapX) / cols;
      const btnH = 112;

      if (!page.items.length) {
        ctx.fillStyle = '#89a7c9';
        ctx.font = '26px system-ui, sans-serif';
        ctx.textAlign = 'center';
        ctx.fillText('No @pin abilities loaded yet.', canvas.width / 2, 330);
      } else {
        page.items.forEach((item, idx) => {
          const col = idx % cols;
          const row = Math.floor(idx / cols);
          const bx = padX + col * (btnW + gapX);
          const by = startY + row * (btnH + gapY);
          const isHover = hoveredId === item.id;
          let bg = '#15253e', border = '#335785';
          if (item.active) { bg = '#0f766e'; border = '#00f0b4'; }
          if (item.warn) { bg = '#611f2b'; border = '#f87171'; }
          if (isHover) { bg = item.warn ? '#852938' : '#244675'; border = '#7dd3fc'; }

          drawRoundedRect(bx, by, btnW, btnH, 18, bg, border);
          ctx.fillStyle = '#f0f7ff';
          ctx.font = 'bold 28px system-ui, sans-serif';
          ctx.textAlign = 'center';
          ctx.fillText(String(item.label).slice(0, 22), bx + btnW / 2, by + btnH / 2);

          hitZones.push({
            id: item.id,
            u0: bx / canvas.width,
            u1: (bx + btnW) / canvas.width,
            v0: 1 - (by + btnH) / canvas.height,
            v1: 1 - by / canvas.height,
            action: item.action
          });
        });
      }

      const footY = canvas.height - 82;
      const prevRect = { x: 34, y: footY, w: 180, h: 56 };
      const nextRect = { x: canvas.width - 214, y: footY, w: 180, h: 56 };

      drawRoundedRect(prevRect.x, prevRect.y, prevRect.w, prevRect.h, 14, hoveredId === 'prev' ? '#234672' : '#13243b', '#3b679e');
      drawRoundedRect(nextRect.x, nextRect.y, nextRect.w, nextRect.h, 14, hoveredId === 'next' ? '#234672' : '#13243b', '#3b679e');

      ctx.fillStyle = '#dcecff';
      ctx.font = 'bold 24px system-ui, sans-serif';
      ctx.textAlign = 'center';
      ctx.fillText('< Prev', prevRect.x + prevRect.w / 2, prevRect.y + prevRect.h / 2);
      ctx.fillText('Next >', nextRect.x + nextRect.w / 2, nextRect.y + nextRect.h / 2);
      ctx.fillStyle = '#8eb0d6';
      ctx.fillText('Page ' + (currentPage + 1) + ' / ' + pages.length, canvas.width / 2, footY + 28);

      hitZones.push({
        id: 'prev',
        u0: prevRect.x / canvas.width,
        u1: (prevRect.x + prevRect.w) / canvas.width,
        v0: 1 - (prevRect.y + prevRect.h) / canvas.height,
        v1: 1 - prevRect.y / canvas.height,
        action: () => { currentPage = (currentPage - 1 + pages.length) % pages.length; }
      });
      hitZones.push({
        id: 'next',
        u0: nextRect.x / canvas.width,
        u1: (nextRect.x + nextRect.w) / canvas.width,
        v0: 1 - (nextRect.y + nextRect.h) / canvas.height,
        v1: 1 - nextRect.y / canvas.height,
        action: () => { currentPage = (currentPage + 1) % pages.length; }
      });

      panelTex.needsUpdate = true;
    }

    function setMinimized(val) {
      minimized = !!val;
      orbGroup.visible = minimized;
      panelGroup.visible = !minimized;
      if (!minimized) redraw();
    }

    function triggerZone(zone, hand) {
      const now = performance.now();
      if (now - lastClickTime < 280) return;
      lastClickTime = now;
      try {
        const gp = hand && hand.gamepad;
        if (gp?.hapticActuators?.[0]) gp.hapticActuators[0].pulse(0.35, 30);
      } catch (_) {}
      try { zone.action(); } catch (e) { console.warn('[vrMenu]', e); }
      syncState();
      redraw();
    }

    // Hook into index.html's APP.pokeTargets raycast clicker
    screenMesh.userData.onPoke = function (hit) {
      if (!root.visible || minimized || !hit || !hit.uv) return;
      const uv = hit.uv;
      for (const z of hitZones) {
        if (uv.x >= z.u0 && uv.x <= z.u1 && uv.y >= z.v0 && uv.y <= z.v1) {
          triggerZone(z, null);
          break;
        }
      }
    };
    orbMesh.userData.onPoke = function () {
      if (!root.visible || !minimized) return;
      const now = performance.now();
      if (now - lastClickTime < 280) return;
      lastClickTime = now;
      setMinimized(false);
    };
    dragBar.userData.onPoke = function () {
      // Handled smoothly in per-frame drag loop
    };

    if (APP.pokeTargets) {
      APP.pokeTargets.add(screenMesh);
      APP.pokeTargets.add(orbMesh);
      APP.pokeTargets.add(dragBar);
    }

    function placeInFrontOfUser() {
      const hp = APP.headPos || new THREE.Vector3(0, 1.5, 0);
      const hf = APP.headFwd || new THREE.Vector3(0, 0, -1);
      const flat = new THREE.Vector3(hf.x, 0, hf.z).normalize();
      if (flat.lengthSq() < 0.01) flat.set(0, 0, -1);
      const right = new THREE.Vector3().crossVectors(flat, new THREE.Vector3(0, 1, 0)).normalize();
      root.position.copy(hp).addScaledVector(flat, 0.85).addScaledVector(right, -0.30);
      root.position.y = Math.max(0.88, hp.y - 0.12);
      root.lookAt(hp.x, root.position.y, hp.z);
    }

    function show(placeFresh) {
      root.visible = true;
      if (placeFresh !== false) placeInFrontOfUser();
      redraw();
    }

    function updateInteraction(dt, elapsed) {
      if (!root.visible) return;
      if (minimized) {
        orbRing.rotation.z = elapsed * 1.4;
        orbRing.rotation.x = Math.sin(elapsed * 0.9) * 0.4;
      }

      const hands = APP.hands || {};
      if (draggingHandKey && hands[draggingHandKey]) {
        const dh = hands[draggingHandKey];
        if (dh.active && (dh.trigger || dh.pinching || dh.gripping)) {
          if (dh.isHand && dh.fingerTips?.index) {
            root.position.copy(dh.fingerTips.index).add(dragOffset);
          } else if (dh.rayOrigin && dh.rayDir) {
            root.position.copy(dh.rayOrigin).addScaledVector(dh.rayDir, 0.85).add(dragOffset);
          }
          if (APP.headPos) root.lookAt(APP.headPos.x, root.position.y, APP.headPos.z);
          return;
        } else {
          draggingHandKey = null;
        }
      }

      let newHover = null;
      const raycaster = new THREE.Raycaster();

      for (const [key, hand] of Object.entries(hands)) {
        if (!hand || !hand.active) continue;
        const tip = hand.isHand ? hand.fingerTips?.index : hand.pos;

        // Fingertip hand-poke & dragbar pinch
        if (hand.isHand && tip) {
          if (minimized) {
            orbMesh.getWorldPosition(tmpVec);
            if (tip.distanceTo(tmpVec) < 0.085) {
              const now = performance.now();
              if (now - lastClickTime > 320) {
                lastClickTime = now;
                setMinimized(false);
                return;
              }
            }
          } else {
            dragBar.getWorldPosition(tmpVec);
            if (hand.pinching && tip.distanceTo(tmpVec) < 0.12) {
              draggingHandKey = key;
              dragOffset.copy(root.position).sub(tip);
              return;
            }
            tmpLocal.copy(tip);
            screenMesh.worldToLocal(tmpLocal);
            if (Math.abs(tmpLocal.z) < 0.042) {
              const u = tmpLocal.x / PANEL_W + 0.5;
              const v = tmpLocal.y / PANEL_H + 0.5;
              if (u >= 0 && u <= 1 && v >= 0 && v <= 1) {
                for (const z of hitZones) {
                  if (u >= z.u0 && u <= z.u1 && v >= z.v0 && v <= z.v1) {
                    newHover = z.id;
                    if (hand.justPinched || tmpLocal.z < 0.015) triggerZone(z, hand);
                    break;
                  }
                }
              }
            }
          }
        }

        // Laser dragbar & hover preview
        if (hand.rayOrigin && hand.rayDir) {
          raycaster.set(hand.rayOrigin, hand.rayDir);
          if (!minimized) {
            const dragHits = raycaster.intersectObject(dragBar, false);
            if (dragHits.length && (hand.trigger || hand.gripping)) {
              draggingHandKey = key;
              dragOffset.copy(root.position).sub(dragHits[0].point);
              return;
            }
            const scrHits = raycaster.intersectObject(screenMesh, false);
            if (scrHits.length && scrHits[0].uv) {
              const uv = scrHits[0].uv;
              for (const z of hitZones) {
                if (uv.x >= z.u0 && uv.x <= z.u1 && uv.y >= z.v0 && uv.y <= z.v1) {
                  newHover = z.id;
                  break;
                }
              }
            }
          }
        }
      }

      if (newHover !== hoveredId) {
        hoveredId = newHover;
        redraw();
      }
    }

    return {
      root,
      screenMesh,
      orbMesh,
      dragBar,
      show,
      hide: () => { root.visible = false; },
      toggle: () => { if (root.visible) root.visible = false; else show(true); },
      minimize: () => setMinimized(true),
      expand: () => setMinimized(false),
      isVisible: () => root.visible,
      redraw,
      update: updateInteraction
    };
  })();

  async function ghApiFetch(path, opts) {
    const cfg = getRepoCfg();
    if (!cfg.owner || !cfg.repo || !cfg.pat) {
      throw new Error('Enter GitHub Owner, Repo Name, and Personal Access Token (PAT) first.');
    }
    const url = `https://api.github.com/repos/${encodeURIComponent(cfg.owner)}/${encodeURIComponent(cfg.repo)}${path}`;
    const res = await fetch(url, Object.assign({}, opts || {}, {
      headers: Object.assign({
        'Accept': 'application/vnd.github+json',
        'Authorization': 'Bearer ' + cfg.pat,
        'Content-Type': 'application/json'
      }, (opts && opts.headers) || {})
    }));
    const txt = await res.text();
    let json = {};
    try { json = JSON.parse(txt || '{}'); } catch (_) {}
    if (!res.ok) throw new Error(`GitHub ${res.status}: ${json.message || txt}`);
    return json;
  }

  function encodeB64(str) {
    const bytes = new TextEncoder().encode(str);
    let bin = '';
    for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
    return btoa(bin);
  }

  function decodeB64(b64) {
    const bin = atob(String(b64 || '').replace(/\s+/g, ''));
    const bytes = new Uint8Array(bin.length);
    for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
    return new TextDecoder().decode(bytes);
  }

  function buildLineDiff(aStr, bStr) {
    const a = String(aStr || '').split('\n');
    const b = String(bStr || '').split('\n');
    const out = [];
    const maxLen = Math.max(a.length, b.length);
    for (let i = 0; i < maxLen; i++) {
      if (a[i] !== b[i]) {
        if (a[i] !== undefined) out.push('- ' + (i + 1) + ': ' + a[i]);
        if (b[i] !== undefined) out.push('+ ' + (i + 1) + ': ' + b[i]);
        if (out.length > 140) { out.push('... (truncated)'); break; }
      }
    }
    return out.length ? out.join('\n') : '(No differences)';
  }
