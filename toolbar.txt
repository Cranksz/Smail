    if (typeof APP.getPinnedAbilities === 'function') return APP.getPinnedAbilities();

  function renderPinChips() {
    if (!pinStrip) return;
    const pins = getPinnedList();
    pinStrip.innerHTML = '';
    if (!pins.length) {
      const span = document.createElement('span');
      span.style.cssText = 'font-size:11px;color:#94a3b8;';
      span.textContent = 'Pinned @pin abilities appear here automatically.';
      pinStrip.appendChild(span);
      return;
    }
    for (const ab of pins) {
      const chip = document.createElement('button');
      chip.type = 'button';
      chip.className = 'tb-chip';
      chip.textContent = ab.pinLabel || ab.label || ab.id;
      chip.title = `${ab.file || ''} — ${ab.matchRaw || ab.id}`;
      chip.addEventListener('click', () => {
        APP.runAbility?.(ab, ab.samplePhrase || ab.pinLabel || ab.id);
      });
      pinStrip.appendChild(chip);
    }

    function getPages() {
      const w = typeof APP.instWeight === 'number' ? APP.instWeight : 50;

    function drawRoundedRect(x, y, w, h, r, fill, stroke) {
      ctx.beginPath();
      ctx.moveTo(x + r, y);
      ctx.arcTo(x + w, y, x + w, y + h, r);
      ctx.arcTo(x + w, y + h, x, y + h, r);
      ctx.arcTo(x, y + h, x, y, r);
      ctx.arcTo(x, y, x + w, y, r);
      ctx.closePath();
      if (fill) { ctx.fillStyle = fill; ctx.fill(); }
      if (stroke) { ctx.strokeStyle = stroke; ctx.lineWidth = 3; ctx.stroke(); }
    }

    function redraw() {
      const pages = getPages();
      if (currentPage >= pages.length) currentPage = 0;
      const page = pages[currentPage];
      hitZones = [];

      ctx.clearRect(0, 0, canvas.width, canvas.height);
      drawRoundedRect(6, 6, canvas.width - 12, canvas.height - 12, 28, '#0b1424', '#00f0b4');

      ctx.fillStyle = '#6ef3d6';
      ctx.font = 'bold 30px system-ui, sans-serif';
      ctx.textAlign = 'left';
      ctx.textBaseline = 'middle';
      ctx.fillText(page.title, 34, 52);

      const minRect = { x: canvas.width - 190, y: 22, w: 158, h: 52 };
      drawRoundedRect(minRect.x, minRect.y, minRect.w, minRect.h, 14, hoveredId === 'minimize' ? '#254a78' : '#162c49', '#5eead4');
      ctx.fillStyle = '#e0f2fe';
      ctx.font = 'bold 22px system-ui, sans-serif';
      ctx.textAlign = 'center';
      ctx.fillText('Minimize', minRect.x + minRect.w / 2, minRect.y + minRect.h / 2);
      hitZones.push({
        id: 'minimize',
        u0: minRect.x / canvas.width, u1: (minRect.x + minRect.w) / canvas.width,
        v0: 1 - (minRect.y + minRect.h) / canvas.height, v1: 1 - minRect.y / canvas.height,
        action: () => setMinimized(true)
      });

      let startY = 104;
      if (page.showBlendSlider) {
        const w = typeof APP.instWeight === 'number' ? APP.instWeight : 50;
        const barX = 34, barY = 92, barW = canvas.width - 68, barH = 44;
        drawRoundedRect(barX, barY, barW, barH, 14, '#091220', '#2b4d78');
        const fillW = Math.max(24, (barW - 12) * (w / 100));
        const grad = ctx.createLinearGradient(barX, 0, barX + barW, 0);
        grad.addColorStop(0, '#38bdf8');
        grad.addColorStop(1, '#00f0b4');
        drawRoundedRect(barX + 6, barY + 6, fillW, barH - 12, 10, grad, null);

        ctx.fillStyle = '#ffffff';
        ctx.font = 'bold 22px system-ui, sans-serif';
        ctx.textAlign = 'left';
        ctx.fillText('AI (100%)', barX + 18, barY + barH / 2);
        ctx.textAlign = 'center';
        ctx.fillText(APP.formatBlendLabel ? APP.formatBlendLabel(w) : `${100 - w}% AI · ${w}% Inst`, canvas.width / 2, barY + barH / 2);
        ctx.textAlign = 'right';
        ctx.fillText('Inst (100%)', barX + barW - 18, barY + barH / 2);

        // Clicking directly along the 3D slider bar sets the exact percentage!
        hitZones.push({
          id: 'vr_blend_bar',
          u0: barX / canvas.width, u1: (barX + barW) / canvas.width,
          v0: 1 - (barY + barH) / canvas.height, v1: 1 - barY / canvas.height,

            const rel = Math.max(0, Math.min(1, (uHit - barX / canvas.width) / (barW / canvas.width)));
            APP.setInstWeight?.(Math.round((rel * 100) / 5) * 5);
          }

        startY = 152;
      }

        id: 'prev', u0: prevRect.x / canvas.width, u1: (prevRect.x + prevRect.w) / canvas.width,
        v0: 1 - (prevRect.y + prevRect.h) / canvas.height, v1: 1 - prevRect.y / canvas.height,

    function setMinimized(val) {
      minimized = !!val;
      orbGroup.visible = minimized;
      panelGroup.visible = !minimized;
      if (!minimized) redraw();
    }

    function triggerZone(zone, hand, uHit) {
      const now = performance.now();
      if (now - lastClickTime < 260) return;
      lastClickTime = now;
      try {
        const gp = hand && hand.gamepad;
        if (gp?.hapticActuators?.[0]) gp.hapticActuators[0].pulse(0.35, 30);
      } catch (_) {}
      try { zone.action(uHit); } catch (e) { console.warn('[vrMenu]', e); }
      syncState();
      redraw();
    }

    screenMesh.userData.onPoke = function (hit) {
      if (!root.visible || minimized || !hit || !hit.uv) return;
      const uv = hit.uv;
      for (const z of hitZones) {
        if (uv.x >= z.u0 && uv.x <= z.u1 && uv.y >= z.v0 && uv.y <= z.v1) {
          triggerZone(z, null, uv.x);
          break;
        }
      }
    };
    orbMesh.userData.onPoke = function () {
      if (!root.visible || !minimized) return;
      const now = performance.now();
      if (now - lastClickTime < 280) return;
      lastClickTime = now;
      setMinimized(false);
    };

    if (APP.pokeTargets) {
      APP.pokeTargets.add(screenMesh);
      APP.pokeTargets.add(orbMesh);
      APP.pokeTargets.add(dragBar);
    }

    function placeInFrontOfUser() {
      const hp = APP.headPos || new THREE.Vector3(0, 1.5, 0);
      const hf = APP.headFwd || new THREE.Vector3(0, 0, -1);
      const flat = new THREE.Vector3(hf.x, 0, hf.z).normalize();
      if (flat.lengthSq() < 0.01) flat.set(0, 0, -1);
      const right = new THREE.Vector3().crossVectors(flat, new THREE.Vector3(0, 1, 0)).normalize();
      root.position.copy(hp).addScaledVector(flat, 0.85).addScaledVector(right, -0.30);
      root.position.y = Math.max(0.88, hp.y - 0.12);
      root.lookAt(hp.x, root.position.y, hp.z);
    }

    function show(placeFresh) {
      root.visible = true;
      if (placeFresh !== false) placeInFrontOfUser();
      redraw();
    }

    function updateInteraction(dt, elapsed) {
      if (!root.visible) return;
      if (minimized) {
        orbRing.rotation.z = elapsed * 1.4;
        orbRing.rotation.x = Math.sin(elapsed * 0.9) * 0.4;
      }

      const hands = APP.hands || {};
      if (draggingHandKey && hands[draggingHandKey]) {
        const dh = hands[draggingHandKey];
        if (dh.active && (dh.trigger || dh.pinching || dh.gripping)) {
          if (dh.isHand && dh.fingerTips?.index) root.position.copy(dh.fingerTips.index).add(dragOffset);
          else if (dh.rayOrigin && dh.rayDir) root.position.copy(dh.rayOrigin).addScaledVector(dh.rayDir, 0.85).add(dragOffset);
          if (APP.headPos) root.lookAt(APP.headPos.x, root.position.y, APP.headPos.z);
          return;
        } else {
          draggingHandKey = null;
        }
      }

      let newHover = null;
      const raycaster = new THREE.Raycaster();

      for (const [key, hand] of Object.entries(hands)) {
        if (!hand || !hand.active) continue;
        const tip = hand.isHand ? hand.fingerTips?.index : hand.pos;

        if (hand.isHand && tip) {
          if (minimized) {
            orbMesh.getWorldPosition(tmpVec);
            if (tip.distanceTo(tmpVec) < 0.085) {
              const now = performance.now();
              if (now - lastClickTime > 320) { lastClickTime = now; setMinimized(false); return; }
            }
          } else {
            dragBar.getWorldPosition(tmpVec);
            if (hand.pinching && tip.distanceTo(tmpVec) < 0.12) {
              draggingHandKey = key;
              dragOffset.copy(root.position).sub(tip);
              return;
            }
            tmpLocal.copy(tip);
            screenMesh.worldToLocal(tmpLocal);
            if (Math.abs(tmpLocal.z) < 0.042) {
              const u = tmpLocal.x / PANEL_W + 0.5;
              const v = tmpLocal.y / PANEL_H + 0.5;
              if (u >= 0 && u <= 1 && v >= 0 && v <= 1) {
                for (const z of hitZones) {
                  if (u >= z.u0 && u <= z.u1 && v >= z.v0 && v <= z.v1) {
                    newHover = z.id;
                    if (hand.justPinched || tmpLocal.z < 0.015) triggerZone(z, hand, u);
                    break;
                  }
                }
              }
            }
          }
        }

        if (hand.rayOrigin && hand.rayDir) {
          raycaster.set(hand.rayOrigin, hand.rayDir);
          if (!minimized) {
            const dragHits = raycaster.intersectObject(dragBar, false);
            if (dragHits.length && (hand.trigger || hand.gripping)) {
              draggingHandKey = key;
              dragOffset.copy(root.position).sub(dragHits[0].point);
              return;
            }
            const scrHits = raycaster.intersectObject(screenMesh, false);
            if (scrHits.length && scrHits[0].uv) {
              const uv = scrHits[0].uv;
              for (const z of hitZones) {
                if (uv.x >= z.u0 && uv.x <= z.u1 && uv.y >= z.v0 && uv.y <= z.v1) {
                  newHover = z.id;
                  break;
                }
              }
            }
          }
        }
      }

      if (newHover !== hoveredId) {
        hoveredId = newHover;
        redraw();
      }
    }

    return {
      root, screenMesh, orbMesh, dragBar, show,
      hide: () => { root.visible = false; },
      toggle: () => { if (root.visible) root.visible = false; else show(true); },
      minimize: () => setMinimized(true),
      expand: () => setMinimized(false),
      isVisible: () => root.visible,
      redraw,
      update: updateInteraction
    };
  })();

  async function ghApiFetch(path, opts) {
    const cfg = getRepoCfg();
    if (!cfg.owner || !cfg.repo || !cfg.pat) throw new Error('Enter GitHub Owner, Repo Name, and PAT first.');
    const url = `https://api.github.com/repos/${encodeURIComponent(cfg.owner)}/${encodeURIComponent(cfg.repo)}${path}`;
    const res = await fetch(url, Object.assign({}, opts || {}, {
      headers: Object.assign({
        'Accept': 'application/vnd.github+json',
        'Authorization': 'Bearer ' + cfg.pat,
        'Content-Type': 'application/json'
      }, (opts && opts.headers) || {})
    }));
    const txt = await res.text();
    let json = {};
    try { json = JSON.parse(txt || '{}'); } catch (_) {}
    if (!res.ok) throw new Error(`GitHub ${res.status}: ${json.message || txt}`);
    return json;
  }

  function encodeB64(str) {
    const bytes = new TextEncoder().encode(str);
    let bin = '';
    for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
    return btoa(bin);
  }

  function decodeB64(b64) {
    const bin = atob(String(b64 || '').replace(/\s+/g, ''));
    const bytes = new Uint8Array(bin.length);
    for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
    return new TextDecoder().decode(bytes);
  }

  function buildLineDiff(aStr, bStr) {
    const a = String(aStr || '').split('\n');
    const b = String(bStr || '').split('\n');
    const out = [];
    const maxLen = Math.max(a.length, b.length);
    for (let i = 0; i < maxLen; i++) {
      if (a[i] !== b[i]) {
        if (a[i] !== undefined) out.push('- ' + (i + 1) + ': ' + a[i]);
        if (b[i] !== undefined) out.push('+ ' + (i + 1) + ': ' + b[i]);
        if (out.length > 140) { out.push('... (truncated)'); break; }
      }
    }
    return out.length ? out.join('\n') : '(No differences)';
  }

  function stripOuterCodeFences(raw) {
    let s = String(raw || '').trim();
    const first = s.indexOf('```');
    const last = s.lastIndexOf('```');
    if (first !== -1 && last > first) {
      const inner = s.slice(first + 3, last).replace(/^[a-zA-Z0-9_-]*\r?\n/, '');
      if (inner.includes('@ability:') || inner.includes('<<<')) return inner.trim();
    }
    return s;
  }

  function applyAIModuleOutput(rawResponse, currentText, filename) {
    const clean = stripOuterCodeFences(rawResponse);

    if (clean.includes('<<<APPEND_ABILITY')) {
      const blockMatch = clean.match(/<<<APPEND_ABILITY\s*([\s\S]*?)<<<END_ABILITY/i);
      const block = blockMatch ? blockMatch[1].trim() : clean.replace(/<<<APPEND_ABILITY|<<<END_ABILITY/gi, '').trim();
      const parsedNew = APP.parseModuleText ? APP.parseModuleText(filename, block) : [];
      let merged = String(currentText || '').trim();

      if (parsedNew.length && APP.parseModuleText) {
        const existing = APP.parseModuleText(filename, merged);
        for (const ab of parsedNew) {
          const idEsc = ab.id.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
          const re = new RegExp(`@ability:\\s*${idEsc}[\\s\\S]*?@end`, 'm');
          const singleBlockMatch = block.match(new RegExp(`@ability:\\s*${idEsc}[\\s\\S]*?@end`, 'm'));
          if (existing.some((e) => e.id === ab.id) && singleBlockMatch) {
            merged = merged.replace(re, singleBlockMatch[0]);
          } else {
            merged = (merged ? merged + '\n\n' : '') + (singleBlockMatch ? singleBlockMatch[0] : block);
          }
        }
        return merged.trim();
      }
      return (merged ? merged + '\n\n' : '') + block;
    }

    if (clean.includes('<<<SEARCH') && clean.includes('<<<REPLACE')) {
      let updated = String(currentText || '');
      const patchRegex = /<<<SEARCH\r?\n([\s\S]*?)\r?\n<<<REPLACE\r?\n([\s\S]*?)\r?\n<<<END_PATCH/g;
      let match, applied = 0;
      while ((match = patchRegex.exec(clean)) !== null) {
        if (updated.includes(match[1])) { updated = updated.replace(match[1], match[2]); applied++; }
      }
      if (applied > 0) return updated;
    }

    const full = clean.replace(/^<<<FULL_FILE\s*/i, '').replace(/<<<END_FILE\s*$/i, '').trim();
    if (APP.parseModuleText && currentText) {
      const curAbilities = APP.parseModuleText(filename, currentText);
      const newAbilities = APP.parseModuleText(filename, full);
      if (curAbilities.length >= 2 && newAbilities.length === 1 && !curAbilities.some((a) => a.id === newAbilities[0].id)) {
        return currentText.trim() + '\n\n' + full;
      }
    }
    return full;
  }

  function validateModuleSyntax(filename, rawText) {
    if (typeof APP.parseModuleText !== 'function') return { ok: true };
    const abilities = APP.parseModuleText(filename, rawText);
    if (!abilities.length) return { ok: false, error: 'Parsed 0 @ability blocks. Ensure @ability: <id>, @code:, and @end tags are present.' };
    const AsyncFn = Object.getPrototypeOf(async function () {}).constructor;
    for (const ab of abilities) {
      try { new AsyncFn('APP', 'THREE', 'command', 'ability', ab.code); }
      catch (e) { return { ok: false, error: `JS SyntaxError inside @ability: ${ab.id}: ${e.message}` }; }
    }
    return { ok: true, count: abilities.length };
  }

  function buildFullAppContextForEdit(targetFilename, requestText) {
    const modEntries = Object.entries(APP.modules || {});
    const moduleIndex = modEntries.map(([fn, txt]) => {
      const abs = APP.parseModuleText ? APP.parseModuleText(fn, txt) : [];
      return `- ${fn} (${txt.length} chars): abilities=[${abs.map(a => a.id).join(', ')}]`;
    }).join('\n');

    const archGuide = [
      'NEXUSXR ARCHITECTURE & GLOBAL HOOKS:',
      '- Shared: `APP`, `THREE`, `APP.scene`, `APP.camera`, `APP.renderer`, `APP.fx` (`water`, `rain`, `snow`, `lavaFloor`).',
      '- Update loop: `APP.onUpdate((dt, elapsed) => { ... })` or `APP.frameHooks.add(fn)`.',
      '- Entities: `APP.registerEntity(group, { name, category, holdable, hasGravity, onUpdate })`.',
      '- Geeny (`bot.txt`): `APP.bot.root`, `APP.bot.bodyGroup`, `APP.bot.getVelocity()`, `APP.bot.showBoard(...)`.',
      '- Module syntax: `@ability: <id>`, `@autorun`, `@pin: <Label>`, `@desc: ...`, `@match: ...`, `@code: ... @end`.',
      '- Use `#` ONLY outside `@code:`; inside `@code:`, use `//` ONLY.'
    ].join('\n');

    const extraFiles = [];
    for (const [fn, txt] of modEntries) {
      if (fn === targetFilename) continue;
      if (fn === 'bot.txt' && /bot|geeny|genie|cape|tentacle|board|visor|wing/i.test(requestText)) {
        extraFiles.push(`=== REFERENCE FILE: ${fn} ===\n${txt}`);
      } else if (fn === 'visual.txt' && /visual|shader|matrix|360|environment|dissolve/i.test(requestText)) {
        extraFiles.push(`=== REFERENCE FILE: ${fn} ===\n${txt}`);
      }
    }
    return [archGuide, '', 'ALL LOADED .TXT MODULES:', moduleIndex || '(none)', extraFiles.length ? '\n' + extraFiles.join('\n\n') : ''].join('\n');
  }

    return currentFile;
  }

    }

    const sel = mount.querySelector('#gh-file-sel');
    const prev = sel.value;
    const files = Array.from(new Set([...Object.keys(APP.modules || {}), 'ai_models.txt', 'visual.txt', 'toolbar.txt', 'bot.txt', 'instruction1.txt']));
    sel.innerHTML = '';
    for (const f of files) { const opt = document.createElement('option'); opt.value = f; opt.textContent = f; sel.appendChild(opt); }
    if (prev && files.includes(prev)) sel.value = prev;
    else if (files.includes('instruction1.txt')) sel.value = 'instruction1.txt';
    mount.querySelector('#gh-code-ta').value = APP.modules?.[sel.value] || '';
  }

  function openGitHubEditor(nlPrompt) {
    buildEditorUI();
    if (nlPrompt) {
      const inp = document.querySelector('#modal-editor-body #gh-nl-prompt');
      if (inp) inp.value = String(nlPrompt);
    }
    APP.ui.openModal('modal-editor');
  }

  function refresh() { renderPinChips(); syncState(); }

  build2DHUD();
  refresh();

  const unsubFrame = APP.onUpdate ? APP.onUpdate((dt, elapsed) => vrMenu.update(dt, elapsed)) : null;
  if (!unsubFrame && APP.frameHooks) APP.frameHooks.add(vrMenu.update);

  const unsubs = [
    APP.on?.('modules:loaded', refresh),
    APP.on?.('ability:registered', refresh),
    APP.on?.('mode:changed', syncState),
    APP.on?.('instWeight:changed', syncState),
    APP.on?.('mic:changed', syncState),
    APP.on?.('drawing:gravity_changed', syncState),
    APP.on?.('bot:state_changed', syncState),
    APP.on?.('visual:env_changed', syncState),
    APP.on?.('ai:changed', syncState),
    APP.on?.('xr:start', () => vrMenu.show(true))
  ].filter(Boolean);

  APP._toolbarCleanup = function () {
    if (typeof unsubFrame === 'function') unsubFrame();
    else APP.frameHooks?.delete(vrMenu.update);
    unsubs.forEach((u) => { try { u(); } catch (_) {} });
    if (APP.pokeTargets) {
      APP.pokeTargets.delete(vrMenu.screenMesh);
      APP.pokeTargets.delete(vrMenu.orbMesh);
      APP.pokeTargets.delete(vrMenu.dragBar);
    }
    if (vrMenu.root.parent) vrMenu.root.parent.remove(vrMenu.root);
  };

  APP.toolbar = {
    vrMenu, refresh, syncState, stripOuterCodeFences, applyAIModuleOutput,
    validateModuleSyntax, stageEditInEditor, openGitHubEditor
  };

  APP.registerAbilityHandler?.('toolbar_hud_and_github', function (text) {
    const low = String(text || '').toLowerCase();
    if (low.includes('github') || low.includes('edit') || low.includes('جيتهب')) { openGitHubEditor(); return { ok: true }; }
    vrMenu.toggle();
    return { ok: true };
  });
})();
