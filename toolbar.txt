  const rawInputState = new Map();
  const releasedFrames = new Map();

  function getRawHandState(key, hand) {
    const prev = rawInputState.get(key) || { trig: false, grip: false, pinch: false };
    let trig = !!hand.trigger;
    let grip = !!hand.gripping;
    let pinch = !!hand.pinching;

    // Read raw WebXR gamepad buttons when available so stuck flags cannot latch
    const btns = hand.gamepad && hand.gamepad.buttons;
    if (btns && btns.length > 0) {
      const b0 = btns[0];
      const b1 = btns[1];
      trig = !!(b0 && (b0.pressed || b0.value > 0.35));
      if (b1) grip = !!(b1.pressed || b1.value > 0.35);
      else grip = false;
    }

    // Read raw thumb-to-index tip distance for hand tracking
    if (hand.isHand && hand.fingerTips?.index && hand.fingerTips?.thumb) {
      const d = hand.fingerTips.index.distanceTo(hand.fingerTips.thumb);
      if (d > 0.036) pinch = false;
      else if (d < 0.021) pinch = true;
    }

    const justTrig = (trig && !prev.trig) || !!hand.justTriggered;
    const justGrip = (grip && !prev.grip) || !!hand.justGripped;
    const justPinch = (pinch && !prev.pinch) || !!hand.justPinched;
    rawInputState.set(key, { trig, grip, pinch });

    return { trig, grip, pinch, justTrig, justGrip, justPinch };
  }

  function releaseStuckGrabs(forceAll) {
    const hands = APP.hands || {};
    for (const [key, hand] of Object.entries(hands)) {
      if (!hand) continue;
      const raw = getRawHandState(key, hand);
      const anyHeld = !forceAll && (raw.trig || raw.grip || raw.pinch);
      if (anyHeld) {
        releasedFrames.set(key, 0);
        continue;
      }
      const count = (releasedFrames.get(key) || 0) + 1;
      releasedFrames.set(key, count);
      if (!forceAll && count < 2) continue;

      // Clear sticky booleans if raw hardware input is unequivocally open
      if (hand.gamepad?.buttons?.length || (hand.isHand && hand.fingerTips?.thumb && hand.fingerTips?.index)) {
        hand.trigger = false;
        hand.gripping = false;
        hand.pinching = false;
      }

      // Clear any latched grab slot on the hand object
      const grabKeys = ['grabbed', 'held', 'heldEntity', 'holding', 'grabTarget', 'attachedEntity'];
      for (const gk of grabKeys) {
        if (hand[gk]) {
          const target = hand[gk];
          if (target.group && target.group.parent && target.group.parent !== APP.scene) {
            try { APP.scene.attach(target.group); } catch (_) {}
          }
          hand[gk] = null;
        }
      }

      if (forceAll || count === 2) {
        try { APP.releaseGrab?.(key); } catch (_) {}
        try { APP.dropHeld?.(key); } catch (_) {}
      }

      // Clear latched entity ownership in APP.entities
      for (const ent of APP.entities || []) {
        if (!ent) continue;
        const ownedByThis = ent.heldBy === key || ent.grabbedBy === key || ent.holder === key || ent.grabHand === key;
        if (ownedByThis || (forceAll && (ent.isHeld || ent.isGrabbed))) {
          ent.heldBy = null;
          ent.grabbedBy = null;
          ent.holder = null;
          ent.grabHand = null;
          ent.isHeld = false;
          ent.isGrabbed = false;
          if (ent.group && ent.group.parent && ent.group.parent !== APP.scene) {
            try { APP.scene.attach(ent.group); } catch (_) {}
          }
        }
      }
    }
  }

  // Native 3D VR Drawing Tool (Works both inside VR and Desktop)
  const DRAW_COLORS = ['#00f0b4', '#38bdf8', '#ef4444', '#f59e0b', '#a855f7', '#f8fafc'];
  const DRAW_SIZES = [
    { label: 'Thin', radius: 0.005 },
    { label: 'Medium', radius: 0.011 },
    { label: 'Thick', radius: 0.022 }
  ];

  const vrDraw = {
    enabled: false,
    colorIdx: 0,
    sizeIdx: 1,
    strokes: [],
    activeByHand: new Map(),
    getColor() { return DRAW_COLORS[this.colorIdx % DRAW_COLORS.length]; },
    getRadius() { return DRAW_SIZES[this.sizeIdx % DRAW_SIZES.length].radius; },
    getSizeLabel() { return DRAW_SIZES[this.sizeIdx % DRAW_SIZES.length].label; },
    cycleColor() { this.colorIdx = (this.colorIdx + 1) % DRAW_COLORS.length; },
    setColor(idx) { this.colorIdx = ((idx % DRAW_COLORS.length) + DRAW_COLORS.length) % DRAW_COLORS.length; },
    cycleSize() { this.sizeIdx = (this.sizeIdx + 1) % DRAW_SIZES.length; },
    toggle() {
      this.enabled = !this.enabled;
      if (!this.enabled) {
        for (const hk of Array.from(this.activeByHand.keys())) this.finishStroke(hk);
      }
      if (typeof APP.setDrawingMode === 'function') {
        try { APP.setDrawingMode(this.enabled); } catch (_) {}
      }
      APP.ui?.toast?.('VR Draw: ' + (this.enabled ? 'ON (Hold Trigger or Pinch to draw in 3D)' : 'OFF'), 2200);
    },
    startStroke(handKey, pt) {
      const grp = new THREE.Group();
      grp.name = 'VR_Stroke_' + (this.strokes.length + 1);
      const col = new THREE.Color(this.getColor());
      const mat = new THREE.MeshStandardMaterial({
        color: col,
        emissive: col,
        emissiveIntensity: 0.25,
        roughness: 0.35,
        metalness: 0.1
      });
      const mesh = new THREE.Mesh(new THREE.BufferGeometry(), mat);
      mesh.castShadow = true;
      grp.add(mesh);
      APP.scene.add(grp);
      this.activeByHand.set(handKey, {
        group: grp,
        mesh,
        mat,
        radius: this.getRadius(),
        points: [pt.clone()]
      });
    },
    extendStroke(handKey, pt) {
      const st = this.activeByHand.get(handKey);
      if (!st) return;
      const last = st.points[st.points.length - 1];
      if (last.distanceTo(pt) < Math.max(0.012, st.radius * 1.1)) return;
      st.points.push(pt.clone());
      if (st.points.length >= 2) {
        const curve = new THREE.CatmullRomCurve3(st.points);
        const segs = Math.min(180, st.points.length * 4);
        const newGeo = new THREE.TubeGeometry(curve, segs, st.radius, 8, false);
        st.mesh.geometry.dispose();
        st.mesh.geometry = newGeo;
      }
      if (st.points.length >= 140) {
        const tail = pt.clone();
        this.finishStroke(handKey);
        this.startStroke(handKey, tail);
      }
    },
    finishStroke(handKey) {
      const st = this.activeByHand.get(handKey);
      this.activeByHand.delete(handKey);
      if (!st) return;
      if (st.points.length < 2) {
        st.mesh.geometry.dispose();
        st.mat.dispose();
        if (st.group.parent) st.group.parent.remove(st.group);
        return;
      }
      const box = new THREE.Box3().setFromObject(st.group);
      const center = new THREE.Vector3();
      box.getCenter(center);
      st.mesh.geometry.translate(-center.x, -center.y, -center.z);
      st.group.position.copy(center);

      const ent = APP.registerEntity ? APP.registerEntity(st.group, {
        name: st.group.name,
        category: 'drawing',
        isPen: false,
        grabbable: true,
        holdable: true,
        hasGravity: !!APP.drawingGravity,
        physics: { enabled: !!APP.drawingGravity, gravity: !!APP.drawingGravity }
      }) : { group: st.group };
      this.strokes.push(ent);
    },
    undo() {
      const last = this.strokes.pop();
      if (last) {
        if (APP.removeEntity && last.id) APP.removeEntity(last, false);
        else if (last.group?.parent) last.group.parent.remove(last.group);
        APP.ui?.toast?.('Undid last 3D stroke.', 1600);
      } else {
        try { APP.undoDrawing?.(); } catch (_) {}
      }
    },
    clearAll() {
      for (const hk of Array.from(this.activeByHand.keys())) this.finishStroke(hk);
      while (this.strokes.length) {
        const st = this.strokes.pop();
        if (APP.removeEntity && st.id) APP.removeEntity(st, false);
        else if (st.group?.parent) st.group.parent.remove(st.group);
      }
      try { APP.clearDrawings?.(); } catch (_) {}
      APP.ui?.toast?.('Cleared 3D drawings.', 1600);
    },
    spawnPen() {
      if (typeof APP.executeCommand === 'function') {
        try {
          APP.executeCommand('make a pen', 'vr_menu');
          return;
        } catch (_) {}
      }
      const penG = new THREE.Group();
      penG.name = 'VR_3D_Pen';
      const body = new THREE.Mesh(
        new THREE.CylinderGeometry(0.011, 0.008, 0.16, 16),
        new THREE.MeshStandardMaterial({ color: 0x1e293b, metalness: 0.7, roughness: 0.25 })
      );
      const nib = new THREE.Mesh(
        new THREE.ConeGeometry(0.008, 0.026, 14),
        new THREE.MeshStandardMaterial({ color: this.getColor(), emissive: this.getColor(), emissiveIntensity: 0.6 })
      );
      nib.position.y = 0.093;
      penG.add(body, nib);
      const pos = APP.getSpawnPosition ? APP.getSpawnPosition(0.65, true) : new THREE.Vector3(0, 1.25, -0.65);
      penG.position.copy(pos);
      APP.scene.add(penG);
      APP.registerEntity?.(penG, {
        name: 'VR 3D Pen',
        category: 'tool',
        isPen: true,
        grabbable: true,
        holdable: true,
        physics: { enabled: false }
      });
    },
    update(isMenuBlockedFn) {
      if (!this.enabled) return;
      const hands = APP.hands || {};
      for (const [hk, hand] of Object.entries(hands)) {
        if (!hand || !hand.active) {
          this.finishStroke(hk);
          continue;
        }
        if (isMenuBlockedFn && isMenuBlockedFn(hk)) {
          this.finishStroke(hk);
          continue;
        }
        const raw = getRawHandState(hk, hand);
        const drawingNow = hand.isHand ? raw.pinch : raw.trig;
        let tipPt = null;
        if (hand.isHand && hand.fingerTips?.index) {
          tipPt = hand.fingerTips.index;
        } else if (hand.rayOrigin && hand.rayDir) {
          tipPt = hand.rayOrigin.clone().addScaledVector(hand.rayDir.clone().normalize(), 0.06);
        } else if (hand.pos) {
          tipPt = hand.pos;
        }
        if (drawingNow && tipPt) {
          if (!this.activeByHand.has(hk)) this.startStroke(hk, tipPt);
          else this.extendStroke(hk, tipPt);
        } else {
          this.finishStroke(hk);
        }
      }
    }
  };

    function drawRoundedRect(x, y, w, h, r, fill, stroke, lineW = 3) {
      ctx.beginPath();
      ctx.moveTo(x + r, y);
      ctx.arcTo(x + w, y, x + w, y + h, r);
      ctx.arcTo(x + w, y + h, x, y + h, r);
      ctx.arcTo(x, y + h, x, y, r);
      ctx.arcTo(x, y, x + w, y, r);
      ctx.closePath();
      if (fill) { ctx.fillStyle = fill; ctx.fill(); }
      if (stroke) { ctx.strokeStyle = stroke; ctx.lineWidth = lineW; ctx.stroke(); }
    }

    function addZone(id, x, y, w, h, action, allowHold) {

        v0: 1 - (y + h) / canvas.height,
        v1: 1 - y / canvas.height,
        action,
        allowHold: !!allowHold
      });
    }

    function triggerZone(zone, hand, uHit, isHold) {
      const now = performance.now();
      const debounce = isHold ? 45 : 240;
      if (now - lastClickTime < debounce) return;
      lastClickTime = now;
      if (!isHold) {
        try {
          const gp = hand && hand.gamepad;
          if (gp?.hapticActuators?.[0]) gp.hapticActuators[0].pulse(0.35, 25);
        } catch (_) {}
      }
      try { zone.action(uHit); } catch (e) { console.warn('[vrMenu]', e); }
      syncState();
      redraw();
    }

      if (!root.visible || minimized || !hit || !hit.uv) return;

          triggerZone(z, null, uv.x, false);
          break;
        }
      }
    };

      if (!root.visible || !minimized) return;

      if (now - lastClickTime < 260) return;
      lastClickTime = now;
      setMinimized(false);
    };

    function placeInFrontOfUser() {
      const hp = APP.headPos || new THREE.Vector3(0, 1.5, 0);
      const hf = APP.headFwd || new THREE.Vector3(0, 0, -1);
      const flat = new THREE.Vector3(hf.x, 0, hf.z).normalize();
      if (flat.lengthSq() < 0.01) flat.set(0, 0, -1);
      const right = new THREE.Vector3().crossVectors(flat, new THREE.Vector3(0, 1, 0)).normalize();
      root.position.copy(hp).addScaledVector(flat, 0.85).addScaledVector(right, -0.26);
      root.position.y = Math.max(0.88, hp.y - 0.10);
      root.lookAt(hp.x, root.position.y, hp.z);
    }

    // Desktop mouse clicking on the 3D Menu when open outside VR
    const desktopRay = new THREE.Raycaster();
    const desktopNdc = new THREE.Vector2();
    function onDesktopPointer(e, isClick) {
      if (!root.visible || !APP.camera) return;
      const el = APP.renderer?.domElement;
      const rect = el ? el.getBoundingClientRect() : { left: 0, top: 0, width: window.innerWidth, height: window.innerHeight };
      desktopNdc.x = ((e.clientX - rect.left) / Math.max(1, rect.width)) * 2 - 1;
      desktopNdc.y = -((e.clientY - rect.top) / Math.max(1, rect.height)) * 2 + 1;
      desktopRay.setFromCamera(desktopNdc, APP.camera);
      if (minimized) {
        const hits = desktopRay.intersectObject(orbMesh, false);
        if (hits.length && isClick) setMinimized(false);
        return;
      }
      const hits = desktopRay.intersectObject(screenMesh, false);
      if (hits.length && hits[0].uv) {
        const uv = hits[0].uv;
        for (const z of hitZones) {
          if (uv.x >= z.u0 && uv.x <= z.u1 && uv.y >= z.v0 && uv.y <= z.v1) {
            if (hoveredId !== z.id) { hoveredId = z.id; redraw(); }
            if (isClick) triggerZone(z, null, uv.x, false);
            return;
          }
        }
      }
    }
    const desktopDownHandler = (e) => { if (e.button === 0) onDesktopPointer(e, true); };
    const desktopMoveHandler = (e) => { onDesktopPointer(e, false); };
    if (typeof window !== 'undefined') {
      window.addEventListener('pointerdown', desktopDownHandler, { passive: true });
      window.addEventListener('pointermove', desktopMoveHandler, { passive: true });
    }

    function updateInteraction(dt, elapsed) {
      // Run sticky-grab guard + VR drawing every frame even if menu is hidden
      releaseStuckGrabs(false);
      menuBlockedHands.clear();

      if (root.visible) {
        if (minimized) {
          orbRing.rotation.z = elapsed * 1.4;
          orbRing.rotation.x = Math.sin(elapsed * 0.9) * 0.4;
        }

        const hands = APP.hands || {};
        if (draggingHandKey) {
          const dh = hands[draggingHandKey];
          const rawD = dh ? getRawHandState(draggingHandKey, dh) : null;
          const stillHolding = !!(dh && dh.active && rawD && (rawD.trig || rawD.grip || rawD.pinch));
          if (stillHolding) {
            menuBlockedHands.add(draggingHandKey);
            if (dh.isHand && dh.fingerTips?.index) {
              root.position.lerp(tmpVec.copy(dh.fingerTips.index).add(dragOffset), 0.55);
            } else if (dh.rayOrigin && dh.rayDir) {
              tmpDir.copy(dh.rayDir).normalize();
              tmpVec.copy(dh.rayOrigin).addScaledVector(tmpDir, dragDistance).add(dragOffset);
              root.position.lerp(tmpVec, 0.55);
            }
            if (APP.headPos) root.lookAt(APP.headPos.x, root.position.y, APP.headPos.z);
          } else {
            draggingHandKey = null;
          }
        }

        let newHover = null;
        let showCursor = false;

        if (!draggingHandKey) {
          for (const [key, hand] of Object.entries(hands)) {
            if (!hand || !hand.active) continue;
            const raw = getRawHandState(key, hand);
            const tip = hand.isHand ? hand.fingerTips?.index : hand.pos;

            // 1) Controller Ray Pointer Interaction (works cleanly with trigger!)
            if (hand.rayOrigin && hand.rayDir) {
              tmpDir.copy(hand.rayDir).normalize();
              raycaster.set(hand.rayOrigin, tmpDir);

              if (minimized) {
                const orbHits = raycaster.intersectObject(orbMesh, false);
                if (orbHits.length) {
                  menuBlockedHands.add(key);
                  showCursor = true;
                  cursorDot.position.copy(orbHits[0].point);
                  root.worldToLocal(cursorDot.position);
                  if (raw.justTrig || raw.justPinch) {
                    setMinimized(false);
                    break;
                  }
                }
              } else {
                const dragHits = raycaster.intersectObject(dragBar, false);
                if (dragHits.length) {
                  menuBlockedHands.add(key);
                  showCursor = true;
                  cursorDot.position.copy(dragHits[0].point);
                  root.worldToLocal(cursorDot.position);
                  newHover = 'drag_bar';
                  if (raw.justTrig || raw.justGrip || raw.justPinch) {
                    draggingHandKey = key;
                    dragDistance = dragHits[0].distance;
                    dragOffset.copy(root.position).sub(dragHits[0].point);
                    break;
                  }
                }

                const scrHits = raycaster.intersectObject(screenMesh, false);
                if (scrHits.length && scrHits[0].uv) {
                  menuBlockedHands.add(key);
                  showCursor = true;
                  cursorDot.position.copy(scrHits[0].point);
                  root.worldToLocal(cursorDot.position);
                  const uv = scrHits[0].uv;
                  for (const z of hitZones) {
                    if (uv.x >= z.u0 && uv.x <= z.u1 && uv.y >= z.v0 && uv.y <= z.v1) {
                      newHover = z.id;
                      if (raw.justTrig || raw.justPinch || (z.allowHold && (raw.trig || raw.pinch))) {
                        triggerZone(z, hand, uv.x, z.allowHold && !raw.justTrig && !raw.justPinch);
                      }
                      break;
                    }
                  }
                }
              }
            }

            // 2) Direct Hand-Tracking Finger Interaction (no sticky auto-drag!)
            if (hand.isHand && tip) {
              if (minimized) {
                orbMesh.getWorldPosition(tmpVec);
                if (tip.distanceTo(tmpVec) < 0.08) {
                  menuBlockedHands.add(key);
                  if (raw.justPinch) { setMinimized(false); break; }
                }
              } else {
                dragBar.getWorldPosition(tmpVec);
                if (tip.distanceTo(tmpVec) < 0.075) {
                  menuBlockedHands.add(key);
                  if (raw.justPinch) {
                    draggingHandKey = key;
                    dragOffset.copy(root.position).sub(tip);
                    break;
                  }
                }

                tmpLocal.copy(tip);
                screenMesh.worldToLocal(tmpLocal);
                const prevZ = prevHandLocalZ.get(key) ?? 1.0;
                prevHandLocalZ.set(key, tmpLocal.z);

                if (Math.abs(tmpLocal.z) < 0.045) {
                  const u = tmpLocal.x / PANEL_W + 0.5;
                  const v = tmpLocal.y / PANEL_H + 0.5;
                  if (u >= 0 && u <= 1 && v >= 0 && v <= 1) {
                    menuBlockedHands.add(key);
                    for (const z of hitZones) {
                      if (u >= z.u0 && u <= z.u1 && v >= z.v0 && v <= z.v1) {
                        newHover = z.id;
                        const crossedPlane = prevZ > 0.008 && tmpLocal.z <= 0.008;
                        if (raw.justPinch || crossedPlane || (z.allowHold && raw.pinch)) {
                          triggerZone(z, hand, u, z.allowHold && !raw.justPinch);
                        }
                        break;
                      }
                    }
                  }
                }
              }
            }
          }
        }

        cursorDot.visible = showCursor && !draggingHandKey;
        if (newHover !== hoveredId) {
          hoveredId = newHover;
          redraw();
        }
      }

      vrDraw.update((hk) => menuBlockedHands.has(hk));
    }

    function dispose() {
      if (typeof window !== 'undefined') {
        window.removeEventListener('pointerdown', desktopDownHandler);
        window.removeEventListener('pointermove', desktopMoveHandler);
      }
    }

    return {
      root, screenMesh, orbMesh, dragBar, show,
      hide: () => { root.visible = false; },
      toggle: () => { if (root.visible) root.visible = false; else show(true); },
      minimize: () => setMinimized(true),
      expand: () => setMinimized(false),
      isVisible: () => root.visible,
      redraw,
      update: updateInteraction,
      dispose
    };
  })();

  function saveRepoCfg() {

  async function ghApiFetch(path, opts) {
    const cfg = getRepoCfg();
    if (!cfg.owner || !cfg.repo || !cfg.pat) throw new Error('Enter GitHub Owner, Repo Name, and PAT first.');
    const url = `https://api.github.com/repos/${encodeURIComponent(cfg.owner)}/${encodeURIComponent(cfg.repo)}${path}`;
    const res = await fetch(url, Object.assign({}, opts || {}, {
      headers: Object.assign({
        'Accept': 'application/vnd.github+json',
        'Authorization': 'Bearer ' + cfg.pat,
        'Content-Type': 'application/json'
      }, (opts && opts.headers) || {})
    }));
    const txt = await res.text();
    let json = {};
    try { json = JSON.parse(txt || '{}'); } catch (_) {}
    if (!res.ok) throw new Error(`GitHub ${res.status}: ${json.message || txt}`);
    return json;
  }

  function encodeB64(str) {
    const bytes = new TextEncoder().encode(str);
    let bin = '';
    for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
    return btoa(bin);
  }

  function decodeB64(b64) {
    const bin = atob(String(b64 || '').replace(/\s+/g, ''));
    const bytes = new Uint8Array(bin.length);
    for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
    return new TextDecoder().decode(bytes);
  }

  function buildLineDiff(aStr, bStr) {
    const a = String(aStr || '').split('\n');
    const b = String(bStr || '').split('\n');
    const out = [];
    const maxLen = Math.max(a.length, b.length);
    for (let i = 0; i < maxLen; i++) {
      if (a[i] !== b[i]) {
        if (a[i] !== undefined) out.push('- ' + (i + 1) + ': ' + a[i]);
        if (b[i] !== undefined) out.push('+ ' + (i + 1) + ': ' + b[i]);
        if (out.length > 140) { out.push('... (truncated)'); break; }
      }
    }
    return out.length ? out.join('\n') : '(No differences)';
  }

  function stripOuterCodeFences(raw) {
    let s = String(raw || '').trim();
    const first = s.indexOf('```');
    const last = s.lastIndexOf('```');
    if (first !== -1 && last > first) {
      const inner = s.slice(first + 3, last).replace(/^[a-zA-Z0-9_-]*\r?\n/, '');
      if (inner.includes('@ability:') || inner.includes('<<<')) return inner.trim();
    }
    return s;
  }

  function applyAIModuleOutput(rawResponse, currentText, filename) {
    const clean = stripOuterCodeFences(rawResponse);

    if (clean.includes('<<<APPEND_ABILITY')) {
      const blockMatch = clean.match(/<<<APPEND_ABILITY\s*([\s\S]*?)<<<END_ABILITY/i);
      const block = blockMatch ? blockMatch[1].trim() : clean.replace(/<<<APPEND_ABILITY|<<<END_ABILITY/gi, '').trim();
      const parsedNew = APP.parseModuleText ? APP.parseModuleText(filename, block) : [];
      let merged = String(currentText || '').trim();

      if (parsedNew.length && APP.parseModuleText) {
        const existing = APP.parseModuleText(filename, merged);
        for (const ab of parsedNew) {
          const idEsc = ab.id.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
          const re = new RegExp(`@ability:\\s*${idEsc}[\\s\\S]*?@end`, 'm');
          const singleBlockMatch = block.match(new RegExp(`@ability:\\s*${idEsc}[\\s\\S]*?@end`, 'm'));
          if (existing.some((e) => e.id === ab.id) && singleBlockMatch) {
            merged = merged.replace(re, singleBlockMatch[0]);
          } else {
            merged = (merged ? merged + '\n\n' : '') + (singleBlockMatch ? singleBlockMatch[0] : block);
          }
        }
        return merged.trim();
      }
      return (merged ? merged + '\n\n' : '') + block;
    }

    if (clean.includes('<<<SEARCH') && clean.includes('<<<REPLACE')) {
      let updated = String(currentText || '');
      const patchRegex = /<<<SEARCH\r?\n([\s\S]*?)\r?\n<<<REPLACE\r?\n([\s\S]*?)\r?\n<<<END_PATCH/g;
      let match, applied = 0;
      while ((match = patchRegex.exec(clean)) !== null) {
        if (updated.includes(match[1])) { updated = updated.replace(match[1], match[2]); applied++; }
      }
      if (applied > 0) return updated;
    }

    const full = clean.replace(/^<<<FULL_FILE\s*/i, '').replace(/<<<END_FILE\s*$/i, '').trim();
    if (APP.parseModuleText && currentText) {
      const curAbilities = APP.parseModuleText(filename, currentText);
      const newAbilities = APP.parseModuleText(filename, full);
      if (curAbilities.length >= 2 && newAbilities.length === 1 && !curAbilities.some((a) => a.id === newAbilities[0].id)) {
        return currentText.trim() + '\n\n' + full;
      }
    }
    return full;
  }

  function validateModuleSyntax(filename, rawText) {
    if (typeof APP.parseModuleText !== 'function') return { ok: true };
    const abilities = APP.parseModuleText(filename, rawText);
    if (!abilities.length) return { ok: false, error: 'Parsed 0 @ability blocks. Ensure @ability: <id>, @code:, and @end tags are present.' };
    const AsyncFn = Object.getPrototypeOf(async function () {}).constructor;
    for (const ab of abilities) {
      try { new AsyncFn('APP', 'THREE', 'command', 'ability', ab.code); }
      catch (e) { return { ok: false, error: `JS SyntaxError inside @ability: ${ab.id}: ${e.message}` }; }
    }
    return { ok: true, count: abilities.length };
  }

  function buildFullAppContextForEdit(targetFilename, requestText) {
    const modEntries = Object.entries(APP.modules || {});
    const moduleIndex = modEntries.map(([fn, txt]) => {
      const abs = APP.parseModuleText ? APP.parseModuleText(fn, txt) : [];
      return `- ${fn} (${txt.length} chars): abilities=[${abs.map(a => a.id).join(', ')}]`;
    }).join('\n');

    const archGuide = [
      'NEXUSXR ARCHITECTURE & GLOBAL HOOKS:',
      '- Shared: `APP`, `THREE`, `APP.scene`, `APP.camera`, `APP.renderer`, `APP.fx` (`water`, `rain`, `snow`, `lavaFloor`).',
      '- Update loop: `APP.onUpdate((dt, elapsed) => { ... })` or `APP.frameHooks.add(fn)`.',
      '- Entities: `APP.registerEntity(group, { name, category, holdable, hasGravity, onUpdate })`.',
      '- Geeny (`bot.txt`): `APP.bot.root`, `APP.bot.bodyGroup`, `APP.bot.getVelocity()`, `APP.bot.showBoard(...)`.',
      '- Module syntax: `@ability: <id>`, `@autorun`, `@pin: <Label>`, `@desc: ...`, `@match: ...`, `@code: ... @end`.',
      '- Use `#` ONLY outside `@code:`; inside `@code:`, use `//` ONLY.'
    ].join('\n');

    const extraFiles = [];
    for (const [fn, txt] of modEntries) {
      if (fn === targetFilename) continue;
      if (fn === 'bot.txt' && /bot|geeny|genie|cape|tentacle|board|visor|wing/i.test(requestText)) {
        extraFiles.push(`=== REFERENCE FILE: ${fn} ===\n${txt}`);
      } else if (fn === 'visual.txt' && /visual|shader|matrix|360|environment|dissolve/i.test(requestText)) {
        extraFiles.push(`=== REFERENCE FILE: ${fn} ===\n${txt}`);
      }
    }
    return [archGuide, '', 'ALL LOADED .TXT MODULES:', moduleIndex || '(none)', extraFiles.length ? '\n' + extraFiles.join('\n\n') : ''].join('\n');
  }

  function autoChooseTargetFile(currentFile, nlRequest) {
    const req = String(nlRequest || '').toLowerCase();
    const explicit = req.match(/\b(ai_models|visual|toolbar|bot|instruction\d+)\.txt\b/i);
    if (explicit) return explicit[0].toLowerCase();
    if (currentFile === 'instruction1.txt' && /\b(bot|geeny|genie|cape|cloak|wing|visor|tentacle|holo.?board)\b/i.test(req)) {
      return 'bot.txt';
    }
    return currentFile;
  }

  function stageEditInEditor(filename, prevText, nextText, logMsg) {
    buildEditorUI();
    const mount = document.getElementById('modal-editor-body');
    if (!mount) return;
    const sel = mount.querySelector('#gh-file-sel');
    if (Array.from(sel.options).some((o) => o.value === filename)) {
      sel.value = filename;
      mount.querySelector('#gh-custom-file').value = '';
    } else {
      mount.querySelector('#gh-custom-file').value = filename;
    }
    sandboxSnapshot = { filename, prevText: prevText || '' };
    mount.querySelector('#gh-code-ta').value = nextText;
    mount.querySelector('#gh-diff').textContent = buildLineDiff(prevText, nextText);
    mount.querySelector('#gh-commit-msg').value = `Update ${filename} via Geeny AI`;
    mount.querySelector('#gh-log').textContent = logMsg || `Staged edit for ${filename}.`;
  }

    injectToolbarCSS();

    if (!mount.dataset.built) {
      mount.dataset.built = '1';
      mount.innerHTML = [
        '<div class="gh-grid">',

        '<div class="tb-row" style="margin-top:8px;">',
        '  <input id="gh-nl-prompt" style="flex:1;" placeholder="Describe any edit across any .txt file..." />',
        '  <label style="font-size:11px;display:flex;align-items:center;gap:4px;cursor:pointer;"><input type="checkbox" id="gh-auto-route" checked> Auto-Pick File</label>',
        '  <button type="button" class="btn active" id="gh-ai-btn">AI Edit</button>',
        '</div>',
        '<textarea id="gh-code-ta" spellcheck="false" style="width:100%;min-height:210px;font-family:ui-monospace,monospace;font-size:12px;margin-top:8px;"></textarea>',
        '<div id="gh-diff" class="gh-diff" style="margin-top:8px;">Diff preview will appear here.</div>',
        '<div class="tb-row" style="margin-top:8px;">',
        '  <input id="gh-commit-msg" style="flex:1;" placeholder="Commit message..." />',
        '  <button type="button" class="btn" id="gh-sandbox-btn">Test in Sandbox</button>',
        '  <button type="button" class="btn warn" id="gh-rollback-btn">Rollback Sandbox</button>',
        '  <button type="button" class="btn active" id="gh-commit-btn">Approve & Commit</button>',
        '</div>',
        '<div id="gh-log" style="font-size:12px;color:#38bdf8;margin-top:6px;">Ready.</div>',
        '<div class="tb-row" style="justify-content:space-between;margin-top:10px;">',
        '  <strong style="font-size:12px;color:#cbd5e1;">Last Updates (GitHub Commits)</strong>',
        '  <button type="button" class="btn" id="gh-commits-btn">Refresh Commits</button>',
        '</div>',
        '<div id="gh-commits-box" class="gh-commits" style="margin-top:5px;"><span style="font-size:11px;color:#94a3b8;">Click Refresh Commits after entering Owner/Repo/PAT.</span></div>'
      ].join('');

      const setLog = (msg) => { mount.querySelector('#gh-log').textContent = String(msg || ''); };
      const activeFilename = () => {
        const custom = mount.querySelector('#gh-custom-file').value.trim();
        return custom || mount.querySelector('#gh-file-sel').value;
      };

      function loadFileIntoTextarea(fnOverride) {
        const fn = fnOverride || mount.querySelector('#gh-file-sel').value;
        mount.querySelector('#gh-file-sel').value = fn;
        mount.querySelector('#gh-custom-file').value = '';
        mount.querySelector('#gh-code-ta').value = APP.modules?.[fn] || '';
        mount.querySelector('#gh-diff').textContent = `(Loaded live "${fn}")`;
      }

      mount.querySelector('#gh-file-sel').addEventListener('change', () => loadFileIntoTextarea());

      mount.querySelector('#gh-pull-btn').addEventListener('click', async () => {
        saveRepoCfg();
        const cfg = getRepoCfg();
        const fn = activeFilename();
        setLog(`Fetching ${fn} from GitHub...`);
        try {
          const data = await ghApiFetch(`/contents/${encodeURIComponent(fn)}?ref=${encodeURIComponent(cfg.branch)}`);
          const txt = decodeB64(data.content);
          const cur = mount.querySelector('#gh-code-ta').value;
          mount.querySelector('#gh-code-ta').value = txt;
          mount.querySelector('#gh-diff').textContent = buildLineDiff(cur, txt);
          setLog(`Fetched ${fn} from GitHub.`);
        } catch (e) { setLog('Fetch error: ' + e.message); }
      });

      mount.querySelector('#gh-ai-btn').addEventListener('click', async () => {
        if (!APP.ai?.chat) { setLog('AI module not loaded.'); return; }
        const req = mount.querySelector('#gh-nl-prompt').value.trim();
        if (!req) { setLog('Enter a natural-language instruction first.'); return; }

        let fn = activeFilename();
        if (mount.querySelector('#gh-auto-route').checked) {
          const routed = autoChooseTargetFile(fn, req);
          if (routed !== fn) {
            fn = routed;
            if (Array.from(mount.querySelector('#gh-file-sel').options).some((o) => o.value === fn)) loadFileIntoTextarea(fn);
            else { mount.querySelector('#gh-custom-file').value = fn; mount.querySelector('#gh-code-ta').value = APP.modules?.[fn] || ''; }
          }
        }

        const cur = mount.querySelector('#gh-code-ta').value;
        const fullContext = buildFullAppContextForEdit(fn, req);
        setLog(`Generating edit for ${fn} via ${APP.ai.getSelectedModel().id}...`);

        const sysPrompt = [
          'You are an expert NexusXR WebXR Three.js .txt module engineer with FULL access to the codebase.',
          fullContext,
          '',
          'OUTPUT FORMAT: Prefer <<<APPEND_ABILITY ... <<<END_ABILITY when adding new abilities, <<<SEARCH / <<<REPLACE / <<<END_PATCH for targeted line edits, or full valid .txt file.',
          'Never put `#` comment lines inside `@code:`; use `//` inside JS.'
        ].join('\n');

        const res = await APP.ai.chat({
          systemPrompt: sysPrompt,
          prompt: `Target file: ${fn}\n\nUser request:\n${req}\n\nCurrent ${fn} content:\n\`\`\`txt\n${cur}\n\`\`\``
        });

        if (!res.ok) { setLog('AI Error: ' + res.error); return; }

        let next = applyAIModuleOutput(res.text, cur, fn);
        let check = validateModuleSyntax(fn, next);
        if (!check.ok) {
          setLog(`Syntax issue (${check.error}), auto-repairing...`);
          const fixRes = await APP.ai.chat({
            systemPrompt: sysPrompt,
            prompt: `Fix this error:\n${check.error}\nReturn ONLY valid <<<APPEND_ABILITY ... <<<END_ABILITY or complete .txt file.`
          });
          if (fixRes.ok) { next = applyAIModuleOutput(fixRes.text, cur, fn); check = validateModuleSyntax(fn, next); }
        }

        sandboxSnapshot = { filename: fn, prevText: cur };
        mount.querySelector('#gh-code-ta').value = next;
        mount.querySelector('#gh-diff').textContent = buildLineDiff(cur, next);
        mount.querySelector('#gh-commit-msg').value = req.slice(0, 68);
        setLog(check.ok ? `Ready for ${fn} (${check.count} abilities validated). Click "Test in Sandbox"!` : `Warning in ${fn}: ${check.error}`);
      });

      mount.querySelector('#gh-sandbox-btn').addEventListener('click', async () => {
        const fn = activeFilename();
        const next = mount.querySelector('#gh-code-ta').value;
        const check = validateModuleSyntax(fn, next);
        if (!check.ok) { setLog('Cannot run Sandbox — ' + check.error); return; }
        if (!sandboxSnapshot || sandboxSnapshot.filename !== fn) sandboxSnapshot = { filename: fn, prevText: APP.modules?.[fn] || '' };
        try {
          await APP.reloadModuleText?.(fn, next);
          refresh();
          setLog(`Sandbox LIVE for "${fn}"! Test now, Rollback, or Approve & Commit.`);
        } catch (e) { setLog('Sandbox runtime error: ' + e.message); }
      });

      mount.querySelector('#gh-rollback-btn').addEventListener('click', async () => {
        if (!sandboxSnapshot) { setLog('No active sandbox snapshot.'); return; }
        const { filename, prevText } = sandboxSnapshot;
        await APP.reloadModuleText?.(filename, prevText);
        mount.querySelector('#gh-code-ta').value = prevText;
        mount.querySelector('#gh-diff').textContent = '(Rolled back)';
        refresh(); sandboxSnapshot = null;
        setLog(`Rolled back "${filename}".`);
      });

      async function loadCommits() {
        saveRepoCfg();
        const cfg = getRepoCfg();
        const box = mount.querySelector('#gh-commits-box');
        box.innerHTML = '<span style="font-size:11px;color:#94a3b8;">Loading commits...</span>';
        try {
          const list = await ghApiFetch(`/commits?sha=${encodeURIComponent(cfg.branch)}&per_page=8`);
          box.innerHTML = '';
          for (const c of list) {
            const row = document.createElement('div');
            row.className = 'gh-c-row';
            const msg = (c.commit?.message || '').split('\n')[0].slice(0, 52);
            row.innerHTML = `<span><code>${c.sha.slice(0, 7)}</code> ${msg}</span><button type="button" class="btn" style="padding:3px 8px;font-size:11px;">Restore File</button>`;
            row.querySelector('button').onclick = async () => {
              const fn = activeFilename();
              setLog(`Loading ${fn} from ${c.sha.slice(0, 7)}...`);
              try {
                const fileObj = await ghApiFetch(`/contents/${encodeURIComponent(fn)}?ref=${encodeURIComponent(c.sha)}`);
                const restored = decodeB64(fileObj.content);
                const cur = mount.querySelector('#gh-code-ta').value;
                mount.querySelector('#gh-code-ta').value = restored;
                mount.querySelector('#gh-diff').textContent = buildLineDiff(cur, restored);
                mount.querySelector('#gh-commit-msg').value = `Revert ${fn} to ${c.sha.slice(0, 7)}`;
                setLog(`Loaded ${fn} from ${c.sha.slice(0, 7)}. Test in Sandbox or Commit.`);
              } catch (err) { setLog('Restore error: ' + err.message); }
            };
            box.appendChild(row);
          }
        } catch (err) { box.innerHTML = `<span style="font-size:11px;color:#fda4af;">${err.message}</span>`; }
      }

      mount.querySelector('#gh-commits-btn').addEventListener('click', loadCommits);

      mount.querySelector('#gh-commit-btn').addEventListener('click', async () => {
        saveRepoCfg();
        const cfg = getRepoCfg();
        const fn = activeFilename();
        const content = mount.querySelector('#gh-code-ta').value;
        const msg = mount.querySelector('#gh-commit-msg').value.trim() || `Update ${fn} via NexusXR`;
        setLog(`Committing ${fn} via Git Data API...`);
        try {
          const refData = await ghApiFetch(`/git/refs/heads/${encodeURIComponent(cfg.branch)}`);
          const headSha = refData.object.sha;
          const headCommit = await ghApiFetch(`/git/commits/${headSha}`);
          const blob = await ghApiFetch('/git/blobs', { method: 'POST', body: JSON.stringify({ content: encodeB64(content), encoding: 'base64' }) });
          const tree = await ghApiFetch('/git/trees', {
            method: 'POST',
            body: JSON.stringify({ base_tree: headCommit.tree.sha, tree: [{ path: fn, mode: '100644', type: 'blob', sha: blob.sha }] })
          });
          const commit = await ghApiFetch('/git/commits', { method: 'POST', body: JSON.stringify({ message: msg, tree: tree.sha, parents: [headSha] }) });
          await ghApiFetch(`/git/refs/heads/${encodeURIComponent(cfg.branch)}`, { method: 'PATCH', body: JSON.stringify({ sha: commit.sha, force: false }) });
          sandboxSnapshot = null;
          await APP.reloadModuleText?.(fn, content);
          setLog(`Committed ${fn} (${commit.sha.slice(0, 7)}).`);
          loadCommits();
        } catch (err) { setLog('Commit error: ' + err.message); }
      });
    }

    const sel = mount.querySelector('#gh-file-sel');
    const prev = sel.value;
    const files = Array.from(new Set([...Object.keys(APP.modules || {}), 'ai_models.txt', 'visual.txt', 'toolbar.txt', 'bot.txt', 'instruction1.txt']));
    sel.innerHTML = '';
    for (const f of files) { const opt = document.createElement('option'); opt.value = f; opt.textContent = f; sel.appendChild(opt); }
    if (prev && files.includes(prev)) sel.value = prev;
    else if (files.includes('instruction1.txt')) sel.value = 'instruction1.txt';
    mount.querySelector('#gh-code-ta').value = APP.modules?.[sel.value] || '';
  }
