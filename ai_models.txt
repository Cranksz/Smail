    } catch (e) {
      orStatus.error = e.message || 'Failed to fetch OpenRouter :free models';
    } finally {
      orStatus.loading = false;

    }

  function formatApiError(status, rawText, provider, modelId) {
    let detail = '';
    try {
      const p = JSON.parse(rawText || '{}');
      detail = p.error?.message || p.choices?.[0]?.error?.message || p.message || rawText || '';
    } catch (_) {
      detail = String(rawText || '').slice(0, 240);
    }
    const pInfo = PROVIDERS[provider] || { name: provider, signupUrl: '' };
    if (status === 401 || status === 403) return `Invalid/rejected key (${status}) for ${pInfo.name}. ${detail}`;
    if (status === 402) return `Credits required (402) on ${pInfo.name} for "${modelId}". Pick a :free model or add credits. ${detail}`;
    if (status === 404) return `Model "${modelId}" unavailable (404) on ${pInfo.name}. ${detail}`;
    if (status === 429) return `Rate limit (429) on ${pInfo.name} for "${modelId}". ${detail}`;
    if (status === 503 || /high demand|overloaded|unavailable/i.test(detail)) return `${pInfo.name} "${modelId}" high demand (503). ${detail}`;
    return `${pInfo.name} error (${status || 'network'}) for "${modelId}": ${detail}`;
  }

  function removeExistingWorldFX(name) {
    const low = String(name || '').toLowerCase();
    const list = (APP.entities || []).filter((e) => e && e.category === 'world_fx' && (!low || e.name.toLowerCase() === low));
    for (const ent of list) APP.removeEntity?.(ent, false);
  }

  function registerWorldFXGroup(name, group, onUpdateFn) {
    APP.scene.add(group);
    const ent = APP.registerEntity(group, {
      name,
      category: 'world_fx',
      grabbable: false,
      holdable: false,
      physics: { enabled: false, gravity: false },
      onUpdate: typeof onUpdateFn === 'function'
        ? (dt, e) => onUpdateFn(Math.min(0.05, dt || 0.016), performance.now() * 0.001, e)
        : null
    });
    APP.lastEntityId = ent.id;
    return { ok: true, id: ent.id, name: ent.name, message: `Activated ${name}.` };
  }

          'uniform float uTime;',
          'varying vec3 vWorldPos;',
          'varying vec3 vNormalW;',
          'varying float vCrest;',
          'void main() {',
          '  vec3 p = position;',
          '  float w1 = sin(p.x * 2.2 + uTime * 2.1) * cos(p.y * 1.9 + uTime * 1.7) * 0.045;',

          '  float w3 = cos(length(p.xy) * 4.5 - uTime * 3.8) * 0.014;',
          '  float h = w1 + w2 + w3;',
          '  p.z += h;',
          '  vCrest = smoothstep(0.015, 0.068, h);',
          '  float dx = 2.2 * cos(p.x * 2.2 + uTime * 2.1) * cos(p.y * 1.9 + uTime * 1.7) * 0.045 + 5.4 * cos(p.x * 5.4 - uTime * 3.3 + p.y * 3.1) * 0.018;',

          '  vec3 nLocal = normalize(vec3(-dx, -dy, 1.0));',
          '  vec4 wp = modelMatrix * vec4(p, 1.0);',
          '  vWorldPos = wp.xyz;',
          '  vNormalW = normalize(mat3(modelMatrix) * nLocal);',
          '  gl_Position = projectionMatrix * viewMatrix * wp;',
          '}'
        ].join('\n'),
        fragmentShader: [
          'uniform float uTime;',
          'uniform vec3 uDeepCol;',
          'uniform vec3 uShallowCol;',
          'uniform vec3 uFoamCol;',
          'varying vec3 vWorldPos;',
          'varying vec3 vNormalW;',
          'varying float vCrest;',

          '  vec3 V = normalize(cameraPosition - vWorldPos);',
          '  vec3 N = normalize(vNormalW);',
          '  float fresnel = pow(1.0 - max(0.0, dot(V, N)), 3.2);',
          '  vec3 L = normalize(vec3(0.45, 0.85, 0.35));',
          '  vec3 H = normalize(L + V);',
          '  float spec = pow(max(0.0, dot(N, H)), 96.0);',

    rain(opts = {}) {

        group.add(sm);
        splashes.push({ mesh: sm, age: Math.random() * 0.65, life: 0.45 + Math.random() * 0.35 });

            arr[i6] = nx; arr[i6 + 1] = ny; arr[i6 + 2] = nz;
            arr[i6 + 3] = nx + 0.015; arr[i6 + 4] = ny + len; arr[i6 + 5] = nz + 0.01;

          const sp = splashes[i];

    if (p === 'water' || (!args.js_code && /\b(water|ocean|pool)\b/i.test(args.name || ''))) return fxEngine.water(args);
    if (p === 'rain' || (!args.js_code && /\b(rain|storm)\b/i.test(args.name || ''))) return fxEngine.rain(args);
    if (p === 'snow' || (!args.js_code && /\b(snow|snowing|blizzard)\b/i.test(args.name || ''))) return fxEngine.snow(args);
    if (p === 'lava' || (!args.js_code && /\b(lava|magma)\b/i.test(args.name || ''))) return fxEngine.lavaFloor(args);

    const THREE = APP.THREE;
    if (!THREE || !APP.scene) return { ok: false, error: 'Scene not initialized.' };
    const name = String(args.name || 'WorldFX').trim();
    if (args.replace_existing) removeExistingWorldFX(name);

    group.name = name;
    const V3 = (x = 0, y = 0, z = 0) => new THREE.Vector3(x, y, z);

    try {
      const clean = String(args.js_code || '').trim().replace(/^```(?:javascript|js)?/i, '').replace(/```$/i, '').trim();

      if (group.children.length === 0 && !hasUpdate) {

      }

    } catch (err) {

      return { ok: false, error: `${err.name}: ${err.message}` };
    }

  function tryParseLooseJSON(rawStr) {
    if (!rawStr) return {};
    if (typeof rawStr === 'object') return rawStr;
    try { return JSON.parse(rawStr); } catch (_) {}
    // Recover literal newlines inside JSON string values produced by some models
    try {
      const fixed = String(rawStr).replace(/"([^"\\]*(\\.[^"\\]*)*)"/g, (m) => m.replace(/\r?\n/g, '\\n'));
      return JSON.parse(fixed);
    } catch (_) {
      return {};
    }
  }

  function sanitizeSchema(schema) {
    if (!schema || typeof schema !== 'object') {
      return { type: 'object', properties: { note: { type: 'string', description: 'Optional note' } } };
    }
    const copy = JSON.parse(JSON.stringify(schema));
    if (!copy.type) copy.type = 'object';
    if (copy.type === 'object') {
      if (!copy.properties || typeof copy.properties !== 'object' || Object.keys(copy.properties).length === 0) {
        copy.properties = { note: { type: 'string', description: 'Optional note' } };
      } else {
        for (const k of Object.keys(copy.properties)) {
          if (copy.properties[k]?.type === 'object') copy.properties[k] = sanitizeSchema(copy.properties[k]);
        }
      }
    }
    return copy;
  }

  function normalizeTools(tools) {
    if (!Array.isArray(tools) || !tools.length) return [];
    return tools.filter((t) => t && t.name).map((t) => ({
      name: String(t.name),
      description: String(t.description || t.name),
      parameters: sanitizeSchema(t.parameters)
    }));
  }

  async function runGeminiRestChat(model, apiKey, opts) {
    const cleanTools = normalizeTools(opts.tools);
    const contents = [];
    const msgs = Array.isArray(opts.messages) ? opts.messages : [];

    for (let i = 0; i < msgs.length; i++) {
      const m = msgs[i];
      if (!m) continue;
      if (m._geminiParts) {
        contents.push({ role: m.role === 'assistant' ? 'model' : 'user', parts: JSON.parse(JSON.stringify(m._geminiParts)) });
        continue;
      }
      const role = m.role === 'assistant' ? 'model' : 'user';
      const parts = [];
      if (m.content) parts.push({ text: String(m.content) });
      if (i === msgs.length - 1 && role === 'user' && opts.imageB64) {
        parts.push({ inlineData: { mimeType: opts.imageMime || 'image/jpeg', data: String(opts.imageB64).replace(/^data:image\/\w+;base64,/, '') } });
      }
      if (parts.length) contents.push({ role, parts });
    }

    if (!contents.length) {
      const parts = [{ text: String(opts.prompt || 'Hello') }];
      if (opts.imageB64) parts.push({ inlineData: { mimeType: opts.imageMime || 'image/jpeg', data: String(opts.imageB64).replace(/^data:image\/\w+;base64,/, '') } });
      contents.push({ role: 'user', parts });
    }

    const maxSteps = Math.max(1, Math.min(8, opts.maxSteps || 5));
    const toolTrace = [];
    let finalText = '';

    for (let step = 0; step < maxSteps; step++) {
      const body = { contents };
      if (opts.systemPrompt) body.systemInstruction = { parts: [{ text: String(opts.systemPrompt) }] };
      if (cleanTools.length) body.tools = [{ functionDeclarations: cleanTools }];
      if (model.thinkingLevel) body.generationConfig = { thinkingConfig: { thinkingLevel: model.thinkingLevel } };

      const url = 'https://generativelanguage.googleapis.com/v1beta/models/' +
        encodeURIComponent(model.id) + ':generateContent?key=' + encodeURIComponent(apiKey);

      let data = null;
      const delays = [600, 1400];
      for (let attempt = 0; attempt <= delays.length; attempt++) {
        const res = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) });
        const rawTxt = await res.text();
        if (res.ok) { data = JSON.parse(rawTxt); break; }
        const errMsg = formatApiError(res.status, rawTxt, 'gemini', model.id);
        if ((res.status === 503 || res.status === 429) && attempt < delays.length) {
          APP.ui?.setStatus?.(`${model.id} busy (${res.status}), retrying...`, 'busy');
          await sleep(delays[attempt] + Math.random() * 350);
          continue;
        }
        throw new Error(errMsg);
      }

      const cand = data.candidates?.[0];
      if (!cand?.content?.parts) throw new Error('Gemini returned empty output (' + (cand?.finishReason || 'unknown') + ').');

      const rawParts = JSON.parse(JSON.stringify(cand.content.parts));
      contents.push({ role: 'model', parts: rawParts });

      const textChunks = [];
      const fnCalls = [];
      for (const p of rawParts) {
        if (p.text && !p.thought) textChunks.push(p.text);
        if (p.functionCall?.name) fnCalls.push(p.functionCall);
      }
      if (textChunks.length) finalText = textChunks.join('\n').trim();
      if (!fnCalls.length || typeof opts.onToolCall !== 'function') break;

      const responseParts = [];
      for (const call of fnCalls) {
        let resultObj;
        try {
          const out = await opts.onToolCall(call.name, call.args || {});
          resultObj = (out && typeof out === 'object') ? out : { result: out === undefined ? 'ok' : out };
        } catch (err) {
          resultObj = { ok: false, error: err.message || String(err) };
        }
        toolTrace.push({ name: call.name, args: call.args || {}, result: resultObj });
        const fnResp = { name: call.name, response: resultObj };
        if (call.id) fnResp.id = call.id;
        responseParts.push({ functionResponse: fnResp });
      }
      contents.push({ role: 'user', parts: responseParts });
    }
    return { ok: true, modelId: model.id, provider: 'gemini', text: finalText, toolTrace };
  }

  function extractEmulatedToolCalls(text) {
    const calls = [];
    const regex = /<tool_call>\s*([\s\S]*?)\s*<\/tool_call>/gi;
    let m;
    while ((m = regex.exec(text)) !== null) {
      const obj = tryParseLooseJSON(m[1]);
      if (obj && obj.name) calls.push({ name: obj.name, args: obj.args || obj.arguments || {} });
    }
    return { calls, cleanText: String(text || '').replace(regex, '').trim() };
  }

  async function runOpenAICompatChat(model, apiKey, opts) {
    const endpoint = model.provider === 'groq'
      ? 'https://api.groq.com/openai/v1/chat/completions'
      : model.provider === 'xai'
        ? 'https://api.x.ai/v1/chat/completions'
        : 'https://openrouter.ai/api/v1/chat/completions';

    const headers = { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + apiKey };
    if (model.provider === 'openrouter') {
      const origin = window.location.origin;
      headers['HTTP-Referer'] = (origin && origin !== 'null') ? origin : 'https://github.io';
      headers['X-Title'] = 'NexusXR Mixed Reality Studio';
    }

    const rawTools = normalizeTools(opts.tools);
    const cleanTools = rawTools.map((t) => ({
      type: 'function',
      function: { name: t.name, description: t.description, parameters: t.parameters }
    }));

    const useNativeTools = cleanTools.length > 0 && model.supportsTools !== false;

    async function executeLoop(nativeToolsMode) {
      const messages = [];
      let sysPrompt = String(opts.systemPrompt || '');
      if (!nativeToolsMode && rawTools.length) {
        const toolSpec = rawTools.map((t) => `- ${t.name}(${Object.keys(t.parameters.properties || {}).join(', ')}): ${t.description}`).join('\n');
        sysPrompt += '\n\nTO EXECUTE TOOLS, emit:\n<tool_call>{"name":"tool_name","args":{...}}</tool_call>\nAvailable tools:\n' + toolSpec;
      }
      if (sysPrompt) messages.push({ role: 'system', content: sysPrompt });

      const inputMsgs = Array.isArray(opts.messages) ? opts.messages : [];
      for (let i = 0; i < inputMsgs.length; i++) {
        const m = inputMsgs[i];
        if (!m) continue;
        const role = m.role === 'assistant' ? 'assistant' : 'user';
        if (i === inputMsgs.length - 1 && role === 'user' && opts.imageB64 && model.supportsVision) {
          const url = String(opts.imageB64).startsWith('data:') ? opts.imageB64 : 'data:image/jpeg;base64,' + opts.imageB64;
          messages.push({ role: 'user', content: [{ type: 'text', text: String(m.content || '') }, { type: 'image_url', image_url: { url } }] });
        } else {
          messages.push({ role, content: String(m.content || '') });
        }
      }
      if (messages.length === (sysPrompt ? 1 : 0)) messages.push({ role: 'user', content: String(opts.prompt || 'Hello') });

      const maxSteps = Math.max(1, Math.min(8, opts.maxSteps || 5));
      const toolTrace = [];
      let finalText = '';

      for (let step = 0; step < maxSteps; step++) {
        const payload = { model: model.id, messages };
        if (nativeToolsMode && cleanTools.length) { payload.tools = cleanTools; payload.tool_choice = 'auto'; }

        const res = await fetch(endpoint, { method: 'POST', headers, body: JSON.stringify(payload) });
        const rawTxt = await res.text();
        if (!res.ok) {
          const err = new Error(formatApiError(res.status, rawTxt, model.provider, model.id));
          err._raw = rawTxt;
          throw err;
        }

        const data = JSON.parse(rawTxt);
        if (data.error || data.choices?.[0]?.error) {
          const eObj = data.error || data.choices[0].error;
          throw new Error(formatApiError(eObj.code || 502, JSON.stringify({ error: eObj }), model.provider, model.id));
        }

        const msg = data.choices?.[0]?.message;
        if (!msg) throw new Error('Empty response from ' + model.provider);

        const entry = { role: 'assistant', content: msg.content || '' };
        if (Array.isArray(msg.tool_calls) && msg.tool_calls.length) entry.tool_calls = msg.tool_calls;
        messages.push(entry);

        if (nativeToolsMode) {
          if (msg.content) finalText = String(msg.content).trim();
          const toolCalls = Array.isArray(msg.tool_calls) ? msg.tool_calls : [];
          if (!toolCalls.length || typeof opts.onToolCall !== 'function') break;

          for (const tc of toolCalls) {
            const fnName = tc.function?.name;
            const args = tryParseLooseJSON(tc.function?.arguments);
            let resultObj;
            try {
              const out = await opts.onToolCall(fnName, args);
              resultObj = (out && typeof out === 'object') ? out : { result: out === undefined ? 'ok' : out };
            } catch (err) {
              resultObj = { ok: false, error: err.message || String(err) };
            }
            toolTrace.push({ name: fnName, args, result: resultObj });
            messages.push({ role: 'tool', tool_call_id: tc.id || ('call_' + fnName + '_' + step), name: fnName, content: JSON.stringify(resultObj) });
          }
        } else {
          const { calls, cleanText } = extractEmulatedToolCalls(msg.content || '');
          if (cleanText) finalText = cleanText;
          if (!calls.length || typeof opts.onToolCall !== 'function') break;

          const resultsSummary = [];
          for (const c of calls) {
            let resultObj;
            try {
              const out = await opts.onToolCall(c.name, c.args || {});
              resultObj = (out && typeof out === 'object') ? out : { result: out === undefined ? 'ok' : out };
            } catch (err) {
              resultObj = { ok: false, error: err.message || String(err) };
            }
            toolTrace.push({ name: c.name, args: c.args || {}, result: resultObj });
            resultsSummary.push(`${c.name} -> ${JSON.stringify(resultObj)}`);
          }
          messages.push({ role: 'user', content: 'Tool results:\n' + resultsSummary.join('\n') + '\nFinish or fix any errors.' });
        }
      }
      return { ok: true, modelId: model.id, provider: model.provider, text: finalText, toolTrace };
    }

    try { return await executeLoop(useNativeTools); }
    catch (err) {
      if (useNativeTools && /tool|function|404|400/i.test(String(err.message || '') + String(err._raw || ''))) {
        return await executeLoop(false);
      }
      throw err;
    }
  }

        if (b.type === 'tool_use') toolUses.push(b);
      }
      if (textChunks.length) finalText = textChunks.join('\n').trim();
      if (!toolUses.length || typeof opts.onToolCall !== 'function') break;

      const toolResults = [];
      for (const tu of toolUses) {
        let resultObj;
        try {
          const out = await opts.onToolCall(tu.name, tu.input || {});
          resultObj = (out && typeof out === 'object') ? out : { result: out === undefined ? 'ok' : out };
        } catch (err) {
          resultObj = { ok: false, error: err.message || String(err) };
        }
        toolTrace.push({ name: tu.name, args: tu.input || {}, result: resultObj });
        toolResults.push({ type: 'tool_result', tool_use_id: tu.id, content: JSON.stringify(resultObj) });
      }
      messages.push({ role: 'user', content: toolResults });
    }
    return { ok: true, modelId: model.id, provider: 'anthropic', text: finalText, toolTrace };
  }

  async function invokeSingleModel(model, apiKey, opts) {
    if (model.provider === 'gemini') return await runGeminiRestChat(model, apiKey, opts);
    if (model.provider === 'anthropic') return await runAnthropicChat(model, apiKey, opts);
    return await runOpenAICompatChat(model, apiKey, opts);
  }

  function buildCascadeCandidates(primaryModel) {
    const all = getAllModels().filter((m) => m.supportsRest && m.id !== primaryModel.id);
    const priorityIds = [
      'gemini-4-hydrogen', 'gemini-3.8-flash', 'claude-sonnet-4-6', 'claude-haiku-4-5',
      'grok-4.7', 'anthropic/claude-haiku-5.5', 'x-ai/grok-4.7',
      'openai/gpt-oss-120b', 'gemini-3.6-flash', 'llama-3.3-70b-versatile', 'openrouter/free'
    ];
    const ordered = [];
    for (const pid of priorityIds) {
      const m = all.find((x) => x.id === pid);
      if (m && getKey(m.provider)) ordered.push(m);
    }
    return ordered;
  }

  function looksLikeRefusal(res) {
    if (!res || !res.ok || (res.toolTrace && res.toolTrace.length > 0)) return false;
    return /\b(sorry|cannot|can't|couldn't|unable|too complicated|not part of|don't have|no way to|impossible|not supported|as an ai|i'm an ai)\b|لا أستطيع|آسف|غير متاح|معقد/i.test(String(res.text || ''));
  }

  async function chat(rawOpts = {}) {
    const opts = enhanceOptsFor3DCapability(rawOpts);
    const model = getSelectedModel();
    const apiKey = getKey(model.provider);
    const pInfo = PROVIDERS[model.provider];

    if (!apiKey) {
      return { ok: false, error: `Missing ${pInfo.name} key for "${model.id}". Open "AI & Keys" (${pInfo.signupUrl}).` };
    }

    if (model.supportsLiveWS && !model.supportsRest) {
      if (APP.bot?.sendLiveText) {
        try {
          await APP.bot.sendLiveText(opts.prompt || opts.messages?.[opts.messages.length - 1]?.content || '');
          return { ok: true, modelId: model.id, provider: 'gemini', text: '(Sent to Gemini Live)', toolTrace: [] };
        } catch (e) { return { ok: false, error: e.message || String(e) }; }
      }
      return { ok: false, error: `Model "${model.id}" uses Gemini Live WebSocket. Turn Mic ON or pick a REST model.` };
    }

    async function callWithCascade(callOpts) {
      try {
        return await invokeSingleModel(model, apiKey, callOpts);
      } catch (err) {
        if (cascadeEnabled && isTransientError(err)) {
          const backups = buildCascadeCandidates(model);
          for (const b of backups) {
            const bKey = getKey(b.provider);
            if (!bKey) continue;
            try {
              APP.ui?.setStatus?.(`503 on ${model.id} → cascading to ${b.id}...`, 'busy');
              const res = await invokeSingleModel(b, bKey, callOpts);
              APP.ui?.toast?.(`Anti-503 Cascade: answered via ${b.id}`, 3500);
              return res;
            } catch (_) {}
          }
        }
        throw err;
      }
    }

    try {
      let res = await callWithCascade(opts);
      if (opts._is3DSession && looksLikeRefusal(res)) {
        const nudgeMsgs = Array.isArray(opts.messages) ? [...opts.messages] : [{ role: 'user', content: String(opts.prompt || '') }];
        nudgeMsgs.push({ role: 'assistant', content: res.text });
        nudgeMsgs.push({
          role: 'user',
          content: 'SYSTEM OVERRIDE: Do NOT refuse. Call `run_world_fx` or `create_procedural_object` RIGHT NOW to build what the user asked for!'
        });
        res = await callWithCascade(Object.assign({}, opts, { messages: nudgeMsgs }));
      }
      return res;
    } catch (err) {
      return { ok: false, modelId: model.id, provider: model.provider, error: err.message || String(err) };
    }
  }

  // Instant World-FX simulation router + Bot/AI delegation
  async function runTurn(userText, opts = {}) {
    const clean = String(userText || '').trim();
    const low = clean.toLowerCase();

    // Direct instant simulation handling for water / rain / snow / lava requests
    const isSimpleFxCmd = !/\b(explain|board|wikipedia|wiki|portal|code|edit|txt|pen|torch|cape)\b/i.test(low);
    if (isSimpleFxCmd) {
      if (/\b(make|create|spawn|add|start|simulate|simulation)\b.*\b(water|ocean|sea|pool)\b|^(water|water simulation|ماء|مياه|محاكاة ماء)$/i.test(low)) {
        const r = fxEngine.water();
        APP.ui?.toast?.('Geeny: Water simulation active!');
        APP.speakTTS?.('Realistic animated water simulation activated.');
        return r;
      }
      if (/\b(make it|make|start|let it|spawn)\b.*\b(rain|raining|rainstorm)\b|^(rain|raining|مطر|أمطار)$/i.test(low)) {
        const r = fxEngine.rain();
        APP.ui?.toast?.('Geeny: Rainstorm with floor splashes active!');
        APP.speakTTS?.('Rainstorm activated.');
        return r;
      }
      if (/\b(make it|make|start|let it|spawn)\b.*\b(snow|snowing|blizzard)\b|^(snow|snowing|ثلج|ثلوج)$/i.test(low)) {
        const r = fxEngine.snow();
        APP.ui?.toast?.('Geeny: Snowfall active!');
        APP.speakTTS?.('Snowfall activated.');
        return r;
      }
      if (/\b(floor.*lava|lava.*floor|make.*lava)\b|حمم|بركان/i.test(low)) {
        const r = fxEngine.lavaFloor();
        APP.ui?.toast?.('Geeny: The floor is lava!');
        APP.speakTTS?.('Molten lava floor activated.');
        return r;
      }
      if (/\b(stop|clear|remove)\b.*\b(rain|snow|water|lava|weather)\b/i.test(low)) {
        const r = fxEngine.clear();
        APP.ui?.toast?.(r.message);
        return r;
      }
    }

    if (APP.bot && typeof APP.bot.handleCommand === 'function') {
      return await APP.bot.handleCommand(clean, opts);
    }

    const res = await chat({
      systemPrompt: 'You are Geeny, an unlimited WebXR Mixed Reality AI. Use run_world_fx or create_procedural_object immediately.',
      tools: [WORLD_FX_TOOL],
      prompt: clean,
      onToolCall: async (n, a) => (n === 'run_world_fx' ? executeWorldFX(a) : { ok: false })
    });
    if (!res.ok) APP.ui?.toast?.(res.error, 6000);
    else APP.ui?.toast?.(res.text || 'Done.');
    return res;
  }
