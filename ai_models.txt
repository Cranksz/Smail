  // Human-readable error formatter for 401/403, 404, 429, 400
  function formatApiError(status, rawText, provider, modelId) {
    let detail = '';
    try {
      const parsed = JSON.parse(rawText || '{}');
      detail = (parsed.error && (parsed.error.message || parsed.error.status)) || parsed.message || rawText || '';
    } catch (_) {
      detail = String(rawText || '').slice(0, 260);
    }

    const pInfo = PROVIDERS[provider] || { name: provider, signupUrl: '' };

    if (status === 401 || status === 403) {
      return (
        'Authentication error (' + status + ') for ' + pInfo.name + '. ' +
        'Your API key is missing, invalid, or lacks permission for "' + modelId + '". ' +
        'Check your key in AI Models & Keys (' + pInfo.signupUrl + ').' +
        (detail ? ' Details: ' + detail : '')
      );
    }
    if (status === 404) {
      return (
        'Model unavailable (404) on ' + pInfo.name + ': "' + modelId + '" was not found or its free endpoint rotated. ' +
        'Select another free model or refresh OpenRouter :free models.' +
        (detail ? ' Details: ' + detail : '')
      );
    }
    if (status === 429) {
      return (
        'Free tier rate limit reached (429) on ' + pInfo.name + ' for "' + modelId + '". ' +
        'Wait a few seconds and retry, or manually pick another free model/provider.' +
        (detail ? ' Details: ' + detail : '')
      );
    }
    if (status === 400) {
      return (
        'Bad request / schema error (400) on ' + pInfo.name + ' ("' + modelId + '"): ' +
        (detail || 'Invalid parameters, unsupported modality, or missing thoughtSignature in multi-turn call.')
      );
    }
    return pInfo.name + ' error (' + (status || 'network') + ') for "' + modelId + '": ' + (detail || 'Unknown error');
  }

  // Sanitize JSON Schema for tool declarations (zero empty properties objects)
  function sanitizeSchema(schema) {
    if (!schema || typeof schema !== 'object') {
      return { type: 'object', properties: { unused: { type: 'string', description: 'Optional placeholder' } } };
    }
    const copy = JSON.parse(JSON.stringify(schema));
    if (!copy.type) copy.type = 'object';
    if (copy.type === 'object') {
      if (!copy.properties || typeof copy.properties !== 'object' || Object.keys(copy.properties).length === 0) {
        copy.properties = { note: { type: 'string', description: 'Optional note' } };
      } else {
        for (const k of Object.keys(copy.properties)) {
          if (copy.properties[k] && copy.properties[k].type === 'object') {
            copy.properties[k] = sanitizeSchema(copy.properties[k]);
          }
        }
      }
    }
    return copy;
  }

  function normalizeTools(tools) {
    if (!Array.isArray(tools) || tools.length === 0) return [];
    return tools
      .filter((t) => t && t.name)
      .map((t) => ({
        name: String(t.name),
        description: String(t.description || t.name),
        parameters: sanitizeSchema(t.parameters)
      }));
  }

  // Deep-clone helper preserving thoughtSignature and all sibling keys

  // Single-provider Gemini REST turn loop with FULL thoughtSignature preservation
  async function runGeminiRestChat(model, apiKey, opts) {
    const cleanTools = normalizeTools(opts.tools);
    const contents = [];

    // Seed prior messages if passed
    const msgs = Array.isArray(opts.messages) ? opts.messages : [];
    for (let i = 0; i < msgs.length; i++) {
      const m = msgs[i];
      if (!m) continue;
      if (m._geminiParts) {
        contents.push({ role: m.role === 'assistant' ? 'model' : 'user', parts: m._geminiParts.map(clonePart) });
        continue;
      }
      const role = m.role === 'assistant' ? 'model' : 'user';
      const parts = [];
      if (m.content) parts.push({ text: String(m.content) });
      // Attach vision image to the last user message if provided
      if (i === msgs.length - 1 && role === 'user' && opts.imageB64) {
        const cleanB64 = String(opts.imageB64).replace(/^data:image\/\w+;base64,/, '');
        parts.push({ inlineData: { mimeType: opts.imageMime || 'image/jpeg', data: cleanB64 } });
      }
      if (parts.length) contents.push({ role, parts });
    }

    // If no messages were provided, create one user turn
    if (contents.length === 0) {
      const parts = [{ text: String(opts.prompt || 'Hello') }];
      if (opts.imageB64) {
        const cleanB64 = String(opts.imageB64).replace(/^data:image\/\w+;base64,/, '');
        parts.push({ inlineData: { mimeType: opts.imageMime || 'image/jpeg', data: cleanB64 } });
      }
      contents.push({ role: 'user', parts });
    }

    const maxSteps = Math.max(1, Math.min(8, opts.maxSteps || 5));
    const toolTrace = [];
    let finalText = '';
    let lastRawParts = [];

    for (let step = 0; step < maxSteps; step++) {
      const body = { contents };

      if (opts.systemPrompt) {
        body.systemInstruction = { parts: [{ text: String(opts.systemPrompt) }] };
      }
      if (cleanTools.length > 0) {
        body.tools = [{ functionDeclarations: cleanTools }];
      }

      // Note: Do NOT send deprecated temperature/topP/topK/candidateCount on Gemini 3.x / 4.x
      if (!model.isGemini3Plus && typeof opts.temperature === 'number') {
        body.generationConfig = { temperature: opts.temperature };
      }

      const url = 'https://generativelanguage.googleapis.com/v1beta/models/' +
        encodeURIComponent(model.id) + ':generateContent?key=' + encodeURIComponent(apiKey);

      let res;
      try {
        res = await fetch(url, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(body),
          signal: opts.signal
        });
      } catch (netErr) {
        throw new Error('Network error contacting Google Gemini API: ' + (netErr.message || netErr));
      }

      const rawTxt = await res.text();
      if (!res.ok) {
        throw new Error(formatApiError(res.status, rawTxt, 'gemini', model.id));
      }

      let data;
      try {
        data = JSON.parse(rawTxt);
      } catch (_) {
        throw new Error('Invalid JSON response from Gemini API.');
      }

      const cand = data.candidates && data.candidates[0];
      if (!cand || !cand.content || !Array.isArray(cand.content.parts)) {
        const blockReason = (data.promptFeedback && data.promptFeedback.blockReason) || (cand && cand.finishReason) || 'empty candidate';
        throw new Error('Gemini returned no usable output (' + blockReason + ').');
      }

      // Preserve the EXACT raw parts array including thoughtSignature and thought flags!
      const rawParts = cand.content.parts.map(clonePart);
      lastRawParts = rawParts;
      contents.push({ role: 'model', parts: rawParts });

      const textChunks = [];
      const fnCalls = [];
      for (const part of rawParts) {
        if (part.text && !part.thought) {
          textChunks.push(part.text);
        }
        if (part.functionCall && part.functionCall.name) {
          fnCalls.push(part.functionCall);
        }
      }
      if (textChunks.length) {
        finalText = textChunks.join('\n').trim();
      }

      if (fnCalls.length === 0 || typeof opts.onToolCall !== 'function') {
        break;
      }

      // Execute tool calls and send functionResponse back while keeping previous model turn's thoughtSignature intact
      const responseParts = [];
      for (const call of fnCalls) {
        let resultObj;
        try {
          const out = await opts.onToolCall(call.name, call.args || {});
          resultObj = (out && typeof out === 'object') ? out : { result: out === undefined ? 'ok' : out };
        } catch (toolErr) {
          resultObj = { ok: false, error: toolErr && toolErr.message ? toolErr.message : String(toolErr) };
        }
        toolTrace.push({ name: call.name, args: call.args || {}, result: resultObj });
        responseParts.push({
          functionResponse: {
            name: call.name,
            response: resultObj
          }
        });
      }

      contents.push({ role: 'user', parts: responseParts });
    }

    return {
      ok: true,
      modelId: model.id,
      provider: 'gemini',
      text: finalText,
      toolTrace,
      rawParts: lastRawParts
    };
  }

  // Single-provider OpenAI-compatible REST turn loop (Groq & OpenRouter)
  async function runOpenAICompatChat(model, apiKey, opts) {
    const endpoint = model.provider === 'groq'
      ? 'https://api.groq.com/openai/v1/chat/completions'
      : 'https://openrouter.ai/api/v1/chat/completions';

    const headers = {
      'Content-Type': 'application/json',
      'Authorization': 'Bearer ' + apiKey
    };
    if (model.provider === 'openrouter') {
      headers['HTTP-Referer'] = window.location.origin || 'https://github.io';
      headers['X-Title'] = 'WebXR Mixed Reality Studio';
    }

    const cleanTools = normalizeTools(opts.tools).map((t) => ({
      type: 'function',
      function: {
        name: t.name,
        description: t.description,
        parameters: t.parameters
      }
    }));

    const messages = [];
    if (opts.systemPrompt) {
      messages.push({ role: 'system', content: String(opts.systemPrompt) });
    }

    const inputMsgs = Array.isArray(opts.messages) ? opts.messages : [];
    for (let i = 0; i < inputMsgs.length; i++) {
      const m = inputMsgs[i];
      if (!m) continue;
      const role = m.role === 'assistant' ? 'assistant' : 'user';
      const isLastUser = i === inputMsgs.length - 1 && role === 'user';

      if (isLastUser && opts.imageB64 && model.supportsVision) {
        const dataUrl = String(opts.imageB64).startsWith('data:')
          ? String(opts.imageB64)
          : 'data:' + (opts.imageMime || 'image/jpeg') + ';base64,' + opts.imageB64;
        messages.push({
          role: 'user',
          content: [
            { type: 'text', text: String(m.content || '') },
            { type: 'image_url', image_url: { url: dataUrl } }
          ]
        });
      } else {
        messages.push({ role, content: String(m.content || '') });
      }
    }

    if (messages.length === (opts.systemPrompt ? 1 : 0)) {
      if (opts.imageB64 && model.supportsVision) {
        const dataUrl = String(opts.imageB64).startsWith('data:')
          ? String(opts.imageB64)
          : 'data:' + (opts.imageMime || 'image/jpeg') + ';base64,' + opts.imageB64;
        messages.push({
          role: 'user',
          content: [
            { type: 'text', text: String(opts.prompt || 'Hello') },
            { type: 'image_url', image_url: { url: dataUrl } }
          ]
        });
      } else {
        messages.push({ role: 'user', content: String(opts.prompt || 'Hello') });
      }
    }

    const maxSteps = Math.max(1, Math.min(8, opts.maxSteps || 5));
    const toolTrace = [];
    let finalText = '';

    for (let step = 0; step < maxSteps; step++) {
      const payload = {
        model: model.id,
        messages
      };
      if (typeof opts.temperature === 'number') {
        payload.temperature = opts.temperature;
      }
      if (cleanTools.length > 0 && model.supportsTools !== false) {
        payload.tools = cleanTools;
        payload.tool_choice = 'auto';
      }

      let res;
      try {
        res = await fetch(endpoint, {
          method: 'POST',
          headers,
          body: JSON.stringify(payload),
          signal: opts.signal
        });
      } catch (netErr) {
        throw new Error('Network error contacting ' + model.provider + ': ' + (netErr.message || netErr));
      }

      const rawTxt = await res.text();
      if (!res.ok) {
        throw new Error(formatApiError(res.status, rawTxt, model.provider, model.id));
      }

      let data;
      try {
        data = JSON.parse(rawTxt);
      } catch (_) {
        throw new Error('Invalid JSON response from ' + model.provider + '.');
      }

      const choice = data.choices && data.choices[0];
      if (!choice || !choice.message) {
        throw new Error(model.provider + ' returned no choices for "' + model.id + '".');
      }

      const msg = choice.message;
      const assistantEntry = {
        role: 'assistant',
        content: msg.content || ''
      };
      if (Array.isArray(msg.tool_calls) && msg.tool_calls.length > 0) {
        assistantEntry.tool_calls = msg.tool_calls;
      }
      // Preserve any extra_content / reasoning fields verbatim if present
      if (msg.extra_content) assistantEntry.extra_content = msg.extra_content;
      messages.push(assistantEntry);

      if (msg.content) {
        finalText = String(msg.content).trim();
      }

      const toolCalls = Array.isArray(msg.tool_calls) ? msg.tool_calls : [];
      if (toolCalls.length === 0 || typeof opts.onToolCall !== 'function') {
        break;
      }

      for (const tc of toolCalls) {
        const fnName = tc.function && tc.function.name;
        let args = {};
        try {
          args = tc.function && tc.function.arguments ? JSON.parse(tc.function.arguments) : {};
        } catch (_) {
          args = {};
        }
        let resultObj;
        try {
          const out = await opts.onToolCall(fnName, args);
          resultObj = (out && typeof out === 'object') ? out : { result: out === undefined ? 'ok' : out };
        } catch (toolErr) {
          resultObj = { ok: false, error: toolErr && toolErr.message ? toolErr.message : String(toolErr) };
        }
        toolTrace.push({ name: fnName, args, result: resultObj });

        messages.push({
          role: 'tool',
          tool_call_id: tc.id || ('call_' + fnName + '_' + step),
          name: fnName,
          content: JSON.stringify(resultObj)
        });
      }
    }

    return {
      ok: true,
      modelId: model.id,
      provider: model.provider,
      text: finalText,
      toolTrace
    };
  }

  // Unified entry point — ZERO hidden cross-provider switching
  async function chat(opts) {
    const model = getSelectedModel();
    const apiKey = getKey(model.provider);
    const pInfo = PROVIDERS[model.provider];

    if (!apiKey) {
      return {
        ok: false,
        error: 'Missing ' + pInfo.name + ' API key for selected model "' + model.id + '". Open AI Models & Keys to paste your free key (' + pInfo.signupUrl + ').'
      };
    }

    if (model.supportsLiveWS && !model.supportsRest) {
      // If Gemini Live WebSocket is active in bot.txt, allow routing text there; otherwise explain clearly
      if (APP.bot && typeof APP.bot.sendLiveText === 'function' && APP.bot.isLiveConnected && APP.bot.isLiveConnected()) {
        try {
          await APP.bot.sendLiveText(opts.prompt || (opts.messages && opts.messages[opts.messages.length - 1]?.content) || '');
          return { ok: true, modelId: model.id, provider: 'gemini', text: '(Sent to active Gemini Live session)', toolTrace: [] };
        } catch (e) {
          return { ok: false, error: e.message || String(e) };
        }
      }
      return {
        ok: false,
        error: 'Selected model "' + model.id + '" is a real-time Gemini Live WebSocket model. Turn ON the Robot Sphere (Gemini Live) or select a REST model such as "gemini-3.8-flash" or "gemini-4-hydrogen".'
      };
    }

    try {
      if (model.provider === 'gemini') {
        return await runGeminiRestChat(model, apiKey, opts || {});
      }
      if (model.provider === 'groq' || model.provider === 'openrouter') {
        return await runOpenAICompatChat(model, apiKey, opts || {});
      }
      return { ok: false, error: 'Unsupported provider: ' + model.provider };
    } catch (err) {
      return {
        ok: false,
        modelId: model.id,
        provider: model.provider,
        error: err && err.message ? err.message : String(err)
      };
    }
  }

  function refreshPanelUI() {
    if (!panelEl) return;
    const sel = panelEl.querySelector('#aim-model-select');
    const meta = panelEl.querySelector('#aim-model-meta');
    const models = getAllModels();
    const cur = getSelectedModel();

    sel.innerHTML = '';
    const groups = {
      gemini: document.createElement('optgroup'),
      groq: document.createElement('optgroup'),
      openrouter: document.createElement('optgroup')
    };
    groups.gemini.label = 'Google Gemini Free Tier';
    groups.groq.label = 'Groq Free Tier';
    groups.openrouter.label = 'OpenRouter :free ($0 Prompt & Completion)';

    for (const m of models) {
      const opt = document.createElement('option');
      opt.value = m.id;
      opt.textContent = m.label;
      if (m.id === cur.id) opt.selected = true;
      (groups[m.provider] || groups.openrouter).appendChild(opt);
    }

    sel.appendChild(groups.gemini);
    sel.appendChild(groups.groq);
    sel.appendChild(groups.openrouter);

    const orNote = openRouterStatus.loading
      ? 'Loading OpenRouter :free list...'
      : openRouterStatus.error
        ? 'OpenRouter fetch note: ' + openRouterStatus.error
        : 'OpenRouter live :free models loaded: ' + openRouterStatus.count;

    meta.innerHTML =
      '<div><strong>ID:</strong> <code>' + cur.id + '</code> | ' +
      '<strong>Vision:</strong> ' + (cur.supportsVision ? 'Yes' : 'Text-only') + ' | ' +
      '<strong>Tools:</strong> ' + (cur.supportsTools ? 'Yes' : 'No') +
      (cur.supportsLiveWS ? ' | <strong>Mode:</strong> Live WebSocket' : '') + '</div>' +
      '<div>' + (cur.notes || '') + '</div>' +
      '<div style="color:#6ef3d6;margin-top:3px;">' + orNote + '</div>';

    const gKey = getKey('gemini');
    const qKey = getKey('groq');
    const oKey = getKey('openrouter');

    panelEl.querySelector('#aim-key-gemini').value = gKey;
    panelEl.querySelector('#aim-key-groq').value = qKey;
    panelEl.querySelector('#aim-key-openrouter').value = oKey;

    panelEl.querySelector('#aim-st-gemini').textContent = gKey ? 'Saved' : 'Not Set';
