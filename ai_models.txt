  function formatApiError(status, rawText, provider, modelId) {
    let detail = '';
    try {
      const p = JSON.parse(rawText || '{}');
      detail = p.error?.message || p.message || rawText || '';
    } catch (_) {
      detail = String(rawText || '').slice(0, 240);
    }
    const pInfo = PROVIDERS[provider] || { name: provider, signupUrl: '' };
    if (status === 401 || status === 403) {
      return `Invalid or missing API key (${status}) for ${pInfo.name}. Get a free key at ${pInfo.signupUrl}. ${detail}`;
    }
    if (status === 404) {
      return `Model "${modelId}" unavailable (404) on ${pInfo.name}. Pick another free model or refresh OpenRouter :free. ${detail}`;
    }
    if (status === 429) {
      return `Free rate limit reached (429) on ${pInfo.name} for "${modelId}". Wait a moment or choose another free model. ${detail}`;
    }
    if (status === 400) {
      return `Bad request/schema (400) on ${pInfo.name} ("${modelId}"): ${detail}`;
    }
    return `${pInfo.name} error (${status || 'network'}) for "${modelId}": ${detail}`;
  }

  function sanitizeSchema(schema) {
    if (!schema || typeof schema !== 'object') {
      return { type: 'object', properties: { note: { type: 'string', description: 'Optional note' } } };
    }
    const copy = JSON.parse(JSON.stringify(schema));
    if (!copy.type) copy.type = 'object';
    if (copy.type === 'object') {
      if (!copy.properties || typeof copy.properties !== 'object' || Object.keys(copy.properties).length === 0) {
        copy.properties = { note: { type: 'string', description: 'Optional note' } };
      } else {
        for (const k of Object.keys(copy.properties)) {
          if (copy.properties[k]?.type === 'object') {
            copy.properties[k] = sanitizeSchema(copy.properties[k]);
          }
        }
      }
    }
    return copy;
  }

  function normalizeTools(tools) {
    if (!Array.isArray(tools) || !tools.length) return [];
    return tools.filter((t) => t && t.name).map((t) => ({
      name: String(t.name),
      description: String(t.description || t.name),
      parameters: sanitizeSchema(t.parameters)
    }));
  }

  async function runGeminiRestChat(model, apiKey, opts) {
    const cleanTools = normalizeTools(opts.tools);
    const contents = [];
    const msgs = Array.isArray(opts.messages) ? opts.messages : [];

    for (let i = 0; i < msgs.length; i++) {
      const m = msgs[i];
      if (!m) continue;
      if (m._geminiParts) {
        contents.push({ role: m.role === 'assistant' ? 'model' : 'user', parts: JSON.parse(JSON.stringify(m._geminiParts)) });
        continue;
      }
      const role = m.role === 'assistant' ? 'model' : 'user';
      const parts = [];
      if (m.content) parts.push({ text: String(m.content) });
      if (i === msgs.length - 1 && role === 'user' && opts.imageB64) {
        parts.push({
          inlineData: {
            mimeType: opts.imageMime || 'image/jpeg',
            data: String(opts.imageB64).replace(/^data:image\/\w+;base64,/, '')
          }
        });
      }
      if (parts.length) contents.push({ role, parts });
    }

    if (contents.length === 0) {
      const parts = [{ text: String(opts.prompt || 'Hello') }];
      if (opts.imageB64) {
        parts.push({
          inlineData: {
            mimeType: opts.imageMime || 'image/jpeg',
            data: String(opts.imageB64).replace(/^data:image\/\w+;base64,/, '')
          }
        });
      }
      contents.push({ role: 'user', parts });
    }

    const maxSteps = Math.max(1, Math.min(8, opts.maxSteps || 5));
    const toolTrace = [];
    let finalText = '';

    for (let step = 0; step < maxSteps; step++) {
      const body = { contents };
      if (opts.systemPrompt) body.systemInstruction = { parts: [{ text: String(opts.systemPrompt) }] };
      if (cleanTools.length) body.tools = [{ functionDeclarations: cleanTools }];

      const url = 'https://generativelanguage.googleapis.com/v1beta/models/' +
        encodeURIComponent(model.id) + ':generateContent?key=' + encodeURIComponent(apiKey);

      const res = await fetch(url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(body)
      });
      const rawTxt = await res.text();
      if (!res.ok) throw new Error(formatApiError(res.status, rawTxt, 'gemini', model.id));

      const data = JSON.parse(rawTxt);
      const cand = data.candidates?.[0];
      if (!cand?.content?.parts) {
        throw new Error('Gemini returned empty output (' + (cand?.finishReason || 'unknown') + ').');
      }

      // Preserve exact rawParts including thoughtSignature & thought parts!
      const rawParts = JSON.parse(JSON.stringify(cand.content.parts));
      contents.push({ role: 'model', parts: rawParts });

      const textChunks = [];
      const fnCalls = [];
      for (const p of rawParts) {
        if (p.text && !p.thought) textChunks.push(p.text);
        if (p.functionCall?.name) fnCalls.push(p.functionCall);
      }
      if (textChunks.length) finalText = textChunks.join('\n').trim();

      if (!fnCalls.length || typeof opts.onToolCall !== 'function') break;

      const responseParts = [];
      for (const call of fnCalls) {
        let resultObj;
        try {
          const out = await opts.onToolCall(call.name, call.args || {});
          resultObj = (out && typeof out === 'object') ? out : { result: out === undefined ? 'ok' : out };
        } catch (err) {
          resultObj = { ok: false, error: err.message || String(err) };
        }
        toolTrace.push({ name: call.name, args: call.args || {}, result: resultObj });
        responseParts.push({ functionResponse: { name: call.name, response: resultObj } });
      }
      contents.push({ role: 'user', parts: responseParts });
    }

    return { ok: true, modelId: model.id, provider: 'gemini', text: finalText, toolTrace };
  }

  async function runOpenAICompatChat(model, apiKey, opts) {
    const endpoint = model.provider === 'groq'
      ? 'https://api.groq.com/openai/v1/chat/completions'
      : 'https://openrouter.ai/api/v1/chat/completions';

    const headers = {
      'Content-Type': 'application/json',
      'Authorization': 'Bearer ' + apiKey
    };
    if (model.provider === 'openrouter') {
      headers['HTTP-Referer'] = window.location.origin || 'https://github.io';
      headers['X-Title'] = 'NexusXR Mixed Reality Studio';
    }

    const cleanTools = normalizeTools(opts.tools).map((t) => ({
      type: 'function',
      function: { name: t.name, description: t.description, parameters: t.parameters }
    }));

    const messages = [];
    if (opts.systemPrompt) messages.push({ role: 'system', content: String(opts.systemPrompt) });

    const inputMsgs = Array.isArray(opts.messages) ? opts.messages : [];
    for (let i = 0; i < inputMsgs.length; i++) {
      const m = inputMsgs[i];
      if (!m) continue;
      const role = m.role === 'assistant' ? 'assistant' : 'user';
      if (i === inputMsgs.length - 1 && role === 'user' && opts.imageB64 && model.supportsVision) {
        const url = String(opts.imageB64).startsWith('data:')
          ? opts.imageB64
          : 'data:image/jpeg;base64,' + opts.imageB64;
        messages.push({
          role: 'user',
          content: [{ type: 'text', text: String(m.content || '') }, { type: 'image_url', image_url: { url } }]
        });
      } else {
        messages.push({ role, content: String(m.content || '') });
      }
    }

    if (messages.length === (opts.systemPrompt ? 1 : 0)) {
      messages.push({ role: 'user', content: String(opts.prompt || 'Hello') });
    }

    const maxSteps = Math.max(1, Math.min(8, opts.maxSteps || 5));
    const toolTrace = [];
    let finalText = '';

    for (let step = 0; step < maxSteps; step++) {
      const payload = { model: model.id, messages };
      if (cleanTools.length && model.supportsTools !== false) {
        payload.tools = cleanTools;
        payload.tool_choice = 'auto';
      }

      const res = await fetch(endpoint, { method: 'POST', headers, body: JSON.stringify(payload) });
      const rawTxt = await res.text();
      if (!res.ok) throw new Error(formatApiError(res.status, rawTxt, model.provider, model.id));

      const data = JSON.parse(rawTxt);
      const msg = data.choices?.[0]?.message;
      if (!msg) throw new Error('Empty response from ' + model.provider);

      const entry = { role: 'assistant', content: msg.content || '' };
      if (Array.isArray(msg.tool_calls) && msg.tool_calls.length) entry.tool_calls = msg.tool_calls;
      messages.push(entry);

      if (msg.content) finalText = String(msg.content).trim();
      const toolCalls = Array.isArray(msg.tool_calls) ? msg.tool_calls : [];
      if (!toolCalls.length || typeof opts.onToolCall !== 'function') break;

      for (const tc of toolCalls) {
        const fnName = tc.function?.name;
        let args = {};
        try { args = tc.function?.arguments ? JSON.parse(tc.function.arguments) : {}; } catch (_) {}
        let resultObj;
        try {
          const out = await opts.onToolCall(fnName, args);
          resultObj = (out && typeof out === 'object') ? out : { result: out === undefined ? 'ok' : out };
        } catch (err) {
          resultObj = { ok: false, error: err.message || String(err) };
        }
        toolTrace.push({ name: fnName, args, result: resultObj });
        messages.push({
          role: 'tool',
          tool_call_id: tc.id || ('call_' + fnName + '_' + step),
          name: fnName,
          content: JSON.stringify(resultObj)
        });
      }
    }

    return { ok: true, modelId: model.id, provider: model.provider, text: finalText, toolTrace };
  }

  async function chat(opts = {}) {
    const model = getSelectedModel();
    const apiKey = getKey(model.provider);
    const pInfo = PROVIDERS[model.provider];

    if (!apiKey) {
      return {
        ok: false,
        error: `Missing ${pInfo.name} key for "${model.id}". Open "AI & Keys" to add your free key (${pInfo.signupUrl}).`
      };
    }

    if (model.supportsLiveWS && !model.supportsRest) {
      if (APP.bot?.sendLiveText && APP.bot?.isLiveConnected?.()) {
        await APP.bot.sendLiveText(opts.prompt || opts.messages?.[opts.messages.length - 1]?.content || '');
        return { ok: true, modelId: model.id, provider: 'gemini', text: '(Sent to Gemini Live)', toolTrace: [] };
      }
      return {
        ok: false,
        error: `Model "${model.id}" uses Gemini Live WebSocket. Turn ON Robot Live or choose a REST model like "gemini-3.8-flash".`
      };
    }

    try {
      if (model.provider === 'gemini') return await runGeminiRestChat(model, apiKey, opts);
      return await runOpenAICompatChat(model, apiKey, opts);
    } catch (err) {
      return { ok: false, modelId: model.id, provider: model.provider, error: err.message || String(err) };
    }
  }

  async function runTurn(userText, opts = {}) {
    if (APP.bot && typeof APP.bot.handleCommand === 'function') {
      return await APP.bot.handleCommand(userText, opts);
    }

    const model = getSelectedModel();
    APP.ui.setStatus(`Thinking (${model.id})...`, 'busy');

    const instSummary = !opts.ignoreInstructions && typeof APP.getInstructionSummaryForAI === 'function'
      ? APP.getInstructionSummaryForAI()
      : '';

    const systemPrompt = [
      'You are NexusXR AI inside a WebXR Mixed Reality studio.',
      'ALWAYS use create_procedural_object to build any requested 3D object with Three.js (APP, group, THREE, V3) and APP.craft helpers.',
      'NEVER refuse a creation request. If create_procedural_object returns an error, fix the code and retry.',
      instSummary ? ('\nACTIVE INSTRUCTIONS (rules & inspiration — be creative when sculpting):\n' + instSummary) : ''
    ].join('\n');

    const tools = [
      {
        name: 'create_procedural_object',
        description: 'Sculpt and materialize a 3D Three.js object in the scene.',
        parameters: {
          type: 'object',
          properties: {
            name: { type: 'string', description: 'Object name' },
            js_code: { type: 'string', description: 'JS body using (APP, group, THREE, V3) adding meshes to group.' },
            holdable: { type: 'boolean', description: 'Whether user can hold it.' }
          },
          required: ['name', 'js_code']
        }
      },
      {
        name: 'spawn_repo_asset',
        description: 'Spawn a .glb, image, or video asset by name.',
        parameters: {
          type: 'object',
          properties: { name: { type: 'string', description: 'Asset name' } },
          required: ['name']
        }
      }
    ];

    standaloneHistory.push({ role: 'user', content: userText });
    if (standaloneHistory.length > 8) standaloneHistory.splice(0, standaloneHistory.length - 8);

    const res = await chat({
      systemPrompt,
      messages: standaloneHistory,
      tools,
      imageB64: model.supportsVision ? (APP.captureVisionFrame?.() || APP.latestVisionB64) : undefined,
      onToolCall: async (name, args) => {
        if (name === 'create_procedural_object') {
          return APP.createProcedural(args.js_code, { name: args.name, holdable: args.holdable !== false });
        }
        if (name === 'spawn_repo_asset') {
          return await APP.spawnAsset(args.name);
        }
        return { ok: false, error: 'Unknown tool: ' + name };
      }
    });

    if (!res.ok) {
      APP.ui.setStatus('AI Error', 'err');
      APP.ui.toast(res.error, 6000);
      return res;
    }

    const reply = res.text || 'Done.';
    standaloneHistory.push({ role: 'assistant', content: reply });
    APP.ui.setStatus('Ready', 'ok');
    APP.ui.toast(reply);
    APP.speakTTS?.(reply);
    return res;
  }
