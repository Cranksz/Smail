  function formatApiError(status, rawText, provider, modelId) {
    let detail = '';
    try { const p = JSON.parse(rawText || '{}'); detail = p.error?.message || p.choices?.[0]?.error?.message || p.message || rawText || ''; }
    catch (_) { detail = String(rawText || '').slice(0, 240); }
    const pInfo = PROVIDERS[provider] || { name: provider };
    if (status === 401 || status === 403) return `Invalid/rejected key (${status}) for ${pInfo.name}. ${detail}`;
    if (status === 402) return `Credits required (402) on ${pInfo.name} for "${modelId}". ${detail}`;
    if (status === 404) return `Model "${modelId}" unavailable (404) on ${pInfo.name}. ${detail}`;
    if (status === 429) return `Rate limit (429) on ${pInfo.name} for "${modelId}". ${detail}`;
    if (status === 503 || /high demand|overloaded|unavailable/i.test(detail)) return `${pInfo.name} "${modelId}" busy (${status || 503}). ${detail}`;
    return `${pInfo.name} error (${status || 'network'}) for "${modelId}": ${detail}`;
  }

  }

  async function transcribeWithGroq(audioBlob, langHint) {
    const gKey = getKey('groq');
    if (!gKey) return { ok: false, error: 'Missing Groq key (gsk_...) for Whisper Voice.' };
    const fd = new FormData();

    if (langCode) fd.append('language', langCode);

    const res = await fetchWithTimeout('https://api.groq.com/openai/v1/audio/transcriptions', {
      method: 'POST',
      headers: { Authorization: 'Bearer ' + gKey },
      body: fd
    }, 15000);
    const txt = await res.text();
    if (!res.ok) return { ok: false, error: formatApiError(res.status, txt, 'groq', 'whisper-large-v3-turbo') };
    try {
      const j = JSON.parse(txt);
      return { ok: true, text: String(j.text || '').trim() };
    } catch (_) {
      return { ok: false, error: 'Invalid Whisper response.' };
    }
  }

  async function stopGroqVoiceMic() {

  async function startGroqVoiceMic() {
    await stopGroqVoiceMic();
    if (!getKey('groq')) return { ok: false, error: 'Add a free Groq key (gsk_...) in AI & Keys to use Groq Whisper Voice.' };
    if (!navigator.mediaDevices?.getUserMedia) return { ok: false, error: 'Microphone unavailable.' };

    groqMicStream = await navigator.mediaDevices.getUserMedia({ audio: { echoCancellation: true, noiseSuppression: true, autoGainControl: true } });
    const AC = window.AudioContext || window.webkitAudioContext;
    groqAudioCtx = new AC();
    const src = groqAudioCtx.createMediaStreamSource(groqMicStream);
    groqAnalyser = groqAudioCtx.createAnalyser();
    groqAnalyser.fftSize = 512;
    src.connect(groqAnalyser);

    const dataArr = new Uint8Array(groqAnalyser.frequencyBinCount);
    groqLoopTimer = setInterval(() => {
      if (!groqAnalyser || APP.botSpeaking) return;
      groqAnalyser.getByteTimeDomainData(dataArr);
      let sum = 0;
      for (let i = 0; i < dataArr.length; i++) { const v = (dataArr[i] - 128) / 128; sum += v * v; }
      const rms = Math.sqrt(sum / dataArr.length);
      APP.micLevel = Math.min(1, rms * 6.0);

      const now = performance.now();
      if (rms > 0.032) {
        groqSilenceStart = 0;
        if (!groqRecording) {
          groqRecording = true; groqChunks = [];
          try {
            groqRecorder = new MediaRecorder(groqMicStream);
            groqRecorder.ondataavailable = (e) => { if (e.data && e.data.size > 0) groqChunks.push(e.data); };
            groqRecorder.onstop = async () => {
              const blob = new Blob(groqChunks, { type: groqRecorder?.mimeType || 'audio/webm' });
              if (blob.size < 1800) return;
              APP.ui?.setStatus?.('Groq Voice transcribing...', 'busy');
              const tr = await transcribeWithGroq(blob);
              if (tr.ok && tr.text) {
                APP.ui?.toast?.('You: ' + tr.text, 2400);
                if (getSelectedModel().supportsLiveWS && APP.bot?.sendLiveText) await APP.bot.sendLiveText(tr.text);
                else if (APP.executeCommand) await APP.executeCommand(tr.text, 'voice');
              }
            };
            groqRecorder.start();
          } catch (_) { groqRecording = false; }
        }
      } else if (groqRecording) {
        if (!groqSilenceStart) groqSilenceStart = now;
        else if (now - groqSilenceStart > 750) { groqRecording = false; try { groqRecorder?.stop(); } catch (_) {} }
      }
    }, 60);

    APP.ui?.toast?.('Groq Whisper Voice Mic active!', 2800);
    return { ok: true };
  }

  function executeWorldFX(args = {}) {

    const name = String(args.name || 'WorldFX').trim();

    const V3 = (x = 0, y = 0, z = 0) => new THREE.Vector3(x, y, z);
    try {
      const beforeSceneCount = APP.scene.children.length;
      new Function('APP', 'scene', 'group', 'THREE', 'V3', clean)(APP, APP.scene, group, THREE, V3);
      if (group.children.length === 0 && APP.scene.children.length > beforeSceneCount) {
        for (const obj of APP.scene.children.slice(beforeSceneCount)) { if (obj && obj !== group) { APP.scene.remove(obj); group.add(obj); } }
      }
      const hasUpdate = typeof group.userData?.onUpdate === 'function';
      if (group.children.length === 0 && !hasUpdate) return { ok: false, error: 'js_code added 0 meshes to `group`.' };

      APP.disposeObject3D?.(group);

    }

  function tryParseLooseJSON(rawStr) {
    if (!rawStr) return {};
    if (typeof rawStr === 'object') return rawStr;
    try { return JSON.parse(rawStr); } catch (_) {}
    try { return JSON.parse(String(rawStr).replace(/"([^"\\]*(\\.[^"\\]*)*)"/g, (m) => m.replace(/\r?\n/g, '\\n'))); }
    catch (_) { return {}; }
  }

  function enhanceOptsFor3DCapability(opts) {
    const is3DSession = Array.isArray(opts.tools) && opts.tools.some((t) => t && t.name === 'create_procedural_object');
    if (!is3DSession) return opts;

    const nextTools = opts.tools.map((t) => (t && t.name === 'run_world_fx' ? WORLD_FX_TOOL_PURE : t));
    if (!nextTools.some((t) => t.name === 'run_world_fx')) nextTools.push(WORLD_FX_TOOL_PURE);

    const origOnToolCall = opts.onToolCall;
    const wrappedOnToolCall = async (name, args) => {
      if (name === 'run_world_fx') return executeWorldFX(args || {});
      return typeof origOnToolCall === 'function' ? await origOnToolCall(name, args) : { ok: false, error: 'No tool handler.' };
    };

    return Object.assign({}, opts, {
      tools: nextTools,
      systemPrompt: String(opts.systemPrompt || '') + '\nUNLIMITED 3D CREATION: Never refuse; write custom Three.js js_code in `create_procedural_object` or `run_world_fx`.',
      onToolCall: wrappedOnToolCall,
      _is3DSession: true
    });
  }

  function sanitizeSchema(schema) {
    if (!schema || typeof schema !== 'object') return { type: 'object', properties: { note: { type: 'string', description: 'Optional note' } } };
    const copy = JSON.parse(JSON.stringify(schema));
    if (!copy.type) copy.type = 'object';
    if (copy.type === 'object') {
      if (!copy.properties || typeof copy.properties !== 'object' || Object.keys(copy.properties).length === 0) copy.properties = { note: { type: 'string', description: 'Optional note' } };
      else for (const k of Object.keys(copy.properties)) if (copy.properties[k]?.type === 'object') copy.properties[k] = sanitizeSchema(copy.properties[k]);
    }
    return copy;
  }

  function normalizeTools(tools) {
    if (!Array.isArray(tools) || !tools.length) return [];
    return tools.filter((t) => t && t.name).map((t) => ({ name: String(t.name), description: String(t.description || t.name), parameters: sanitizeSchema(t.parameters) }));
  }

  function extractEmulatedToolCalls(text) {
    const calls = [];
    const regex = /<tool_call>\s*([\s\S]*?)\s*<\/tool_call>/gi;
    let m;
    while ((m = regex.exec(text)) !== null) {
      const obj = tryParseLooseJSON(m[1]);
      if (obj && obj.name) calls.push({ name: obj.name, args: obj.args || obj.arguments || {} });
    }
    return { calls, cleanText: String(text || '').replace(regex, '').trim() };
  }

    const contents = [];
    const msgs = Array.isArray(opts.messages) ? opts.messages : [];

    for (let i = 0; i < msgs.length; i++) {
      const m = msgs[i]; if (!m) continue;
      const role = m.role === 'assistant' ? 'model' : 'user', parts = [];
      if (m.content) parts.push({ text: String(m.content) });
      if (i === msgs.length - 1 && role === 'user' && opts.imageB64) {
        parts.push({ inlineData: { mimeType: opts.imageMime || 'image/jpeg', data: String(opts.imageB64).replace(/^data:image\/\w+;base64,/, '') } });
      }
      if (parts.length) contents.push({ role, parts });
    }
    if (!contents.length) {
      const parts = [{ text: String(opts.prompt || 'Hello') }];
      if (opts.imageB64) parts.push({ inlineData: { mimeType: opts.imageMime || 'image/jpeg', data: String(opts.imageB64).replace(/^data:image\/\w+;base64,/, '') } });
      contents.push({ role: 'user', parts });
    }

    const maxSteps = Math.max(1, Math.min(8, opts.maxSteps || 5)), toolTrace = [];
    let finalText = '', activeKeyIdx = 0;

    for (let step = 0; step < maxSteps; step++) {
      const body = { contents };
      if (opts.systemPrompt) body.systemInstruction = { parts: [{ text: String(opts.systemPrompt) }] };
      if (cleanTools.length) body.tools = [{ functionDeclarations: cleanTools }];
      if (model.thinkingLevel) body.generationConfig = { thinkingConfig: { thinkingLevel: String(model.thinkingLevel).toLowerCase() } };

      let data = null, lastErr = '';
      const totalTries = Math.max(1, keyList.length);
      for (let attempt = 0; attempt < totalTries; attempt++) {
        const curKey = keyList[activeKeyIdx % keyList.length];
        const url = 'https://generativelanguage.googleapis.com/v1beta/models/' + encodeURIComponent(model.id) + ':generateContent?key=' + encodeURIComponent(curKey);
        const res = await fetchWithTimeout(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) }, 16000);
        const rawTxt = await res.text();
        if (res.ok) { data = JSON.parse(rawTxt); break; }
        lastErr = formatApiError(res.status, rawTxt, 'gemini', model.id);
        if (res.status === 400 && body.generationConfig && /thinking/i.test(rawTxt)) {
          delete body.generationConfig;
          continue;
        }
        if (res.status === 503 || res.status === 429) {
          if (keyList.length > 1 && attempt + 1 < totalTries) { activeKeyIdx++; continue; }
        }
        throw new Error(lastErr);
      }
      if (!data) throw new Error(lastErr || 'Gemini request failed.');

      const cand = data.candidates?.[0];
      if (!cand?.content?.parts) throw new Error('Gemini returned empty output.');
      const rawParts = JSON.parse(JSON.stringify(cand.content.parts));
      contents.push({ role: 'model', parts: rawParts });

      const textChunks = [], fnCalls = [];
      for (const p of rawParts) {
        if (p.text && !p.thought) textChunks.push(p.text);
        if (p.functionCall?.name) fnCalls.push(p.functionCall);
      }
      if (textChunks.length) finalText = textChunks.join('\n').trim();
      if (!fnCalls.length || typeof opts.onToolCall !== 'function') break;

      const responseParts = [];
      for (const call of fnCalls) {
        let resultObj;
        try {
          const out = await opts.onToolCall(call.name, call.args || {});
          resultObj = (out && typeof out === 'object') ? out : { result: out === undefined ? 'ok' : out };
        } catch (err) { resultObj = { ok: false, error: err.message || String(err) }; }
        toolTrace.push({ name: call.name, args: call.args || {}, result: resultObj });
        const fnResp = { name: call.name, response: resultObj };
        if (call.id) fnResp.id = call.id;
        responseParts.push({ functionResponse: fnResp });
      }
      contents.push({ role: 'user', parts: responseParts });
    }
    return { ok: true, modelId: model.id, provider: 'gemini', text: finalText, toolTrace };
  }

  async function runOpenRouterChat(model, apiKey, opts) {
    const endpoint = 'https://openrouter.ai/api/v1/chat/completions';
    const origin = window.location.origin;
    const headers = {
      'Content-Type': 'application/json',
      'Authorization': 'Bearer ' + apiKey,
      'HTTP-Referer': (origin && origin !== 'null') ? origin : 'https://github.io',
      'X-Title': 'NexusXR Mixed Reality Studio'
    };

    const rawTools = normalizeTools(opts.tools);
    const cleanTools = rawTools.map((t) => ({ type: 'function', function: { name: t.name, description: t.description, parameters: t.parameters } }));
    const useNativeTools = cleanTools.length > 0 && model.supportsTools !== false;

    async function executeLoop(nativeToolsMode, allowVision) {
      const messages = [];
      let sysPrompt = String(opts.systemPrompt || '');
      if (!nativeToolsMode && rawTools.length) {
        const toolSpec = rawTools.map((t) => `- ${t.name}(${Object.keys(t.parameters.properties || {}).join(', ')}): ${t.description}`).join('\n');
        sysPrompt += '\n\nTO EXECUTE TOOLS, emit:\n<tool_call>{"name":"tool_name","args":{...}}</tool_call>\nAvailable tools:\n' + toolSpec;
      }
      if (sysPrompt) messages.push({ role: 'system', content: sysPrompt });

      const inputMsgs = Array.isArray(opts.messages) ? opts.messages : [];
      for (let i = 0; i < inputMsgs.length; i++) {
        const m = inputMsgs[i]; if (!m) continue;
        const role = m.role === 'assistant' ? 'assistant' : 'user';
        if (i === inputMsgs.length - 1 && role === 'user' && opts.imageB64 && allowVision) {
          const url = String(opts.imageB64).startsWith('data:') ? opts.imageB64 : 'data:image/jpeg;base64,' + opts.imageB64;
          messages.push({ role: 'user', content: [{ type: 'text', text: String(m.content || '') }, { type: 'image_url', image_url: { url } }] });
        } else {
          messages.push({ role, content: String(m.content || '') });
        }
      }
      if (messages.length === (sysPrompt ? 1 : 0)) {
        const pTxt = String(opts.prompt || 'Hello');
        if (opts.imageB64 && allowVision) {
          const url = String(opts.imageB64).startsWith('data:') ? opts.imageB64 : 'data:image/jpeg;base64,' + opts.imageB64;
          messages.push({ role: 'user', content: [{ type: 'text', text: pTxt }, { type: 'image_url', image_url: { url } }] });
        } else {
          messages.push({ role: 'user', content: pTxt });
        }
      }

      const maxSteps = Math.max(1, Math.min(8, opts.maxSteps || 5)), toolTrace = [];
      let finalText = '';

      for (let step = 0; step < maxSteps; step++) {
        const payload = { model: model.id, messages };
        if (nativeToolsMode && cleanTools.length) { payload.tools = cleanTools; payload.tool_choice = 'auto'; }

        const res = await fetchWithTimeout(endpoint, { method: 'POST', headers, body: JSON.stringify(payload) }, 16000);
        const rawTxt = await res.text();
        if (!res.ok) {
          const err = new Error(formatApiError(res.status, rawTxt, 'openrouter', model.id));
          err._raw = rawTxt; err._status = res.status; throw err;
        }
        const data = JSON.parse(rawTxt);
        if (data.error || data.choices?.[0]?.error) {
          const eObj = data.error || data.choices[0].error;
          throw new Error(formatApiError(eObj.code || 502, JSON.stringify({ error: eObj }), 'openrouter', model.id));
        }

        const msg = data.choices?.[0]?.message;
        if (!msg) throw new Error('Empty response from OpenRouter');
        const entry = { role: 'assistant', content: msg.content || '' };
        if (Array.isArray(msg.tool_calls) && msg.tool_calls.length) entry.tool_calls = msg.tool_calls;
        messages.push(entry);

        if (nativeToolsMode) {
          if (msg.content) finalText = String(msg.content).trim();
          const toolCalls = Array.isArray(msg.tool_calls) ? msg.tool_calls : [];
          if (!toolCalls.length || typeof opts.onToolCall !== 'function') break;
          for (const tc of toolCalls) {
            const fnName = tc.function?.name, args = tryParseLooseJSON(tc.function?.arguments);
            let resultObj;
            try {
              const out = await opts.onToolCall(fnName, args);
              resultObj = (out && typeof out === 'object') ? out : { result: out === undefined ? 'ok' : out };
            } catch (err) { resultObj = { ok: false, error: err.message || String(err) }; }
            toolTrace.push({ name: fnName, args, result: resultObj });
            messages.push({ role: 'tool', tool_call_id: tc.id || ('call_' + fnName + '_' + step), name: fnName, content: JSON.stringify(resultObj) });
          }
        } else {
          const { calls, cleanText } = extractEmulatedToolCalls(msg.content || '');
          if (cleanText) finalText = cleanText;
          if (!calls.length || typeof opts.onToolCall !== 'function') break;
          const resultsSummary = [];
          for (const c of calls) {
            let resultObj;
            try {
              const out = await opts.onToolCall(c.name, c.args || {});
              resultObj = (out && typeof out === 'object') ? out : { result: out === undefined ? 'ok' : out };
            } catch (err) { resultObj = { ok: false, error: err.message || String(err) }; }
            toolTrace.push({ name: c.name, args: c.args || {}, result: resultObj });
            resultsSummary.push(`${c.name} -> ${JSON.stringify(resultObj)}`);
          }
          messages.push({ role: 'user', content: 'Tool results:\n' + resultsSummary.join('\n') + '\nFinish or fix any errors.' });
        }
      }
      return { ok: true, modelId: model.id, provider: 'openrouter', text: finalText, toolTrace };
    }

    try { return await executeLoop(useNativeTools, !!model.supportsVision); }
    catch (err) {
      const msgStr = String(err.message || '') + String(err._raw || '');
      if (model.supportsVision && opts.imageB64 && /image|vision|multimodal|content.*array|400|404/i.test(msgStr)) {
        try { return await executeLoop(false, false); } catch (_) {}
      }
      if (useNativeTools && /tool|function|404|400/i.test(msgStr)) {
        return await executeLoop(false, !!model.supportsVision);
      }
      throw err;
    }
  }

  async function invokeSingleModel(model, apiKeyOrPool, opts) {
    if (model.provider === 'gemini') return await runGeminiRestChat(model, apiKeyOrPool, opts);
    return await runOpenRouterChat(model, apiKeyOrPool, opts);
  }

  function buildCascadeCandidates(primaryModel, options = {}) {
    // Automatic fallback is restricted to the two approved Gemini models.
    // OpenRouter Free Router is manual-only because it may internally choose
    // different free models of variable quality.
    if (primaryModel.provider !== 'gemini') return [];
    return ['gemini-4-hydrogen', 'gemini-3.8-flash']
      .filter((id) => id !== primaryModel.id)
      .map((id) => STATIC_MODELS.find((m) => m.id === id))
      .filter((m) => m && getKey('gemini') &&
        (!options.excludeProvider || m.provider !== options.excludeProvider));
  }

  function looksLikeRefusal(res) {
    if (!res || !res.ok || (res.toolTrace && res.toolTrace.length > 0)) return false;
    return /\b(sorry|cannot|can't|couldn't|unable|too complicated|not part of|don't have|no way to|impossible|not supported|as an ai|i'm an ai)\b|لا أستطيع|آسف|غير متاح|معقد/i.test(String(res.text || ''));
  }

  async function executeRestChat(targetModel, rawOpts) {
    const opts = enhanceOptsFor3DCapability(rawOpts);
    const apiKeyPool = targetModel.provider === 'gemini' ? getKeyPool('gemini') : getKey(targetModel.provider);
    const hasKey = Array.isArray(apiKeyPool) ? apiKeyPool.length > 0 : !!apiKeyPool;
    const pInfo = PROVIDERS[targetModel.provider] || { name: targetModel.provider, signupUrl: '' };

    if (!hasKey) {
      if (cascadeEnabled) {
        const backups = buildCascadeCandidates(targetModel);
        let lastError = '';
        for (const b of backups) {
          const bPool = b.provider === 'gemini' ? getKeyPool('gemini') : getKey(b.provider);
          try {
            APP.ui?.setStatus?.(`Fallback → ${b.id}...`, 'busy');
            return await invokeSingleModel(b, bPool, opts);
          } catch (e) { lastError = e.message || String(e); }
        }
        if (lastError) return { ok: false, error: lastError };
      }
      return { ok: false, error: `No usable API key for "${targetModel.id}". Add a Gemini or OpenRouter key in AI & Keys.` };
    }

    const showGenFX = !!opts._is3DSession;
    if (showGenFX) {
      const lbl = String(opts.prompt || opts.messages?.[opts.messages.length - 1]?.content || 'AI 3D').slice(0, 26);
      APP.visual?.startGenerationFX?.(lbl);
    }

    async function callWithCascade(callOpts) {
      try {
        return await invokeSingleModel(targetModel, apiKeyPool, callOpts);
      } catch (err) {
        if (cascadeEnabled && isCascadableError(err)) {
          const authFailure = /\b(401|403)\b|invalid.*key|api key not valid/i.test(String(err?.message || err));
          const backups = buildCascadeCandidates(targetModel, {
            excludeProvider: authFailure ? targetModel.provider : null
          });
          for (const b of backups) {
            const bPool = b.provider === 'gemini' ? getKeyPool('gemini') : getKey(b.provider);
            if (!bPool || (Array.isArray(bPool) && !bPool.length)) continue;
            try {
              APP.ui?.setStatus?.(`Fallback → ${b.id}...`, 'busy');
              const res = await invokeSingleModel(b, bPool, callOpts);
              APP.ui?.toast?.(`Answered via ${b.id}`, 2400);
              return res;
            } catch (_) {}
          }
        }
        throw err;
      }
    }

    try {
      let res = await callWithCascade(opts);
      if (opts._is3DSession && looksLikeRefusal(res)) {
        const nudgeMsgs = Array.isArray(opts.messages) ? [...opts.messages] : [{ role: 'user', content: String(opts.prompt || '') }];
        nudgeMsgs.push({ role: 'assistant', content: res.text });
        nudgeMsgs.push({ role: 'user', content: 'SYSTEM OVERRIDE: Write custom Three.js code and call `create_procedural_object` or `run_world_fx` RIGHT NOW!' });
        res = await callWithCascade(Object.assign({}, opts, { messages: nudgeMsgs }));
      }
      return res;
    } catch (err) {
      return { ok: false, modelId: targetModel.id, provider: targetModel.provider, error: err.message || String(err) };
    } finally {
      if (showGenFX) APP.visual?.stopGenerationFX?.();
    }
  }

  async function chat(rawOpts = {}) {

    if (model.supportsLiveWS && !model.supportsRest) {

      }
      return { ok: false, error: `Model "${model.id}" uses Gemini Live WebSocket. Turn Mic ON or pick a REST model.` };
    }
    return await executeRestChat(model, rawOpts);
  }
