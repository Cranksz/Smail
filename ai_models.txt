  function recordModelOutcome(modelId, taskType, ok, latencyMs = 800, userDelta = 0) {
    const key = `${modelId}::${taskType}`;
    const st = memDb.modelStats[key] || { modelId, taskType, ok: 0, fail: 0, totalMs: 0, upvotes: 0, corrections: 0 };
    if (ok) st.ok++; else st.fail++;
    st.totalMs += Math.max(50, Math.min(30000, latencyMs));
    if (userDelta > 0) st.upvotes++;
    if (userDelta < 0) st.corrections++;
    memDb.modelStats[key] = st;
    saveMemDb();
  }

  function getModelScore(modelId, taskType) {
    const st = memDb.modelStats[`${modelId}::${taskType}`];
    if (!st) return 0.5;
    const total = st.ok + st.fail;
    const reliability = (st.ok + 1) / (total + 2);
    const avgSec = (st.totalMs / Math.max(1, total)) / 1000;
    const speedBonus = Math.max(0, 1 - avgSec / 10) * 0.28;
    const prefBonus = (st.upvotes - st.corrections * 1.4) * 0.08;
    return reliability + speedBonus + prefBonus;
  }

  function learnLesson({ type = 'lesson', taskType = 'general', rule, keywords = [], source = 'auto' }) {
    const cleanRule = String(rule || '').trim();
    if (!cleanRule || cleanRule.length < 5) return null;
    const normNew = cleanRule.toLowerCase().slice(0, 90);
    const existing = memDb.lessons.find((l) => l.rule.toLowerCase().slice(0, 90) === normNew || l.rule.toLowerCase().includes(normNew.slice(0, 48)));
    if (existing) {
      existing.weight = Math.min(10, (existing.weight || 1) + 1);
      existing.updatedAt = Date.now();
      saveMemDb(); renderUI();
      return existing;
    }
    const entry = {
      id: 'mem_' + Date.now().toString(36),
      type,
      taskType,
      rule: cleanRule.slice(0, 320),
      keywords: Array.from(new Set([...keywords, ...tokenizeKeywords(cleanRule)])).slice(0, 14),
      weight: type === 'correction' || type === 'training' ? 4 : 1,
      source,
      updatedAt: Date.now()
    };
    memDb.lessons.unshift(entry);
    saveMemDb(); renderUI();
    APP.emit?.('ai:memory_updated', entry);
    return entry;
  }

  function trainBehavior({ input, expected, domain = 'general' }) {
    const q = String(input || '').trim(), a = String(expected || '').trim();
    if (!a) return { ok: false, error: 'Missing expected rule/behavior.' };
    const pair = { id: 'trn_' + Date.now().toString(36), input: q, expected: a, domain, ts: Date.now() };
    memDb.trainingPairs.unshift(pair);
    learnLesson({
      type: 'training',
      taskType: domain,
      rule: q ? `When user says/asks "${q}" -> ${a}` : a,
      keywords: [...tokenizeKeywords(q), ...tokenizeKeywords(a)],
      source: 'user_training'
    });
    saveMemDb();
    return { ok: true, id: pair.id, message: `Trained & saved into persistent brain (${memDb.lessons.length} total lessons).` };
  }

  function forgetMemory(idOrQuery) {
    const q = String(idOrQuery || '').toLowerCase().trim();
    if (!q) return { ok: false, error: 'Specify id or keyword to forget.' };
    const before = memDb.lessons.length;
    memDb.lessons = memDb.lessons.filter((l) => l.id.toLowerCase() !== q && !l.rule.toLowerCase().includes(q));
    memDb.trainingPairs = memDb.trainingPairs.filter((t) => t.id.toLowerCase() !== q && !t.input.toLowerCase().includes(q) && !t.expected.toLowerCase().includes(q));
    saveMemDb(); renderUI();
    return { ok: true, removed: before - memDb.lessons.length };
  }

  function retrieveRelevantLessons(queryText, taskType, limit = 6) {
    if (!selfImproveEnabled || !memDb.lessons.length) return [];
    const qTokens = new Set(tokenizeKeywords(queryText));
    const scored = memDb.lessons.map((l) => {
      let score = (l.weight || 1) * 1.5;
      if (!taskType || l.taskType === taskType) score += 2.8;
      if (l.type === 'correction' || l.type === 'training' || l.type === 'error_fix') score += 2.2;
      for (const kw of l.keywords || []) if (qTokens.has(kw)) score += 2.6;
      const ageDays = (Date.now() - (l.updatedAt || Date.now())) / 86400000;
      score -= Math.min(2.0, ageDays * 0.1);
      return { lesson: l, score };
    });
    scored.sort((a, b) => b.score - a.score);
    return scored.slice(0, limit).filter((x) => x.score > 1.0).map((x) => x.lesson);
  }

  function detectUserFeedbackSignal(text) {

    const isPositive = /^(good|great|awesome|perfect|nice|yes|correct|thanks|bravo|ممتاز|احسنت|رائع|صح|جيد|تمام)\b/i.test(low);
    return { isCorrection, isPositive };
  }

  function recordFeedback(isPositiveOrText, extraNote = '') {
    if (!lastTurnContext) {
      if (typeof isPositiveOrText === 'string' && isPositiveOrText.trim()) {
        return learnLesson({ type: 'preference', taskType: 'general', rule: isPositiveOrText.trim(), source: 'manual' });
      }
      return null;
    }
    const tType = lastTurnContext.taskType, mId = lastTurnContext.modelId;
    if (isPositiveOrText === true) {
      recordModelOutcome(mId, tType, true, 500, +1);
      const l = learnLesson({ type: 'success', taskType: tType, rule: `User praised approach for "${lastTurnContext.prompt.slice(0, 60)}"` + (extraNote ? `: ${extraNote}` : ''), keywords: tokenizeKeywords(lastTurnContext.prompt), source: 'feedback_up' });
      APP.ui?.toast?.('Learned positive feedback!', 1800);
      return l;
    }
    const corrText = typeof isPositiveOrText === 'string' ? isPositiveOrText : (extraNote || 'Previous output rejected by user.');
    recordModelOutcome(mId, tType, false, 1000, -1);
    const l = learnLesson({ type: 'correction', taskType: tType, rule: `Correction on "${lastTurnContext.prompt.slice(0, 48)}": ${corrText}`, keywords: [...tokenizeKeywords(lastTurnContext.prompt), ...tokenizeKeywords(corrText)], source: 'user_correction' });
    if (!memDb.instructionNotes.some((n) => n.note === corrText)) {
      memDb.instructionNotes.unshift({ taskType: tType, note: corrText.slice(0, 200), ts: Date.now() });
      saveMemDb();
    }
    APP.ui?.toast?.('Learned from your correction!', 2200);
    return l;
  }

  function formatApiError(status, rawText, provider, modelId) {
    let detail = '';
    try { const p = JSON.parse(rawText || '{}'); detail = p.error?.message || p.choices?.[0]?.error?.message || p.message || rawText || ''; }
    catch (_) { detail = String(rawText || '').slice(0, 220); }
    const pInfo = PROVIDERS[provider] || { name: provider };
    if (status === 401 || status === 403) return `Invalid key (${status}) for ${pInfo.name}. ${detail}`;
    if (status === 402) return `Quota/Credits required (402) on ${pInfo.name} for "${modelId}".`;
    if (status === 404) return `Model "${modelId}" unavailable (404) on ${pInfo.name}.`;
    if (status === 429) return `Rate limit (429) on ${pInfo.name} for "${modelId}".`;
    if (status === 503 || /high demand|overloaded|unavailable/i.test(detail)) return `${pInfo.name} "${modelId}" busy (${status || 503}).`;
    return `${pInfo.name} error (${status || 'network'}) for "${modelId}": ${detail}`;
  }

  function isCascadableError(err) {
    const msg = String(err?.message || err || '');
    if (/\b(401|403)\b|api key not valid|incorrect api key/i.test(msg)) return false;
    return /\b(400|402|404|429|500|502|503|504)\b|high demand|overloaded|temporarily|rate limit|unavailable|resource_exhausted|provider returned error|timeout|abort|failed to fetch|network|quota/i.test(msg);
  }

  async function fetchWithTimeout(url, init, timeoutMs = 14000) {
    const ctrl = new AbortController();
    const timer = setTimeout(() => ctrl.abort(new Error('Request timeout (' + Math.round(timeoutMs / 1000) + 's)')), timeoutMs);
    try { return await fetch(url, Object.assign({}, init, { signal: ctrl.signal })); }
    finally { clearTimeout(timer); }
  }

  async function startGroqVoiceMic() {
    await stopGroqVoiceMic();
    if (!getKey('groq')) return { ok: false, error: 'Add a free Groq key (gsk_...) in AI & Keys to use Groq Whisper Voice.' };
    if (!navigator.mediaDevices?.getUserMedia) return { ok: false, error: 'Microphone unavailable.' };
    groqMicStream = await navigator.mediaDevices.getUserMedia({ audio: { echoCancellation: true, noiseSuppression: true, autoGainControl: true } });
    const AC = window.AudioContext || window.webkitAudioContext;
    groqAudioCtx = new AC();
    const src = groqAudioCtx.createMediaStreamSource(groqMicStream);
    groqAnalyser = groqAudioCtx.createAnalyser(); groqAnalyser.fftSize = 512; src.connect(groqAnalyser);
    const dataArr = new Uint8Array(groqAnalyser.frequencyBinCount);
    groqLoopTimer = setInterval(() => {
      if (!groqAnalyser || APP.botSpeaking) return;
      groqAnalyser.getByteTimeDomainData(dataArr);
      let sum = 0; for (let i = 0; i < dataArr.length; i++) { const v = (dataArr[i] - 128) / 128; sum += v * v; }
      const rms = Math.sqrt(sum / dataArr.length); APP.micLevel = Math.min(1, rms * 6.0);
      const now = performance.now();
      if (rms > 0.032) {
        groqSilenceStart = 0;
        if (!groqRecording) {
          groqRecording = true; groqChunks = [];
          try {
            groqRecorder = new MediaRecorder(groqMicStream);
            groqRecorder.ondataavailable = (e) => { if (e.data && e.data.size > 0) groqChunks.push(e.data); };
            groqRecorder.onstop = async () => {
              const blob = new Blob(groqChunks, { type: groqRecorder?.mimeType || 'audio/webm' });
              if (blob.size < 1800) return;
              APP.ui?.setStatus?.('Groq Voice transcribing...', 'busy');
              const tr = await transcribeWithGroq(blob);
              if (tr.ok && tr.text) {
                APP.ui?.toast?.('You: ' + tr.text, 2400);
                if (getSelectedModel().supportsLiveWS && APP.bot?.sendLiveText) await APP.bot.sendLiveText(tr.text);
                else if (APP.executeCommand) await APP.executeCommand(tr.text, 'voice');
              }
            };
            groqRecorder.start();
          } catch (_) { groqRecording = false; }
        }
      } else if (groqRecording) {
        if (!groqSilenceStart) groqSilenceStart = now;
        else if (now - groqSilenceStart > 750) { groqRecording = false; try { groqRecorder?.stop(); } catch (_) {} }
      }
    }, 60);
    APP.ui?.toast?.('Groq Whisper Voice Mic active!', 2800);
    return { ok: true };
  }

  function sanitizeSchema(schema) {
    if (!schema || typeof schema !== 'object') return { type: 'object', properties: { note: { type: 'string' } } };
    const copy = JSON.parse(JSON.stringify(schema));
    if (!copy.type) copy.type = 'object';
    if (copy.type === 'object') {
      if (!copy.properties || typeof copy.properties !== 'object' || Object.keys(copy.properties).length === 0) copy.properties = { note: { type: 'string' } };
      else for (const k of Object.keys(copy.properties)) if (copy.properties[k]?.type === 'object') copy.properties[k] = sanitizeSchema(copy.properties[k]);
    }
    return copy;
  }

  async function runGeminiRestChat(model, apiKeyPool, opts) {
    const keyList = Array.isArray(apiKeyPool) ? apiKeyPool : [apiKeyPool];
    const cleanTools = normalizeTools(opts.tools);
    const contents = [];
    const msgs = Array.isArray(opts.messages) ? opts.messages : [];
    for (let i = 0; i < msgs.length; i++) {
      const m = msgs[i]; if (!m) continue;
      const role = m.role === 'assistant' ? 'model' : 'user', parts = [];
      if (m.content) parts.push({ text: String(m.content) });
      if (i === msgs.length - 1 && role === 'user' && opts.imageB64) parts.push({ inlineData: { mimeType: opts.imageMime || 'image/jpeg', data: String(opts.imageB64).replace(/^data:image\/\w+;base64,/, '') } });
      if (parts.length) contents.push({ role, parts });
    }
    if (!contents.length) {
      const parts = [{ text: String(opts.prompt || 'Hello') }];
      if (opts.imageB64) parts.push({ inlineData: { mimeType: opts.imageMime || 'image/jpeg', data: String(opts.imageB64).replace(/^data:image\/\w+;base64,/, '') } });
      contents.push({ role: 'user', parts });
    }

    const maxSteps = Math.max(1, Math.min(8, opts.maxSteps || 5)), toolTrace = [];
    let finalText = '', activeKeyIdx = 0;

    for (let step = 0; step < maxSteps; step++) {
      const body = { contents };
      if (opts.systemPrompt) body.systemInstruction = { parts: [{ text: String(opts.systemPrompt) }] };
      if (cleanTools.length) body.tools = [{ functionDeclarations: cleanTools }];
      if (model.thinkingLevel) body.generationConfig = { thinkingConfig: { thinkingLevel: String(model.thinkingLevel).toLowerCase() } };

      let data = null, lastErr = '';
      const totalTries = Math.max(1, keyList.length);
      for (let attempt = 0; attempt < totalTries; attempt++) {
        const curKey = keyList[activeKeyIdx % keyList.length];
        const url = 'https://generativelanguage.googleapis.com/v1beta/models/' + encodeURIComponent(model.id) + ':generateContent?key=' + encodeURIComponent(curKey);
        const res = await fetchWithTimeout(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) }, 14000);
        const rawTxt = await res.text();
        if (res.ok) { data = JSON.parse(rawTxt); break; }
        lastErr = formatApiError(res.status, rawTxt, 'gemini', model.id);
        if (res.status === 400 && body.generationConfig && /thinking/i.test(rawTxt)) { delete body.generationConfig; attempt--; continue; }
        if ((res.status === 503 || res.status === 429) && keyList.length > 1) { activeKeyIdx++; continue; }
        if (res.status === 503 || res.status === 429) putModelOnCooldown(model.id, 90000);
        throw new Error(lastErr);
      }
      if (!data) throw new Error(lastErr || 'Gemini failed.');

      const cand = data.candidates?.[0];
      if (!cand?.content?.parts) throw new Error('Gemini returned empty output.');
      const rawParts = JSON.parse(JSON.stringify(cand.content.parts));
      contents.push({ role: 'model', parts: rawParts });

      const textChunks = [], fnCalls = [];
      for (const p of rawParts) { if (p.text && !p.thought) textChunks.push(p.text); if (p.functionCall?.name) fnCalls.push(p.functionCall); }
      if (textChunks.length) finalText = textChunks.join('\n').trim();
      if (!fnCalls.length || typeof opts.onToolCall !== 'function') break;

      const responseParts = [];
      for (const call of fnCalls) {
        let resultObj;
        try { const out = await opts.onToolCall(call.name, call.args || {}); resultObj = (out && typeof out === 'object') ? out : { result: out ?? 'ok' }; }
        catch (err) { resultObj = { ok: false, error: err.message || String(err) }; }
        toolTrace.push({ name: call.name, args: call.args || {}, result: resultObj });
        const fnResp = { name: call.name, response: resultObj };
        if (call.id) fnResp.id = call.id;
        responseParts.push({ functionResponse: fnResp });
      }
      contents.push({ role: 'user', parts: responseParts });
    }
    return { ok: true, modelId: model.id, provider: 'gemini', text: finalText, toolTrace };
  }

  async function base64ToSizedBlob(b64, mime = 'image/jpeg', maxDim = 512, mult64 = false) {
    const img = new Image();
    await new Promise((ok, bad) => { img.onload = ok; img.onerror = bad; img.src = `data:${mime};base64,${b64}`; });
    const sc = Math.min(1, maxDim / Math.max(img.width, img.height));
    let w = Math.max(64, Math.round(img.width * sc));
    let h = Math.max(64, Math.round(img.height * sc));
    if (mult64) { w = Math.max(64, Math.round(w / 64) * 64); h = Math.max(64, Math.round(h / 64) * 64); }
    const c = document.createElement('canvas'); c.width = w; c.height = h;
    c.getContext('2d').drawImage(img, 0, 0, w, h);
    const outB64 = c.toDataURL('image/png').split(',')[1];
    const blob = await new Promise((res) => c.toBlob(res, 'image/png'));
    return { blob, b64: outB64, width: w, height: h };
  }

  async function urlOrB64ToBase64(val) {
    if (!val) return null;
    if (typeof val === 'string') {
      if (val.startsWith('data:image/')) return val.split('base64,')[1];
      if (!val.startsWith('http')) return val;
      const res = await fetchWithTimeout(val, {}, 18000);
      const blob = await res.blob();
      return await new Promise((r) => { const fr = new FileReader(); fr.onloadend = () => r(String(fr.result || '').split(',')[1]); fr.readAsDataURL(blob); });
    }
    if (val.url) return await urlOrB64ToBase64(val.url);
    return null;
  }

  async function runGradioImageEdit(spaceId, cleanB64, imageMime, promptText) {
    const { Client } = await import('https://cdn.jsdelivr.net/npm/@gradio/client/+esm');
    const sized = await base64ToSizedBlob(cleanB64, imageMime, 768, false);
    const opts = keysObj.hf ? { hf_token: keysObj.hf } : {};
    const client = await Client.connect(spaceId, opts);
    const apiInfo = await client.view_api().catch(() => null);
    const endpoints = apiInfo?.named_endpoints || {};
    let epName = Object.keys(endpoints).find((k) => /infer|edit|predict|run/i.test(k)) || Object.keys(endpoints)[0] || '/infer';
    const params = endpoints[epName]?.parameters || [];
    let payload;
    if (params.length) {
      payload = {};
      for (const p of params) {
        const pn = String(p.parameter_name || p.label || '').toLowerCase();
        if (/image|img|input/.test(pn) && !/images/.test(pn)) payload[p.parameter_name] = sized.blob;
        else if (/images/.test(pn)) payload[p.parameter_name] = [{ image: sized.blob }];
        else if (/prompt|instruction|text/.test(pn)) payload[p.parameter_name] = promptText;
        else if (p.parameter_default !== undefined) payload[p.parameter_name] = p.parameter_default;
      }
    } else {
      payload = [sized.blob, promptText];
    }
    const out = await client.predict(epName, payload);
    const first = Array.isArray(out?.data) ? out.data[0] : out?.data;
    const outB64 = await urlOrB64ToBase64(first);
    if (!outB64) throw new Error('Empty Gradio output');
    return outB64;
  }

      try { const r = await fetchWithTimeout('https://openrouter.ai/api/v1/auth/key', { headers: { Authorization: 'Bearer ' + keysObj.openrouter } }, 8000); results.push(r.ok ? 'OpenRouter: OK' : `OpenRouter: HTTP ${r.status}`); }
      catch (e) { results.push('OpenRouter: ' + e.message); }
    }
    if (keysObj.pollinations) results.push('Pollinations Key: Saved');
    if (keysObj.hf) results.push('HuggingFace Token: Saved');
    return results.join(' | ');
  }

  async function runTurn(userText, opts = {}) {
    const clean = String(userText || '').trim();
    if (selfImproveEnabled && lastTurnContext && (Date.now() - lastTurnContext.ts < 120000)) {
      const sig = detectUserFeedbackSignal(clean);
      if (sig.isPositive && clean.split(/\s+/).length <= 5) { recordFeedback(true); return { ok: true, text: 'Lesson saved!' }; }
      if (sig.isCorrection) recordFeedback(clean);
    }

    if (APP.bot && typeof APP.bot.handleCommand === 'function') return await APP.bot.handleCommand(clean, opts);

    const res = await chat({
      ignoreInstructions: !!opts.ignoreInstructions,
      systemPrompt: 'You are Geeny, an unlimited WebXR Mixed Reality AI. Write custom Three.js code using create_procedural_object or run_world_fx.',
      tools: [WORLD_FX_TOOL_PURE],
      prompt: clean,
      onToolCall: async (n, a) => (n === 'run_world_fx' ? executeWorldFX(a) : { ok: false })
    });
    if (!res.ok) APP.ui?.toast?.(res.error, 6000); else APP.ui?.toast?.(res.text || 'Done.');
    return res;
  }
