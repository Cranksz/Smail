        var A = T[i * 3] * 3, B = T[i * 3 + 1] * 3, Cc = T[i * 3 + 2] * 3;
        var e1x = P[B] - P[A], e1y = P[B + 1] - P[A + 1], e1z = P[B + 2] - P[A + 2], e2x = P[Cc] - P[A], e2y = P[Cc + 1] - P[A + 1], e2z = P[Cc + 2] - P[A + 2];
        var nxv = e1y * e2z - e1z * e2y, nyv = e1z * e2x - e1x * e2z, nzv = e1x * e2y - e1y * e2x;
        fn[i * 3] = nxv; fn[i * 3 + 1] = nyv; fn[i * 3 + 2] = nzv;
        for (var cc = 0; cc < 3; cc++) { var vi2 = T[i * 3 + cc] * 3; vn[vi2] += nxv; vn[vi2 + 1] += nyv; vn[vi2 + 2] += nzv; }
      }
      var uv = new Float32Array(nV * 2), md = Math.max(bs[0], bs[1], bs[2], 1e-6);
      for (i = 0; i < nV; i++) {
        var l3 = Math.sqrt(vn[i * 3] * vn[i * 3] + vn[i * 3 + 1] * vn[i * 3 + 1] + vn[i * 3 + 2] * vn[i * 3 + 2]) || 1;
        vn[i * 3] /= l3; vn[i * 3 + 1] /= l3; vn[i * 3 + 2] /= l3;
        var ax_ = Math.abs(vn[i * 3]), ay_ = Math.abs(vn[i * 3 + 1]), az_ = Math.abs(vn[i * 3 + 2]);
        uv[i * 2] = (ay_ >= ax_ && ay_ >= az_ ? P[i * 3] : (ax_ >= az_ ? P[i * 3 + 2] : P[i * 3])) / md + 0.5;
        uv[i * 2 + 1] = (ay_ >= ax_ && ay_ >= az_ ? P[i * 3 + 2] : P[i * 3 + 1]) / md + 0.5;
      }
      var out = {
        positions: P, normals: vn, uvs: uv, indices: T,
        bbox: { center: bc, size: bs }, pca: { axes: axes, extents: pe },
        symmetry: symm, sliceY: sliceY,
        ascii: { front: buildAscii('front'), side: buildAscii('side'), top: buildAscii('top') },
        yaw: { ang: ang, u: yu, v: yv, ratio: yu / Math.max(yv, 1e-6) }, tris: nT
      };
      return { out: out, tr: [P.buffer, vn.buffer, uv.buffer, T.buffer] };
    }
  }

  let worker = null, wid = 0;
  const pend = new Map();
  function getWorker() {
    if (worker) return worker;
    const url = URL.createObjectURL(new Blob(['(' + workerMain.toString() + ')();'], { type: 'application/javascript' }));
    worker = new Worker(url); setTimeout(() => URL.revokeObjectURL(url), 8000);
    worker.onmessage = (e) => { const m = e.data, p = pend.get(m.id); if (!p) return; pend.delete(m.id); m.ok ? p.res(m.out) : p.rej(new Error(m.err)); };
    worker.onerror = () => { pend.forEach((p) => p.rej(new Error('worker error'))); pend.clear(); };
    return worker;
  }
  const runWorker = (payload) => new Promise((res, rej) => {
    const id = ++wid;
    const t = setTimeout(() => { pend.delete(id); try { worker && worker.terminate(); } catch (_) {} worker = null; rej(new Error('mesh worker timeout')); }, 15000);
    pend.set(id, { res: (v) => { clearTimeout(t); res(v); }, rej: (e) => { clearTimeout(t); rej(e); } });
    getWorker().postMessage(Object.assign({ id }, payload));
  });
  const withTimeout = (p, ms, msg) => new Promise((res, rej) => { const t = setTimeout(() => rej(new Error(msg)), ms); p.then((v) => { clearTimeout(t); res(v); }, (e) => { clearTimeout(t); rej(e); }); });

  function buildTubeGeo(pts, radii, radial = 10) {
    const curve = pts.length > 2 ? new THREE.CatmullRomCurve3(pts, false, 'centripetal') : new THREE.LineCurve3(pts[0], pts[pts.length - 1]);
    const segs = Math.min(Math.max(pts.length * 2, 4), 480), fr = curve.computeFrenetFrames(segs, false);
    const pos = [], idx = [], p = new V3();
    const rAt = (t) => { const f = t * (radii.length - 1), i = Math.floor(f), j = Math.min(i + 1, radii.length - 1); return radii[i] + (radii[j] - radii[i]) * (f - i); };
    for (let i = 0; i <= segs; i++) {
      const t = i / segs; curve.getPointAt(t, p); const r = rAt(t), N = fr.normals[i], B = fr.binormals[i];
      for (let k = 0; k < radial; k++) { const a = k / radial * Math.PI * 2, c = Math.cos(a), s = Math.sin(a); pos.push(p.x + (N.x * c + B.x * s) * r, p.y + (N.y * c + B.y * s) * r, p.z + (N.z * c + B.z * s) * r); }
    }
    for (let i = 0; i < segs; i++) for (let k = 0; k < radial; k++) {
      const a = i * radial + k, b = i * radial + (k + 1) % radial, c = (i + 1) * radial + k, d = (i + 1) * radial + (k + 1) % radial; idx.push(a, b, c, b, d, c);
    }
    const c0 = pos.length / 3; pos.push(pts[0].x, pts[0].y, pts[0].z); const c1 = c0 + 1; pos.push(pts[pts.length - 1].x, pts[pts.length - 1].y, pts[pts.length - 1].z);
    for (let k = 0; k < radial; k++) { idx.push(c0, (k + 1) % radial, k); idx.push(c1, segs * radial + k, segs * radial + (k + 1) % radial); }
    const g = new THREE.BufferGeometry(); g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3)); g.setIndex(idx); g.computeVertexNormals(); return g;
  }

  function strokeMesh(s) {
    if (s.pts.length < 2) return;
    if (!S.rawGroup || !S.rawGroup.parent) newRawGroup();
    const g = buildTubeGeo(s.pts, s.radii);
    if (!s.mesh) {

      s.mesh = new THREE.Mesh(g, s.mat);
      s.mesh.frustumCulled = false; s.mesh.userData.grabbable = false;
      S.rawGroup.add(s.mesh);

      s.mesh.geometry.dispose(); s.mesh.geometry = g;
    }
  }
  function startStroke(id, pos, pr) {
    if (!S.strokes.length && S.cur.size === 0) S.viewFwd = flatFwd();
    const s = { pts: [], radii: [], color: activeColor().hex, mesh: null, mat: null, lastBuild: 0 };
    S.cur.set(id, s); addPoint(s, pos, pr, true);
  }
  function addPoint(s, pos, pr, force) {
    const r = S.brushRadius * (0.80 + 0.40 * clamp(pr, 0, 1)), last = s.pts[s.pts.length - 1];
    if (last && !force && last.distanceTo(pos) < Math.max(0.004, r * 0.45)) return;
    s.pts.push(last ? last.clone().lerp(pos, 0.6) : pos.clone()); s.radii.push(r);
    const now = performance.now();
    if (s.pts.length >= 2 && now - s.lastBuild > 55) { s.lastBuild = now; strokeMesh(s); }
  }
  function endStroke(id) {
    const s = S.cur.get(id); if (!s) return; S.cur.delete(id);
    if (s.pts.length < 2) { if (s.mesh) { S.rawGroup?.remove(s.mesh); s.mesh.geometry.dispose(); s.mat?.dispose(); } return; }
    strokeMesh(s); S.strokes.push(s); S.dirty = true;
  }
  const endAllStrokes = () => { for (const id of Array.from(S.cur.keys())) endStroke(id); };
  const strokeLen = (s) => { let l = 0; for (let i = 1; i < s.pts.length; i++) l += s.pts[i].distanceTo(s.pts[i - 1]); return l; };
  function discardStrokes() {
    for (const s of S.strokes) if (s.mesh) { S.rawGroup?.remove(s.mesh); s.mesh.geometry.dispose(); s.mat?.dispose(); }
    S.strokes = []; S.cur.clear(); S.dirty = true;
  }

  // ------------------------------------------------------------------ Volumetric FBM Magic Smoke + Dissolving Holo + Embers
  function makeMagicSmokeMat() {
    return new THREE.ShaderMaterial({
      transparent: true, depthWrite: false,
      uniforms: { uTime: { value: 0 }, uPx: { value: window.devicePixelRatio || 1 } },
      vertexShader: [
        'attribute float aSize; attribute float aLife; attribute float aSeed; uniform float uPx;',
        'varying float vLife; varying float vSeed;',
        'void main(){ vLife=aLife; vSeed=aSeed; vec4 mv=modelViewMatrix*vec4(position,1.0);',
        '  gl_PointSize = aLife>0.01 ? clamp((aSize*uPx)/max(0.22,-mv.z), 14.0, 160.0) : 0.0;',
        '  gl_Position = projectionMatrix*mv; }'
      ].join('\n'),
      fragmentShader: [
        'uniform float uTime; varying float vLife; varying float vSeed;',
        'float sh2(vec2 p){ return fract(sin(dot(p,vec2(127.1,311.7)))*43758.5453); }',
        'float sn2(vec2 p){ vec2 i=floor(p),f=fract(p); f=f*f*(3.0-2.0*f); return mix(mix(sh2(i),sh2(i+vec2(1,0)),f.x),mix(sh2(i+vec2(0,1)),sh2(i+vec2(1,1)),f.x),f.y); }',
        'float fbm(vec2 p){ return 0.52*sn2(p)+0.28*sn2(p*2.13)+0.14*sn2(p*4.41)+0.06*sn2(p*8.1); }',
        'void main(){',
        '  if(vLife<0.01) discard;',
        '  vec2 uv=gl_PointCoord-0.5; float r2=dot(uv,uv)*4.0; if(r2>1.0) discard;',
        '  float ang=vSeed*6.28+uTime*(0.45+fract(vSeed*3.1)*0.6); mat2 rot=mat2(cos(ang),-sin(ang),sin(ang),cos(ang));',
        '  vec2 ruv=rot*uv; float w=fbm(ruv*3.6+vec2(vSeed*5.2,uTime*0.38));',
        '  float puff=exp(-r2*2.2)*smoothstep(0.14,0.82,w);',
        '  float fade=sin(clamp(vLife,0.0,1.0)*3.14159);',
        '  vec3 cDark=vec3(0.06,0.16,0.32), cCyan=vec3(0.05,0.88,0.82), cViolet=vec3(0.62,0.42,1.0);',
        '  vec3 col=mix(cDark,cCyan,w); col=mix(col,cViolet,smoothstep(0.55,0.92,w)*0.5);',
        '  gl_FragColor=vec4(col, puff*fade*0.54);',
        '}'
      ].join('\n')
    });
  }
  function makeSparkDiscMat(colorHex) {
    return new THREE.ShaderMaterial({
      transparent: true, depthWrite: false, blending: THREE.AdditiveBlending,
      uniforms: { uPx: { value: window.devicePixelRatio || 1 }, uCol: { value: new THREE.Color(colorHex || 0x7ef7d4) } },
      vertexShader: 'attribute float aLife; uniform float uPx; varying float vLife; void main(){ vLife=aLife; vec4 mv=modelViewMatrix*vec4(position,1.0); gl_PointSize=aLife>0.01?clamp((18.0*uPx*aLife)/max(0.25,-mv.z),2.0,15.0):0.0; gl_Position=projectionMatrix*mv; }',
      fragmentShader: 'uniform vec3 uCol; varying float vLife; void main(){ if(vLife<0.01)discard; float d=length(gl_PointCoord-0.5)*2.0; if(d>1.0)discard; float core=pow(1.0-d,2.1); gl_FragColor=vec4(mix(uCol,vec3(1.0),core*0.8),core*clamp(vLife*1.5,0.0,1.0)); }'

  }
  function spawnMagicVFXSystem(center, radius) {
    const SM_COUNT = 72, SP_COUNT = 160;
    const smPos = new Float32Array(SM_COUNT * 3).fill(-1000), smVel = new Float32Array(SM_COUNT * 3);
    const smSize = new Float32Array(SM_COUNT), smLife = new Float32Array(SM_COUNT), smSeed = new Float32Array(SM_COUNT);
    const smGeo = new THREE.BufferGeometry();
    smGeo.setAttribute('position', new THREE.BufferAttribute(smPos, 3));
    smGeo.setAttribute('aSize', new THREE.BufferAttribute(smSize, 1));
    smGeo.setAttribute('aLife', new THREE.BufferAttribute(smLife, 1));
    smGeo.setAttribute('aSeed', new THREE.BufferAttribute(smSeed, 1));
    const smMat = makeMagicSmokeMat();
    const smokePts = new THREE.Points(smGeo, smMat); smokePts.frustumCulled = false; APP.scene.add(smokePts);

    const spPos = new Float32Array(SP_COUNT * 3).fill(-1000), spVel = new Float32Array(SP_COUNT * 3), spLife = new Float32Array(SP_COUNT);
    const spGeo = new THREE.BufferGeometry();
    spGeo.setAttribute('position', new THREE.BufferAttribute(spPos, 3));
    spGeo.setAttribute('aLife', new THREE.BufferAttribute(spLife, 1));
    const spMat = makeSparkDiscMat(0x5efce8);
    const sparkPts = new THREE.Points(spGeo, spMat); sparkPts.frustumCulled = false; APP.scene.add(sparkPts);

    let smHead = 0, spHead = 0;
    const rad = Math.max(0.08, radius || 0.25);

    return {
      emitBurst(px, py, pz, countSmoke = 3, countSparks = 6) {
        for (let i = 0; i < countSmoke; i++) {
          const s = smHead; smHead = (smHead + 1) % SM_COUNT;
          const a = Math.random() * Math.PI * 2, r = rad * Math.random() * 0.85;
          smPos[s * 3] = px + Math.cos(a) * r; smPos[s * 3 + 1] = py + (Math.random() - 0.5) * 0.08; smPos[s * 3 + 2] = pz + Math.sin(a) * r;
          smVel[s * 3] = Math.cos(a) * 0.16; smVel[s * 3 + 1] = 0.14 + Math.random() * 0.24; smVel[s * 3 + 2] = Math.sin(a) * 0.16;
          smSize[s] = 52 + Math.random() * 48; smLife[s] = 1.0; smSeed[s] = Math.random();
        }
        for (let i = 0; i < countSparks; i++) {
          const s = spHead; spHead = (spHead + 1) % SP_COUNT;
          const a = Math.random() * Math.PI * 2, r = rad * (0.25 + 0.75 * Math.random());
          spPos[s * 3] = px + Math.cos(a) * r; spPos[s * 3 + 1] = py; spPos[s * 3 + 2] = pz + Math.sin(a) * r;
          spVel[s * 3] = Math.cos(a) * 0.30; spVel[s * 3 + 1] = 0.22 + Math.random() * 0.44; spVel[s * 3 + 2] = Math.sin(a) * 0.30;
          spLife[s] = 0.5 + Math.random() * 0.55;
        }
      },
      step(dt, tNow) {
        smMat.uniforms.uTime.value = tNow;
        let alive = false;
        for (let i = 0; i < SM_COUNT; i++) {
          if (smLife[i] <= 0) continue; alive = true; smLife[i] -= dt * 0.65;
          const swirl = dt * 0.85, vx = smVel[i * 3], vz = smVel[i * 3 + 2];
          smVel[i * 3] = (vx - vz * swirl) * 0.97; smVel[i * 3 + 2] = (vz + vx * swirl) * 0.97;
          smPos[i * 3] += smVel[i * 3] * dt; smPos[i * 3 + 1] += smVel[i * 3 + 1] * dt; smPos[i * 3 + 2] += smVel[i * 3 + 2] * dt;
          if (smLife[i] <= 0) smPos[i * 3 + 1] = -1000;
        }
        for (let i = 0; i < SP_COUNT; i++) {
          if (spLife[i] <= 0) continue; alive = true; spLife[i] -= dt;
          spPos[i * 3] += spVel[i * 3] * dt; spPos[i * 3 + 1] += spVel[i * 3 + 1] * dt; spPos[i * 3 + 2] += spVel[i * 3 + 2] * dt;
          if (spLife[i] <= 0) spPos[i * 3 + 1] = -1000;
        }
        smGeo.attributes.position.needsUpdate = true; smGeo.attributes.aLife.needsUpdate = true;
        smGeo.attributes.aSize.needsUpdate = true; smGeo.attributes.aSeed.needsUpdate = true;
        spGeo.attributes.position.needsUpdate = true; spGeo.attributes.aLife.needsUpdate = true;
        return alive;
      },
      dispose() {
        if (smokePts.parent) smokePts.parent.remove(smokePts);
        if (sparkPts.parent) sparkPts.parent.remove(sparkPts);
        smGeo.dispose(); smMat.dispose(); spGeo.dispose(); spMat.dispose();
      }
    };
  }

  function makeHolo(rec, bounds) {
    rec.planeAbove = new THREE.Plane(new V3(0, 1, 0), 1e4);
    rec.holoMat = new THREE.ShaderMaterial({
      transparent: true, depthWrite: false, side: THREE.DoubleSide, blending: THREE.AdditiveBlending,
      uniforms: {
        uTime: { value: 0 }, uMinY: { value: bounds.min.y }, uH: { value: Math.max(bounds.max.y - bounds.min.y, 0.05) },
        uReveal: { value: -1e4 }, uDissolve: { value: 0.0 },
        uCol: { value: new THREE.Color(0x00f0b4) }, uCol2: { value: new THREE.Color(0x38bdf8) }
      },
      vertexShader: 'varying vec3 vW; varying vec3 vN; void main(){ vec4 w=modelMatrix*vec4(position,1.0); vW=w.xyz; vN=normalize(mat3(modelMatrix)*normal); gl_Position=projectionMatrix*viewMatrix*w; }',
      fragmentShader: [
        'uniform float uTime,uMinY,uH,uReveal,uDissolve; uniform vec3 uCol,uCol2; varying vec3 vW; varying vec3 vN;',
        'float hn(vec3 p){ return fract(sin(dot(p, vec3(12.9898, 78.233, 37.719))) * 43758.5453); }',
        'void main(){',
        '  if(vW.y < uReveal) discard;',
        '  float n = hn(floor(vW * 55.0));',
        '  if(uDissolve > 0.01 && n < uDissolve) discard;',
        '  float edge = (uDissolve > 0.01) ? smoothstep(uDissolve, uDissolve + 0.14, n) : 1.0;',
        '  vec3 V = normalize(cameraPosition - vW);',
        '  float fr = pow(1.0 - abs(dot(normalize(vN), V)), 2.0);',
        '  float sy = uMinY + fract(uTime * 0.65) * uH;',
        '  float band = exp(-abs(vW.y - sy) / (0.06 * uH + 0.005));',
        '  float seam = (uReveal > -999.0) ? exp(-abs(vW.y - uReveal) / (0.05 * uH + 0.005)) : 0.0;',
        '  vec3 dissolveRim = mix(vec3(1.0, 0.72, 0.25), uCol, edge);',
        '  vec3 col = mix(uCol2, dissolveRim, clamp(band + seam + (1.0 - edge), 0.0, 1.0)) * (0.55 + fr * 1.4 + band * 1.8 + seam * 2.8);',
        '  float a = clamp((0.20 + fr * 0.85 + band * 1.1 + seam * 1.4) * (1.0 - uDissolve * 0.85), 0.0, 1.0);',
        '  gl_FragColor = vec4(col, a);',
        '}'
      ].join('\n')
    });
    rec.wireMat = new THREE.MeshBasicMaterial({
      color: 0x7ef7d4, wireframe: true, transparent: true, opacity: 0.45,
      depthWrite: false, blending: THREE.AdditiveBlending, clippingPlanes: [rec.planeAbove]
    });
    const sz = bounds.getSize(new V3()), ctr = bounds.getCenter(new V3());
    const ringR = Math.max(0.08, Math.max(sz.x, sz.z) * 0.70);
    rec.scanRing = new THREE.Mesh(
      new THREE.TorusGeometry(ringR, 0.005, 8, 40),
      new THREE.MeshBasicMaterial({ color: 0x00f0b4, transparent: true, opacity: 0.92, blending: THREE.AdditiveBlending, depthWrite: false })
    );
    rec.scanRing.rotation.x = Math.PI / 2;
    rec.scanRing.position.set(ctr.x, bounds.min.y, ctr.z);
    APP.scene.add(rec.scanRing);
    rec.magicVFX = spawnMagicVFXSystem(ctr, ringR);
    rec.temps = rec.temps || [];
    rec.temps.push(rec.scanRing);
  }
  function holoize(rec, mesh) {
    mesh.userData.origMat = mesh.material; mesh.material = rec.holoMat;
    const w = new THREE.Mesh(mesh.geometry, rec.wireMat); w.frustumCulled = false; mesh.add(w); rec.temps.push(w); rec.wired = rec.wired || []; rec.wired.push(mesh);
  }
  function holoBoundsUpdate(rec, obj) {
    const b = new THREE.Box3().setFromObject(obj);
    rec.holoMat.uniforms.uMinY.value = b.min.y;
    rec.holoMat.uniforms.uH.value = Math.max(b.max.y - b.min.y, 0.05);
    if (rec.scanRing) {
      const c = b.getCenter(new V3()), sz = b.getSize(new V3());
      rec.scanRing.position.set(c.x, b.min.y, c.z);
      const sc = Math.max(0.08, Math.max(sz.x, sz.z) * 0.70) / Math.max(0.08, rec.scanRing.geometry.parameters.radius);

    }

  function render4Views(out) {
    const PW = 256, cv = document.createElement('canvas'); cv.width = PW * 2; cv.height = PW * 2;
    const c = cv.getContext('2d'), Pp = out.positions, I = out.indices, nT = I.length / 3, nV = Pp.length / 3;
    const az = 0.62, el = 0.44;
    const views = [
      { n: 'FRONT (XY)', ox: 0,  oy: 0,  f: new V3(0, 0, -1), up: UP },
      { n: 'SIDE (ZY)',  ox: PW, oy: 0,  f: new V3(-1, 0, 0), up: UP },
      { n: 'TOP (XZ)',   ox: 0,  oy: PW, f: new V3(0, -1, 0), up: new V3(0, 0, -1) },
      { n: '3/4 VIEW',   ox: PW, oy: PW, f: new V3(-Math.sin(az) * Math.cos(el), -Math.sin(el), -Math.cos(az) * Math.cos(el)), up: UP }
    ];
    const sx = new Float32Array(nV), sy = new Float32Array(nV), sd = new Float32Array(nV);
    let scale = 1e9;
    views.forEach((v) => {
      v.r = new V3().crossVectors(v.f, v.up).normalize();
      v.u = new V3().crossVectors(v.r, v.f).normalize();
      let a0 = 1e9, a1 = -1e9, b0 = 1e9, b1 = -1e9;
      for (let i = 0; i < nV; i++) {
        const x = Pp[i * 3], y = Pp[i * 3 + 1], z = Pp[i * 3 + 2];
        const u = x * v.r.x + y * v.r.y + z * v.r.z, w = x * v.u.x + y * v.u.y + z * v.u.z;
        if (u < a0) a0 = u; if (u > a1) a1 = u; if (w < b0) b0 = w; if (w > b1) b1 = w;
      }
      scale = Math.min(scale, PW * 0.78 / Math.max(a1 - a0, b1 - b0, 1e-4));
      v.cx = (a0 + a1) / 2; v.cy = (b0 + b1) / 2;
    });
    c.fillStyle = '#090f1a'; c.fillRect(0, 0, PW * 2, PW * 2);
    const stride = Math.max(1, Math.ceil(nT / 6000));
    views.forEach((v) => {
      const ox = v.ox, oy = v.oy;
      c.save(); c.beginPath(); c.rect(ox, oy, PW, PW); c.clip();
      for (let i = 0; i < nV; i++) {
        const x = Pp[i * 3], y = Pp[i * 3 + 1], z = Pp[i * 3 + 2];
        sx[i] = ox + PW / 2 + ((x * v.r.x + y * v.r.y + z * v.r.z) - v.cx) * scale;
        sy[i] = oy + PW / 2 - ((x * v.u.x + y * v.u.y + z * v.u.z) - v.cy) * scale;
        sd[i] = x * v.f.x + y * v.f.y + z * v.f.z;
      }
      const L = new V3().copy(v.f).multiplyScalar(-0.7).addScaledVector(v.r, -0.45).addScaledVector(v.u, 0.58).normalize();
      const list = [];
      for (let t = 0; t < nT; t += stride) {
        const a = I[t * 3], b = I[t * 3 + 1], d = I[t * 3 + 2];
        const e1x = Pp[b * 3] - Pp[a * 3], e1y = Pp[b * 3 + 1] - Pp[a * 3 + 1], e1z = Pp[b * 3 + 2] - Pp[a * 3 + 2];
        const e2x = Pp[d * 3] - Pp[a * 3], e2y = Pp[d * 3 + 1] - Pp[a * 3 + 1], e2z = Pp[d * 3 + 2] - Pp[a * 3 + 2];
        let nx = e1y * e2z - e1z * e2y, ny = e1z * e2x - e1x * e2z, nz = e1x * e2y - e1y * e2x;
        const nl = Math.hypot(nx, ny, nz) || 1; nx /= nl; ny /= nl; nz /= nl;
        if (nx * v.f.x + ny * v.f.y + nz * v.f.z > 0) continue;
        const lam = Math.max(0, nx * L.x + ny * L.y + nz * L.z);
        const rim = Math.pow(1 - Math.abs(nx * v.f.x + ny * v.f.y + nz * v.f.z), 2.0);
        list.push([(sd[a] + sd[b] + sd[d]) / 3, a, b, d, Math.min(1, lam * 0.78 + rim * 0.35 + 0.12)]);
      }
      list.sort((p, q) => q[0] - p[0]);
      c.lineWidth = 0.85; c.lineJoin = 'round';
      for (const t of list) {
        const g2 = 75 + t[4] * 180;
        const col = 'rgb(' + (g2 * 0.85 | 0) + ',' + (g2 * 0.96 | 0) + ',' + (Math.min(255, g2 + 18) | 0) + ')';
        c.fillStyle = col; c.strokeStyle = col;
        c.beginPath(); c.moveTo(sx[t[1]], sy[t[1]]); c.lineTo(sx[t[2]], sy[t[2]]); c.lineTo(sx[t[3]], sy[t[3]]); c.closePath();
        c.fill(); c.stroke();
      }
      c.fillStyle = '#38bdf8'; c.font = 'bold 14px system-ui, sans-serif'; c.fillText(v.n, ox + 10, oy + 20);
      c.restore(); c.strokeStyle = '#334155'; c.strokeRect(ox + 0.5, oy + 0.5, PW - 1, PW - 1);
    });
    return cv.toDataURL('image/jpeg', 0.86).split(',')[1];
  }

  // ------------------------------------------------------------------ Multi-Layer Procedural PBR Texture Synthesizer
  const hash2 = (x, y) => { const h = Math.sin(x * 127.1 + y * 311.7) * 43758.5453; return h - Math.floor(h); };
  function vnoise(x, y) { const xi = Math.floor(x), yi = Math.floor(y), xf = x - xi, yf = y - yi, u = xf * xf * (3 - 2 * xf), v = yf * yf * (3 - 2 * yf); const a = hash2(xi, yi), b = hash2(xi + 1, yi), c = hash2(xi, yi + 1), d = hash2(xi + 1, yi + 1); return a + (b - a) * u + (c - a) * v + (a - b - c + d) * u * v; }
  function fbm(x, y, o) { let s = 0, a = 0.5; for (let i = 0; i < (o || 5); i++) { s += a * vnoise(x, y); x *= 2.04; y *= 2.04; a *= 0.5; } return s; }
  function smoothstep(a, b, x) { const t = clamp((x - a) / (b - a), 0, 1); return t * t * (3 - 2 * t); }

  function patternDetail(kind, u, v, o) {
    switch (kind) {
      case 'wood':
      case 'bark': {
        const knot = Math.exp(-Math.pow((u - 0.35) * 4.5, 2) - Math.pow((v - 0.55) * 3.0, 2)) * 2.2;
        const g = u * (kind === 'bark' ? 18 : 10) + fbm(u * 4, v * 0.9, 5) * 5.2 + knot;
        const ring = 0.5 + 0.5 * Math.sin(g * 6.2832);
        const micro = fbm(u * 64, v * 4, 3);
        o[0] = ring * 0.5 + micro * 0.38 + knot * 0.2;
        o[1] = clamp(ring * 0.82 + micro * 0.25, 0, 1);
        o[2] = (kind === 'bark' ? 0.78 : 0.48) + 0.26 * micro;
        break;
      }
      case 'metal':
      case 'brushed_metal': {
        const streak = fbm(u * 3, v * 190, 3);
        const pit = hash2(Math.floor(u * 140), Math.floor(v * 140)) > 0.985 ? 0.4 : 0;
        const scratch = Math.abs(Math.sin((u * 17 + v * 23 + fbm(u * 8, v * 8, 2)) * 9)) > 0.992 ? 0.35 : 0;
        o[0] = streak * 0.28 + scratch * 0.5 - pit * 0.4;
        o[1] = clamp(streak * 0.55 + scratch * 0.6, 0, 1);
        o[2] = 0.16 + 0.24 * streak + scratch * 0.35;
        break;
      }
      case 'rusted_metal': {
        const rust = fbm(u * 9, v * 9, 5), pit = fbm(u * 40, v * 40, 3);
        o[0] = rust * 0.65 + pit * 0.35; o[1] = smoothstep(0.42, 0.75, rust); o[2] = 0.25 + 0.68 * smoothstep(0.4, 0.78, rust);
        break;
      }
      case 'stone':
      case 'marble': {
        const n = fbm(u * 8, v * 8, 5);
        const vein = kind === 'marble' ? Math.pow(1 - Math.abs(Math.sin((u * 5 + v * 3 + n * 4.5) * 3.14)), 6) : fbm(u * 32, v * 32, 3) * 0.45;
        const speck = hash2(Math.floor(u * 260), Math.floor(v * 260)) > 0.96 ? 0.35 : 0;
        o[0] = n * 0.55 + vein * 0.35 + speck * 0.2; o[1] = clamp(n * 0.65 + vein * 0.6, 0, 1); o[2] = (kind === 'marble' ? 0.24 : 0.76) + 0.18 * n;
        break;
      }
      case 'ceramic': {
        const glaze = fbm(u * 5, v * 5, 4), speck = hash2(Math.floor(u * 220), Math.floor(v * 220)) > 0.988 ? 0.95 : 0;
        o[0] = 0.08 + 0.06 * glaze + speck * 0.15; o[1] = clamp(glaze * 0.25 + speck, 0, 1); o[2] = 0.12 + 0.10 * glaze + speck * 0.2;
        break;
      }
      case 'organic':
      case 'skin':
      case 'foliage': {
        const macro = fbm(u * 6, v * 6, 5), pores = fbm(u * 48, v * 48, 3);
        const vein = Math.pow(1 - Math.abs(Math.sin((u * 4 + macro * 3.2) * 3.14)), 5) * 0.45;
        o[0] = macro * 0.45 + pores * 0.38 + vein * 0.25; o[1] = clamp(macro * 0.72 + vein * 0.5, 0, 1); o[2] = (kind === 'skin' ? 0.52 : 0.60) + 0.20 * pores;
        break;
      }
      case 'fabric':
      case 'leather': {
        if (kind === 'leather') {
          const cell = fbm(u * 28, v * 28, 4), crease = Math.abs(Math.sin((u * 12 + cell * 4) * 3.14)) * 0.25;
          o[0] = cell * 0.7 + crease * 0.3; o[1] = clamp(cell * 0.8, 0, 1); o[2] = 0.48 + 0.25 * cell;
        } else {
          const wx = Math.sin(u * 420), wy = Math.sin(v * 420), weave = 0.5 + 0.25 * wx + 0.25 * wy, m = fbm(u * 8, v * 8, 3);
          o[0] = weave * 0.65 + m * 0.2; o[1] = clamp(m * 0.7 + weave * 0.2, 0, 1); o[2] = 0.84 + 0.1 * weave;
        }
        break;
      }
      case 'rubber': { const g = fbm(u * 55, v * 55, 3); o[0] = g * 0.45; o[1] = g * 0.35; o[2] = 0.86 + 0.10 * g; break; }
      case 'glass': o[0] = 0.5; o[1] = 0.05; o[2] = 0.05; break;
      default: { const g = fbm(u * 16, v * 16, 4); o[0] = g * 0.4; o[1] = g * 0.6; o[2] = 0.48 + 0.16 * g; }
    }
  }

  const pbrCanvasCache = new Map();
  function bakePBRTextures(kind, c1Hex, c2Hex) {
    let c1, c2;
    try { c1 = new THREE.Color(c1Hex != null ? c1Hex : '#c8d0d8'); } catch (_) { c1 = new THREE.Color('#c8d0d8'); }
    try { c2 = new THREE.Color(c2Hex != null ? c2Hex : c1.clone().multiplyScalar(0.55)); } catch (_) { c2 = c1.clone().multiplyScalar(0.55); }
    const key = `${kind}:${c1.getHexString()}:${c2.getHexString()}`;
    let cvSet = pbrCanvasCache.get(key);
    if (!cvSet) {
      const N = 256, mk = () => { const c = document.createElement('canvas'); c.width = c.height = N; return c; };
      const ca = mk(), cn = mk(), cr = mk();
      const ia = ca.getContext('2d').createImageData(N, N), inn = cn.getContext('2d').createImageData(N, N), ir = cr.getContext('2d').createImageData(N, N);
      const H = new Float32Array(N * N), o = [0, 0, 0], h1 = c1.getHex(), h2 = c2.getHex();
      const A = [(h1 >> 16) & 255, (h1 >> 8) & 255, h1 & 255], B = [(h2 >> 16) & 255, (h2 >> 8) & 255, h2 & 255];
      for (let y = 0; y < N; y++) for (let x = 0; x < N; x++) {
        patternDetail(kind, x / N, y / N, o);
        const i = y * N + x, m = clamp(o[1], 0, 1), sh = 0.70 + 0.30 * clamp(o[0], 0, 1);
        H[i] = o[0];
        ia.data[i * 4] = (A[0] + (B[0] - A[0]) * m) * sh;
        ia.data[i * 4 + 1] = (A[1] + (B[1] - A[1]) * m) * sh;
        ia.data[i * 4 + 2] = (A[2] + (B[2] - A[2]) * m) * sh;
        ia.data[i * 4 + 3] = 255;
        const rr = clamp(o[2], 0.04, 1) * 255;
        ir.data[i * 4] = ir.data[i * 4 + 1] = ir.data[i * 4 + 2] = rr; ir.data[i * 4 + 3] = 255;
      }
      const str = { stone: 5.5, bark: 6.0, wood: 3.4, metal: 1.8, brushed_metal: 2.0, rusted_metal: 4.8, organic: 3.6, skin: 2.6, foliage: 3.8, fabric: 4.5, leather: 4.2, ceramic: 1.4, rubber: 3.2, glass: 0 }[kind] || 2.5;
      const hh = (x, y) => H[((y + N) % N) * N + ((x + N) % N)];
      for (let y = 0; y < N; y++) for (let x = 0; x < N; x++) {
        const gx = (hh(x + 1, y - 1) + 2 * hh(x + 1, y) + hh(x + 1, y + 1)) - (hh(x - 1, y - 1) + 2 * hh(x - 1, y) + hh(x - 1, y + 1));
        const gy = (hh(x - 1, y + 1) + 2 * hh(x, y + 1) + hh(x + 1, y + 1)) - (hh(x - 1, y - 1) + 2 * hh(x, y - 1) + hh(x + 1, y - 1));
        let nx = -gx * str * 0.25, ny = gy * str * 0.25, nz = 1; const l = Math.hypot(nx, ny, nz); const i = (y * N + x) * 4;
        inn.data[i] = (nx / l * 0.5 + 0.5) * 255; inn.data[i + 1] = (ny / l * 0.5 + 0.5) * 255; inn.data[i + 2] = (nz / l * 0.5 + 0.5) * 255; inn.data[i + 3] = 255;
      }
      ca.getContext('2d').putImageData(ia, 0, 0); cn.getContext('2d').putImageData(inn, 0, 0); cr.getContext('2d').putImageData(ir, 0, 0);
      cvSet = { ca, cn, cr };
      if (pbrCanvasCache.size > 48) pbrCanvasCache.clear();
      pbrCanvasCache.set(key, cvSet);
    }
    const wrap = (cv, srgb) => {
      const t = new THREE.CanvasTexture(cv);
      t.wrapS = t.wrapT = THREE.RepeatWrapping; t.repeat.set(2, 2); t.anisotropy = 4;
      if (srgb && THREE.SRGBColorSpace) t.colorSpace = THREE.SRGBColorSpace;
      t.needsUpdate = true; return t;
    };
    return { map: wrap(cvSet.ca, true), normalMap: wrap(cvSet.cn, false), roughnessMap: wrap(cvSet.cr, false) };
  }

  function makeDetailedPBRMaterial(kind, primaryHex, secondaryHex, opts = {}) {
    const k = String(kind || 'organic').toLowerCase();
    const t = bakePBRTextures(k, primaryHex, secondaryHex);
    const env = APP.cubeRenderTarget?.texture || APP.roomEnvTexture || null;
    if (k === 'glass') {
      return new THREE.MeshPhysicalMaterial(Object.assign({
        color: new THREE.Color(primaryHex != null ? primaryHex : '#cceeff'), map: t.map, normalMap: t.normalMap,
        roughness: 0.08, metalness: 0.0, transmission: 0.90, ior: 1.5, thickness: 0.25, clearcoat: 1.0,
        envMap: env, envMapIntensity: 1.5, side: THREE.DoubleSide
      }, opts));
    }
    const isMetal = k === 'metal' || k === 'brushed_metal' || k === 'rusted_metal';
    const isGlossy = k === 'ceramic' || k === 'car_paint' || opts.clearcoat;
    const MatCls = isGlossy && THREE.MeshPhysicalMaterial ? THREE.MeshPhysicalMaterial : THREE.MeshStandardMaterial;
    return new MatCls(Object.assign({
      map: t.map, normalMap: t.normalMap, normalScale: new THREE.Vector2(1.25, 1.25),
      roughnessMap: t.roughnessMap, roughness: 1.0,
      metalness: isMetal ? (k === 'rusted_metal' ? 0.55 : 0.88) : 0.04,
      envMap: env, envMapIntensity: isMetal ? 1.5 : 1.0,
      clearcoat: isGlossy ? 0.85 : 0.0, clearcoatRoughness: 0.12
    }, opts));
  }

  function reconHelpers() {
    const C = APP.craft || {};
    return {
      pbrMat: (kind, hex1, hex2, opts) => makeDetailedPBRMaterial(kind, hex1, hex2, opts),
      mat: (c, o) => {
        const k = o?.kind || (o?.metalness > 0.5 ? 'brushed_metal' : (o?.transmission > 0.3 ? 'glass' : (o?.roughness < 0.28 ? 'ceramic' : 'organic')));
        return makeDetailedPBRMaterial(k, c, o?.secondaryColor, o);
      },
      bevelBox: (w, h, d, r, s) => {
        const ww = Math.max(0.004, w || 0.1), hh = Math.max(0.004, h || 0.1), dd = Math.max(0.004, d || 0.1);
        const rr = Math.min(r || 0.008, Math.min(ww, hh, dd) * 0.45);
        return C.bevelBox ? C.bevelBox(ww, hh, dd, rr, s || 3) : new THREE.BoxGeometry(ww, hh, dd);
      },
      roundedBox: (w, h, d, r, s) => {
        const ww = Math.max(0.004, w || 0.1), hh = Math.max(0.004, h || 0.1), dd = Math.max(0.004, d || 0.1);
        const rr = Math.min(r || 0.008, Math.min(ww, hh, dd) * 0.45);
        return C.bevelBox ? C.bevelBox(ww, hh, dd, rr, s || 3) : new THREE.BoxGeometry(ww, hh, dd);
      },
      lathe: (pts, segs) => {
        const v2 = (pts || []).map((p) => (p.isVector2 ? p : new THREE.Vector2(Math.max(0, p[0]), p[1])));
        return new THREE.LatheGeometry(v2, segs || 32);
      },
      tube: (pts, r, ts, rs, closed, taperFn) => {
        const v3s = (pts || []).map((p) => (p.isVector3 ? p : new THREE.Vector3(p[0], p[1], p[2])));
        const curve = new THREE.CatmullRomCurve3(v3s, !!closed);
        const geo = new THREE.TubeGeometry(curve, ts || 32, r || 0.02, rs || 12, !!closed);
        if (typeof taperFn === 'function') {
          const pos = geo.attributes.position, rSegs = rs || 12, tSegs = ts || 32;
          const cp = new V3(), rad = new V3();
          for (let i = 0; i < pos.count; i++) {
            const ring = Math.floor(i / (rSegs + 1)), t = clamp(ring / tSegs, 0, 1);
            curve.getPoint(t, cp);
            rad.fromBufferAttribute(pos, i).sub(cp).multiplyScalar(Math.max(0.04, taperFn(t))).add(cp);
            pos.setXYZ(i, rad.x, rad.y, rad.z);
          }
          geo.computeVertexNormals();
        }
        return geo;
      },
      extrude: (pts2D, depth = 0.04, bevel = 0.008) => {
        const shape = new THREE.Shape();
        pts2D.forEach((p, idx) => { const x = p.x ?? p[0], y = p.y ?? p[1]; if (idx === 0) shape.moveTo(x, y); else shape.lineTo(x, y); });
        shape.closePath();
        const g = new THREE.ExtrudeGeometry(shape, { depth, bevelEnabled: bevel > 0, bevelThickness: bevel, bevelSize: bevel, bevelSegments: 3 });
        g.translate(0, 0, -depth * 0.5);
        return g;
      },
      organic: (rx = 0.1, ry = 0.1, rz = 0.1, detail = 28, amp = 0.08, seed = 1) => {
        const g = new THREE.SphereGeometry(1, detail, detail);
        const pos = g.attributes.position, v = new V3();
        for (let i = 0; i < pos.count; i++) {
          v.fromBufferAttribute(pos, i);
          const n = Math.sin(v.x * 5.1 + seed) * Math.cos(v.y * 4.7 - seed) * Math.sin(v.z * 6.3 + seed);
          v.multiplyScalar(1 + n * amp);
          pos.setXYZ(i, v.x * rx, v.y * ry, v.z * rz);
        }
        g.computeVertexNormals(); return g;
      },
      noiseTex: (hexA, hexB) => bakePBRTextures('stone', hexA, hexB).map,
      bumpTex: (kind) => bakePBRTextures(kind || 'stone', '#888888', '#aaaaaa').normalMap
    };
  }

  function stripThinking(t) {
    return String(t || '').replace(/<think>[\s\S]*?<\/think>/gi, '').trim();
  }

  function extractJSON(t) {
    const raw = stripThinking(t).replace(/```(?:json)?/gi, '').replace(/```/g, '').trim();
    // Try balanced brace extraction from last valid JSON object backward
    for (let end = raw.lastIndexOf('}'); end >= 0; end = raw.lastIndexOf('}', end - 1)) {
      let depth = 0;
      for (let i = end; i >= 0; i--) {
        if (raw[i] === '}') depth++;
        else if (raw[i] === '{') {
          depth--;
          if (depth === 0) {
            const sub = raw.slice(i, end + 1);
            try { return JSON.parse(sub); } catch (_) {}
            try { return JSON.parse(sub.replace(/,\s*([}\]])/g, '$1')); } catch (_) {}
            break;
          }
        }
      }
    }
    return null;
  }

  function extractJSCode(t) {
    const raw = stripThinking(t);
    const fenceMatches = [...raw.matchAll(/```(?:javascript|js)?\s*\r?\n([\s\S]*?)```/gi)];
    if (fenceMatches.length) return fenceMatches[fenceMatches.length - 1][1].trim();
    const asJson = extractJSON(raw);
    if (asJson && typeof asJson.code === 'string') return asJson.code.trim();
    if (raw.includes('THREE.') && (raw.includes('return ') || raw.includes('new THREE.Group'))) {
      return raw.replace(/^```[a-z]*\s*/i, '').replace(/```$/i, '').trim();
    }
    return '';
  }

  async function callAI(systemPrompt, userPrompt, imageB64) {
    if (!APP.ai) throw new Error('APP.ai not loaded');
    const aiCaller = (APP.ai?.isLiveModelSelected?.() && APP.ai?.chatWithBuilder) ? APP.ai.chatWithBuilder : APP.ai.chat;
    const runCall = (withImg) => withTimeout(aiCaller.call(APP.ai, {
      systemPrompt,
      messages: [{ role: 'user', content: userPrompt }],
      prompt: userPrompt,
      imageB64: withImg ? imageB64 : undefined,
      maxSteps: 1
    }), 42000, 'AI timeout');

    try {
      const res = await runCall(!!imageB64);
      if (res && res.ok === false) throw new Error(res.error || 'AI error');
      return String(res?.text || res?.content || '');

  const r3 = (a) => a.map((x) => +x.toFixed(3));
  function describeGeometry(base, strokes) {
    const strokeInfo = strokes.map((s, idx) => {
      const b = new THREE.Box3().setFromPoints(s.pts);
      const sz = b.getSize(new V3());
      const closed = s.pts[0].distanceTo(s.pts[s.pts.length - 1]) < Math.max(0.06, sz.length() * 0.18);
      return `stroke#${idx + 1}: len=${strokeLen(s).toFixed(2)}m, span=[${sz.x.toFixed(2)},${sz.y.toFixed(2)},${sz.z.toFixed(2)}], ${closed ? 'closed-loop' : 'open-curve'}`;
    }).join('; ');
    return {
      size_m: r3(base.bbox.size),
      aspect: { w_h: +(base.bbox.size[0] / Math.max(base.bbox.size[1], 1e-4)).toFixed(2), d_h: +(base.bbox.size[2] / Math.max(base.bbox.size[1], 1e-4)).toFixed(2) },
      sym: { mirrorX: +base.symmetry.mirrorX.toFixed(2), radialY: +base.symmetry.radialY.toFixed(2) },
      sliceY: base.sliceY.map((s) => [+s.w.toFixed(2), +s.d.toFixed(2)]),
      strokeInfo,
      asciiFront: base.ascii?.front || '',
      asciiSide: base.ascii?.side || '',
      asciiTop: base.ascii?.top || ''
    };
  }

  function safeColor(c, d) { try { return new THREE.Color(c).getHexString(); } catch (_) { return d; } }
  function fallbackLabelFromGeometry(d) {
    const [sx, sy, sz] = d.size_m;

    if (d.sym.radialY > 0.78) return 'ceramic mug';
    if (d.sym.mirrorX > 0.80) return 'sculpted figure';
    return 'wooden vessel';
  }

  function parseHint(text) {
    let raw = String(text || '').trim();

        /^(يا جيني |جيني |راني رايح نرسم|راني باغي نرسم|راني حاب نرسم|راني نرسم في|راني نرسم|باغي نرسم|حاب نرسم|راح نرسم|غادي نرسم|سارسم|ارسم|نرسم|ندير|نخدم|اريد رسم|هيا نرسم|بغيت نرسم|حبيت نرسم)\s+/,

      let mappedEn = '';
      for (const w of words) {
        const wNoAl = w.replace(/^ال/, '');
        if (AR_NOUN_MAP[w]) { mappedEn = AR_NOUN_MAP[w]; break; }
        if (AR_NOUN_MAP[wNoAl]) { mappedEn = AR_NOUN_MAP[wNoAl]; break; }
        for (const [k, v] of Object.entries(AR_NOUN_MAP)) {
          if (normalizeAr(k) === w || normalizeAr(k) === wNoAl) { mappedEn = v; break; }
        }
        if (mappedEn) break;
      }
      const label = mappedEn || words.join(' ');
      return { label, keywords: [label, words.join(' ')], text: raw, rawVoice: raw };
    }

      /^(hey geeny |geeny |ok |okay )/,

    for (const kw of kws) for (const [k, v] of map) if (k.includes(kw) || (k.length > 3 && kw.includes(k))) return v;
    for (const kw of kws) {
      const w = kw.replace(/\s+/g, '');
      for (const url of ['./assets/' + w + '.glb', './assets/' + w + '.gltf', './' + w + '.glb']) { try { const r = await fetch(url, { method: 'HEAD' }); if (r.ok) return { name: w, url }; } catch (_) {} }
    }
    return null;
  }
  const glbCache = new Map();
  const loadGLB = (url) => {
    if (!glbCache.has(url)) glbCache.set(url, new Promise((res, rej) => APP.gltfLoader.load(url, (g) => res(g.scene || g.scenes[0]), undefined, rej)).catch((e) => { glbCache.delete(url); throw e; }));
    return glbCache.get(url).then((s) => s.clone(true));
  };
  function fitGLB(model, spec, base) {
    const holder = new THREE.Group(), scaler = new THREE.Group(), pre = new THREE.Group();
    pre.add(model); scaler.add(pre); holder.add(scaler); holder.updateMatrixWorld(true);
    let box = new THREE.Box3().setFromObject(pre), sz = box.getSize(new V3()); const bs = base.bbox.size;
    if (spec.lying && sz.y > Math.max(sz.x, sz.z) * 1.4 && Math.max(bs[0], bs[2]) > bs[1] * 1.4) { pre.rotation.x = Math.PI / 2; holder.updateMatrixWorld(true); box.setFromObject(pre); sz = box.getSize(new V3()); }
    pre.position.copy(box.getCenter(new V3())).negate(); holder.updateMatrixWorld(true);
    let laZ = sz.z >= sz.x; const la = laZ ? new V3(0, 0, 1) : new V3(1, 0, 0);
    const faceDir = { toward: new V3(0, 0, 1), away: new V3(0, 0, -1), right: new V3(1, 0, 0), left: new V3(-1, 0, 0) }[spec.facing] || null, yw = base.yaw;
    const elong = yw.ratio >= 1.25; let D = null;
    if (elong) { D = new V3(Math.cos(yw.ang), 0, Math.sin(yw.ang)); if (faceDir && D.dot(faceDir) < 0) D.negate(); }
    else if (faceDir) { D = faceDir.clone(); la.set(0, 0, 1); laZ = true; }
    if (D) { if (la.dot(D) < -0.9999) holder.quaternion.setFromAxisAngle(UP, Math.PI); else holder.quaternion.setFromUnitVectors(la, D); }
    const gLa = laZ ? sz.z : sz.x, gOt = laZ ? sz.x : sz.z, tLa = elong ? yw.u : (yw.u + yw.v) / 2, tOt = elong ? yw.v : (yw.u + yw.v) / 2;
    const rLa = tLa / Math.max(gLa, 1e-4), rOt = tOt / Math.max(gOt, 1e-4), rY = bs[1] / Math.max(sz.y, 1e-4), su = Math.cbrt(rLa * rOt * rY), lo = 0.80, hi = 1.25;
    const fLa = su * clamp(rLa / su, lo, hi), fOt = su * clamp(rOt / su, lo, hi), fY = su * clamp(rY / su, lo, hi);
    scaler.scale.set(laZ ? fOt : fLa, fY, laZ ? fLa : fOt);
    return holder;
  }
  function fitCentered(obj, size, lo, hi) {
    const holder = new THREE.Group(), scaler = new THREE.Group(); scaler.add(obj); holder.add(scaler); holder.updateMatrixWorld(true);
    const box = new THREE.Box3().setFromObject(obj), sz = box.getSize(new V3()); obj.position.sub(box.getCenter(new V3()));
    const rx = size[0] / Math.max(sz.x, 1e-4), ry = size[1] / Math.max(sz.y, 1e-4), rz = size[2] / Math.max(sz.z, 1e-4), su = Math.cbrt(rx * ry * rz);
    scaler.scale.set(su * clamp(rx / su, lo, hi), su * clamp(ry / su, lo, hi), su * clamp(rz / su, lo, hi)); return holder;
  }

  // ------------------------------------------------------------------ Stable 3D Rigid-Body Physics Solver
  const rigidBodies = [];
  const tmpSupportPt = new V3(), tmpVpt = new V3(), tmpR = new V3(), tmpTorque = new V3(), tmpAngQ = new THREE.Quaternion();

  function initRigidBody(ent, root) {
    root.updateMatrixWorld(true);
    const invRoot = root.matrixWorld.clone().invert();
    const localBox = new THREE.Box3();
    root.traverse((m) => {
      if (m.isMesh && m.geometry) {
        m.geometry.computeBoundingBox();
        if (m.geometry.boundingBox) {
          const b = m.geometry.boundingBox.clone().applyMatrix4(invRoot.clone().multiply(m.matrixWorld));
          localBox.union(b);
        }
      }
    });
    if (localBox.isEmpty()) localBox.set(new V3(-0.1, -0.1, -0.1), new V3(0.1, 0.1, 0.1));
    const sz = localBox.getSize(new V3());
    const lx0 = localBox.min.x, lx1 = localBox.max.x;
    const ly0 = localBox.min.y, ly1 = localBox.max.y;
    const lz0 = localBox.min.z, lz1 = localBox.max.z;
    const lcx = (lx0 + lx1) * 0.5, lcy = (ly0 + ly1) * 0.5, lcz = (lz0 + lz1) * 0.5;

    const supportPts = [
      new V3(lx0, ly0, lz0), new V3(lx1, ly0, lz0), new V3(lx0, ly1, lz0), new V3(lx1, ly1, lz0),
      new V3(lx0, ly0, lz1), new V3(lx1, ly0, lz1), new V3(lx0, ly1, lz1), new V3(lx1, ly1, lz1),
      new V3(lcx, ly0, lcz), new V3(lcx, ly1, lcz), new V3(lx0, lcy, lcz), new V3(lx1, lcy, lcz),
      new V3(lcx, lcy, lz0), new V3(lcx, lcy, lz1)
    ];

    const mass = clamp((sz.x * sz.y * sz.z) * 40.0, 0.4, 10.0);
    const radius = Math.max(0.05, sz.length() * 0.44);
    // Stable clamped rotational inertia (never explodes on small objects!)
    const inertia = Math.max(0.018, mass * radius * radius * 0.55);

    ent.rb = {
      localBox, supportPts, radius, mass, invMass: 1 / mass, invInertia: 1 / inertia,
      vel: new V3(), angVel: new V3(), bounce: 0.28, friction: 0.65,
      handGrabbed: false, sleeping: false, sleepTimer: 0,
      prevMinY: root.position.y + ly0
    };
    if (!rigidBodies.includes(ent)) rigidBodies.push(ent);
  }

  function computeWorldMinY(ent) {
    ent.group.updateMatrixWorld(true);
    let minY = Infinity;
    const pts = ent.rb.supportPts;
    for (let k = 0; k < pts.length; k++) {
      tmpSupportPt.copy(pts[k]).applyMatrix4(ent.group.matrixWorld);
      if (tmpSupportPt.y < minY) minY = tmpSupportPt.y;
    }
    return minY;
  }

  function stepRigidBodies(dt) {
    const safeDt = Math.min(0.025, Math.max(0.004, dt || 0.016));
    const usingHands = !!(APP.hands?.left?.isHand || APP.hands?.right?.isHand);

    for (let i = rigidBodies.length - 1; i >= 0; i--) {
      const ent = rigidBodies[i];
      if (!ent || !ent.group?.parent) { rigidBodies.splice(i, 1); continue; }
      ent.grabbable = !usingHands;
      const rb = ent.rb;
      if (!rb) continue;

      if (ent.isHeld || rb.handGrabbed) {
        rb.sleeping = false; rb.sleepTimer = 0;
        rb.prevMinY = computeWorldMinY(ent);
        if (ent.physics) ent.physics.vel.copy(rb.vel);
        continue;
      }
      if (!S.gravityOn) {
        rb.vel.set(0, 0, 0); rb.angVel.set(0, 0, 0);
        continue;
      }
      if (rb.sleeping) continue;

      rb.vel.y -= 9.81 * safeDt;
      rb.vel.multiplyScalar(Math.pow(0.97, safeDt * 60));
      rb.angVel.multiplyScalar(Math.pow(0.90, safeDt * 60));
      rb.vel.clampLength(0, 7.5);
      rb.angVel.clampLength(0, 9.0);

      ent.group.position.addScaledVector(rb.vel, safeDt);
      const angSpd = rb.angVel.length();
      if (angSpd > 1e-4) {
        tmpAngQ.setFromAxisAngle(tmpVpt.copy(rb.angVel).divideScalar(angSpd), angSpd * safeDt);
        ent.group.quaternion.premultiply(tmpAngQ).normalize();
      }

      // Anti-tunneling support plane resolution (Floor y=0, Hand-Height Shelf, XR Planes)
      let supportY = 0;
      const px = ent.group.position.x, pz = ent.group.position.z;
      if (workFloor && workFloor.visible) {
        const wfTop = workFloor.position.y + 0.012;
        if (Math.hypot(px - workFloor.position.x, pz - workFloor.position.z) <= 0.64 && rb.prevMinY >= wfTop - 0.06) {
          supportY = Math.max(supportY, wfTop);
        }
      }
      for (const pl of APP.planes || []) {
        if (pl.orientation === 'horizontal' && px >= pl.box.min.x && px <= pl.box.max.x && pz >= pl.box.min.z && pz <= pl.box.max.z) {
          if (rb.prevMinY >= pl.pos.y - 0.08) supportY = Math.max(supportY, pl.pos.y);
        }
      }

      ent.group.updateMatrixWorld(true);
      let deepestPen = 0, hitCount = 0, minWorldY = Infinity;
      const avgContact = new V3();
      for (let k = 0; k < rb.supportPts.length; k++) {
        tmpSupportPt.copy(rb.supportPts[k]).applyMatrix4(ent.group.matrixWorld);
        if (tmpSupportPt.y < minWorldY) minWorldY = tmpSupportPt.y;
        const pen = supportY - tmpSupportPt.y;
        if (pen > 0) {
          if (pen > deepestPen) deepestPen = pen;
          avgContact.add(tmpSupportPt);
          hitCount++;
        }
      }

      if (hitCount > 0 && deepestPen > 0) {
        ent.group.position.y += deepestPen;
        minWorldY += deepestPen;
        avgContact.divideScalar(hitCount);
        avgContact.y = supportY;
        tmpR.subVectors(avgContact, ent.group.position);

        tmpVpt.crossVectors(rb.angVel, tmpR).add(rb.vel);
        if (tmpVpt.y < 0) {
          const e = tmpVpt.y < -0.55 ? rb.bounce : 0.0;
          const rCrossN = new V3().crossVectors(tmpR, UP);
          const denom = rb.invMass + rCrossN.lengthSq() * rb.invInertia;
          const jn = -(1 + e) * tmpVpt.y / Math.max(1e-4, denom);
          rb.vel.y += jn * rb.invMass;
          tmpTorque.crossVectors(tmpR, UP).multiplyScalar(jn * 0.45);
          rb.angVel.addScaledVector(tmpTorque, rb.invInertia);
        }

        const vt = new V3(tmpVpt.x, 0, tmpVpt.z);
        const vtLen = vt.length();
        if (vtLen > 1e-4) {
          const frictionImp = Math.min(vtLen * rb.mass * 0.45, 9.81 * rb.mass * safeDt * rb.friction);
          vt.divideScalar(vtLen).multiplyScalar(-frictionImp);
          rb.vel.addScaledVector(vt, rb.invMass);
          tmpTorque.crossVectors(tmpR, vt).multiplyScalar(0.35);
          rb.angVel.addScaledVector(tmpTorque, rb.invInertia);
        }

        if (hitCount === 1) {
          // Gentle tipping torque toward stable base
          const tip = new V3().crossVectors(tmpR, new V3(0, 9.81 * rb.mass * safeDt * 0.35, 0));
          rb.angVel.addScaledVector(tip, rb.invInertia);
        } else {
          rb.vel.x *= 0.80; rb.vel.z *= 0.80;
          rb.angVel.multiplyScalar(0.75);
        }

        if (rb.vel.lengthSq() < 0.015 && rb.angVel.lengthSq() < 0.08) {
          rb.sleepTimer += safeDt;
          if (rb.sleepTimer > 0.25) {
            rb.vel.set(0, 0, 0); rb.angVel.set(0, 0, 0);
            rb.sleeping = true;
          }
        } else {
          rb.sleepTimer = 0;
        }
      } else {
        rb.sleepTimer = 0;
      }

      rb.prevMinY = minWorldY;
      if (ent.physics) ent.physics.vel.copy(rb.vel);
    }

    // Object-to-object rigid-body collisions
    for (let i = 0; i < rigidBodies.length; i++) {
      const a = rigidBodies[i]; if (!a.rb || !S.gravityOn) continue;
      for (let j = i + 1; j < rigidBodies.length; j++) {
        const b = rigidBodies[j]; if (!b.rb) continue;
        if (a.rb.sleeping && b.rb.sleeping) continue;
        if ((a.isHeld || a.rb.handGrabbed) && (b.isHeld || b.rb.handGrabbed)) continue;
        const delta = tmpR.subVectors(b.group.position, a.group.position);
        const minDist = (a.rb.radius + b.rb.radius) * 0.68;
        const dSq = delta.lengthSq();
        if (dSq > 1e-6 && dSq < minDist * minDist) {
          const d = Math.sqrt(dSq), n = delta.divideScalar(d), pen = minDist - d;
          a.rb.sleeping = false; b.rb.sleeping = false;
          const aStatic = a.isHeld || a.rb.handGrabbed, bStatic = b.isHeld || b.rb.handGrabbed;
          if (!aStatic && !bStatic) { a.group.position.addScaledVector(n, -pen * 0.5); b.group.position.addScaledVector(n, pen * 0.5); }
          else if (!aStatic) a.group.position.addScaledVector(n, -pen);
          else if (!bStatic) b.group.position.addScaledVector(n, pen);

          const relV = tmpVpt.subVectors(b.rb.vel, a.rb.vel);
          const vn = relV.dot(n);
          if (vn < 0) {
            const jImp = -(1 + 0.30) * vn / ((aStatic ? 0 : a.rb.invMass) + (bStatic ? 0 : b.rb.invMass) || 1);
            if (!aStatic) { a.rb.vel.addScaledVector(n, -jImp * a.rb.invMass); a.rb.angVel.add(new V3(n.z, 0, -n.x).multiplyScalar(jImp * 0.25)); }
            if (!bStatic) { b.rb.vel.addScaledVector(n, jImp * b.rb.invMass); b.rb.angVel.add(new V3(-n.z, 0, n.x).multiplyScalar(jImp * 0.25)); }
          }
        }
      }
    }
  }

  // ------------------------------------------------------------------ Realistic Hand Grab (Real WebXR Hand Always Visible + Multi-Finger Surface Enclosure + Finger Poke)
  const handGrabs = {
    L: { ent: null, offPos: new V3(), offQuat: new THREE.Quaternion(), prevPos: new V3(), prevQuat: new THREE.Quaternion(), vel: new V3(), angVel: new V3() },
    R: { ent: null, offPos: new V3(), offQuat: new THREE.Quaternion(), prevPos: new V3(), prevQuat: new THREE.Quaternion(), vel: new V3(), angVel: new V3() }
  };

  function releaseProceduralHandGrab(side) {
    const hg = handGrabs[side];
    if (!hg || !hg.ent) return;
    if (hg.ent.rb) {
      hg.ent.rb.handGrabbed = false;
      hg.ent.rb.sleeping = false;
      hg.ent.rb.vel.copy(hg.vel).clampLength(0, 5.5);
      hg.ent.rb.angVel.copy(hg.angVel).clampLength(0, 8.0);
    }
    hg.ent = null;
  }

  function getJointPos(xrHand, name, out) {
    const j = xrHand?.joints?.[name];
    if (!j || !j.visible) return null;
    return j.getWorldPosition(out || new V3());
  }

  function tickProceduralHandGrab(dt) {
    const safeDt = Math.max(0.004, Math.min(0.05, dt || 0.016));
    const hands = [['L', APP.hands?.left], ['R', APP.hands?.right]];
    const isDrawingNow = S.sculptOn && S.state === 1;

    for (const [side, slot] of hands) {
      const hg = handGrabs[side];
      if (!slot || !slot.active || !slot.isHand || !slot.xrHand) {
        if (hg.ent) releaseProceduralHandGrab(side);
        continue;
      }

      const wrist = getJointPos(slot.xrHand, 'wrist');
      const midMeta = getJointPos(slot.xrHand, 'middle-finger-metacarpal');
      const thumbTip = getJointPos(slot.xrHand, 'thumb-tip');
      const idxTip = getJointPos(slot.xrHand, 'index-finger-tip');
      const midTip = getJointPos(slot.xrHand, 'middle-finger-tip');
      const rngTip = getJointPos(slot.xrHand, 'ring-finger-tip');
      const pnkTip = getJointPos(slot.xrHand, 'pinky-finger-tip');
      if (!wrist || !midMeta || !thumbTip || !midTip) continue;

      const handScale = Math.max(0.055, wrist.distanceTo(midMeta));
      const palmPos = new V3().lerpVectors(wrist, midMeta, 0.58);
      const palmQuat = slot.quat ? slot.quat.clone() : new THREE.Quaternion();

      // Scale-invariant curl check for ALL power fingers (never triggers on index+thumb pinch alone!)
      const curlThresh = handScale * 1.24;
      const openThresh = handScale * 1.44;
      let curledPower = 0, openPower = 0;
      for (const tip of [midTip, rngTip, pnkTip]) {
        if (!tip) continue;
        const d = tip.distanceTo(palmPos);
        if (d < curlThresh) curledPower++;
        if (d > openThresh) openPower++;
      }
      const idxCurled = idxTip ? (idxTip.distanceTo(palmPos) < curlThresh) : false;
      const isPinchOnly = idxTip && (idxTip.distanceTo(thumbTip) < 0.036) && curledPower === 0;

      if (hg.ent) {
        if (!hg.ent.group?.parent || isDrawingNow) {
          releaseProceduralHandGrab(side);
          continue;
        }
        const surfBox = new THREE.Box3().setFromObject(hg.ent.group).expandByScalar(0.06);
        const tipsTouching = [thumbTip, idxTip, midTip, rngTip, pnkTip].filter((t) => t && surfBox.containsPoint(t)).length;
        if (openPower >= 2 && tipsTouching < 2) {
          releaseProceduralHandGrab(side);
          continue;
        }

        const targetPos = hg.offPos.clone().applyQuaternion(palmQuat).add(palmPos);
        const targetQuat = palmQuat.clone().multiply(hg.offQuat);
        hg.ent.group.position.lerp(targetPos, 0.78);
        hg.ent.group.quaternion.slerp(targetQuat, 0.78);

        const instVel = hg.ent.group.position.clone().sub(hg.prevPos).divideScalar(safeDt);
        hg.vel.lerp(instVel, 0.45);
        hg.prevPos.copy(hg.ent.group.position);

        const qDelta = hg.ent.group.quaternion.clone().multiply(hg.prevQuat.clone().invert());
        const angAxis = new V3(qDelta.x, qDelta.y, qDelta.z), angLen = angAxis.length();
        if (angLen > 1e-5) {
          hg.angVel.lerp(angAxis.divideScalar(angLen).multiplyScalar((2 * Math.atan2(angLen, qDelta.w)) / safeDt), 0.35);
        }
        hg.prevQuat.copy(hg.ent.group.quaternion);
        continue;
      }

      if (isDrawingNow || isPinchOnly) continue;

      for (const ent of rigidBodies) {
        if (!ent.group?.parent || !ent.rb || ent.rb.handGrabbed || ent.isHeld) continue;
        const distPalm = ent.group.position.distanceTo(palmPos);
        if (distPalm > ent.rb.radius + 0.22) continue;

        const surfBox = new THREE.Box3().setFromObject(ent.group).expandByScalar(0.042);
        const tipsTouching = [thumbTip, idxTip, midTip, rngTip, pnkTip].filter((t) => t && surfBox.containsPoint(t)).length;
        const thumbNear = surfBox.containsPoint(thumbTip) || thumbTip.distanceTo(palmPos) < handScale * 1.25;

        // Grab ONLY on multi-finger power enclosure (thumb + >=2 power/non-pinch fingers curled or wrapped around surface)
        if (thumbNear && ((curledPower >= 2 && (idxCurled || tipsTouching >= 2)) || tipsTouching >= 3)) {
          hg.ent = ent;
          ent.rb.handGrabbed = true;
          ent.rb.sleeping = false;
          const invQ = palmQuat.clone().invert();
          // Keep object seated cleanly outside the palm so fingers wrap around it instead of clipping inside palm
          const objCenter = ent.group.position.clone();
          const palmNormal = new V3(0, -1, 0).applyQuaternion(palmQuat).normalize();
          const toObj = objCenter.clone().sub(palmPos);
          if (toObj.length() < Math.min(0.06, ent.rb.radius * 0.5)) {
            objCenter.copy(palmPos).addScaledVector(palmNormal, Math.min(0.08, ent.rb.radius * 0.65));
          }
          hg.offPos.copy(objCenter).sub(palmPos).applyQuaternion(invQ);
          hg.offQuat.copy(invQ).multiply(ent.group.quaternion);
          hg.prevPos.copy(ent.group.position);
          hg.prevQuat.copy(ent.group.quaternion);
          hg.vel.set(0, 0, 0); hg.angVel.set(0, 0, 0);
          break;
        }

        // Physical surface contact poke/push when fingers touch an ungrabbed object
        if (S.gravityOn) {
          ent.group.updateMatrixWorld(true);
          const invM = ent.group.matrixWorld.clone().invert();
          for (const tip of [idxTip, midTip, rngTip, thumbTip]) {
            if (!tip) continue;
            const localTip = tip.clone().applyMatrix4(invM);
            const padded = ent.rb.localBox.clone().expandByScalar(0.015);
            if (padded.containsPoint(localTip)) {
              ent.rb.sleeping = false;
              const push = ent.group.position.clone().sub(tip).normalize();
              ent.rb.vel.addScaledVector(push, 0.22);
              ent.rb.angVel.add(new V3(push.z, 0, -push.x).multiplyScalar(0.9));
              break;
            }
          }
        }
      }
    }
  }

  function registerSculptedEntity(rec, root, label) {
    root.updateMatrixWorld(true);
    let ent = null;
    try {
      ent = APP.registerEntity(root, {
        name: label, category: 'sculpted', grabbable: true, holdable: false,
        physics: { enabled: false, gravity: false },
        onGrab(slot) {
          ent._prevGrabPos = ent.group.position.clone();
          ent._prevGrabQuat = ent.group.quaternion.clone();
          ent._throwVel = new V3(); ent._throwAngVel = new V3();
          if (ent.rb) ent.rb.sleeping = false;
          try { slot?.gamepad?.hapticActuators?.[0]?.pulse(0.35, 30); } catch (_) {}
        },
        onRelease(slot) {
          if (ent.rb) {
            ent.rb.sleeping = false;
            if (ent._throwVel) ent.rb.vel.copy(ent._throwVel).clampLength(0, 5.5);
            if (ent._throwAngVel) ent.rb.angVel.copy(ent._throwAngVel).clampLength(0, 8.0);
          }
          try { slot?.gamepad?.hapticActuators?.[0]?.pulse(0.2, 20); } catch (_) {}
        },
        onUpdate(dt) {
          if (ent.isHeld && ent._prevGrabPos) {
            const safeDt = Math.max(0.004, Math.min(0.05, dt || 0.016));
            ent._throwVel.lerp(ent.group.position.clone().sub(ent._prevGrabPos).divideScalar(safeDt), 0.45);
            ent._prevGrabPos.copy(ent.group.position);
            const qd = ent.group.quaternion.clone().multiply(ent._prevGrabQuat.clone().invert());
            const ax = new V3(qd.x, qd.y, qd.z); const al = ax.length();
            if (al > 1e-5) ent._throwAngVel.lerp(ax.divideScalar(al).multiplyScalar((2 * Math.atan2(al, qd.w)) / safeDt), 0.35);
            ent._prevGrabQuat.copy(ent.group.quaternion);
          }
        }
      });
      initRigidBody(ent, root);
      APP.emit?.('entity:created', ent);
    } catch (_) {}
    rec.root = root; rec.entity = ent; rec.label = label;
    return ent;
  }

  // ------------------------------------------------------------------ Finalize & Zero-Lag GPU Vertex Morph (Mode B)
  function startReveal(rec, root, planeBelow, dur) {
    const b = new THREE.Box3().setFromObject(root);
    S.reveals.push({ rec, t: 0, dur: dur || 1.1, y0: b.min.y - 0.04, y1: b.max.y + 0.07, planeBelow, root, ctr: b.getCenter(new V3()) });
  }
  function finalize(rec, obj, label, centerF, dur, skipClip) {
    const root = new THREE.Group(); root.name = 'sculpt_' + label; root.add(obj);
    root.quaternion.copy(rec.yawQ);
    root.position.set(centerF[0], centerF[1], centerF[2]).applyQuaternion(rec.yawQ);
    APP.scene.add(root);
    if (skipClip) {
      registerSculptedEntity(rec, root, label);
      cleanHolo(rec);
      toast('✓ ' + label + ' manifested', 2400);
      return;
    }
    const plane = new THREE.Plane(new V3(0, -1, 0), -1e4);
    root.traverse((o) => { if (o.material) (Array.isArray(o.material) ? o.material : [o.material]).forEach((m) => { m.clippingPlanes = [plane]; m.needsUpdate = true; }); });
    registerSculptedEntity(rec, root, label);
    rec.planeBelow = plane;
    startReveal(rec, root, plane, dur);
  }

  function morphInto(rec, holder, base, label, alive) {
    return new Promise((resolve) => {
      holder.updateMatrixWorld(true);
      const P0 = base.positions, nP = P0.length / 3, stp = Math.max(1, Math.floor(nP / 500)), src = [];
      for (let i = 0; i < nP; i += stp) src.push(P0[i * 3], P0[i * 3 + 1], P0[i * 3 + 2]);
      if (!src.length) { finalize(rec, holder, label, base.bbox.center, 0.9, false); return resolve(); }

      const bSz = base.bbox.size, G = 8, buckets = new Array(G * G * G);
      for (let i = 0; i < src.length; i += 3) {
        const bx = clamp(Math.floor((src[i] / Math.max(bSz[0], 0.01) + 0.5) * G), 0, G - 1);
        const by = clamp(Math.floor((src[i + 1] / Math.max(bSz[1], 0.01) + 0.5) * G), 0, G - 1);
        const bz = clamp(Math.floor((src[i + 2] / Math.max(bSz[2], 0.01) + 0.5) * G), 0, G - 1);
        const bi = bx + G * (by + G * bz); if (buckets[bi] === undefined) buckets[bi] = i;
      }

      const morphTime = { value: 0.0 };
      const morphRoot = new THREE.Group();
      const morphOrigin = new V3(base.bbox.center[0], base.bbox.center[1], base.bbox.center[2]).applyQuaternion(rec.yawQ);
      morphRoot.quaternion.copy(rec.yawQ);
      morphRoot.position.copy(morphOrigin);

      const parts = [];
      let yMin = Infinity, yMax = -Infinity;

      holder.traverse((m) => {
        if (!m.isMesh || !m.geometry || !m.geometry.attributes.position) return;
        const g = m.geometry.clone();
        g.applyMatrix4(m.matrixWorld);
        const posArr = g.attributes.position.array;
        const n = posArr.length / 3;
        const startArr = new Float32Array(posArr.length);
        const delayArr = new Float32Array(n);

        for (let i = 0; i < n; i++) {
          const x = posArr[i * 3], y = posArr[i * 3 + 1], z = posArr[i * 3 + 2];
          const bx = clamp(Math.floor((x / Math.max(bSz[0], 0.01) + 0.5) * G), 0, G - 1);
          const by = clamp(Math.floor((y / Math.max(bSz[1], 0.01) + 0.5) * G), 0, G - 1);
          const bz = clamp(Math.floor((z / Math.max(bSz[2], 0.01) + 0.5) * G), 0, G - 1);
          let j = buckets[bx + G * (by + G * bz)];
          if (j === undefined) j = ((i * 7) % (src.length / 3)) * 3;
          startArr[i * 3] = src[j]; startArr[i * 3 + 1] = src[j + 1]; startArr[i * 3 + 2] = src[j + 2];
          if (y < yMin) yMin = y; if (y > yMax) yMax = y;
        }
        g.setAttribute('aMorphStart', new THREE.BufferAttribute(startArr, 3));
        g.setAttribute('aMorphDelay', new THREE.BufferAttribute(delayArr, 1));
        parts.push({ g, delayArr, posArr, n, origMat: m.material, mesh: null });
      });

      if (!parts.length) { finalize(rec, holder, label, base.bbox.center, 0.9, false); return resolve(); }
      const H = Math.max(yMax - yMin, 1e-4);

      for (const p of parts) {
        for (let i = 0; i < p.n; i++) p.delayArr[i] = 0.25 * ((p.posArr[i * 3 + 1] - yMin) / H);
        p.g.attributes.aMorphDelay.needsUpdate = true;

        // GPU Morph Proxy Material (Zero CPU vertex loops per frame!)
        const gpuMat = new THREE.MeshStandardMaterial({
          color: (p.origMat && !Array.isArray(p.origMat) && p.origMat.color) ? p.origMat.color.clone() : new THREE.Color(0xb0c4d8),
          map: (p.origMat && !Array.isArray(p.origMat)) ? p.origMat.map : null,
          roughness: 0.45, metalness: 0.25
        });
        gpuMat.customProgramCacheKey = () => 'nxr_gpu_morph_v5';
        gpuMat.onBeforeCompile = (shader) => {
          shader.uniforms.uMorphT = morphTime;
          shader.vertexShader = shader.vertexShader
            .replace('#include <common>', '#include <common>\nattribute vec3 aMorphStart;\nattribute float aMorphDelay;\nuniform float uMorphT;\nvarying float vMorphE;')
            .replace('#include <begin_vertex>', '#include <begin_vertex>\nfloat mE = clamp((uMorphT - aMorphDelay) / 0.72, 0.0, 1.0);\nmE = 1.0 - pow(1.0 - mE, 3.0);\ntransformed = mix(aMorphStart, transformed, mE);\nvMorphE = mE;');
          shader.fragmentShader = shader.fragmentShader
            .replace('#include <common>', '#include <common>\nvarying float vMorphE;')
            .replace('#include <emissivemap_fragment>', '#include <emissivemap_fragment>\ntotalEmissiveRadiance += vec3(0.0, 0.95, 0.74) * (1.0 - vMorphE) * 1.4;');
        };
        const mesh = new THREE.Mesh(p.g, gpuMat);
        mesh.frustumCulled = false;
        p.mesh = mesh; p.gpuMat = gpuMat;
        morphRoot.add(mesh);
      }

      APP.scene.add(morphRoot);
      if (rec.magicVFX) rec.magicVFX.emitBurst(morphOrigin.x, morphOrigin.y, morphOrigin.z, 20, 26);

      const TOT = 1.05;
      let t = 0, settled = false, unsub = null;

      const stop = () => { if (settled) return false; settled = true; try { unsub && unsub(); } catch (_) {} unsub = null; return true; };
      const cancel = () => { if (stop()) { APP.scene.remove(morphRoot); parts.forEach((p) => { p.g.dispose(); p.gpuMat.dispose(); }); } resolve(); };
      const done = () => {
        if (!stop()) return;
        parts.forEach((p) => {
          p.mesh.material = p.origMat;
          p.gpuMat.dispose();
        });
        APP.scene.remove(morphRoot);
        // Reset morphRoot local transform BEFORE finalize() so position/rotation are NOT doubled!
        morphRoot.position.set(0, 0, 0);
        morphRoot.quaternion.identity();
        if (alive()) finalize(rec, morphRoot, label, base.bbox.center, 0.1, true);
        resolve();
      };

      unsub = APP.onUpdate((dt) => {
        if (settled) return;
        if (!alive()) return cancel();
        const step = Math.min(dt, 0.05);
        t += step;
        const k = clamp(t / TOT, 0, 1);
        morphTime.value = t;
        if (rec.holoMat) { rec.holoMat.uniforms.uDissolve.value = clamp(k * 1.4, 0, 1); rec.holoMat.uniforms.uTime.value = performance.now() / 1000; }
        rec.wireMat.opacity = 0.65 * (1 - k);
        const worldSweepY = morphOrigin.y + (yMin - 0.04) + (H + 0.08) * k;
        if (rec.scanRing) { rec.scanRing.position.y = worldSweepY; rec.scanRing.rotation.z = t * 4.0; }
        if (rec.magicVFX) { if (k < 0.88) rec.magicVFX.emitBurst(morphOrigin.x, worldSweepY, morphOrigin.z, 2, 4); rec.magicVFX.step(step, performance.now() / 1000); }
        if (t >= TOT) done();
      });
    });
  }

  // ------------------------------------------------------------------ Manifest Pipeline (AI-First)
  async function manifest() {
    const strokes = S.strokes.slice(), group = S.rawGroup;
    let total = 0; strokes.forEach((s) => { total += strokeLen(s); });
    if (!strokes.length || total < MIN_LENGTH) { discardStrokes(); S.state = 0; if (strokes.length) toast('Drawing too small — cancelled'); return; }
    S.state = 2;
    const hint = S.nameMode === 'said' ? S.hint : null;
    S.hint = null; S.dirty = true;
    const fwd = S.viewFwd || flatFwd(), yawQ = yawQuatFor(fwd), inv = yawQ.clone().invert(), tmp = new V3();
    const rec = { id: ++S.jobId, strokes, group, temps: [], wired: [], yawQ, viewFwd: fwd, cancelled: false, mode: S.mode, root: null, entity: null, hinted: !!hint };
    rec.payload = strokes.map((s) => { const p = new Float32Array(s.pts.length * 3), r = new Float32Array(s.pts.length); s.pts.forEach((v, i) => { tmp.copy(v).applyQuaternion(inv); p[i * 3] = tmp.x; p[i * 3 + 1] = tmp.y; p[i * 3 + 2] = tmp.z; r[i] = s.radii[i]; }); return { p, r }; });
    let rs = 0, rn = 0; strokes.forEach((s) => s.radii.forEach((r) => { rs += r; rn++; })); rec.avgR = rs / Math.max(rn, 1);
    S.undo.push(rec); if (S.undo.length > 10) { const old = S.undo.shift(); if (old.group && old.group !== S.rawGroup) disposeDeep(old.group); }
    S.active = rec; S.busy = true; S.strokes = []; S.viewFwd = null; newRawGroup(); group.userData.locked = true;
    const alive = () => S.active === rec && !rec.cancelled;

    try {
      const wb = new THREE.Box3().setFromObject(group); makeHolo(rec, wb); strokes.forEach((s) => s.mesh && holoize(rec, s.mesh));
      const ctrW = wb.getCenter(new V3());
      if (rec.magicVFX) rec.magicVFX.emitBurst(ctrW.x, ctrW.y, ctrW.z, 16, 22);

      setStatus(hint ? `Preparing "${hint.label}"…` : 'Analyzing 3D silhouette…', 'busy');
      const base = await runWorker({ strokes: rec.payload, res: 38, smoothK: rec.avgR, iter: 3 });
      if (!alive()) return;
      const bGeo = geoFromOut(base), hm = new THREE.Mesh(bGeo, rec.holoMat); hm.frustumCulled = false; hm.userData.own = true;
      hm.quaternion.copy(yawQ); hm.position.set(base.bbox.center[0], base.bbox.center[1], base.bbox.center[2]).applyQuaternion(yawQ); APP.scene.add(hm); rec.temps.push(hm);
      const wm = new THREE.Mesh(bGeo, rec.wireMat); wm.frustumCulled = false; hm.add(wm); rec.temps.push(wm);
      group.visible = false; hm.updateMatrixWorld(true); holoBoundsUpdate(rec, hm);

      const snap = render4Views(base);
      let spec;
      if (hint) {
        spec = specFromHint(hint, base);
      } else {
        setStatus('AI recognizing drawing…', 'busy');
        spec = await stage1Recognize(snap, base, strokes);
      }
      if (!alive()) return;
      toast(`Recognized: "${spec.label}" — AI modeling...`, 2200);

      if (S.mode === 'B') {
        setStatus(`Matching GLB for "${spec.label}"…`, 'busy');
        const hit = await findGLB(spec.keywords.concat([spec.label]));
        if (!alive()) return;
        if (hit) {
          try {
            const model = await loadGLB(hit.url);
            if (!alive()) { disposeDeep(model); return; }
            toast('Morphing into ' + hit.name, 2000);
            await morphInto(rec, fitGLB(model, spec, base), base, spec.label, alive);
            if (alive() && rec.root) return;
          } catch (_) {}
        }
      }

      const builtObj = await stage2BuildRealistic3D(spec, base, snap, S.mode, alive);
      if (!alive()) return;
      if (builtObj) {
        finalize(rec, fitCentered(builtObj, base.bbox.size, 0.80, 1.24), spec.label, base.bbox.center, 1.10, false);
      } else {
        throw new Error('AI returned empty 3D model');
      }
    } catch (e) {
      if (!rec.cancelled) {
        console.error('[inst4] manifest failed', e);
        toast('AI Sculpt Error: ' + (e && e.message || e), 5000);
        undoLast();
      }
    } finally {
      if (S.active === rec && rec.root) { S.busy = false; S.active = null; setStatus('Done: ' + (rec.label || 'Model'), 'ok'); }
      if (S.active === rec && !rec.root) { S.busy = false; S.active = null; }
      S.state = 0; S.dirty = true;
    }
  }

  // ------------------------------------------------------------------ Non-Destructive Undo & Clean Clear
  function removeManifestedRecord(rec) {
    rec.cancelled = true;
    if (S.active === rec) { S.active = null; S.busy = false; }
    S.reveals = S.reveals.filter((r) => r.rec !== rec);
    cleanHolo(rec);
    if (rec.entity) {
      if (handGrabs.L.ent === rec.entity) releaseProceduralHandGrab('L');
      if (handGrabs.R.ent === rec.entity) releaseProceduralHandGrab('R');
      const rbIdx = rigidBodies.indexOf(rec.entity);
      if (rbIdx !== -1) rigidBodies.splice(rbIdx, 1);
      try { APP.removeEntity(rec.entity, false); }
      catch (_) { if (rec.root?.parent) rec.root.parent.remove(rec.root); disposeDeep(rec.root); }
    } else if (rec.root) {
      if (rec.root.parent) rec.root.parent.remove(rec.root);
      disposeDeep(rec.root);
    }
  }

  function undoLast() {
    endAllStrokes();
    const rec = S.undo.pop(); if (!rec) { toast('Nothing to undo'); return; }
    removeManifestedRecord(rec);

    const restored = rec.group, cur = S.strokes;
    cur.forEach((s) => { if (s.mesh) restored.add(s.mesh); });
    if (S.rawGroup && S.rawGroup !== restored) {
      if (S.rawGroup.parent) S.rawGroup.parent.remove(S.rawGroup);
      disposeDeep(S.rawGroup);
    }
    restored.visible = true; restored.userData.locked = false;
    if (!restored.parent) APP.scene.add(restored);
    rec.strokes.forEach((s) => { if (s.mesh && s.mat) s.mesh.material = s.mat; });
    S.rawGroup = restored; S.strokes = rec.strokes.concat(cur); S.viewFwd = rec.viewFwd;
    S.busy = false; S.active = null; S.state = 0; S.dirty = true;
    setStatus('Raw strokes restored', 'ok');
  }

  function clearAll() {
    S.jobId++;
    pend.forEach((p) => p.rej(new Error('cleared')));
    pend.clear();
    endAllStrokes(); S.cur.clear();
    if (S.active) { S.active.cancelled = true; S.active = null; }
    S.busy = false; S.state = 0; S.keyG = false; S.mouse.down = false;
    releaseProceduralHandGrab('L');
    releaseProceduralHandGrab('R');
while (S.undo.length) {
      const rec = S.undo.pop();
      removeManifestedRecord(rec);
      if (rec.group) {
        if (rec.group.parent) rec.group.parent.remove(rec.group);
        disposeDeep(rec.group);
      }
    }
    S.reveals = [];

    discardStrokes();
    if (S.rawGroup) {
      if (S.rawGroup.parent) S.rawGroup.parent.remove(S.rawGroup);
      disposeDeep(S.rawGroup);
    }
    newRawGroup();
    S.hint = null;
    S.dirty = true;
    setStatus('Cleared — ready to draw!', 'ok');
    toast('Cleared all drawings & models');
  }

  function interceptSculptVoiceText(rawText) {
    if (!S.sculptOn) return false;
    const t = String(rawText || '').trim();
    if (!t || SCULPT_CMD_IGNORE_RE.test(t)) return false;

    const explicitDraw = SCULPT_INTENT_RE.test(t);
    const shortNounWhileActive = t.split(/\s+/).length <= 5;
    if (!explicitDraw && !shortNounWhileActive) return false;

    const h = setHint(t, true);
    if (h) {
      try {
        const msg = /[\u0600-\u06FF]/.test(t) ? 'تمام، ارسم الآن' : ('Got it, draw your ' + h.label + ' now.');
        APP.speakTTS?.(msg);
      } catch (_) {}
      return true;
    }
    return false;
  }

  // ------------------------------------------------------------------ Input Helpers
  function vecOf(o) {
    if (!o) return null; if (o.isVector3) return o; if (o.position && o.position.isVector3) return o.position;
    if (o.getWorldPosition) { const v = new V3(); o.getWorldPosition(v); return v; } if (typeof o.x === 'number') return new V3(o.x, o.y, o.z); return null;
  }
  function readHand(h, side) {
    if (!h || !h.active) return null;
    const b = h.gamepad && h.gamepad.buttons;
    const isHand = !!h.isHand;
    const pinch = !!h.pinching;
    const grip = isHand ? (side === 'L' ? pinch : false) : (!!(b && b[1] && b[1].pressed) || !!h.gripping);
    const trig = isHand ? (side === 'R' ? pinch : false) : (!!(b && b[0] && b[0].pressed) || !!h.trigger);
    const val = (b && b[0]) ? (b[0].value != null ? b[0].value : (b[0].pressed ? 1 : 0)) : 1;
    let tip = isHand ? vecOf(h.fingerTips && h.fingerTips.index) : null;
    if (!tip && h.rayOrigin && h.rayDir) tip = new V3().copy(h.rayOrigin).addScaledVector(h.rayDir, 0.04);
    if (!tip) tip = vecOf(h.pos);
    return { h, side, isHand, grip, trig, rawTrig: !!(b && b[0]?.pressed) || pinch, val, tip: tip ? tip.clone() : null };
  }
  function mouseWorld() {
    const el = APP.renderer.domElement, r = el.getBoundingClientRect(), cam = APP.camera, o = cam.getWorldPosition(new V3());
    const p = new V3(((S.mouse.x - r.left) / r.width) * 2 - 1, -((S.mouse.y - r.top) / r.height) * 2 + 1, 0.5).unproject(cam);
    return o.addScaledVector(p.sub(o).normalize(), S.mouse.depth);
  }

  // ------------------------------------------------------------------ Two-Stage Sculpt State Machine
  function tickSculpt() {
    if (!S.rawGroup || !S.rawGroup.parent) newRawGroup();

      const k = clamp(rv.t / rv.dur, 0, 1), e = k * k * (3 - 2 * k), y = rv.y0 + (rv.y1 - rv.y0) * e;
      rv.planeBelow.constant = y;
      if (rv.rec.holoMat) { rv.rec.holoMat.uniforms.uReveal.value = y; rv.rec.holoMat.uniforms.uDissolve.value = Math.max(0, (k - 0.35) * 1.5); rv.rec.planeAbove.constant = -y; }
      if (rv.rec.scanRing) { rv.rec.scanRing.position.y = y; rv.rec.scanRing.rotation.z = tm * 3.0; }
      if (rv.rec.magicVFX) {
        if (k < 0.88) rv.rec.magicVFX.emitBurst(rv.ctr.x, y, rv.ctr.z, 3, 6);
        rv.rec.magicVFX.step(safeDt, tm);
      }
      if (k >= 1) {
        rv.root.traverse((o) => { if (o.material) (Array.isArray(o.material) ? o.material : [o.material]).forEach((m) => { m.clippingPlanes = null; m.needsUpdate = true; }); });
        cleanHolo(rv.rec); S.reveals.splice(i, 1);
        if (S.undo.includes(rv.rec)) toast('✓ ' + (rv.rec.label || 'Model') + ' manifested', 2400);
      }
    }
    try { tickSculpt(); } catch (e) { console.warn('[inst4] sculpt tick', e); }
    try { tickProceduralHandGrab(safeDt); } catch (e) { console.warn('[inst4] hand grab', e); }
    try { stepRigidBodies(safeDt); } catch (e) { console.warn('[inst4] physics', e); }
    try { tickPanel(); } catch (e) { console.warn('[inst4] panel tick', e); }
    const now = performance.now();
    if (S.dirty && now - S.lastDraw > 75) { S.lastDraw = now; S.dirty = false; drawPanel(); refreshHud(); }
    hud.style.display = (APP.xrMode === 'none') ? 'block' : 'none';
  });

  APP._inst4 = { toggle: toggleSculpt, toggleGravity, undo: undoLast, clear: clearAll, setMode, cycleMode, setHint, setNameMode, setBrushRadius, setColorIndex, listenForSculptName, state: S };
  APP._inst4Cleanup = () => {

    try { unsubMods && unsubMods(); } catch (_) {}
    if (origExecuteCommand) APP.executeCommand = origExecuteCommand;

    releaseProceduralHandGrab('R');

    try { if (worker) worker.terminate(); } catch (_) {} pend.forEach((p) => p.rej(new Error('disposed'))); pend.clear(); worker = null;
    S.reveals.forEach((rv) => {

    });
    S.reveals = [];
    endAllStrokes(); S.cursors.forEach((c) => APP.scene.remove(c)); S.cursors.clear();
    S.undo.forEach((r) => { cleanHolo(r); if (r.group) disposeDeep(r.group); if (r.root) disposeDeep(r.root); });
    if (S.rawGroup) disposeDeep(S.rawGroup);
    if (workFloor) { if (workFloor.parent) workFloor.parent.remove(workFloor); disposeDeep(workFloor); workFloor = null; }
    pbrCanvasCache.clear();
    try { [P.screen, P.bar, P.orb].forEach((m) => APP.pokeTargets && APP.pokeTargets.delete(m)); } catch (_) {}
    try { APP.scene.remove(P.root); P.tex.dispose(); disposeDeep(P.root); } catch (_) {}
    try { hud.remove(); } catch (_) {}
    disposers.forEach((fn) => { try { fn(); } catch (_) {} });
    try { APP.renderer.localClippingEnabled = !!S.prevLocal; } catch (_) {}
    if (APP._inst4 && APP._inst4.state === S) APP._inst4 = null;
  };
  toast('Sculpt v5 ready — AI Builder + RigidBody + Realistic Hand Grab', 3200);
})();
