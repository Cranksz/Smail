  function isManipulable(e) {
    if (!e || !e.group || !e.group.parent) return false;
    if (e.isPen || e.isStroke || e.userData?.isManipTool) return false;
    if (['world_fx', 'board', 'portal', 'tool', 'tool_pen', 'smoke_holo'].includes(e.category)) return false;
    const n = e.group.name || '';

    return true;

  function resolveTarget(arg) {
    if (arg && typeof arg === 'object' && arg.group) return arg;
    const q = String(arg || '').trim();
    if (q && q !== 'latest') {
      const found = APP.findEntity?.(q);
      if (isManipulable(found)) return found;
    }
    if (APP.lastEntityId) {
      const last = (APP.entities || []).find((e) => e && e.id === APP.lastEntityId);
      if (isManipulable(last)) return last;
    }
    const cands = (APP.entities || []).filter(isManipulable);
    return cands[cands.length - 1] || null;
  }

  function meshesOf(entity) {
    const out = [];
    entity.group.traverse((c) => {
      if (c.isMesh && c.geometry?.attributes?.position && !c.userData?.nxrDecal) out.push(c);
    });
    return out;
  }

  function firstMeshColor(ent) {
    for (const m of meshesOf(ent)) {
      const mat = Array.isArray(m.material) ? m.material[0] : m.material;
      if (mat?.color) return mat.color.clone();
    }
    return new THREE.Color(0x9aa5b1);

  // Give a mesh its own material copies so color/emissive changes don't leak to other objects.
  function ownMaterials(mesh) {
    if (!mesh.userData.__ownMat) {
      const cl = (m) => (m && m.clone ? m.clone() : m);
      mesh.material = Array.isArray(mesh.material) ? mesh.material.map(cl) : cl(mesh.material);
      mesh.userData.__ownMat = true;
    }
    return Array.isArray(mesh.material) ? mesh.material : [mesh.material];
  }

  function worldTriangles(entity, maxTris) {
    entity.group.updateMatrixWorld(true);
    const tris = [], box = new THREE.Box3();
    for (const m of meshesOf(entity)) {
      const p = m.geometry.attributes.position, idx = m.geometry.index;
      const tc = idx ? idx.count / 3 : p.count / 3;
      const stride = maxTris && tc > maxTris ? Math.ceil(tc / maxTris) : 1;
      for (let t = 0; t < tc; t += stride) {
        const ia = idx ? idx.getX(t * 3) : t * 3;
        const ib = idx ? idx.getX(t * 3 + 1) : t * 3 + 1;
        const ic = idx ? idx.getX(t * 3 + 2) : t * 3 + 2;
        const a = V().fromBufferAttribute(p, ia).applyMatrix4(m.matrixWorld);
        const b = V().fromBufferAttribute(p, ib).applyMatrix4(m.matrixWorld);
        const c = V().fromBufferAttribute(p, ic).applyMatrix4(m.matrixWorld);
        tris.push([a, b, c]);
        box.expandByPoint(a); box.expandByPoint(b); box.expandByPoint(c);
      }
    }
    return { tris, box };
  }

  function volumeOf(b) {
    const s = b.getSize(V());
    return s.x * s.y * s.z;
  }

  function boxGap(a, b) {
    const dx = Math.max(0, a.min.x - b.max.x, b.min.x - a.max.x);
    const dy = Math.max(0, a.min.y - b.max.y, b.min.y - a.max.y);
    const dz = Math.max(0, a.min.z - b.max.z, b.min.z - a.max.z);
    return Math.hypot(dx, dy, dz);
  }

  function touchedEntity(p) {
    let best = null, bestVol = Infinity;
    for (const e of APP.entities || []) {
      if (!isManipulable(e)) continue;
      const b = new THREE.Box3().setFromObject(e.group).expandByScalar(0.02);
      if (!b.containsPoint(p)) continue;
      const v = volumeOf(b);
      if (v < bestVol) { bestVol = v; best = e; }
    }
    return best;
  }

  function setRay(e) {
    const el = APP.renderer?.domElement;
    const r = el ? el.getBoundingClientRect() : { left: 0, top: 0, width: window.innerWidth, height: window.innerHeight };
    ndc.x = ((e.clientX - r.left) / Math.max(1, r.width)) * 2 - 1;
    ndc.y = -((e.clientY - r.top) / Math.max(1, r.height)) * 2 + 1;
    ray.setFromCamera(ndc, APP.camera);
  }

  function pick(e) {
    if (!APP.camera) return null;
    setRay(e);
    let best = null;
    for (const ent of APP.entities || []) {
      if (!isManipulable(ent)) continue;
      const hit = ray.intersectObject(ent.group, true).find((h) => h.object.isMesh);
      if (hit && (!best || hit.distance < best.distance)) best = { point: hit.point.clone(), ent, distance: hit.distance };
    }
    return best;
  }

  function buildGeo(tris, origin) {
    const arr = new Float32Array(tris.length * 9);
    let k = 0;
    for (const t of tris) for (const v of t) {
      arr[k++] = v.x - origin.x; arr[k++] = v.y - origin.y; arr[k++] = v.z - origin.z;
    }
    const g = new THREE.BufferGeometry();
    g.setAttribute('position', new THREE.BufferAttribute(arr, 3));
    g.computeVertexNormals();
    return g;
  }

  // Keeps the half-space where dot(x - p, n) <= 0 and returns cut segments on the plane.
  function clipTri(tri, p, n) {
    const d = [0, 1, 2].map((i) => (tri[i].x - p.x) * n.x + (tri[i].y - p.y) * n.y + (tri[i].z - p.z) * n.z);
    const inside = [], outside = [];
    for (let i = 0; i < 3; i++) (d[i] <= 1e-6 ? inside : outside).push(i);
    if (!outside.length) return { tris: [tri], cut: null };
    if (!inside.length) return { tris: [], cut: null };
    const at = (a, b) => {
      const den = d[a] - d[b];
      const t = Math.abs(den) > 1e-9 ? clamp(d[a] / den, 0, 1) : 0.5;
      return V().lerpVectors(tri[a], tri[b], t);
    };
    if (inside.length === 1) {
      const i0 = inside[0], i1 = (i0 + 1) % 3, i2 = (i0 + 2) % 3;
      const pA = at(i0, i1), pB = at(i0, i2);
      return { tris: [[tri[i0].clone(), pA, pB]], cut: [pA, pB] };
    }
    const o = outside[0], i1 = (o + 1) % 3, i2 = (o + 2) % 3;
    const pA = at(i1, o), pB = at(i2, o);
    return {
      tris: [[tri[i1].clone(), tri[i2].clone(), pB], [tri[i1].clone(), pB.clone(), pA]],
      cut: [pA, pB]
    };
  }

      const dust = i < 70;
      kind[i] = dust ? 1 : 0;
      pos[i * 3] = origin.x + (Math.random() - 0.5) * scale * 0.4;
      pos[i * 3 + 1] = Math.max(FLOOR, origin.y + (Math.random() - 0.5) * scale * 0.4);
      pos[i * 3 + 2] = origin.z + (Math.random() - 0.5) * scale * 0.4;
      const dir = randDir();
      vel.push(dust
        ? dir.multiplyScalar(0.18 + Math.random() * 0.25).add(V(0, 0.12, 0))
        : dir.multiplyScalar(0.8 + Math.random() * 1.4).add(V(0, 0.5, 0)));
      maxL[i] = life[i] = dust ? 1.2 + Math.random() * 0.9 : 0.7 + Math.random() * 0.6;
      size[i] = dust ? 36 + Math.random() * 40 : 5 + Math.random() * 7;
    }
    const geo = new THREE.BufferGeometry();

    geo.setAttribute('aLife', new THREE.BufferAttribute(life, 1));
    geo.setAttribute('aMax', new THREE.BufferAttribute(maxL, 1));

      transparent: true, depthWrite: false,

        uPx: { value: window.devicePixelRatio || 1 },

      fragmentShader: [

        '  if(vK>0.5){ gl_FragColor=vec4(uDust,(1.0-r*r)*vT*0.32); }',
        '  else { gl_FragColor=vec4(uChip,(1.0-smoothstep(0.7,1.0,r))*vT); } }'

  function fractureEntity(targetArg, hitPt) {
    const target = resolveTarget(targetArg);
    if (!target) return { ok: false, error: 'No target object found to break.' };
    const { tris, box } = worldTriangles(target, 9000);
    if (tris.length < 4) return { ok: false, error: 'Not enough geometry to break.' };
    const size = box.getSize(V()), center = box.getCenter(V());
    const maxDim = Math.max(size.x, size.y, size.z, 0.02);
    const impact = hitPt ? V().copy(hitPt) : center.clone();

    // Irregular seeds: a dense cluster around the hit (small chips) plus sparse outer seeds (big pieces).
    const want = maxDim < 0.2 ? 5 + Math.floor(Math.random() * 3) : 8 + Math.floor(Math.random() * 5);
    const seeds = [];
    for (let a = 0; a < 600 && seeds.length < want; a++) {
      const near = Math.random() < 0.55;
      const p = near
        ? impact.clone().addScaledVector(randDir(), Math.pow(Math.random(), 0.8) * maxDim * 0.45)
        : V(center.x + (Math.random() - 0.5) * size.x * 1.1,
            center.y + (Math.random() - 0.5) * size.y * 1.1,
            center.z + (Math.random() - 0.5) * size.z * 1.1);
      p.clamp(box.min, box.max);
      const minSep = maxDim * (near ? 0.12 + Math.random() * 0.1 : 0.22 + Math.random() * 0.2);
      if (seeds.every((s) => s.distanceTo(p) > minSep)) seeds.push(p);
    }
    if (seeds.length < 2) return { ok: false, error: 'Could not plan the fracture.' };

    const srcMesh = meshesOf(target)[0];
    const srcMat = srcMesh ? (Array.isArray(srcMesh.material) ? srcMesh.material[0] : srcMesh.material) : null;
    const baseCol = srcMat?.color ? srcMat.color.clone() : new THREE.Color(0x9aa5b1);
    const outerMat = srcMat ? srcMat.clone() : new THREE.MeshStandardMaterial({ color: baseCol, roughness: 0.5 });
    outerMat.side = THREE.DoubleSide;
    outerMat.map = null; // shards carry no UVs, so drop the texture and keep the color
    const capMat = new THREE.MeshStandardMaterial({
      color: baseCol.clone().multiplyScalar(0.45), roughness: 0.95, metalness: 0, side: THREE.DoubleSide
    });

    const shards = [];
    for (let i = 0; i < seeds.length; i++) {
      const s = seeds[i];
      let outer = tris, inner = [];
      for (let j = 0; j < seeds.length && (outer.length || inner.length); j++) {
        if (j === i) continue;
        const o = seeds[j];
        const n = V().subVectors(o, s).normalize();
        const p = V().addVectors(s, o).multiplyScalar(0.5);
        const nO = [], nI = [], cut = [];
        for (const t of outer) { const r = clipTri(t, p, n); for (const x of r.tris) nO.push(x); if (r.cut) cut.push(r.cut); }
        for (const t of inner) { const r = clipTri(t, p, n); for (const x of r.tris) nI.push(x); if (r.cut) cut.push(r.cut); }
        outer = nO; inner = nI;
        // Close the cell with cap triangles on this bisector plane (they get clipped by later planes too).
        if (cut.length >= 2) {
          const capC = V();
          for (const seg of cut) capC.add(seg[0]).add(seg[1]);
          capC.divideScalar(cut.length * 2);
          for (const seg of cut) {
            const a = seg[0], b = seg[1];
            if (a.distanceToSquared(b) < 1e-10) continue;
            const nrm = V().crossVectors(V().subVectors(b, a), V().subVectors(capC, a));
            if (nrm.dot(n) >= 0) inner.push([capC.clone(), a.clone(), b.clone()]);
            else inner.push([capC.clone(), b.clone(), a.clone()]);
          }
        }
      }
      if (outer.length < 2) continue;

      const c = V();
      let cnt = 0;
      for (const t of outer) for (const v of t) { c.add(v); cnt++; }
      c.divideScalar(cnt);

      const grp = new THREE.Group();
      grp.name = `${target.name}_Shard_${i}`;
      grp.position.copy(c);
      const om = new THREE.Mesh(buildGeo(outer, c), outerMat);
      om.castShadow = true; om.receiveShadow = true;
      grp.add(om);
      if (inner.length) {
        const im = new THREE.Mesh(buildGeo(inner, c), capMat);
        im.castShadow = true;
        grp.add(im);
      }
      APP.scene.add(grp);

      const out = V().subVectors(c, impact);
      if (out.lengthSq() < 1e-6) out.copy(randDir());
      out.normalize();
      const speed = 0.4 + Math.random() * 1.1;
      const velocity = out.multiplyScalar(speed).add(V(0, 0.3 + Math.random() * 0.5, 0));
      const spin = V((Math.random() - 0.5) * 7, (Math.random() - 0.5) * 7, (Math.random() - 0.5) * 7);
      let ent = null;
      ent = APP.registerEntity(grp, {
        name: grp.name, category: 'shard', grabbable: true, holdable: true, hasGravity: true,
        restitution: 0.28, velocity,
        physics: { gravity: true, enabled: true, bounce: 0.28 },
        onUpdate: (dt) => {
          if (ent?.isHeld || ent?.heldBy) return;
          const d = Math.min(0.04, dt || 0.016);
          if (grp.position.y > 0.04) {
            grp.rotation.x += spin.x * d; grp.rotation.y += spin.y * d; grp.rotation.z += spin.z * d;
          } else spin.multiplyScalar(0.85);
        }
      });
      shards.push(ent);
    }
    if (!shards.length) return { ok: false, error: 'Fracture produced no pieces.' };

    spawnBreakFX(impact, maxDim, baseCol);
    APP.removeEntity?.(target, false);
    APP.lastEntityId = shards[shards.length - 1].id;
    return { ok: true, shards: shards.length, message: `Broke into ${shards.length} pieces!` };
  }

  function spawnGranules(worldPt, col, count) {
    const geo = new THREE.DodecahedronGeometry(0.012, 0);
    const mat = new THREE.MeshStandardMaterial({ color: col.clone(), roughness: 0.85 });
    const inst = new THREE.InstancedMesh(geo, mat, count);
    const grp = new THREE.Group();
    grp.name = 'Tear_Granules';
    grp.add(inst);
    APP.scene.add(grp);
    const p = [], v = [], s = [];
    for (let i = 0; i < count; i++) {
      p.push(worldPt.clone().add(randDir().multiplyScalar(0.02)));
      v.push(randDir().multiplyScalar(0.8 + Math.random() * 1.2).add(V(0, 0.6, 0)));
      s.push(0.6 + Math.random());
    }
    const dummy = new THREE.Object3D();
    let age = 0, done = false;
    const ent = APP.registerEntity(grp, {
      name: 'Tear Granules', category: 'world_fx', grabbable: false, physics: { enabled: false },
      onUpdate: (dt) => {
        if (done) return;
        const d = Math.min(0.04, Math.max(0.004, dt || 0.016));
        age += d;
        for (let i = 0; i < count; i++) {
          v[i].y -= 9.8 * d;
          p[i].addScaledVector(v[i], d);
          if (p[i].y < FLOOR) { p[i].y = FLOOR; v[i].y *= -0.3; v[i].x *= 0.6; v[i].z *= 0.6; }
          dummy.position.copy(p[i]);
          dummy.rotation.set(age * 3 + i, i * 0.7, 0);
          dummy.scale.setScalar(s[i]);
          dummy.updateMatrix();
          inst.setMatrixAt(i, dummy.matrix);
        }
        inst.instanceMatrix.needsUpdate = true;
        if (age > 6) {
          done = true;
          geo.dispose(); mat.dispose();
          APP.removeEntity?.(ent, false);
        }
      }
    });
  }

  function deformState(mesh) {
    let st = deformStates.get(mesh);
    if (st) return st;
    const p = mesh.geometry.attributes.position, n = p.count;
    const rest = new Float32Array(n * 3), seed = new Float32Array(n);
    for (let i = 0; i < n; i++) {
      rest[i * 3] = p.getX(i); rest[i * 3 + 1] = p.getY(i); rest[i * 3 + 2] = p.getZ(i);
      seed[i] = Math.random() * 6.28;
    }
    const mats = ownMaterials(mesh);
    st = {
      mesh, n, rest, seed,
      disp: new Float32Array(n * 3), vel: new Float32Array(n * 3), heat: new Float32Array(n),
      mats, origRough: mats[0]?.roughness ?? 0.5,
      dirty: false, tick: 0, calm: 0, timer: 0, lastSrc: null, inSet: false
    };
    deformStates.set(mesh, st);
    return st;
  }

  function flushDeform() {
    for (const st of dirtyStates) {
      const mesh = st.mesh;
      dirtyStates.delete(st);
      if (!mesh.parent) continue;
      const p = mesh.geometry.attributes.position;
      for (let i = 0; i < st.n; i++) {
        const i3 = i * 3;
        p.setXYZ(i, st.rest[i3] + st.disp[i3], st.rest[i3 + 1] + st.disp[i3 + 1], st.rest[i3 + 2] + st.disp[i3 + 2]);
      }
      p.needsUpdate = true;
      st.tick++;
      if (st.tick % 2 === 0) mesh.geometry.computeVertexNormals();
      mesh.geometry.computeBoundingSphere();
      mesh.geometry.computeBoundingBox();
      st.dirty = false;
    }
  }

  function spawnDrop(p, col) {
    if (drops.length > 45) return;
    const c = new THREE.Color(0xff6a1a);
    if (col && col.isColor) c.lerp(col, 0.35);
    const m = new THREE.Mesh(
      new THREE.SphereGeometry(0.007 + Math.random() * 0.006, 10, 8),
      new THREE.MeshStandardMaterial({ color: c, emissive: 0xff4a10, emissiveIntensity: 0.8, roughness: 0.12, transparent: true })
    );
    m.position.copy(p);
    APP.scene.add(m);
    drops.push({ m, vy: -0.05, splat: false, life: 2.5 });
  }

  function stepDrops(dt) {
    for (let i = drops.length - 1; i >= 0; i--) {
      const d = drops[i];
      if (!d.splat) {
        d.vy -= 9.8 * dt;
        d.m.position.y += d.vy * dt;
        if (d.m.position.y <= FLOOR) {
          d.m.position.y = FLOOR;
          d.m.scale.set(2.6, 0.2, 2.6);
          d.splat = true;
        }
      }
      d.life -= dt;
      if (d.splat) d.m.material.opacity = clamp(d.life / 0.8, 0, 1);
      if (d.life <= 0) {
        APP.scene.remove(d.m);
        d.m.geometry.dispose();
        d.m.material.dispose();
        drops.splice(i, 1);
      }
    }
  }

  function stepMeltMesh(st, srcsWorld, dt) {
    const mesh = st.mesh, M = mesh.matrixWorld;
    const invM = M4().copy(M).invert();
    const down = V(0, -1, 0).transformDirection(invM);            // gravity in mesh-local space
    const sc = V().setFromMatrixScale(M);
    const sY = Math.max(1e-4, sc.y), sAvg = Math.max(1e-4, (sc.x + sc.y + sc.z) / 3);
    const srcs = srcsWorld.map((s) => V().copy(s).applyMatrix4(invM));
    const r = CFG.meltRadius / sAvg;
    if (srcs.length) st.lastSrc = srcs[0];
    const cen = st.lastSrc;
    const { rest, disp, vel, heat, seed, n } = st;
    let maxH = 0, energy = 0, lowest = null, lowestY = Infinity;
    const w = V();

    for (let i = 0; i < n; i++) {
      const i3 = i * 3;
      const lx = rest[i3] + disp[i3], ly = rest[i3 + 1] + disp[i3 + 1], lz = rest[i3 + 2] + disp[i3 + 2];

      for (const s of srcs) {
        const d = Math.hypot(lx - s.x, ly - s.y, lz - s.z);
        if (d < r) { const k = 1 - d / r; heat[i] = Math.min(1, heat[i] + k * k * dt * CFG.meltSpeed); }
      }

      const h = heat[i];
      if (h > 0.003) {
        // gravity pulls the molten part down
        vel[i3] += down.x * 3.0 * h * dt;
        vel[i3 + 1] += down.y * 3.0 * h * dt;
        vel[i3 + 2] += down.z * 3.0 * h * dt;
        // lateral flow away from the heat source
        if (cen) {
          const rx = lx - cen.x, ry = ly - cen.y, rz = lz - cen.z, rl = Math.hypot(rx, ry, rz) || 1;
          const f = 0.35 * h * dt;
          vel[i3] += (rx / rl) * f; vel[i3 + 1] += (ry / rl) * f * 0.4; vel[i3 + 2] += (rz / rl) * f;
        }
        // wobble gives the liquid distortion
        const wv = clockT * 5 + seed[i] + lx * 11;
        vel[i3] += Math.sin(wv) * 0.18 * h * dt;
        vel[i3 + 2] += Math.cos(wv * 1.13) * 0.18 * h * dt;
        // viscosity: hot parts flow more freely
        const damp = Math.exp(-(1.2 + 1.4 * (1 - h)) * dt);
        vel[i3] *= damp; vel[i3 + 1] *= damp; vel[i3 + 2] *= damp;
        disp[i3] += vel[i3] * dt; disp[i3 + 1] += vel[i3 + 1] * dt; disp[i3 + 2] += vel[i3 + 2] * dt;

        // floor: molten material pools and spreads instead of passing through
        w.set(rest[i3] + disp[i3], rest[i3 + 1] + disp[i3 + 1], rest[i3 + 2] + disp[i3 + 2]).applyMatrix4(M);
        if (w.y < FLOOR) {
          disp[i3 + 1] += (FLOOR - w.y) / sY;
          vel[i3 + 1] = 0; vel[i3] *= 0.85; vel[i3 + 2] *= 0.85;
        } else if (w.y < lowestY) {
          lowestY = w.y; lowest = w.clone();
        }
        const mag = Math.hypot(disp[i3], disp[i3 + 1], disp[i3 + 2]);
        if (mag > 0.35) { const k = 0.35 / mag; disp[i3] *= k; disp[i3 + 1] *= k; disp[i3 + 2] *= k; }
        if (h > maxH) maxH = h;
      }
      energy += Math.abs(vel[i3]) + Math.abs(vel[i3 + 1]) + Math.abs(vel[i3 + 2]);
    }

    if (maxH > 0.003) {
      st.dirty = true;
      dirtyStates.add(st);
      for (const mat of st.mats) {
        if (mat.emissive) { mat.emissive.setRGB(maxH * 0.85, maxH * 0.22, maxH * 0.03); mat.emissiveIntensity = 1; }
        if ('roughness' in mat) mat.roughness = lerp(st.origRough, 0.12, maxH);
      }
      st.timer += dt;
      if (lowest && lowestY > 0.1 && st.timer > 0.22) {
        st.timer = 0;
        spawnDrop(lowest, st.mats[0]?.color);
      }
    }
    st.calm = energy / Math.max(1, n) < 0.0015 ? st.calm + dt : 0;
  }

  function stepMeltAll(dt, srcs) {
    const box = new THREE.Box3();
    if (srcs.length) {
      for (const e of APP.entities || []) {
        if (!isManipulable(e) || e === meltTool) continue;
        e.group.updateMatrixWorld(true);
        for (const mesh of meshesOf(e)) {
          if (deformStates.get(mesh)?.inSet) continue;
          box.setFromObject(mesh).expandByScalar(CFG.meltRadius);
          if (srcs.some((s) => box.containsPoint(s))) {
            const st = deformState(mesh);
            st.inSet = true; st.calm = 0;
            meltSet.add(st);
          }
        }
      }
    }
    for (const st of [...meltSet]) {
      if (!st.mesh.parent) { st.inSet = false; meltSet.delete(st); continue; }
      st.mesh.updateWorldMatrix(true, false);
      box.setFromObject(st.mesh).expandByScalar(CFG.meltRadius);
      const near = srcs.filter((s) => box.containsPoint(s));
      stepMeltMesh(st, near, dt);
      if (st.calm > 3) { st.inSet = false; meltSet.delete(st); }
    }
  }

  function ensureMeltTool(on) {
    if (on) {
      if (!meltTool || !meltTool.group?.parent) {
        const g = new THREE.Group();
        g.name = 'Thermal_Melt_Sphere';
        const core = new THREE.Mesh(
          new THREE.SphereGeometry(0.06, 24, 18),
          new THREE.MeshStandardMaterial({ color: 0xff5a1a, emissive: 0xff3b00, emissiveIntensity: 2.0, roughness: 0.25 })

        const halo = new THREE.Mesh(

        );
        const light = new THREE.PointLight(0xff5a1a, 1.4, 1.6);
        g.add(core, halo, light);
        const sp = APP.getSpawnPosition ? APP.getSpawnPosition(0.6, true) : V(0, 1.2, -0.6);
        g.position.copy(sp);
        APP.scene.add(g);
        meltTool = APP.registerEntity(g, {

      }
      meltTool.group.visible = true;
    } else if (meltTool?.group) {
      meltTool.group.visible = false;
    }
  }

  // ------------------------------------------------------------------
  // 3. SOFTBODY (lattice-driven, pokeable, stretchable, tearable)
  // ------------------------------------------------------------------
  function makeSoftbody(targetArg) {
    const target = resolveTarget(targetArg);
    if (!target) return { ok: false, error: 'No target object found for Softbody.' };
    if (target.userData?.isSoftbody) return { ok: true, id: target.id, message: `${target.name} is already soft.` };
    const gW = target.group;
    gW.updateMatrixWorld(true);
    const meshes = meshesOf(target);
    if (!meshes.length) return { ok: false, error: 'Target has no deformable geometry.' };

    const invG0 = M4().copy(gW.matrixWorld).invert();
    const bmin = V(Infinity, Infinity, Infinity), bmax = V(-Infinity, -Infinity, -Infinity);
    const parts = meshes.map((m) => {
      const rel = M4().multiplyMatrices(invG0, m.matrixWorld);   // mesh-local -> group-local, constant
      const p = m.geometry.attributes.position, count = p.count;
      const localRest = new Float32Array(count * 3), rootRest = new Float32Array(count * 3);
      const tmp = V();
      for (let i = 0; i < count; i++) {
        localRest[i * 3] = p.getX(i); localRest[i * 3 + 1] = p.getY(i); localRest[i * 3 + 2] = p.getZ(i);
        tmp.set(localRest[i * 3], localRest[i * 3 + 1], localRest[i * 3 + 2]).applyMatrix4(rel);
        rootRest[i * 3] = tmp.x; rootRest[i * 3 + 1] = tmp.y; rootRest[i * 3 + 2] = tmp.z;
        bmin.min(tmp); bmax.max(tmp);
      }
      const src = m.geometry.index ? m.geometry.index.array : null;
      const triCount = src ? src.length / 3 : count / 3;
      const origIdx = new Uint32Array(triCount * 3);
      for (let k = 0; k < triCount * 3; k++) origIdx[k] = src ? src[k] : k;
      return {
        mesh: m, count, triCount, localRest, rootRest, origIdx,
        relInv3: M3().setFromMatrix4(M4().copy(rel).invert())
      };
    });

    const N = CFG.softRes, NN = N * N * N;
    const span = V().subVectors(bmax, bmin).max(V(0.05, 0.05, 0.05));
    const ni = (x, y, z) => (z * N + y) * N + x;
    const rest = new Float32Array(NN * 3), cur = new Float32Array(NN * 3), prev = new Float32Array(NN * 3);
    const torn = new Uint8Array(NN);
    for (let z = 0; z < N; z++) for (let y = 0; y < N; y++) for (let x = 0; x < N; x++) {
      const i = ni(x, y, z) * 3;
      rest[i] = bmin.x + (x / (N - 1)) * span.x;
      rest[i + 1] = bmin.y + (y / (N - 1)) * span.y;
      rest[i + 2] = bmin.z + (z / (N - 1)) * span.z;
    }
    cur.set(rest); prev.set(rest);
    const spacing = Math.max(span.x, span.y, span.z) / (N - 1);

    const springs = [];
    const addS = (a, b, k) => {
      const A = a * 3, B = b * 3;
      const d = Math.hypot(rest[A] - rest[B], rest[A + 1] - rest[B + 1], rest[A + 2] - rest[B + 2]);
      if (d > 1e-6) springs.push({ a, b, rest: d, tear: d * CFG.tearStretch, k, broken: false });
    };
    for (let z = 0; z < N; z++) for (let y = 0; y < N; y++) for (let x = 0; x < N; x++) {
      const a = ni(x, y, z);
      if (x < N - 1) addS(a, ni(x + 1, y, z), 0.85);
      if (y < N - 1) addS(a, ni(x, y + 1, z), 0.85);
      if (z < N - 1) addS(a, ni(x, y, z + 1), 0.85);
      if (x < N - 1 && y < N - 1) { addS(a, ni(x + 1, y + 1, z), 0.4); addS(ni(x + 1, y, z), ni(x, y + 1, z), 0.4); }
      if (y < N - 1 && z < N - 1) { addS(a, ni(x, y + 1, z + 1), 0.4); addS(ni(x, y + 1, z), ni(x, y, z + 1), 0.4); }
      if (x < N - 1 && z < N - 1) { addS(a, ni(x + 1, y, z + 1), 0.4); addS(ni(x + 1, y, z), ni(x, y, z + 1), 0.4); }
    }

    // Lattice cell lookup for a root-space point.
    const cellOf = (px, py, pz) => {
      const fx = clamp((px - bmin.x) / span.x * (N - 1), 0, N - 1);
      const fy = clamp((py - bmin.y) / span.y * (N - 1), 0, N - 1);
      const fz = clamp((pz - bmin.z) / span.z * (N - 1), 0, N - 1);
      const ix = Math.min(N - 2, Math.floor(fx)), iy = Math.min(N - 2, Math.floor(fy)), iz = Math.min(N - 2, Math.floor(fz));
      return { ix, iy, iz, tx: fx - ix, ty: fy - iy, tz: fz - iz };
    };
    for (const p of parts) {
      p.bIdx = new Int16Array(p.count * 3);
      p.bT = new Float32Array(p.count * 3);
      for (let i = 0; i < p.count; i++) {
        const c = cellOf(p.rootRest[i * 3], p.rootRest[i * 3 + 1], p.rootRest[i * 3 + 2]);
        p.bIdx[i * 3] = c.ix; p.bIdx[i * 3 + 1] = c.iy; p.bIdx[i * 3 + 2] = c.iz;
        p.bT[i * 3] = c.tx; p.bT[i * 3 + 1] = c.ty; p.bT[i * 3 + 2] = c.tz;
      }
      p.triCell = new Int32Array(p.triCount);
      for (let t = 0; t < p.triCount; t++) {
        let cx = 0, cy = 0, cz = 0;
        for (let k = 0; k < 3; k++) {
          const id = p.origIdx[t * 3 + k];
          cx += p.rootRest[id * 3]; cy += p.rootRest[id * 3 + 1]; cz += p.rootRest[id * 3 + 2];
        }
        const c = cellOf(cx / 3, cy / 3, cz / 3);
        p.triCell[t] = (c.iz * (N - 1) + c.iy) * (N - 1) + c.ix;
      }
    }

    const rebuildIndex = (p) => {
      const out = new Uint32Array(p.triCount * 3);
      let k = 0;
      for (let t = 0; t < p.triCount; t++) {
        const base = p.triCell[t];
        const cx = base % (N - 1), cy = Math.floor(base / (N - 1)) % (N - 1), cz = Math.floor(base / ((N - 1) * (N - 1)));
        let tornN = 0;
        for (let c = 0; c < 8; c++) if (torn[ni(cx + (c & 1), cy + ((c >> 1) & 1), cz + ((c >> 2) & 1))]) tornN++;
        if (tornN >= 2) continue;   // cells cut by a tear are removed -> visible crack
        out[k++] = p.origIdx[t * 3]; out[k++] = p.origIdx[t * 3 + 1]; out[k++] = p.origIdx[t * 3 + 2];
      }
      p.mesh.geometry.setIndex(new THREE.BufferAttribute(out.slice(0, k), 1));
      p.mesh.geometry.computeBoundingSphere();
    };

    const writeMesh = (p) => {
      const pos = p.mesh.geometry.attributes.position, e = p.relInv3.elements;
      for (let i = 0; i < p.count; i++) {
        const b = i * 3, ix = p.bIdx[b], iy = p.bIdx[b + 1], iz = p.bIdx[b + 2];
        const tx = p.bT[b], ty = p.bT[b + 1], tz = p.bT[b + 2];
        let dx = 0, dy = 0, dz = 0;
        for (let c = 0; c < 8; c++) {
          const ox = c & 1, oy = (c >> 1) & 1, oz = (c >> 2) & 1;
          const wgt = (ox ? tx : 1 - tx) * (oy ? ty : 1 - ty) * (oz ? tz : 1 - tz);
          const n = ni(ix + ox, iy + oy, iz + oz) * 3;
          dx += wgt * (cur[n] - rest[n]); dy += wgt * (cur[n + 1] - rest[n + 1]); dz += wgt * (cur[n + 2] - rest[n + 2]);
        }
        pos.setXYZ(i,
          p.localRest[b] + e[0] * dx + e[3] * dy + e[6] * dz,
          p.localRest[b + 1] + e[1] * dx + e[4] * dy + e[7] * dz,
          p.localRest[b + 2] + e[2] * dx + e[5] * dy + e[8] * dz);
      }
      pos.needsUpdate = true;
      p.mesh.geometry.computeVertexNormals();
      p.mesh.geometry.computeBoundingSphere();
    };

    const grabs = new Map();
    const sampleCol = firstMeshColor(target);
    const sb = {
      entity: target,
      dragNode: null,
      world: (i) => V(cur[i * 3], cur[i * 3
      world: (i) => V(cur[i * 3], cur[i * 3 + 1], cur[i * 3 + 2]).applyMatrix4(gW.matrixWorld),
      toRoot: (p) => V().copy(p).applyMatrix4(M4().copy(gW.matrixWorld).invert()),
      pick: (wp) => {
        const r = sb.toRoot(wp);
        let best = -1, bd = Infinity;
        for (let i = 0; i < NN; i++) {
          const d = Math.hypot(cur[i * 3] - r.x, cur[i * 3 + 1] - r.y, cur[i * 3 + 2] - r.z);
          if (d < bd) { bd = d; best = i; }
        }
        return best >= 0 && bd < spacing * 1.5 ? { i: best, root: r } : null;
      },
      step: null
    };

    sb.step = (dt, tips) => {
      if (!gW.parent) { softbodies.delete(sb); return; }
      gW.updateMatrixWorld(true);
      const G = gW.matrixWorld;
      const invG = M4().copy(G).invert();
      const m3 = M3().setFromMatrix4(invG);
      const gRoot = V(0, -1, 0).applyMatrix3(m3).normalize();
      const upRoot = V(0, 1, 0).applyMatrix3(m3).normalize();
      const floorRoot = V(0, FLOOR, 0).applyMatrix4(invG);
      const h = Math.min(0.02, Math.max(0.004, dt));
      const h2 = h * h;

      // Hands: pinch to grab nodes; otherwise the hand pushes them (poke / squeeze)
      for (const tp of tips) {
        const rp = V().copy(tp.pos).applyMatrix4(invG);
        if (tp.pressed) {
          let g = grabs.get(tp.key);
          if (!g) {
            g = [];
            for (let i = 0; i < NN; i++) {
              const off = V(cur[i * 3] - rp.x, cur[i * 3 + 1] - rp.y, cur[i * 3 + 2] - rp.z);
              if (off.length() < spacing * 0.9) g.push({ i, off });
            }
            grabs.set(tp.key, g);
          }
          for (const c of g) {
            const i3 = c.i * 3;
            const tx = rp.x + c.off.x, ty = rp.y + c.off.y, tz = rp.z + c.off.z;
            cur[i3] += (tx - cur[i3]) * 0.5;
            cur[i3 + 1] += (ty - cur[i3 + 1]) * 0.5;
            cur[i3 + 2] += (tz - cur[i3 + 2]) * 0.5;
            prev[i3] = cur[i3]; prev[i3 + 1] = cur[i3 + 1]; prev[i3 + 2] = cur[i3 + 2];
          }
        } else {
          grabs.delete(tp.key);
          const rv = V().copy(tp.vel).applyMatrix3(m3);
          const R = spacing * 0.75;
          for (let i = 0; i < NN; i++) {
            const i3 = i * 3;
            const dx = cur[i3] - rp.x, dy = cur[i3 + 1] - rp.y, dz = cur[i3 + 2] - rp.z;
            const d = Math.hypot(dx, dy, dz);
            if (d < R && d > 1e-5) {
              const push = (R - d) * 0.9, f = 1 - d / R;
              cur[i3] += (dx / d) * push + rv.x * 0.4 * f;
              cur[i3 + 1] += (dy / d) * push + rv.y * 0.4 * f;
              cur[i3 + 2] += (dz / d) * push + rv.z * 0.4 * f;
            }
          }
        }
      }

      // Mouse drag on a grabbed node
      if (sb.dragNode) {
        const dn = sb.dragNode, i3 = dn.i * 3;
        cur[i3] += (dn.target.x - cur[i3]) * 0.35;
        cur[i3 + 1] += (dn.target.y - cur[i3 + 1]) * 0.35;
        cur[i3 + 2] += (dn.target.z - cur[i3 + 2]) * 0.35;
        prev[i3] = cur[i3]; prev[i3 + 1] = cur[i3 + 1]; prev[i3 + 2] = cur[i3 + 2];
      }

      // Verlet integration with gravity
      const GRAV = 6.5 * h2;
      for (let i = 0; i < NN; i++) {
        const i3 = i * 3;
        const vx = (cur[i3] - prev[i3]) * 0.985;
        const vy = (cur[i3 + 1] - prev[i3 + 1]) * 0.985;
        const vz = (cur[i3 + 2] - prev[i3 + 2]) * 0.985;
        prev[i3] = cur[i3]; prev[i3 + 1] = cur[i3 + 1]; prev[i3 + 2] = cur[i3 + 2];
        cur[i3] += vx + gRoot.x * GRAV;
        cur[i3 + 1] += vy + gRoot.y * GRAV;
        cur[i3 + 2] += vz + gRoot.z * GRAV;
      }

      // Spring constraints (with tearing) + floor collision
      const tearPts = [];
      let newTear = false;
      for (let it = 0; it < 3; it++) {
        for (const s of springs) {
          if (s.broken) continue;
          const a = s.a * 3, b = s.b * 3;
          const dx = cur[b] - cur[a], dy = cur[b + 1] - cur[a + 1], dz = cur[b + 2] - cur[a + 2];
          const d = Math.hypot(dx, dy, dz);
          if (d < 1e-6) continue;
          if (d > s.tear) {
            s.broken = true;
            torn[s.a] = 1; torn[s.b] = 1;
            newTear = true;
            tearPts.push(V((cur[a] + cur[b]) / 2, (cur[a + 1] + cur[b + 1]) / 2, (cur[a + 2] + cur[b + 2]) / 2));
            continue;
          }
          const k = ((d - s.rest) / d) * s.k * 0.5;
          cur[a] += dx * k; cur[a + 1] += dy * k; cur[a + 2] += dz * k;
          cur[b] -= dx * k; cur[b + 1] -= dy * k; cur[b + 2] -= dz * k;
        }
        for (let i = 0; i < NN; i++) {
          const i3 = i * 3;
          const sd = (cur[i3] - floorRoot.x) * upRoot.x + (cur[i3 + 1] - floorRoot.y) * upRoot.y + (cur[i3 + 2] - floorRoot.z) * upRoot.z;
          if (sd < 0) { cur[i3] -= upRoot.x * sd; cur[i3 + 1] -= upRoot.y * sd; cur[i3 + 2] -= upRoot.z * sd; }
        }
      }

      if (newTear) {
        for (const p of parts) rebuildIndex(p);
        for (const tp of tearPts) spawnGranules(V().copy(tp).applyMatrix4(G), sampleCol, 20);
        APP.ui?.toast?.('Softbody tore! Granular chunks flying.', 1600);
      }
      for (const p of parts) writeMesh(p);
    };

    for (const p of parts) {
      const mats = Array.isArray(p.mesh.material) ? p.mesh.material : [p.mesh.material];
      mats.forEach((mt) => { if (mt) mt.side = THREE.DoubleSide; });
    }
    target.userData = target.userData || {};
    target.userData.isSoftbody = true;
    target.grabbable = false;
    if (target.physics) target.physics.enabled = false;
    softbodies.add(sb);
    APP.lastEntityId = target.id;
    return { ok: true, id: target.id, message: `"${target.name}" is now a softbody. Poke, stretch, squeeze it; pull hard to tear.` };
  }

  // ------------------------------------------------------------------
  // 4. SMOKE CLOUD (volumetric particles that part around your hand)
  // ------------------------------------------------------------------
  function makeSmoke(targetArg) {
    const target = resolveTarget(targetArg);
    if (!target) return { ok: false, error: 'No target object found for Smoke Cloud.' };
    const { tris, box } = worldTriangles(target, 14000);
    if (!tris.length) return { ok: false, error: 'Could not sample target geometry.' };

    // Area-weighted cumulative distribution over triangles
    const cum = new Float64Array(tris.length);
    let total = 0;
    tris.forEach((t, i) => {
      total += V().subVectors(t[1], t[0]).cross(V().subVectors(t[2], t[0])).length() * 0.5;
      cum[i] = total;
    });

    const centroid = box.getCenter(V());
    const base = firstMeshColor(target);
    const N = CFG.smokeCount;
    const rest = new Float32Array(N * 3), pos = new Float32Array(N * 3), vel = new Float32Array(N * 3);
    const col = new Float32Array(N * 3), seed = new Float32Array(N), size = new Float32Array(N);
    const cool = new THREE.Color(0xcfeaff), tc = new THREE.Color();

    for (let i = 0; i < N; i++) {
      // Pick a random surface point...
      const r = Math.random() * total;
      let lo = 0, hi = tris.length - 1;
      while (lo < hi) { const mid = (lo + hi) >> 1; if (cum[mid] < r) lo = mid + 1; else hi = mid; }
      const [a, b, c] = tris[lo];
      let u = Math.random(), v = Math.random();
      if (u + v > 1) { u = 1 - u; v = 1 - v; }
      const p = V().copy(a).addScaledVector(V().subVectors(b, a), u).addScaledVector(V().subVectors(c, a), v);
      // ...then pull some inward so the cloud has volume, not just a shell
      p.lerp(centroid, Math.pow(Math.random(), 1.5) * 0.55);
      p.addScaledVector(randDir(), 0.012);

      rest[i * 3] = pos[i * 3] = p.x;
      rest[i * 3 + 1] = pos[i * 3 + 1] = p.y;
      rest[i * 3 + 2] = pos[i * 3 + 2] = p.z;

      tc.copy(base).lerp(cool, 0.45 + Math.random() * 0.3);
      col[i * 3] = tc.r; col[i * 3 + 1] = tc.g; col[i * 3 + 2] = tc.b;
      seed[i] = Math.random() * 100;
      size[i] = 22 + Math.random() * 30;
    }

    const grp = new THREE.Group();
    grp.name = `${target.name}_Smoky_Hologram`;
    const geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
    geo.setAttribute('aCol', new THREE.BufferAttribute(col, 3));
    geo.setAttribute('aSize', new THREE.BufferAttribute(size, 1));
    geo.setAttribute('aSeed', new THREE.BufferAttribute(seed, 1));
    const mat = new THREE.ShaderMaterial({
      transparent: true,
      depthWrite: false,
      uniforms: {
        uTime: { value: 0 },
        uPx: { value: window.devicePixelRatio || 1 },
        uAlpha: { value: 0.9 }
      },
      vertexShader: [
        'attribute vec3 aCol; attribute float aSize; attribute float aSeed; uniform float uPx;',
        'varying vec3 vCol; varying float vSeed;',
        'void main(){ vCol=aCol; vSeed=aSeed; vec4 mv=modelViewMatrix*vec4(position,1.0);',
        '  gl_PointSize=clamp((aSize*uPx)/max(0.25,-mv.z), 3.0, 56.0);',
        '  gl_Position=projectionMatrix*mv; }'
      ].join('\n'),
      fragmentShader: [
        'uniform float uTime; uniform float uAlpha; varying vec3 vCol; varying float vSeed;',
        'float hash(vec2 p){ return fract(sin(dot(p, vec2(127.1,311.7)))*43758.5453); }',
        'float noise(vec2 p){ vec2 i=floor(p); vec2 f=fract(p); f=f*f*(3.0-2.0*f);',
        '  return mix(mix(hash(i),hash(i+vec2(1.0,0.0)),f.x), mix(hash(i+vec2(0.0,1.0)),hash(i+vec2(1.0,1.0)),f.x), f.y); }',
        'void main(){ vec2 uv=gl_PointCoord-0.5; float r=length(uv)*2.0; if(r>1.0) discard;',
        '  float wisp=noise(uv*3.0+vec2(vSeed, uTime*0.6));',
        '  float core=pow(1.0-r,1.8);',
        '  float a=core*(0.3+0.45*wisp)*uAlpha;',
        '  gl_FragColor=vec4(vCol*(0.9+0.25*wisp), a); }'
      ].join('\n')
    });
    const pts = new THREE.Points(geo, mat);
    pts.frustumCulled = false;
    grp.add(pts);
    APP.scene.add(grp);

    const name = target.name;
    const center = centroid.clone();
    APP.removeEntity?.(target, false);
    const ent = APP.registerEntity(grp, {
      name: `${name} (Smoke Cloud)`, category: 'smoke_holo', grabbable: false, physics: { enabled: false }
    });
    smokes.add({ ent, geo, mat, pos, rest, vel, seed, N, center });
    APP.lastEntityId = ent.id;
    return { ok: true, id: ent.id, message: `"${name}" is now a smoke cloud. Pass your hand through it.` };
  }

  // dist = [{ p: Vector3 (world), v: Vector3 (world velocity, m/s) }]
  function stepSmoke(st, dt, dist) {
    const { pos, rest, vel, seed, N } = st;
    const R = 0.28, R2 = R * R;
    st.mat.uniforms.uTime.value = clockT;
    const k = Math.min(dt, 0.03) * 2.5;
    for (let i = 0; i < N; i++) {
      const i3 = i * 3, sd = seed[i];
      // Gentle breathing around the rest pose
      const bx = rest[i3] + Math.sin(clockT * 0.8 + sd) * 0.006;
      const by = rest[i3 + 1] + Math.sin(clockT * 0.6 + sd * 1.3) * 0.008 + 0.004 * Math.sin(clockT * 0.3 + sd);
      const bz = rest[i3 + 2] + Math.cos(clockT * 0.7 + sd) * 0.006;
      vel[i3] += (bx - pos[i3]) * 1.4 * dt;
      vel[i3 + 1] += (by - pos[i3 + 1]) * 1.4 * dt;
      vel[i3 + 2] += (bz - pos[i3 + 2]) * 1.4 * dt;

      // Hands / pointer push the mist outward, swirl it, and lift it slightly
      for (const hd of dist) {
        const dx = pos[i3] - hd.p.x, dy = pos[i3 + 1] - hd.p.y, dz = pos[i3 + 2] - hd.p.z;
        const d2 = dx * dx + dy * dy + dz * dz;
        if (d2 < R2 && d2 > 1e-6) {
          const d = Math.sqrt(d2), f = Math.pow(1 - d / R, 1.6);
          const nx = dx / d, ny = dy / d, nz = dz / d;
          const vx = hd.v.x, vy = hd.v.y, vz = hd.v.z;
          const sp = 1.6 + Math.min(4, Math.hypot(vx, vy, vz)) * 1.2;
          const sx = ny * vz - nz * vy, sy = nz * vx - nx * vz, sz = nx * vy - ny * vx;
          vel[i3] += (nx * sp + vx * 0.8 + sx * 1.5) * f * k;
          vel[i3 + 1] += (ny * sp + vy * 0.8 + sy * 1.5 + 0.35) * f * k;
          vel[i3 + 2] += (nz * sp + vz * 0.8 + sz * 1.5) * f * k;
        }
      }

      const damp = Math.exp(-2.4 * dt);
      vel[i3] *= damp; vel[i3 + 1] *= damp; vel[i3 + 2] *= damp;
      pos[i3] += vel[i3] * dt;
      pos[i3 + 1] += vel[i3 + 1] * dt;
      pos[i3 + 2] += vel[i3 + 2] * dt;
    }
    st.geo.attributes.position.needsUpdate = true;
  }
