      axes[2] = [axes[0][1] * axes[1][2] - axes[0][2] * axes[1][1], axes[0][2] * axes[1][0] - axes[0][0] * axes[1][2], axes[0][0] * axes[1][1] - axes[0][1] * axes[1][0]];
      var pmin = [1e9, 1e9, 1e9], pmax = [-1e9, -1e9, -1e9];
      for (i = 0; i < nV; i++) for (k = 0; k < 3; k++) { var pr = P[i * 3] * axes[k][0] + P[i * 3 + 1] * axes[k][1] + P[i * 3 + 2] * axes[k][2]; if (pr < pmin[k]) pmin[k] = pr; if (pr > pmax[k]) pmax[k] = pr; }
      var pe = [pmax[0] - pmin[0], pmax[1] - pmin[1], pmax[2] - pmin[2]];
      var vox = cnt * cell * cell * cell;
      // slices (Y and principal axis)
      var NB = 10, sY = [], sA = [], b;
      for (b = 0; b < NB; b++) { sY.push([1e9, -1e9, 1e9, -1e9]); sA.push([1e9, -1e9, 1e9, -1e9]); }
      for (i = 0; i < nV; i++) {
        var px = P[i * 3], py = P[i * 3 + 1], pz = P[i * 3 + 2];

        var p0 = px * axes[0][0] + py * axes[0][1] + pz * axes[0][2], p1 = px * axes[1][0] + py * axes[1][1] + pz * axes[1][2], p2 = px * axes[2][0] + py * axes[2][1] + pz * axes[2][2];
        var ba = Math.max(0, Math.min(NB - 1, Math.floor((p0 - pmin[0]) / Math.max(pe[0], 1e-6) * NB))), r2 = sA[ba];
        if (p1 < r2[0]) r2[0] = p1; if (p1 > r2[1]) r2[1] = p1; if (p2 < r2[2]) r2[2] = p2; if (p2 > r2[3]) r2[3] = p2;
      }

      var sliceAxis = sA.map(function (r, bi) { return r[1] < r[0] ? { t: (bi + 0.5) / NB, w1: 0, w2: 0 } : { t: (bi + 0.5) / NB, w1: (r[1] - r[0]) / Math.max(pe[1], 1e-6), w2: (r[3] - r[2]) / Math.max(pe[2], 1e-6) }; });
      // horizontal principal axis (yaw)
      var sxx = 0, szz = 0, sxz = 0;
      for (i = 0; i < nV; i++) { sxx += P[i * 3] * P[i * 3]; szz += P[i * 3 + 2] * P[i * 3 + 2]; sxz += P[i * 3] * P[i * 3 + 2]; }

      var vn = new Float32Array(nV * 3);

      }
      function uvOf(x, y, z, nx_, ny_, nz_, out, oi) {
        var ax_ = Math.abs(nx_), ay_ = Math.abs(ny_), az_ = Math.abs(nz_), md = Math.max(bs[0], bs[1], bs[2], 1e-6), u, v;
        if (ay_ >= ax_ && ay_ >= az_) { u = x; v = z; } else if (ax_ >= az_) { u = z; v = y; } else { u = x; v = y; }
        out[oi] = u / md + 0.5; out[oi + 1] = v / md + 0.5;
      }
      var out, tr;
      if (sharpen > 0.15) {

        for (i = 0; i < nT * 3; i++) vf[T[i] + 1]++;

        for (i = 0; i < nT; i++) for (var cc2 = 0; cc2 < 3; cc2++) { var vv2 = T[i * 3 + cc2]; fl[vf[vv2] + fc[vv2]++] = i; }
        var oP = new Float32Array(nT * 9), oN = new Float32Array(nT * 9), oU = new Float32Array(nT * 6);
        for (i = 0; i < nT; i++) {

          for (var c3 = 0; c3 < 3; c3++) {
            var vv3 = T[i * 3 + c3], sx3 = 0, sy3 = 0, sz3 = 0;
            for (var m2 = vf[vv3]; m2 < vf[vv3 + 1]; m2++) {
              var fj = fl[m2], gx = fn[fj * 3], gy = fn[fj * 3 + 1], gz = fn[fj * 3 + 2], gl = Math.sqrt(gx * gx + gy * gy + gz * gz) || 1;
              if ((gx * fx + gy * fy + gz * fz) / (gl * fl_) > crease) { sx3 += gx; sy3 += gy; sz3 += gz; }
            }
            var nl = Math.sqrt(sx3 * sx3 + sy3 * sy3 + sz3 * sz3) || 1, oi = i * 9 + c3 * 3;
            oP[oi] = P[vv3 * 3]; oP[oi + 1] = P[vv3 * 3 + 1]; oP[oi + 2] = P[vv3 * 3 + 2];
            oN[oi] = sx3 / nl; oN[oi + 1] = sy3 / nl; oN[oi + 2] = sz3 / nl;
            uvOf(oP[oi], oP[oi + 1], oP[oi + 2], fx / fl_, fy / fl_, fz / fl_, oU, i * 6 + c3 * 2);
          }
        }
        out = { positions: oP, normals: oN, uvs: oU, indices: null }; tr = [oP.buffer, oN.buffer, oU.buffer];
      } else {
        var uv = new Float32Array(nV * 2);
        for (i = 0; i < nV; i++) {
          var l3 = Math.sqrt(vn[i * 3] * vn[i * 3] + vn[i * 3 + 1] * vn[i * 3 + 1] + vn[i * 3 + 2] * vn[i * 3 + 2]) || 1;
          vn[i * 3] /= l3; vn[i * 3 + 1] /= l3; vn[i * 3 + 2] /= l3;
          uvOf(P[i * 3], P[i * 3 + 1], P[i * 3 + 2], vn[i * 3], vn[i * 3 + 1], vn[i * 3 + 2], uv, i * 2);
        }
        out = { positions: P, normals: vn, uvs: uv, indices: T }; tr = [P.buffer, vn.buffer, uv.buffer, T.buffer];
      }
      out.bbox = { center: bc, size: bs };
      out.pca = { axes: axes, extents: pe };
      out.aspect = { e0e1: pe[0] / Math.max(pe[1], 1e-6), e1e2: pe[1] / Math.max(pe[2], 1e-6), e0e2: pe[0] / Math.max(pe[2], 1e-6) };
      out.fill = { aabb: vox / Math.max(bs[0] * bs[1] * bs[2], 1e-9), obb: vox / Math.max(pe[0] * pe[1] * pe[2], 1e-9) };
      out.symmetry = symm; out.sliceY = sliceY; out.sliceAxis = sliceAxis;
      out.yaw = { ang: ang, u: yu, v: yv, ratio: yu / Math.max(yv, 1e-6) };
      out.tris = nT;
      return { out: out, tr: tr };
    }
  }

  function buildTubeGeo(pts, radii, radial) {
    radial = radial || 8;
    const curve = pts.length > 2 ? new THREE.CatmullRomCurve3(pts, false, 'centripetal') : new THREE.LineCurve3(pts[0], pts[pts.length - 1]);
    const segs = Math.min(Math.max(pts.length * 2, 4), 700), fr = curve.computeFrenetFrames(segs, false);
    const pos = [], idx = [], p = new V3();
    const rAt = (t) => { const f = t * (radii.length - 1), i = Math.floor(f), j = Math.min(i + 1, radii.length - 1); return radii[i] + (radii[j] - radii[i]) * (f - i); };
    for (let i = 0; i <= segs; i++) {
      const t = i / segs; curve.getPointAt(t, p); const r = rAt(t), N = fr.normals[i], B = fr.binormals[i];
      for (let k = 0; k < radial; k++) { const a = k / radial * Math.PI * 2, c = Math.cos(a), s = Math.sin(a); pos.push(p.x + (N.x * c + B.x * s) * r, p.y + (N.y * c + B.y * s) * r, p.z + (N.z * c + B.z * s) * r); }
    }
    for (let i = 0; i < segs; i++) for (let k = 0; k < radial; k++) {
      const a = i * radial + k, b = i * radial + (k + 1) % radial, c = (i + 1) * radial + k, d = (i + 1) * radial + (k + 1) % radial; idx.push(a, b, c, b, d, c);
    }
    const c0 = pos.length / 3; pos.push(pts[0].x, pts[0].y, pts[0].z); const c1 = c0 + 1; pos.push(pts[pts.length - 1].x, pts[pts.length - 1].y, pts[pts.length - 1].z);
    for (let k = 0; k < radial; k++) { idx.push(c0, (k + 1) % radial, k); idx.push(c1, segs * radial + k, segs * radial + (k + 1) % radial); }
    const g = new THREE.BufferGeometry(); g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3)); g.setIndex(idx); g.computeVertexNormals(); return g;
  }

  function addPoint(s, pos, pr, force) {
    const r = BR().r * (0.78 + 0.44 * clamp(pr, 0, 1)), last = s.pts[s.pts.length - 1];
    if (last && !force && last.distanceTo(pos) < Math.max(0.004, r * 0.45)) return;
    s.pts.push(last ? last.clone().lerp(pos, 0.6) : pos.clone()); s.radii.push(r);
    const now = performance.now();
    if (s.pts.length >= 2 && now - s.lastBuild > 70) { s.lastBuild = now; strokeMesh(s); }
  }

  function makeHolo(rec, bounds) {
    rec.planeAbove = new THREE.Plane(new V3(0, 1, 0), 1e4);
    rec.holoMat = new THREE.ShaderMaterial({
      transparent: true, depthWrite: false, side: THREE.DoubleSide, blending: THREE.AdditiveBlending,
      uniforms: { uTime: { value: 0 }, uMinY: { value: bounds.min.y }, uH: { value: Math.max(bounds.max.y - bounds.min.y, 0.05) }, uReveal: { value: -1e4 }, uCol: { value: new THREE.Color(0x2fe0ff) }, uCol2: { value: new THREE.Color(0xc8fbff) } },
      vertexShader: 'varying vec3 vW; varying vec3 vN; void main(){ vec4 w = modelMatrix * vec4(position,1.0); vW = w.xyz; vN = normalize(mat3(modelMatrix) * normal); gl_Position = projectionMatrix * viewMatrix * w; }',
      fragmentShader: 'uniform float uTime,uMinY,uH,uReveal; uniform vec3 uCol,uCol2; varying vec3 vW; varying vec3 vN;' +
        'void main(){ if(vW.y < uReveal) discard; vec3 V = normalize(cameraPosition - vW); float fr = pow(1.0 - abs(dot(normalize(vN),V)), 2.2);' +
        'float sy = uMinY + fract(uTime*0.45)*uH; float band = exp(-abs(vW.y - sy)/(0.05*uH + 0.004));' +
        'float lines = 0.55 + 0.45*sin(vW.y*220.0 - uTime*6.0); float pulse = 0.78 + 0.22*sin(uTime*5.0);' +
        'vec3 col = mix(uCol,uCol2,clamp(band,0.0,1.0)) * (0.45 + fr*1.1 + band*1.6) * (0.7+0.3*lines);' +
        'float a = clamp((0.16 + fr*0.85 + band*1.2) * pulse, 0.0, 1.0); gl_FragColor = vec4(col, a); }'
    });
    rec.wireMat = new THREE.MeshBasicMaterial({ color: 0x8ff6ff, wireframe: true, transparent: true, opacity: 0.4, depthWrite: false, blending: THREE.AdditiveBlending, clippingPlanes: [rec.planeAbove] });
    rec.temps = rec.temps || [];
  }
  function holoize(rec, mesh) {

    const w = new THREE.Mesh(mesh.geometry, rec.wireMat); w.frustumCulled = false; mesh.add(w); rec.temps.push(w); rec.wired = rec.wired || []; rec.wired.push(mesh);

  }

  }

  const hash2 = (x, y) => { const h = Math.sin(x * 127.1 + y * 311.7) * 43758.5453; return h - Math.floor(h); };
  function vnoise(x, y) { const xi = Math.floor(x), yi = Math.floor(y), xf = x - xi, yf = y - yi, u = xf * xf * (3 - 2 * xf), v = yf * yf * (3 - 2 * yf); const a = hash2(xi, yi), b = hash2(xi + 1, yi), c = hash2(xi, yi + 1), d = hash2(xi + 1, yi + 1); return a + (b - a) * u + (c - a) * v + (a - b - c + d) * u * v; }
  function fbm(x, y, o) { let s = 0, a = 0.5; for (let i = 0; i < (o || 4); i++) { s += a * vnoise(x, y); x *= 2.03; y *= 2.03; a *= 0.5; } return s; }
  function pattern(kind, u, v, o) { // o = [height, colorMix, roughness]
    switch (kind) {
      case 'wood': { const g = u * 14 + fbm(u * 3, v * 0.6, 4) * 6, ring = 0.5 + 0.5 * Math.sin(g * 6.2832); o[0] = ring * 0.4 + fbm(u * 40, v * 2, 3) * 0.3; o[1] = ring; o[2] = 0.55 + 0.2 * fbm(u * 30, v * 3, 2); break; }
      case 'metal': { const n = fbm(u * 2, v * 180, 3); o[0] = n * 0.3; o[1] = n; o[2] = 0.28 + 0.22 * n; break; }
      case 'stone': { const n = fbm(u * 9, v * 9, 5); o[0] = n; o[1] = n; o[2] = 0.82 + 0.12 * n; break; }
      case 'ceramic': { const sp = hash2(Math.floor(u * 300), Math.floor(v * 300)) > 0.985 ? 1 : 0; o[0] = 0.05 + 0.05 * fbm(u * 20, v * 20, 3); o[1] = fbm(u * 3, v * 3, 3) * 0.35 + sp; o[2] = 0.2 + 0.1 * fbm(u * 12, v * 12, 3); break; }
      case 'organic': { const n = fbm(u * 7, v * 7, 5), ve = Math.abs(Math.sin(n * 12)); o[0] = n * 0.6 + ve * 0.2; o[1] = n; o[2] = 0.7 + 0.1 * n; break; }
      case 'glass': o[0] = 0.5; o[1] = 0; o[2] = 0.05; break;
      default: { const w = Math.sin(u * 400) * Math.sin(v * 400); o[0] = 0.5 + 0.25 * w + 0.1 * fbm(u * 30, v * 30, 2); o[1] = fbm(u * 5, v * 5, 3); o[2] = 0.9; }
    }
  }

  function bakePBRRaw(kind, c1, c2) {
    const N = 256, mk = () => { const c = document.createElement('canvas'); c.width = c.height = N; return c; };
    const ca = mk(), cn = mk(), cr = mk(), ia = ca.getContext('2d').createImageData(N, N), inn = cn.getContext('2d').createImageData(N, N), ir = cr.getContext('2d').createImageData(N, N);
    const H = new Float32Array(N * N), o = [0, 0, 0], h1 = c1.getHex(), h2 = c2.getHex();
    const A = [(h1 >> 16) & 255, (h1 >> 8) & 255, h1 & 255], B = [(h2 >> 16) & 255, (h2 >> 8) & 255, h2 & 255];
    for (let y = 0; y < N; y++) for (let x = 0; x < N; x++) {
      pattern(kind, x / N, y / N, o); const i = y * N + x, m = clamp(o[1], 0, 1), sh = 0.72 + 0.28 * clamp(o[0], 0, 1);
      H[i] = o[0]; ia.data[i * 4] = (A[0] + (B[0] - A[0]) * m) * sh; ia.data[i * 4 + 1] = (A[1] + (B[1] - A[1]) * m) * sh; ia.data[i * 4 + 2] = (A[2] + (B[2] - A[2]) * m) * sh; ia.data[i * 4 + 3] = 255;
      const rr = clamp(o[2], 0.02, 1) * 255; ir.data[i * 4] = ir.data[i * 4 + 1] = ir.data[i * 4 + 2] = rr; ir.data[i * 4 + 3] = 255;
    }
    const str = { stone: 5, wood: 3, metal: 1.4, organic: 4, fabric: 5, ceramic: 2, glass: 0 }[kind] || 2, hh = (x, y) => H[((y + N) % N) * N + ((x + N) % N)];
    for (let y = 0; y < N; y++) for (let x = 0; x < N; x++) {
      const gx = (hh(x + 1, y - 1) + 2 * hh(x + 1, y) + hh(x + 1, y + 1)) - (hh(x - 1, y - 1) + 2 * hh(x - 1, y) + hh(x - 1, y + 1));
      const gy = (hh(x - 1, y + 1) + 2 * hh(x, y + 1) + hh(x + 1, y + 1)) - (hh(x - 1, y - 1) + 2 * hh(x, y - 1) + hh(x + 1, y - 1));
      let nx = -gx * str * 0.25, ny = gy * str * 0.25, nz = 1; const l = Math.hypot(nx, ny, nz); const i = (y * N + x) * 4;
      inn.data[i] = (nx / l * 0.5 + 0.5) * 255; inn.data[i + 1] = (ny / l * 0.5 + 0.5) * 255; inn.data[i + 2] = (nz / l * 0.5 + 0.5) * 255; inn.data[i + 3] = 255;
    }
    ca.getContext('2d').putImageData(ia, 0, 0); cn.getContext('2d').putImageData(inn, 0, 0); cr.getContext('2d').putImageData(ir, 0, 0);
    const tex = (cv, srgb) => { const t = new THREE.CanvasTexture(cv); t.wrapS = t.wrapT = THREE.RepeatWrapping; t.anisotropy = 4; if (srgb && THREE.SRGBColorSpace) t.colorSpace = THREE.SRGBColorSpace; return t; };
    return { map: tex(ca, true), normalMap: tex(cn, false), roughnessMap: tex(cr, false) };
  }

  // Mode B morph: each GLB vertex starts on its nearest sample of the drawn volume and eases into its own position.
  // The morph runs as a hologram over the drawing, the drawing fades out, then the real materials reveal bottom-up.
  function morphInto(rec, holder, base, label, alive) {
    return new Promise((resolve) => {
      holder.updateMatrixWorld(true);
      const P0 = base.positions, nP = P0.length / 3, stp = Math.max(1, Math.floor(nP / 1500)), src = [];
      for (let i = 0; i < nP; i += stp) src.push(P0[i * 3], P0[i * 3 + 1], P0[i * 3 + 2]);
      const parts = [];
      holder.traverse((m) => {
        if (!m.isMesh || !m.geometry || !m.geometry.attributes.position) return;
        const g = m.geometry.clone(); g.applyMatrix4(m.matrixWorld);
        const rest = Float32Array.from(g.attributes.position.array), n = rest.length / 3, start = new Float32Array(rest.length);
        for (let i = 0; i < n; i++) {
          const x = rest[i * 3], y = rest[i * 3 + 1], z = rest[i * 3 + 2]; let best = Infinity, bi = 0;
          for (let j = 0; j < src.length; j += 3) { const dx = src[j] - x, dy = src[j + 1] - y, dz = src[j + 2] - z, d = dx * dx + dy * dy + dz * dz; if (d < best) { best = d; bi = j; } }
          start[i * 3] = src[bi]; start[i * 3 + 1] = src[bi + 1]; start[i * 3 + 2] = src[bi + 2];
        }
        g.attributes.position.array.set(start); g.attributes.position.needsUpdate = true;
        const mesh = new THREE.Mesh(g, rec.holoMat), wire = new THREE.Mesh(g, rec.wireMat);
        mesh.frustumCulled = false; wire.frustumCulled = false; mesh.add(wire);
        parts.push({ mesh, wire, g, rest, start, orig: m.material });
      });
      if (!parts.length || !src.length) { finalize(rec, holder, label, base.bbox.center); return resolve(); }

      const morphRoot = new THREE.Group(); parts.forEach((p) => morphRoot.add(p.mesh));
      const bootTemps = rec.temps.slice(), DUR = 2.4;
      holoBoundsUpdate(rec, morphRoot);
      APP.scene.add(morphRoot);
      let t = 0, unsub = null, settled = false;
      const ease = (k) => (k < 0.5 ? 4 * k * k * k : 1 - Math.pow(-2 * k + 2, 3) / 2);
      const stop = () => { if (settled) return false; settled = true; try { unsub && unsub(); } catch (_) {} unsub = null; APP.scene.remove(morphRoot); return true; };
      const cancel = () => { if (stop()) parts.forEach((p) => p.g.dispose()); resolve(); };
      const done = () => {
        if (!stop()) return;
        parts.forEach((p) => {
          p.g.attributes.position.array.set(p.rest); p.g.attributes.position.needsUpdate = true; p.g.computeVertexNormals();
          p.mesh.remove(p.wire);
          const om = p.orig;
          p.mesh.material = Array.isArray(om) ? om.map((x) => x.clone()) : (om ? om.clone() : new THREE.MeshStandardMaterial({ color: 0xcfd8dc }));
        });
        bootTemps.forEach((o) => { o.visible = false; });
        if (alive()) finalize(rec, morphRoot, label, base.bbox.center);
        resolve();
      };
      unsub = APP.onUpdate((dt) => {
        if (settled) return;
        if (!alive()) return cancel();
        t += dt; const k = Math.min(1, t / DUR), e = ease(k);
        for (const p of parts) {
          const a = p.g.attributes.position.array, s = p.start, r = p.rest;
          for (let i = 0; i < a.length; i++) a[i] = s[i] + (r[i] - s[i]) * e;
          p.g.attributes.position.needsUpdate = true;
        }
        bootTemps.forEach((o) => { o.visible = e < 0.5; });
        if (k >= 1) done();
      });
    });
  }
