(function () {
  const APP = window.APP || (window.APP = {});
  const THREE = APP.THREE;
  if (!THREE || !APP.scene) return;

  if (typeof APP._inst4Cleanup === 'function') {
    try { APP._inst4Cleanup(); } catch (_) {}
  }

  const manipCfg = {
    mode: 'off', // 'off' | 'voronoi' | 'melt' | 'softbody' | 'smoke' | 'merge'
    meltRadius: 0.24,
    tearThreshold: 1.85
  };

  function isManipulableEntity(e) {
    if (!e || !e.group?.parent) return false;
    if (e.isPen || e.isStroke || e.userData?.isManipTool) return false;
    if (['world_fx', 'board', 'portal', 'tool', 'tool_pen', 'shard', 'smoke_holo'].includes(e.category)) return false;
    if (e.group.name && (e.group.name.includes('APP_GEENY') || e.group.name.includes('TOOLBAR') || e.group.name.includes('Thermal_Melt'))) return false;
    return true;
  }

  function resolveTargetEntity(targetArg) {
    if (targetArg && typeof targetArg === 'object' && targetArg.group) return targetArg;
    const q = String(targetArg || '').trim();
    if (q && q !== 'latest') {
      const found = APP.findEntity?.(q);
      if (isManipulableEntity(found)) return found;
    }
    if (APP.lastEntityId) {
      const last = (APP.entities || []).find((e) => e && e.id === APP.lastEntityId);
      if (isManipulableEntity(last)) return last;
    }
    const cands = (APP.entities || []).filter(isManipulableEntity);
    return cands[cands.length - 1] || null;
  }

  // ====================================================================
  // 1. REALISTIC VORONOI FRACTURE & SHATTER (With Interior Caps & Dust)

  function clipTriangleAgainstPlane(tri, pPoint, pNorm) {
    // Returns { inside: [[v0,v1,v2], ...], cutEdge: [a, b] | null }
    // Keep side: dot(v - pPoint, pNorm) <= 0
    const d = [
      (tri[0].x - pPoint.x) * pNorm.x + (tri[0].y - pPoint.y) * pNorm.y + (tri[0].z - pPoint.z) * pNorm.z,
      (tri[1].x - pPoint.x) * pNorm.x + (tri[1].y - pPoint.y) * pNorm.y + (tri[1].z - pPoint.z) * pNorm.z,
      (tri[2].x - pPoint.x) * pNorm.x + (tri[2].y - pPoint.y) * pNorm.y + (tri[2].z - pPoint.z) * pNorm.z
    ];
    const inIdx = [], outIdx = [];
    for (let i = 0; i < 3; i++) {
      if (d[i] <= 1e-5) inIdx.push(i);
      else outIdx.push(i);
    }
    if (inIdx.length === 3) return { tris: [tri], cut: null };
    if (inIdx.length === 0) return { tris: [], cut: null };

    function interp(ia, ib) {
      const t = d[ia] / (d[ia] - d[ib]);
      return new THREE.Vector3().lerpVectors(tri[ia], tri[ib], THREE.MathUtils.clamp(t, 0, 1));
    }

    if (inIdx.length === 1) {
      const i0 = inIdx[0], i1 = (i0 + 1) % 3, i2 = (i0 + 2) % 3;
      const pA = interp(i0, i1), pB = interp(i0, i2);
      return { tris: [[tri[i0].clone(), pA, pB]], cut: [pA.clone(), pB.clone()] };
    }
    // 2 inside, 1 outside
    const o0 = outIdx[0], i1 = (o0 + 1) % 3, i2 = (o0 + 2) % 3;
    const pA = interp(i1, o0), pB = interp(i2, o0);
    return {
      tris: [
        [tri[i1].clone(), tri[i2].clone(), pB],
        [tri[i1].clone(), pB.clone(), pA]
      ],
      cut: [pA.clone(), pB.clone()]
    };
  }

  function spawnFractureDustAndDebris(center, size, baseCol) {
    const dustGroup = new THREE.Group();
    dustGroup.name = 'Voronoi_Dust_FX';

    // 1. Fine mineral dust puffs + 140 sharp micro-chips
    const COUNT = 180;
    const pos = new Float32Array(COUNT * 3);
    const vel = Array.from({ length: COUNT }, () => new THREE.Vector3());
    const life = new Float32Array(COUNT);
    const isDust = new Float32Array(COUNT);
    const spread = Math.max(0.15, Math.max(size.x, size.y, size.z) * 0.45);

    for (let i = 0; i < COUNT; i++) {
      pos[i * 3] = center.x + (Math.random() - 0.5) * spread;
      pos[i * 3 + 1] = Math.max(0.03, center.y + (Math.random() - 0.5) * spread);
      pos[i * 3 + 2] = center.z + (Math.random() - 0.5) * spread;
      const d = i < 55;
      isDust[i] = d ? 1.0 : 0.0;
      const spd = d ? 0.55 : 1.65;
      vel[i].set((Math.random() - 0.5) * spd, (d ? 0.25 : 0.75) + Math.random() * spd * 0.7, (Math.random() - 0.5) * spd);
      life[i] = d ? (1.1 + Math.random() * 0.8) : (0.7 + Math.random() * 0.6);
    }

    const geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
    geo.setAttribute('aLife', new THREE.BufferAttribute(life, 1));
    geo.setAttribute('aDust', new THREE.BufferAttribute(isDust, 1));

    const pts = new THREE.Points(geo, new THREE.ShaderMaterial({
      transparent: true, depthWrite: false,
      uniforms: { uCol: { value: baseCol.clone().lerp(new THREE.Color(0x94a3b8), 0.45) }, uPx: { value: window.devicePixelRatio || 1 } },
      vertexShader: 'attribute float aLife; attribute float aDust; uniform float uPx; varying float vLife; varying float vDust; void main(){ vLife=aLife; vDust=aDust; vec4 mv=modelViewMatrix*vec4(position,1.0); float sz=aDust>0.5 ? mix(54.0, 22.0, clamp(aLife*0.5,0.0,1.0)) : 8.5; gl_PointSize=aLife>0.01?clamp((sz*uPx)/max(0.3,-mv.z), 1.5, 34.0):0.0; gl_Position=projectionMatrix*mv; }',
      fragmentShader: 'uniform vec3 uCol; varying float vLife; varying float vDust; void main(){ float d=length(gl_PointCoord-0.5)*2.0; if(d>1.0)discard; float a=vDust>0.5 ? (1.0-d*d)*clamp(vLife*0.26,0.0,0.26) : (1.0-d)*clamp(vLife,0.0,1.0); gl_FragColor=vec4(uCol, a); }'
    }));
    pts.frustumCulled = false;
    dustGroup.add(pts);
    APP.scene.add(dustGroup);

    const fxEnt = APP.registerEntity(dustGroup, {
      name: 'Fracture Dust',
      category: 'world_fx',
      grabbable: false,
      physics: { enabled: false },
      onUpdate: (dt) => {
        const delta = Math.min(0.04, dt || 0.016);
        let alive = false;
        for (let i = 0; i < COUNT; i++) {
          if (life[i] <= 0) { pos[i * 3 + 1] = -99; continue; }
          alive = true;
          life[i] -= delta;
          if (isDust[i] > 0.5) {
            vel[i].multiplyScalar(0.95);
            vel[i].y += 0.14 * delta;
          } else {
            vel[i].y -= 8.8 * delta;
          }
          pos[i * 3] += vel[i].x * delta;
          pos[i * 3 + 1] += vel[i].y * delta;
          pos[i * 3 + 2] += vel[i].z * delta;
          if (isDust[i] < 0.5 && pos[i * 3 + 1] < 0.01) {
            pos[i * 3 + 1] = 0.01;
            vel[i].y *= -0.35; vel[i].x *= 0.6; vel[i].z *= 0.6;
          }
        }
        geo.attributes.position.needsUpdate = true;
        geo.attributes.aLife.needsUpdate = true;
        if (!alive) APP.removeEntity?.(fxEnt, false);
      }
    });
  }

  function fractureEntity(targetArg, impactWorldPt) {
    const target = resolveTargetEntity(targetArg);
    if (!target) return { ok: false, error: 'No target object found to fracture.' };

    target.group.updateMatrixWorld(true);
    const worldBox = new THREE.Box3().setFromObject(target.group);
    if (worldBox.isEmpty()) return { ok: false, error: 'Target has no mesh geometry.' };
    const size = new THREE.Vector3(), center = new THREE.Vector3();
    worldBox.getSize(size); worldBox.getCenter(center);

    const impactPt = impactWorldPt ? impactWorldPt.clone() : center.clone().add(new THREE.Vector3(0, size.y * 0.15, 0));

    // Extract triangles in world space (decimate gently only if super high-poly > 2800 tris)
    const rawTris = [];
    let sampleMat = null;
    const v0 = new THREE.Vector3(), v1 = new THREE.Vector3(), v2 = new THREE.Vector3();

    target.group.traverse((c) => {
      if (!c.isMesh || !c.geometry?.attributes?.position || c.userData?.nxrDecal) return;
      const geo = c.geometry, pos = geo.attributes.position, idx = geo.index;
      if (!sampleMat) sampleMat = Array.isArray(c.material) ? c.material[0] : c.material;
      const triCount = idx ? Math.floor(idx.count / 3) : Math.floor(pos.count / 3);
      const stride = Math.max(1, Math.ceil(triCount / 2200));
      for (let t = 0; t < triCount; t += stride) {
        const i0 = idx ? idx.getX(t * 3) : t * 3;
        const i1 = idx ? idx.getX(t * 3 + 1) : t * 3 + 1;
        const i2 = idx ? idx.getX(t * 3 + 2) : t * 3 + 2;
        v0.fromBufferAttribute(pos, i0).applyMatrix4(c.matrixWorld);
        v1.fromBufferAttribute(pos, i1).applyMatrix4(c.matrixWorld);
        v2.fromBufferAttribute(pos, i2).applyMatrix4(c.matrixWorld);
        rawTris.push([v0.clone(), v1.clone(), v2.clone()]);
      }
    });

    if (rawTris.length < 4) return { ok: false, error: 'Not enough geometry to fracture.' };

    // Generate anisotropic Voronoi seeds (more tightly clustered near impact point for realistic shard variety!)
    const SEED_COUNT = 12;
    const seeds = [];
    for (let i = 0; i < SEED_COUNT; i++) {
      if (i < 5) {
        // Dense small shards near impact point
        seeds.push(new THREE.Vector3(
          THREE.MathUtils.clamp(impactPt.x + (Math.random() - 0.5) * size.x * 0.48, worldBox.min.x, worldBox.max.x),
          THREE.MathUtils.clamp(impactPt.y + (Math.random() - 0.5) * size.y * 0.48, worldBox.min.y, worldBox.max.y),
          THREE.MathUtils.clamp(impactPt.z + (Math.random() - 0.5) * size.z * 0.48, worldBox.min.z, worldBox.max.z)
        ));
      } else {
        seeds.push(new THREE.Vector3(
          THREE.MathUtils.lerp(worldBox.min.x, worldBox.max.x, 0.1 + Math.random() * 0.8),
          THREE.MathUtils.lerp(worldBox.min.y, worldBox.max.y, 0.1 + Math.random() * 0.8),
          THREE.MathUtils.lerp(worldBox.min.z, worldBox.max.z, 0.1 + Math.random() * 0.8)
        ));
      }
    }

    const baseCol = sampleMat?.color ? sampleMat.color.clone() : new THREE.Color(0x94a3b8);
    const outerMat = sampleMat ? sampleMat.clone() : new THREE.MeshStandardMaterial({ color: baseCol, roughness: 0.55 });
    outerMat.side = THREE.DoubleSide;
    const innerMat = new THREE.MeshStandardMaterial({
      color: baseCol.clone().multiplyScalar(0.58),
      roughness: 0.94,
      metalness: 0.05,
      side: THREE.DoubleSide
    });

    const shardsSpawned = [];

    for (let si = 0; si < seeds.length; si++) {
      const sA = seeds[si];
      // Sort neighbors by proximity and clip against top 6 Voronoi bisector planes
      const neighbors = seeds
        .map((sB, idx) => ({ idx, sB, d: sA.distanceToSquared(sB) }))
        .filter((n) => n.idx !== si && n.d > 1e-6)
        .sort((a, b) => a.d - b.d)
        .slice(0, 6);

      let cellTris = rawTris;
      const capTris = [];

      for (let ni = 0; ni < neighbors.length; ni++) {
        const sB = neighbors[ni].sB;
        const pMid = new THREE.Vector3().addVectors(sA, sB).multiplyScalar(0.5);
        // Subtle normal jitter makes cracks slightly irregular like real brittle fracture
        const pNorm = new THREE.Vector3().subVectors(sB, sA).normalize();
        const nextTris = [];
        const cutSegments = [];

        for (let ti = 0; ti < cellTris.length; ti++) {
          const res = clipTriangleAgainstPlane(cellTris[ti], pMid, pNorm);
          for (let k = 0; k < res.tris.length; k++) nextTris.push(res.tris[k]);
          if (res.cut) cutSegments.push(res.cut);
        }
        cellTris = nextTris;
        if (!cellTris.length) break;

        // Triangulate interior cap on the fracture plane from cut segments
        if (cutSegments.length >= 3) {
          const capCenter = new THREE.Vector3();
          for (let k = 0; k < cutSegments.length; k++) capCenter.add(cutSegments[k][0]).add(cutSegments[k][1]);
          capCenter.divideScalar(cutSegments.length * 2);
          for (let k = 0; k < cutSegments.length; k++) {
            const a = cutSegments[k][0], b = cutSegments[k][1];
            if (a.distanceToSquared(b) > 1e-7) capTris.push([capCenter.clone(), a, b]);
          }
        }
      }

      if (cellTris.length < 2) continue;

      const shardCenter = new THREE.Vector3();
      let vCnt = 0;
      for (const t of cellTris) { shardCenter.add(t[0]).add(t[1]).add(t[2]); vCnt += 3; }
      shardCenter.divideScalar(vCnt);

      const shardGroup = new THREE.Group();
      shardGroup.name = `${target.name}_Shard_${si + 1}`;
      shardGroup.position.copy(shardCenter);

      function buildGeoFromTris(tris) {
        const arr = new Float32Array(tris.length * 9);
        for (let i = 0; i < tris.length; i++) {
          for (let k = 0; k < 3; k++) {
            const idx = (i * 3 + k) * 3;
            arr[idx] = tris[i][k].x - shardCenter.x;
            arr[idx + 1] = tris[i][k].y - shardCenter.y;
            arr[idx + 2] = tris[i][k].z - shardCenter.z;
          }
        }
        const g = new THREE.BufferGeometry();
        g.setAttribute('position', new THREE.BufferAttribute(arr, 3));
        g.computeVertexNormals();
        return g;
      }

      const outerMesh = new THREE.Mesh(buildGeoFromTris(cellTris), outerMat);
      outerMesh.castShadow = true; outerMesh.receiveShadow = true;
      shardGroup.add(outerMesh);

      if (capTris.length) {
        const innerMesh = new THREE.Mesh(buildGeoFromTris(capTris), innerMat);
        innerMesh.castShadow = true; innerMesh.receiveShadow = true;
        shardGroup.add(innerMesh);
      }

      APP.scene.add(shardGroup);
      const outDir = shardCenter.clone().sub(impactPt);
      if (outDir.lengthSq() < 0.001) outDir.set(Math.random() - 0.5, 0.5, Math.random() - 0.5);
      outDir.normalize();
      const impulseSpd = 0.8 + Math.random() * 1.6;
      const angVel = new THREE.Vector3((Math.random() - 0.5) * 6, (Math.random() - 0.5) * 6, (Math.random() - 0.5) * 6);

      const sEnt = APP.registerEntity(shardGroup, {
        name: shardGroup.name,
        category: 'shard',
        grabbable: true,
        holdable: true,
        hasGravity: true,
        restitution: 0.32,
        velocity: new THREE.Vector3(outDir.x * impulseSpd, Math.abs(outDir.y) * impulseSpd + 0.65, outDir.z * impulseSpd),
        physics: { gravity: true, enabled: true, bounce: 0.32 },
        onUpdate: (dt) => {
          if (sEnt.isHeld || sEnt.heldBy) return;
          const delta = Math.min(0.04, dt || 0.016);
          if (shardGroup.position.y > 0.05) {
            shardGroup.rotation.x += angVel.x * delta;
            shardGroup.rotation.y += angVel.y * delta;
            shardGroup.rotation.z += angVel.z * delta;
          } else {
            angVel.multiplyScalar(0.82);
          }
        }
      });
      shardsSpawned.push(sEnt);
    }

    spawnFractureDustAndDebris(impactPt, size, baseCol);
    APP.removeEntity?.(target, false);
    return { ok: true, shards: shardsSpawned.length, message: `Shattered "${target.name}" into ${shardsSpawned.length} Voronoi shards.` };
  }

  // ====================================================================
  // 2. THERMAL MELT SPHERE (Localized Liquid Sag, Tendril Drip & Pooling)
  // ====================================================================
  let meltSphereEnt = null;
  let meltSphereActive = false;
  const meltStates = new WeakMap();

  function prepareMeltGeometry(mesh) {
    if (meltStates.has(mesh)) return meltStates.get(mesh);
    const geo = mesh.geometry;
    if (!geo?.attributes?.position) return null;
    const pos = geo.attributes.position;
    const orig = pos.array.slice();
    const heat = new Float32Array(pos.count);
    const droopSeed = new Float32Array(pos.count);
    for (let i = 0; i < pos.count; i++) droopSeed[i] = Math.random();
    geo.setAttribute('aMeltHeat', new THREE.BufferAttribute(heat, 1));

    // Patch material for glossy wet molten shading
    const mats = Array.isArray(mesh.material) ? mesh.material : [mesh.material];
    mesh.material = mats.map((m) => {
      if (!m) return m;
      const cl = m.clone();
      cl.customProgramCacheKey = () => 'nxr_melt_v1';
      cl.onBeforeCompile = (shader) => {
        shader.vertexShader = shader.vertexShader
          .replace('#include <common>', '#include <common>\nattribute float aMeltHeat;\nvarying float vMeltHeat;')
          .replace('#include <begin_vertex>', '#include <begin_vertex>\nvMeltHeat = aMeltHeat;');
        shader.fragmentShader = shader.fragmentShader
          .replace('#include <common>', '#include <common>\nvarying float vMeltHeat;')
          .replace('#include <roughnessmap_fragment>', '#include <roughnessmap_fragment>\nroughnessFactor = mix(roughnessFactor, 0.06, clamp(vMeltHeat * 1.3, 0.0, 1.0));')
          .replace('#include <emissivemap_fragment>', '#include <emissivemap_fragment>\nfloat hot = smoothstep(0.15, 0.85, vMeltHeat) * (1.0 - smoothstep(0.85, 1.2, vMeltHeat));\ntotalEmissiveRadiance += vec3(1.0, 0.32, 0.04) * hot * 1.1;');
      };
      cl.needsUpdate = true;
      return cl;
    });
    if (!Array.isArray(mesh.material)) mesh.material = mesh.material[0];

    const st = { orig, heat, droopSeed };
    meltStates.set(mesh, st);
    return st;
  }

  function applyMeltAtWorldPoint(worldPt, radius, dt) {
    const tmpW = new THREE.Vector3(), tmpL = new THREE.Vector3(), box = new THREE.Box3();
    for (const ent of APP.entities || []) {
      if (!isManipulableEntity(ent)) continue;
      ent.group.updateMatrixWorld(true);
      box.setFromObject(ent.group).expandByScalar(radius);
      if (!box.containsPoint(worldPt)) continue;

      const centerW = new THREE.Vector3(); box.getCenter(centerW);
      ent.group.traverse((mesh) => {
        if (!mesh.isMesh || !mesh.geometry?.attributes?.position || mesh.userData?.nxrDecal) return;
        const st = prepareMeltGeometry(mesh);
        if (!st) return;
        const pos = mesh.geometry.attributes.position;
        const heatAttr = mesh.geometry.attributes.aMeltHeat;
        let dirty = false;

        for (let i = 0; i < pos.count; i++) {
          const i3 = i * 3;
          tmpL.set(pos.array[i3], pos.array[i3 + 1], pos.array[i3 + 2]);
          tmpW.copy(tmpL).applyMatrix4(mesh.matrixWorld);
          const d = tmpW.distanceTo(worldPt);
          if (d < radius) {
            const prox = 1.0 - d / radius;
            st.heat[i] = Math.min(1.0, st.heat[i] + prox * dt * 1.85);
          }
          const h = st.heat[i];
          if (h > 0.02) {
            // Viscous liquid sag + tendril lobes + puddle lateral spreading at floor y=0.008
            const ripple = 1.0 + 0.35 * Math.sin(tmpW.x * 24.0 + tmpW.z * 21.0 + st.droopSeed[i] * 6.28);
            const dropRate = h * h * 0.42 * ripple * dt;
            if (tmpW.y > 0.012) {
              tmpW.y = Math.max(0.008, tmpW.y - dropRate);
              // Slight inward pull toward melt axis while falling, then lateral puddle spread near floor!
              const floorProx = 1.0 - THREE.MathUtils.clamp(tmpW.y / 0.14, 0, 1);
              const outX = tmpW.x - centerW.x, outZ = tmpW.z - centerW.z;
              tmpW.x += outX * (floorProx * 0.85 - 0.12) * h * dt;
              tmpW.z += outZ * (floorProx * 0.85 - 0.12) * h * dt;
            } else {
              // Spread puddle horizontally on floor
              const outX = tmpW.x - worldPt.x, outZ = tmpW.z - worldPt.z;
              tmpW.x += outX * 0.75 * h * dt;
              tmpW.z += outZ * 0.75 * h * dt;
              tmpW.y = 0.008 + h * 0.004;
            }
            tmpL.copy(tmpW);
            mesh.worldToLocal(tmpL);
            pos.array[i3] = tmpL.x; pos.array[i3 + 1] = tmpL.y; pos.array[i3 + 2] = tmpL.z;
            dirty = true;
          }
        }
        if (dirty) {
          pos.needsUpdate = true;
          heatAttr.needsUpdate = true;
          mesh.geometry.computeVertexNormals();
          mesh.geometry.computeBoundingBox();
        }
      });
    }
  }

  function toggleMeltSphere(on) {
    meltSphereActive = on !== undefined ? !!on : !meltSphereActive;
    if (meltSphereActive) {
      if (!meltSphereEnt || !meltSphereEnt.group?.parent) {
        const g = new THREE.Group();
        g.name = 'Thermal_Melt_Sphere';
        const core = new THREE.Mesh(
          new THREE.SphereGeometry(0.068, 24, 20),
          new THREE.MeshStandardMaterial({ color: 0xff5500, emissive: 0xff3b00, emissiveIntensity: 2.2, roughness: 0.2 })
        );
        const halo = new THREE.Mesh(
          new THREE.SphereGeometry(0.11, 20, 16),
          new THREE.MeshBasicMaterial({ color: 0xff8833, transparent: true, opacity: 0.22, wireframe: true })
        );
        const light = new THREE.PointLight(0xff5500, 1.8, 2.5);
        g.add(core, halo, light);
        const sp = APP.getSpawnPosition ? APP.getSpawnPosition(0.68, true) : new THREE.Vector3(0, 1.25, -0.68);
        g.position.set(sp.x, 1.25, sp.z);
        APP.scene.add(g);
        meltSphereEnt = APP.registerEntity(g, {
          name: 'Thermal Melt Sphere',
          category: 'tool',
          grabbable: true,
          holdable: true,
          physics: { enabled: false },
          userData: { isManipTool: true },
          onUpdate: (dt) => {
            if (!meltSphereActive) { g.visible = false; return; }
            g.visible = true;
            halo.rotation.y += dt * 1.6;
            applyMeltAtWorldPoint(g.position, manipCfg.meltRadius, Math.min(0.04, dt || 0.016));
          }
        });
        meltSphereEnt.userData = { isManipTool: true };
      } else {
        meltSphereEnt.group.visible = true;
      }
      return { ok: true, message: 'Thermal Melt Sphere active! Move it near any model (or drag on model) to melt it.' };
    } else if (meltSphereEnt?.group) {
      meltSphereEnt.group.visible = false;
      return { ok: true, message: 'Melt Sphere hidden.' };
    }
  }

  // ====================================================================
  // 3. VOLUMETRIC SOFTBODY CONVERTER (Stretch, Poke, Squeeze + Granular Tear)
  // ====================================================================
  const activeSoftbodies = new Set();

  function spawnGranularTearClumps(worldPt, baseCol, count = 36) {
    const grp = new THREE.Group();
    grp.name = 'Softbody_Tear_Granules';
    const geo = new THREE.DodecahedronGeometry(0.016, 0);
    const inst = new THREE.InstancedMesh(geo, new THREE.MeshStandardMaterial({ color: baseCol, roughness: 0.82 }), count);
    const pos = Array.from({ length: count }, () => worldPt.clone().add(new THREE.Vector3((Math.random() - 0.5) * 0.06, (Math.random() - 0.5) * 0.06, (Math.random() - 0.5) * 0.06)));
    const vel = Array.from({ length: count }, () => new THREE.Vector3((Math.random() - 0.5) * 1.1, 0.3 + Math.random() * 0.9, (Math.random() - 0.5) * 1.1));
    const scale = Array.from({ length: count }, () => 0.6 + Math.random() * 0.9);
    const dummy = new THREE.Object3D();
    grp.add(inst);
    APP.scene.add(grp);

    let age = 0;
    const fxEnt = APP.registerEntity(grp, {
      name: 'Tear Granules',
      category: 'world_fx',
      grabbable: false,
      physics: { enabled: false },
      onUpdate: (dt) => {
        const delta = Math.min(0.04, dt || 0.016);
        age += delta;
        for (let i = 0; i < count; i++) {
          vel[i].y -= 8.4 * delta;
          pos[i].addScaledVector(vel[i], delta);
          if (pos[i].y < 0.015) {
            pos[i].y = 0.015;
            vel[i].y *= -0.25; vel[i].x *= 0.65; vel[i].z *= 0.65;
          }
          dummy.position.copy(pos[i]);
          dummy.rotation.set(i + age * 2, i * 0.7, 0);
          dummy.scale.setScalar(scale[i]);
          dummy.updateMatrix();
          inst.setMatrixAt(i, dummy.matrix);
        }
        inst.instanceMatrix.needsUpdate = true;
        if (age > 7.5) APP.removeEntity?.(fxEnt, false);
      }
    });
  }

  function convertToSoftbody(targetArg) {
    const target = resolveTargetEntity(targetArg);
    if (!target) return { ok: false, error: 'No target object found for Softbody.' };
    if (target.userData?.isSoftbody) return { ok: true, message: `${target.name} is already a Softbody!` };

    target.group.updateMatrixWorld(true);
    const localBox = new THREE.Box3();
    const invRoot = target.group.matrixWorld.clone().invert();
    const meshes = [];
    let sampleCol = new THREE.Color(0x94a3b8);

    target.group.traverse((c) => {
      if (!c.isMesh || !c.geometry?.attributes?.position || c.userData?.nxrDecal) return;
      if (c.material?.color) sampleCol.copy(c.material.color);
      c.geometry.computeBoundingBox();
      const b = c.geometry.boundingBox.clone().applyMatrix4(invRoot.clone().multiply(c.matrixWorld));
      localBox.union(b);
      meshes.push(c);
    });
    if (!meshes.length || localBox.isEmpty()) return { ok: false, error: 'Target has no deformable geometry.' };

    const min = localBox.min, max = localBox.max;
    const span = new THREE.Vector3().subVectors(max, min).max(new THREE.Vector3(0.08, 0.08, 0.08));
    const NX = 5, NY = 5, NZ = 5;
    const N = NX * NY * NZ;
    const cur = [], prev = [], rest = [], tornNode = new Uint8Array(N);

    for (let z = 0; z < NZ; z++) for (let y = 0; y < NY; y++) for (let x = 0; x < NX; x++) {
      const pLocal = new THREE.Vector3(
        min.x + (x / (NX - 1)) * span.x,
        min.y + (y / (NY - 1)) * span.y,
        min.z + (z / (NZ - 1)) * span.z
      );
      rest.push(pLocal.clone());
      const pWorld = pLocal.clone().applyMatrix4(target.group.matrixWorld);
      cur.push(pWorld.clone());
      prev.push(pWorld.clone());
    }

    const springs = [];
    function nodeIdx(x, y, z) { return (z * NY + y) * NX + x; }
    function addSp(a, b, stiff) {
      const d = cur[a].distanceTo(cur[b]);
      if (d > 1e-5) springs.push({ a, b, rest: d, tearDist: d * manipCfg.tearThreshold, stiff, broken: false });
    }
    for (let z = 0; z < NZ; z++) for (let y = 0; y < NY; y++) for (let x = 0; x < NX; x++) {
      const i = nodeIdx(x, y, z);
      if (x < NX - 1) addSp(i, nodeIdx(x + 1, y, z), 0.72);
      if (y < NY - 1) addSp(i, nodeIdx(x, y + 1, z), 0.72);
      if (z < NZ - 1) addSp(i, nodeIdx(x, y, z + 1), 0.72);
      if (x < NX - 1 && y < NY - 1) { addSp(i, nodeIdx(x + 1, y + 1, z), 0.45); addSp(nodeIdx(x + 1, y, z), nodeIdx(x, y + 1, z), 0.45); }
      if (y < NY - 1 && z < NZ - 1) { addSp(i, nodeIdx(x, y + 1, z + 1), 0.45); addSp(nodeIdx(x, y + 1, z), nodeIdx(x, y, z + 1), 0.45); }
      if (x < NX - 1 && z < NZ - 1) { addSp(i, nodeIdx(x + 1, y, z + 1), 0.45); addSp(nodeIdx(x + 1, y, z), nodeIdx(x, y, z + 1), 0.45); }
    }

    // Bind every mesh vertex to its surrounding 8 lattice nodes via trilinear coordinates
    const meshBindings = [];
    const tmpP = new THREE.Vector3();
    for (const m of meshes) {
      const pos = m.geometry.attributes.position;
      const relMat = invRoot.clone().multiply(m.matrixWorld);
      const relInv = relMat.clone().invert();
      const binds = [];
      const tearAttr = new Float32Array(pos.count);
      m.geometry.setAttribute('aSoftTear', new THREE.BufferAttribute(tearAttr, 1));

      const mats = Array.isArray(m.material) ? m.material : [m.material];
      m.material = mats.map((mat) => {
        if (!mat) return mat;
        const cl = mat.clone();
        cl.side = THREE.DoubleSide;
        cl.customProgramCacheKey = () => 'nxr_soft_tear_v1';
        cl.onBeforeCompile = (shader) => {
          shader.vertexShader = shader.vertexShader
            .replace('#include <common>', '#include <common>\nattribute float aSoftTear;\nvarying float vSoftTear;')
            .replace('#include <begin_vertex>', '#include <begin_vertex>\nvSoftTear = aSoftTear;');
          shader.fragmentShader = shader.fragmentShader
            .replace('#include <common>', '#include <common>\nvarying float vSoftTear;')
            .replace('#include <map_fragment>', '#include <map_fragment>\nif (vSoftTear > 0.5) discard;');
        };
        cl.needsUpdate = true;
        return cl;
      });
      if (!Array.isArray(m.material)) m.material = m.material[0];

      for (let i = 0; i < pos.count; i++) {
        tmpP.fromBufferAttribute(pos, i).applyMatrix4(relMat);
        const fx = THREE.MathUtils.clamp(((tmpP.x - min.x) / span.x) * (NX - 1), 0, NX - 1.0001);
        const fy = THREE.MathUtils.clamp(((tmpP.y - min.y) / span.y) * (NY - 1), 0, NY - 1.0001);
        const fz = THREE.MathUtils.clamp(((tmpP.z - min.z) / span.z) * (NZ - 1), 0, NZ - 1.0001);
        const ix = Math.floor(fx), iy = Math.floor(fy), iz = Math.floor(fz);
        binds.push({ ix, iy, iz, tx: fx - ix, ty: fy - iy, tz: fz - iz });
      }
      meshBindings.push({ mesh: m, relInv, binds, tearAttr });
    }

    // Reset target group transform to world identity so lattice nodes drive world positions cleanly
    target.group.position.set(0, 0, 0);
    target.group.rotation.set(0, 0, 0);
    target.group.scale.set(1, 1, 1);
    target.group.updateMatrixWorld(true);
    for (const mb of meshBindings) { mb.relInv.copy(mb.mesh.matrixWorld).invert(); }

    target.userData = target.userData || {};
    target.userData.isSoftbody = true;
    target.grabbable = false; // Handled directly by softbody pinch/stretch!
    if (target.physics) target.physics.enabled = false;

    const grabbedByHand = new Map();
    const tmpDelta = new THREE.Vector3(), tmpW = new THREE.Vector3(), tmpL = new THREE.Vector3();
    const c000 = new THREE.Vector3(), c100 = new THREE.Vector3(), c010 = new THREE.Vector3(), c110 = new THREE.Vector3();
    const c001 = new THREE.Vector3(), c101 = new THREE.Vector3(), c011 = new THREE.Vector3(), c111 = new THREE.Vector3();

    const sbRecord = {
      entity: target,
      applyPointerPull: (worldPt, worldVel, isDown) => {
        if (!isDown) return;
        for (let i = 0; i < N; i++) {
          const d = cur[i].distanceTo(worldPt);
          if (d < 0.20) {
            const w = 1.0 - d / 0.20;
            cur[i].addScaledVector(worldVel, w * 1.15);
          }
        }
      },
      step: (dt) => {
        if (!target.group?.parent) { activeSoftbodies.delete(sbRecord); return; }
        const safeDt = Math.min(0.025, Math.max(0.004, dt || 0.016));
        const grav = -7.2 * safeDt * safeDt;

        // 1. VR Hand Poke / Squeeze / Two-Hand Stretch!
        for (const [hk, h] of Object.entries(APP.hands || {})) {
          if (!h || !h.active) { grabbedByHand.delete(hk); continue; }
          const tip = h.isHand ? h.fingerTips?.index : h.pos;
          if (!tip) continue;
          const pinching = !!(h.pinching || h.trigger || h.gripping);
          if (pinching) {
            if (!grabbedByHand.has(hk)) {
              const caught = [];
              for (let i = 0; i < N; i++) if (cur[i].distanceTo(tip) < 0.16) caught.push({ idx: i, off: cur[i].clone().sub(tip) });
              if (caught.length) grabbedByHand.set(hk, caught);
            } else {
              for (const c of grabbedByHand.get(hk)) {
                cur[c.idx].lerp(tmpW.copy(tip).add(c.off), 0.78);
                prev[c.idx].copy(cur[c.idx]);
              }
            }
          } else {
            grabbedByHand.delete(hk);
            // Poke / Squeeze collision
            for (let i = 0; i < N; i++) {
              tmpDelta.subVectors(cur[i], tip);
              const d = tmpDelta.length();
              if (d < 0.11 && d > 1e-4) {
                cur[i].addScaledVector(tmpDelta.normalize(), 0.11 - d);
              }
            }
          }
        }

        // 2. Verlet integration + floor collision
        for (let i = 0; i < N; i++) {
          const p = cur[i], pr = prev[i];
          const vx = (p.x - pr.x) * 0.975, vy = (p.y - pr.y) * 0.975, vz = (p.z - pr.z) * 0.975;
          pr.copy(p);
          p.x += vx; p.y += vy + grav; p.z += vz;
          if (p.y < 0.02) { p.y = 0.02; pr.x = THREE.MathUtils.lerp(pr.x, p.x, 0.25); pr.z = THREE.MathUtils.lerp(pr.z, p.z, 0.25); }
        }

        // 3. Spring constraints + Over-Stretch Granular Tearing!
        let toreHere = null;
        for (let iter = 0; iter < 5; iter++) {
          for (let i = 0; i < springs.length; i++) {
            const s = springs[i]; if (s.broken) continue;
            const pa = cur[s.a], pb = cur[s.b];
            tmpDelta.subVectors(pb, pa);
            const d = tmpDelta.length();
            if (d < 1e-6) continue;
            if (iter === 0 && d > s.tearDist) {
              s.broken = true;
              tornNode[s.a] = 1; tornNode[s.b] = 1;
              if (!toreHere) toreHere = pa.clone().add(pb).multiplyScalar(0.5);
              continue;
            }
            const diff = ((d - s.rest) / d) * s.stiff;
            pa.addScaledVector(tmpDelta, diff * 0.5);
            pb.addScaledVector(tmpDelta, -diff * 0.5);
          }
        }
        if (toreHere) {
          spawnGranularTearClumps(toreHere, sampleCol, 34);
          APP.ui?.toast?.('Softbody tore into granular chunks!', 1800);
        }

        // 4. Trilinear Lattice-to-Mesh Vertex Deformation
        for (const mb of meshBindings) {
          const pos = mb.mesh.geometry.attributes.position;
          const tearA = mb.mesh.geometry.attributes.aSoftTear;
          let tearDirty = false;
          for (let i = 0; i < mb.binds.length; i++) {
            const b = mb.binds[i];
            const i000 = nodeIdx(b.ix, b.iy, b.iz), i100 = nodeIdx(b.ix + 1, b.iy, b.iz);
            const i010 = nodeIdx(b.ix, b.iy + 1, b.iz), i110 = nodeIdx(b.ix + 1, b.iy + 1, b.iz);
            const i001 = nodeIdx(b.ix, b.iy, b.iz + 1), i101 = nodeIdx(b.ix + 1, b.iy, b.iz + 1);
            const i011 = nodeIdx(b.ix, b.iy + 1, b.iz + 1), i111 = nodeIdx(b.ix + 1, b.iy + 1, b.iz + 1);

            if (tornNode[i000] && tornNode[i111] && mb.tearAttr[i] < 0.5) {
              mb.tearAttr[i] = 1.0; tearDirty = true;
            }

            c000.copy(cur[i000]).lerp(cur[i100], b.tx);
            c010.copy(cur[i010]).lerp(cur[i110], b.tx);
            c001.copy(cur[i001]).lerp(cur[i101], b.tx);
            c011.copy(cur[i011]).lerp(cur[i111], b.tx);
            c100.copy(c000).lerp(c010, b.ty);
            c101.copy(c001).lerp(c011, b.ty);
            tmpW.copy(c100).lerp(c101, b.tz);
            tmpL.copy(tmpW).applyMatrix4(mb.relInv);
            pos.setXYZ(i, tmpL.x, tmpL.y, tmpL.z);
          }
          pos.needsUpdate = true;
          if (tearDirty) tearA.needsUpdate = true;
          mb.mesh.geometry.computeVertexNormals();
          mb.mesh.geometry.computeBoundingBox();
        }
      }
    };

    activeSoftbodies.add(sbRecord);
    return { ok: true, id: target.id, message: `Converted "${target.name}" into a tearable Softbody! Poke, squeeze, or stretch to tear.` };
  }

  // ====================================================================
  // 4. SMOKY HOLOGRAM CLOUD (Hand-Dispersing Vortex Mist)
  // ====================================================================
  function convertToSmokeHologram(targetArg) {
    const target = resolveTargetEntity(targetArg);
    if (!target) return { ok: false, error: 'No target object found for Smoky Hologram.' };

    target.group.updateMatrixWorld(true);
    const tris = [];
    let totalArea = 0;
    const v0 = new THREE.Vector3(), v1 = new THREE.Vector3(), v2 = new THREE.Vector3(), e1 = new THREE.Vector3(), e2 = new THREE.Vector3();

    target.group.traverse((c) => {
      if (!c.isMesh || !c.geometry?.attributes?.position || c.userData?.nxrDecal) return;
      const geo = c.geometry, pos = geo.attributes.position, idx = geo.index;
      const mat = Array.isArray(c.material) ? c.material[0] : c.material;
      const col = mat?.color ? mat.color.clone() : new THREE.Color(0x94a3b8);
      const tCnt = idx ? Math.floor(idx.count / 3) : Math.floor(pos.count / 3);
      for (let t = 0; t < tCnt; t++) {
        const i0 = idx ? idx.getX(t * 3) : t * 3, i1 = idx ? idx.getX(t * 3 + 1) : t * 3 + 1, i2 = idx ? idx.getX(t * 3 + 2) : t * 3 + 2;
        v0.fromBufferAttribute(pos, i0).applyMatrix4(c.matrixWorld);
        v1.fromBufferAttribute(pos, i1).applyMatrix4(c.matrixWorld);
        v2.fromBufferAttribute(pos, i2).applyMatrix4(c.matrixWorld);
        const area = e1.subVectors(v1, v0).cross(e2.subVectors(v2, v0)).length() * 0.5;
        if (area > 1e-8) { totalArea += area; tris.push({ a: v0.clone(), b: v1.clone(), c: v2.clone(), cum: totalArea, col }); }
      }
    });

    if (!tris.length) return { ok: false, error: 'Could not sample target geometry.' };

    const P_COUNT = 3800;
    const restPos = new Float32Array(P_COUNT * 3);
    const posArr = new Float32Array(P_COUNT * 3);
    const velArr = new Float32Array(P_COUNT * 3);
    const colArr = new Float32Array(P_COUNT * 3);
    const seedArr = new Float32Array(P_COUNT);
    const sizeArr = new Float32Array(P_COUNT);

    for (let i = 0; i < P_COUNT; i++) {
      const r = Math.random() * totalArea;
      let lo = 0, hi = tris.length - 1;
      while (lo < hi) { const mid = (lo + hi) >> 1; if (tris[mid].cum < r) lo = mid + 1; else hi = mid; }
      const tr = tris[lo];
      let u = Math.random(), v = Math.random();
      if (u + v > 1) { u = 1 - u; v = 1 - v; }
      const w = 1 - u - v;
      const px = tr.a.x * w + tr.b.x * u + tr.c.x * v + (Math.random() - 0.5) * 0.018;
      const py = tr.a.y * w + tr.b.y * u + tr.c.y * v + (Math.random() - 0.5) * 0.018;
      const pz = tr.a.z * w + tr.b.z * u + tr.c.z * v + (Math.random() - 0.5) * 0.018;
      restPos[i * 3] = px; restPos[i * 3 + 1] = py; restPos[i * 3 + 2] = pz;
      posArr[i * 3] = px; posArr[i * 3 + 1] = py; posArr[i * 3 + 2] = pz;

      const cMix = tr.col.clone().lerp(new THREE.Color(0xcbd5e1), 0.45);
      colArr[i * 3] = cMix.r; colArr[i * 3 + 1] = cMix.g; colArr[i * 3 + 2] = cMix.b;
      seedArr[i] = Math.random() * 100.0;
      sizeArr[i] = 28.0 + Math.random() * 26.0;
    }

    const smokeGeo = new THREE.BufferGeometry();
    smokeGeo.setAttribute('position', new THREE.BufferAttribute(posArr, 3));
    smokeGeo.setAttribute('aCol', new THREE.BufferAttribute(colArr, 3));
    smokeGeo.setAttribute('aSize', new THREE.BufferAttribute(sizeArr, 1));
    smokeGeo.setAttribute('aSeed', new THREE.BufferAttribute(seedArr, 1));

    const smokeMat = new THREE.ShaderMaterial({
      transparent: true,
      depthWrite: false,
      uniforms: { uTime: { value: 0 }, uPx: { value: window.devicePixelRatio || 1 } },
      vertexShader: [
        'attribute vec3 aCol; attribute float aSize; attribute float aSeed; uniform float uPx;',
        'varying vec3 vCol; varying float vSeed;',
        'void main(){ vCol=aCol; vSeed=aSeed; vec4 mv=modelViewMatrix*vec4(position,1.0); gl_PointSize=clamp((aSize*uPx)/max(0.3,-mv.z), 3.0, 42.0); gl_Position=projectionMatrix*mv; }'
      ].join('\n'),
      fragmentShader: [
        'uniform float uTime; varying vec3 vCol; varying float vSeed;',
        'float sHash(vec2 p){ p=fract(p*vec2(127.1,311.7)); p+=dot(p,p+34.5); return fract(p.x*p.y); }',
        'float sNoise(vec2 p){ vec2 i=floor(p); vec2 f=fract(p); f=f*f*(3.0-2.0*f); return mix(mix(sHash(i),sHash(i+vec2(1,0)),f.x),mix(sHash(i+vec2(0,1)),sHash(i+vec2(1,1)),f.x),f.y); }',
        'void main(){',
        '  vec2 uv = gl_PointCoord - 0.5;',
        '  float r = length(uv) * 2.0;',
        '  if (r > 1.0) discard;',
        '  float wisp = sNoise(uv * 3.8 + vec2(vSeed, uTime * 0.6));',
        '  float alpha = pow(1.0 - r, 1.9) * (0.14 + 0.18 * wisp);',
        '  gl_FragColor = vec4(vCol, alpha);',
        '}'
      ].join('\n')
    });

    const smokeGroup = new THREE.Group();
    smokeGroup.name = `${target.name}_Smoky_Hologram`;
    const pts = new THREE.Points(smokeGeo, smokeMat);
    pts.frustumCulled = false;
    smokeGroup.add(pts);
    APP.scene.add(smokeGroup);

    // Remove solid target cleanly
    const origName = target.name;
    APP.removeEntity?.(target, false);

    const prevHands = new Map(), hVel = new THREE.Vector3(), tmpD = new THREE.Vector3(), swirl = new THREE.Vector3();

    const sEnt = APP.registerEntity(smokeGroup, {
      name: `${origName} (Smoke Cloud)`,
      category: 'smoke_holo',
      grabbable: false,
      physics: { enabled: false },
      onUpdate: (dt, elapsed) => {
        const delta = Math.min(0.035, Math.max(0.004, dt || 0.016));
        const now = elapsed || performance.now() * 0.001;
        smokeMat.uniforms.uTime.value = now;

        // Collect active disturbers (VR hands/controllers + desktop pointer!)
        const disturbers = [];
        for (const [hk, h] of Object.entries(APP.hands || {})) {
          if (!h || !h.active) continue;
          const hp = h.isHand ? h.fingerTips?.index : h.pos;
          if (!hp) continue;
          const last = prevHands.get(hk);
          if (last) hVel.subVectors(hp, last).divideScalar(delta);
          else hVel.set(0, 0, 0);
          prevHands.set(hk, hp.clone());
          disturbers.push({ pos: hp.clone(), vel: hVel.clone() });
        }
        if (lastMouseSmokePt && now - lastMouseSmokeTime < 0.2) {
          disturbers.push({ pos: lastMouseSmokePt.clone(), vel: lastMouseSmokeVel.clone() });
        }

        const rSq = 0.28 * 0.28;
        for (let i = 0; i < P_COUNT; i++) {
          const i3 = i * 3;
          const px = posArr[i3], py = posArr[i3 + 1], pz = posArr[i3 + 2];
          const rx = restPos[i3], ry = restPos[i3 + 1], rz = restPos[i3 + 2];

          // Gentle turbulent simmering around rest silhouette
          const turbX = Math.sin(now * 1.4 + seedArr[i] + py * 5.0) * 0.012;
          const turbY = Math.cos(now * 1.2 + seedArr[i] * 1.3 + px * 5.0) * 0.012;
          const turbZ = Math.sin(now * 1.3 + seedArr[i] * 0.7 + pz * 5.0) * 0.012;

          velArr[i3] += ((rx + turbX) - px) * 2.2 * delta;
          velArr[i3 + 1] += ((ry + turbY) - py) * 2.2 * delta;
          velArr[i3 + 2] += ((rz + turbZ) - pz) * 2.2 * delta;

          for (let d = 0; d < disturbers.length; d++) {
            const hb = disturbers[d];
            const dx = px - hb.pos.x, dy = py - hb.pos.y, dz = pz - hb.pos.z;
            const distSq = dx * dx + dy * dy + dz * dz;
            if (distSq < rSq && distSq > 1e-6) {
              const dist = Math.sqrt(distSq);
              const f = Math.pow(1.0 - dist / 0.28, 1.5);
              tmpD.set(dx / dist, dy / dist, dz / dist);
              swirl.crossVectors(tmpD, hb.vel);
              const pushStr = 4.2 + hb.vel.length() * 2.4;
              velArr[i3] += (tmpD.x * pushStr + hb.vel.x * 1.4 + swirl.x * 2.8) * f * delta * 8.0;
              velArr[i3 + 1] += (tmpD.y * pushStr + hb.vel.y * 1.4 + swirl.y * 2.8 + 0.6) * f * delta * 8.0;
              velArr[i3 + 2] += (tmpD.z * pushStr + hb.vel.z * 1.4 + swirl.z * 2.8) * f * delta * 8.0;
            }
          }

          velArr[i3] *= 0.92; velArr[i3 + 1] *= 0.92; velArr[i3 + 2] *= 0.92;
          posArr[i3] += velArr[i3] * delta;
          posArr[i3 + 1] = Math.max(0.02, posArr[i3 + 1] + velArr[i3 + 1] * delta);
          posArr[i3 + 2] += velArr[i3 + 2] * delta;
        }
        smokeGeo.attributes.position.needsUpdate = true;
      }
    });

    APP.lastEntityId = sEnt.id;
    return { ok: true, id: sEnt.id, message: `Transformed "${origName}" into a Smoky Hologram! Pass your hand through it to disperse.` };
  }

  const activeMerges = new Map(); // key: `${idA}_${idB}` -> mergeState

  function createSymbioteTendrils(pairKey, colA, colB) {
    const grp = new THREE.Group();
    grp.name = 'Venom_Merge_Tendrils_' + pairKey;
    const mat = new THREE.MeshPhysicalMaterial({
      color: colA.clone().lerp(colB, 0.5).multiplyScalar(0.35),
      roughness: 0.08,
      metalness: 0.25,
      clearcoat: 1.0,
      clearcoatRoughness: 0.05
    });
    const TENDRIL_COUNT = 8, SEGS = 14, RAD_SEGS = 6;
    const tendrils = [];
    for (let i = 0; i < TENDRIL_COUNT; i++) {
      const geo = new THREE.CylinderGeometry(1, 1, 1, RAD_SEGS, SEGS, false);
      const basePos = geo.attributes.position.array.slice();
      const mesh = new THREE.Mesh(geo, mat);
      mesh.castShadow = true;
      grp.add(mesh);
      tendrils.push({
        mesh, geo, basePos,
        offA: new THREE.Vector3((Math.random() - 0.5) * 0.16, (Math.random() - 0.5) * 0.16, (Math.random() - 0.5) * 0.16),
        offB: new THREE.Vector3((Math.random() - 0.5) * 0.16, (Math.random() - 0.5) * 0.16, (Math.random() - 0.5) * 0.16),
        phase: Math.random() * 6.28
      });
    }
    APP.scene.add(grp);
    return { grp, mat, tendrils, SEGS, RAD_SEGS };
  }

  function updateTendrilTubes(rig, pA, pB, prog, now) {
    const tmpP0 = new THREE.Vector3(), tmpP1 = new THREE.Vector3(), tmpT = new THREE.Vector3(), tmpN = new THREE.Vector3(), tmpB = new THREE.Vector3();
    const sA = new THREE.Vector3(), sB = new THREE.Vector3(), mid = new THREE.Vector3();

    for (const td of rig.tendrils) {
      sA.copy(pA).add(td.offA);
      sB.copy(pB).add(td.offB);
      mid.addVectors(sA, sB).multiplyScalar(0.5);
      mid.x += Math.sin(now * 5.2 + td.phase) * 0.04 * (1 - prog * 0.5);
      mid.y += Math.cos(now * 4.6 + td.phase) * 0.04 * (1 - prog * 0.5);
      mid.z += Math.sin(now * 3.8 + td.phase * 1.4) * 0.04 * (1 - prog * 0.5);

      const posA = td.geo.attributes.position;
      const thick = (0.011 + prog * 0.014);

      for (let iy = 0; iy <= rig.SEGS; iy++) {
        const t = iy / rig.SEGS;
        const u = 1 - t;
        tmpP0.set(
          u * u * sA.x + 2 * u * t * mid.x + t * t * sB.x,
          u * u * sA.y + 2 * u * t * mid.y + t * t * sB.y,
          u * u * sA.z + 2 * u * t * mid.z + t * t * sB.z
        );
        const t2 = Math.min(1, t + 0.03), u2 = 1 - t2;
        tmpP1.set(
          u2 * u2 * sA.x + 2 * u2 * t2 * mid.x + t2 * t2 * sB.x,
          u2 * u2 * sA.y + 2 * u2 * t2 * mid.y + t2 * t2 * sB.y,
          u2 * u2 * sA.z + 2 * u2 * t2 * mid.z + t2 * t2 * sB.z
        );
        tmpT.subVectors(tmpP1, tmpP0).normalize();
        if (tmpT.lengthSq() < 0.001) tmpT.set(0, 1, 0);
        tmpN.set(0, 0, 1).cross(tmpT);
        if (tmpN.lengthSq() < 0.001) tmpN.set(1, 0, 0).cross(tmpT);
        tmpN.normalize();
        tmpB.crossVectors(tmpT, tmpN).normalize();

        // Thick root flares at A and B, pulsing liquid neck in middle!
        const r = thick * (0.45 + 0.9 * Math.pow(Math.abs(t - 0.5) * 2.0, 1.8) + Math.sin(t * 14 - now * 8 + td.phase) * 0.14);
        for (let ir = 0; ir <= rig.RAD_SEGS; ir++) {
          const idx = (iy * (rig.RAD_SEGS + 1) + ir) * 3;
          const bx = td.basePos[idx], bz = td.basePos[idx + 2];
          posA.array[idx] = tmpP0.x + (tmpN.x * bx + tmpB.x * bz) * r;
          posA.array[idx + 1] = tmpP0.y + (tmpN.y * bx + tmpB.y * bz) * r;
          posA.array[idx + 2] = tmpP0.z + (tmpN.z * bx + tmpB.z * bz) * r;
        }
      }
      posA.needsUpdate = true;
      td.geo.computeVertexNormals();
    }
  }

  function stepMagicMerge(dt, now) {
    if (!magicMergeEnabled) return;
    const list = (APP.entities || []).filter(isManipulableEntity);
    const boxA = new THREE.Box3(), boxB = new THREE.Box3(), cA = new THREE.Vector3(), cB = new THREE.Vector3(), szA = new THREE.Vector3(), szB = new THREE.Vector3();
    const closestA = new THREE.Vector3(), closestB = new THREE.Vector3();

    // Detect close pairs
    for (let i = 0; i < list.length; i++) {
      const eA = list[i];
      eA.group.updateMatrixWorld(true);
      boxA.setFromObject(eA.group);
      if (boxA.isEmpty()) continue;
      boxA.getCenter(cA); boxA.getSize(szA);

      for (let j = i + 1; j < list.length; j++) {
        const eB = list[j];
        const key = `${eA.id}_${eB.id}`;
        if (activeMerges.has(key)) continue;

        eB.group.updateMatrixWorld(true);
        boxB.setFromObject(eB.group);
        if (boxB.isEmpty()) continue;
        boxB.getCenter(cB); boxB.getSize(szB);

        boxA.clampPoint(cB, closestA);
        boxB.clampPoint(cA, closestB);
        const gap = closestA.distanceTo(closestB);
        if (gap < 0.24) {
          // Determine host (larger or stationary) and guest (smaller or held)
          const volA = szA.x * szA.y * szA.z, volB = szB.x * szB.y * szB.z;
          const host = (eB.isHeld || volA >= volB) ? eA : eB;
          const guest = host === eA ? eB : eA;
          const colA = new THREE.Color(0x111827), colB = new THREE.Color(0x1e293b);
          host.group.traverse((m) => { if (m.isMesh && m.material?.color) colA.copy(m.material.color); });
          guest.group.traverse((m) => { if (m.isMesh && m.material?.color) colB.copy(m.material.color); });
          activeMerges.set(key, {
            host, guest, progress: 0.0,
            rig: createSymbioteTendrils(key, colA, colB),
            guestOrigPos: guest.group.position.clone()
          });
        }
      }
    }

    // Step active Venom merges
    for (const [key, m] of activeMerges.entries()) {
      if (!m.host.group?.parent || !m.guest.group?.parent) {
        if (m.rig.grp.parent) m.rig.grp.parent.remove(m.rig.grp);
        activeMerges.delete(key);
        continue;
      }

      m.host.group.updateMatrixWorld(true);
      m.guest.group.updateMatrixWorld(true);
      boxA.setFromObject(m.host.group); boxB.setFromObject(m.guest.group);
      boxA.getCenter(cA); boxB.getCenter(cB);
      boxA.clampPoint(cB, closestA); boxB.clampPoint(cA, closestB);

      const gap = closestA.distanceTo(closestB);
      if (gap > 0.45 && m.progress < 0.35) {
        if (m.rig.grp.parent) m.rig.grp.parent.remove(m.rig.grp);
        activeMerges.delete(key);
        continue;
      }

      m.progress = Math.min(1.0, m.progress + dt * 0.48);

      // Pull guest smoothly into host as symbiote swallows it!
      if (!m.guest.isHeld) {
        m.guest.group.position.lerp(cA, dt * (1.2 + m.progress * 3.5));
      }

      // Organic liquid vertex distortion on closer facing parts
      const pullDist = 0.28 + m.progress * 0.25;
      const tmpW = new THREE.Vector3(), tmpL = new THREE.Vector3();
      [ [m.host, cB], [m.guest, cA] ].forEach(([ent, otherC]) => {
        ent.group.traverse((mesh) => {
          if (!mesh.isMesh || !mesh.geometry?.attributes?.position) return;
          const pos = mesh.geometry.attributes.position;
          let dirty = false;
          for (let k = 0; k < pos.count; k++) {
            tmpL.fromBufferAttribute(pos, k);
            tmpW.copy(tmpL).applyMatrix4(mesh.matrixWorld);
            const d = tmpW.distanceTo(otherC);
            if (d < pullDist) {
              const f = (1.0 - d / pullDist) * m.progress * dt * 1.4;
              tmpW.lerp(otherC, Math.min(0.18, f));
              tmpL.copy(tmpW);
              mesh.worldToLocal(tmpL);
              pos.setXYZ(k, tmpL.x, tmpL.y, tmpL.z);
              dirty = true;
            }
          }
          if (dirty) { pos.needsUpdate = true; mesh.geometry.computeVertexNormals(); }
        });
      });

      updateTendrilTubes(m.rig, closestA, closestB, m.progress, now);

      if (m.progress >= 1.0) {
        // Complete Fusion into ONE unified object!
        const host = m.host, guest = m.guest;
        const moveChildren = [...guest.group.children];
        for (const ch of moveChildren) host.group.attach(ch);
        if (m.rig.grp.parent) m.rig.grp.parent.remove(m.rig.grp);
        APP.removeEntity?.(guest, false);
        host.name = `${host.name} + ${guest.name}`;
        APP.lastEntityId = host.id;
        activeMerges.delete(key);
        APP.ui?.toast?.(`Magic-Merged into "${host.name}"!`, 2600);
      }
    }
  }

  function triggerPairMerge(targetA, targetB) {
    magicMergeEnabled = true;
    const eA = resolveTargetEntity(targetA);
    const eB = targetB ? resolveTargetEntity(targetB) : (APP.entities || []).filter((e) => isManipulableEntity(e) && e !== eA).pop();
    if (!eA || !eB || eA === eB) {
      syncUI();
      return { ok: true, message: 'Magic-Merge ON! Move any two objects close together to watch them Venom-merge.' };
    }
    const key = `${eA.id}_${eB.id}`;
    if (!activeMerges.has(key)) {
      const colA = new THREE.Color(0x111827), colB = new THREE.Color(0x1e293b);
      activeMerges.set(key, { host: eA, guest: eB, progress: 0.05, rig: createSymbioteTendrils(key, colA, colB) });
    }
    syncUI();
    return { ok: true, message: `Magic-Merging "${eA.name}" & "${eB.name}"!` };
  }

    function placeNearUser() {

                newHov = z.id; if (raw.justTrig || raw.justPinch) triggerZone(z, hand); break;
              }
            }
          }
          const tip = hand.isHand ? hand.fingerTips?.index : null;
          if (tip) {
            dragBar.getWorldPosition(tmpVec);
            if (tip.distanceTo(tmpVec) < 0.07 && raw.justPinch) { dragHandKey = k; dragOffset.copy(root.position).sub(tip); break; }

              }
            }
          }
        }
      }
      cursorDot.visible = showCur && !dragHandKey;
      if (newHov !== hoveredId) { hoveredId = newHov; redraw(); }
    }

    return { root, screen, dragBar, redraw, update, show: (val) => { root.visible = !!val; if (root.visible) { placeNearUser(); redraw(); } } };

    domToolbar.querySelectorAll('[data-m]').forEach((b) => {
      const m = b.dataset.m;
      const on = (m === 'melt' && meltSphereActive) || (m === 'merge' && magicMergeEnabled) || (m === manipCfg.mode);
      b.classList.toggle('active', on);
    });
    const hint = domToolbar.querySelector('#nxr-m-hint');
    if (hint) {
      hint.textContent = manipCfg.mode === 'voronoi' ? 'Click or punch any model to Voronoi shatter it!'
        : meltSphereActive ? 'Drag Thermal Sphere (or click/drag model) to melt!'
        : manipCfg.mode === 'softbody' ? 'Drag/stretch Softbody in VR or mouse until it tears!'
        : manipCfg.mode === 'smoke' ? 'Sweep hand or cursor through Smoke Cloud!'
        : magicMergeEnabled ? 'Move two models close together for Venom Magic-Merge!'
        : 'Pick a mode above to manipulate any model or GLB.';
    }
  }

  function setMode(m) {
    manipCfg.mode = String(m || 'off').toLowerCase();
    if (manipCfg.mode === 'melt') toggleMeltSphere(true);
    else if (manipCfg.mode === 'off') { toggleMeltSphere(false); magicMergeEnabled = false; }
    if (manipCfg.mode === 'merge') magicMergeEnabled = true;
    syncUI();
  }

  function showToolbar(show = true) {
    const el = ensureDomToolbar();
    el.style.display = show ? 'block' : 'none';
    vrPanel.show(show);
    syncUI();
  }

  // Desktop Mouse Raycast Interaction for Manipulations
  const dRay = new THREE.Raycaster(), dNdc = new THREE.Vector2();
  let dDown = false, dLastPt = null, dSoftTarget = null, dSoftPlane = new THREE.Plane();

  function getPointerHits(e) {
    if (!APP.camera) return [];
    const el = APP.renderer?.domElement;
    const rect = el ? el.getBoundingClientRect() : { left: 0, top: 0, width: window.innerWidth, height: window.innerHeight };
    dNdc.x = ((e.clientX - rect.left) / Math.max(1, rect.width)) * 2 - 1;
    dNdc.y = -((e.clientY - rect.top) / Math.max(1, rect.height)) * 2 + 1;
    dRay.setFromCamera(dNdc, APP.camera);
    const meshes = [];
    for (const ent of APP.entities || []) {
      if (isManipulableEntity(ent)) ent.group.traverse((c) => { if (c.isMesh) meshes.push({ mesh: c, ent }); });
    }
    const hits = dRay.intersectObjects(meshes.map((x) => x.mesh), false);
    if (!hits.length) return [];
    const match = meshes.find((x) => x.mesh === hits[0].object);
    return match ? [{ point: hits[0].point, ent: match.ent }] : [];
  }

  function onPtrDown(e) {
    if (e.button !== 0) return;
    dDown = true;
    const hits = getPointerHits(e);
    if (!hits.length) return;
    const { point, ent } = hits[0];
    dLastPt = point.clone();
    if (manipCfg.mode === 'voronoi') {
      fractureEntity(ent, point);
    } else if (manipCfg.mode === 'melt' || meltSphereActive) {
      applyMeltAtWorldPoint(point, manipCfg.meltRadius, 0.08);
    } else if (ent.userData?.isSoftbody) {
      for (const sb of activeSoftbodies) if (sb.entity === ent) {
        dSoftTarget = sb;
        dSoftPlane.setFromNormalAndCoplanarPoint(APP.camera.getWorldDirection(new THREE.Vector3()).negate(), point);
      }
    }
  }

  function onPtrMove(e) {
    const t = performance.now() * 0.001;
    if (APP.camera) {
      const el = APP.renderer?.domElement;
      const rect = el ? el.getBoundingClientRect() : { left: 0, top: 0, width: window.innerWidth, height: window.innerHeight };
      dNdc.x = ((e.clientX - rect.left) / Math.max(1, rect.width)) * 2 - 1;
      dNdc.y = -((e.clientY - rect.top) / Math.max(1, rect.height)) * 2 + 1;
      dRay.setFromCamera(dNdc, APP.camera);

      // Smoke cloud cursor parting
      const smokeEnts = (APP.entities || []).filter((en) => en && en.category === 'smoke_holo' && en.group?.parent);
      if (smokeEnts.length) {
        const plane = new THREE.Plane().setFromNormalAndCoplanarPoint(APP.camera.getWorldDirection(new THREE.Vector3()).negate(), smokeEnts[0].group.position);
        const pt = new THREE.Vector3();
        if (dRay.ray.intersectPlane(plane, pt)) {
          if (lastMouseSmokePt) lastMouseSmokeVel.copy(pt).sub(lastMouseSmokePt).multiplyScalar(30.0);
          lastMouseSmokePt = pt.clone(); lastMouseSmokeTime = t;
        }
      }
    }

    if (!dDown) return;
    if (dSoftTarget && APP.camera) {
      const pt = new THREE.Vector3();
      if (dRay.ray.intersectPlane(dSoftPlane, pt)) {
        const vel = dLastPt ? pt.clone().sub(dLastPt) : new THREE.Vector3();
        dLastPt = pt.clone();
        dSoftTarget.applyPointerPull(pt, vel, true);
      }
      return;
    }
    if (manipCfg.mode === 'melt' || meltSphereActive) {
      const hits = getPointerHits(e);
      if (hits.length) applyMeltAtWorldPoint(hits[0].point, manipCfg.meltRadius, 0.045);
    }
  }

  function onPtrUp() { dDown = false; dSoftTarget = null; dLastPt = null; }

  if (typeof window !== 'undefined') {
    window.addEventListener('pointerdown', onPtrDown, { passive: true });
    window.addEventListener('pointermove', onPtrMove, { passive: true });
    window.addEventListener('pointerup', onPtrUp, { passive: true });
  }

  // Per-frame Update: VR Hand Voronoi Punch + Softbodies + Magic-Merge + VR Panel
  const prevVRHands = new Map(), tmpHVel = new THREE.Vector3(), tmpBoxCheck = new THREE.Box3();
  let voronoiCooldown = 0;

  function update(dt, elapsed) {
    const delta = Math.min(0.04, Math.max(0.004, dt || 0.016));
    const now = elapsed || performance.now() * 0.001;
    voronoiCooldown = Math.max(0, voronoiCooldown - delta);

    vrPanel.update();

    for (const sb of activeSoftbodies) sb.step(delta);
    stepMagicMerge(delta, now);

    // VR Hand Triggers for Voronoi Shatter & Softbody Touch
    if (manipCfg.mode === 'voronoi' && voronoiCooldown <= 0) {
      for (const [hk, h] of Object.entries(APP.hands || {})) {
        if (!h || !h.active) continue;
        const hp = h.isHand ? h.fingerTips?.index : h.pos;
        if (!hp) continue;
        const last = prevVRHands.get(hk);
        const spd = last ? tmpHVel.subVectors(hp, last).length() / delta : 0;
        prevVRHands.set(hk, hp.clone());
        if (spd > 0.25 || h.trigger || h.pinching) {
          for (const ent of [...(APP.entities || [])]) {
            if (!isManipulableEntity(ent)) continue;
            tmpBoxCheck.setFromObject(ent.group).expandByScalar(0.05);
            if (tmpBoxCheck.containsPoint(hp)) {
              voronoiCooldown = 0.65;
              fractureEntity(ent, hp);
              break;
            }
          }
        }
      }
    }
  }

  const unsubFrame = APP.onUpdate ? APP.onUpdate(update) : null;
  if (!unsubFrame && APP.frameHooks) APP.frameHooks.add(update);

  APP._inst4Cleanup = function () {
    if (typeof unsubFrame === 'function') unsubFrame();
    else APP.frameHooks?.delete(update);
    if (typeof window !== 'undefined') {
      window.removeEventListener('pointerdown', onPtrDown);
      window.removeEventListener('pointermove', onPtrMove);
      window.removeEventListener('pointerup', onPtrUp);
    }
    APP.pokeTargets?.delete(vrPanel.screen); APP.pokeTargets?.delete(vrPanel.dragBar);
    if (vrPanel.root.parent) vrPanel.root.parent.remove(vrPanel.root);
    if (domToolbar && domToolbar.parentNode) domToolbar.parentNode.removeChild(domToolbar);
    if (meltSphereEnt) APP.removeEntity?.(meltSphereEnt, false);
  };

  APP.manip = {
    config: manipCfg,
    showToolbar,
    setMode,
    fracture: fractureEntity,
    melt: toggleMeltSphere,
    softbody: convertToSoftbody,
    smokeHologram: convertToSmokeHologram,
    magicMerge: triggerPairMerge
  };

  // Register AI Tool with Geeny
  const MANIP_TOOL = {
    name: 'manipulate_model',
    description: 'Apply advanced model/GLB manipulation: "voronoi" (realistic Voronoi shatter with interior cracks & dust), "melt" (spawn thermal melting sphere), "softbody" (stretch/poke/squeeze + granular tear), "smoke" (smoky hologram that parts on hand swipe), or "merge" (Venom liquid symbiote merging between two objects).',
    parameters: {
      type: 'object',
      properties: {
        mode: { type: 'string', description: 'voronoi | melt | softbody | smoke | merge | toolbar' },
        target: { type: 'string', description: 'Entity ID, name, or "latest".' },
        target_b: { type: 'string', description: 'Optional second entity for mode="merge".' }
      },
      required: ['mode']
    }
  };
  APP.registerBotTool?.(MANIP_TOOL, (a = {}) => {
    const m = String(a.mode || '').toLowerCase();
    if (m === 'voronoi' || m === 'shatter' || m === 'break') return fractureEntity(a.target || 'latest');
    if (m === 'melt') { setMode('melt'); return toggleMeltSphere(true); }
    if (m === 'softbody') { setMode('softbody'); return convertToSoftbody(a.target || 'latest'); }
    if (m === 'smoke' || m === 'hologram') { setMode('smoke'); return convertToSmokeHologram(a.target || 'latest'); }
    if (m === 'merge' || m === 'venom') return triggerPairMerge(a.target || 'latest', a.target_b);
    showToolbar(true);
    return { ok: true, message: 'Opened Model Manipulation Lab.' };
  });

  APP.registerAbilityHandler?.('model_manipulation_engine', function (text) {
    const low = String(text || '').toLowerCase();
    if (low.includes('voronoi') || low.includes('shatter') || low.includes('break') || low.includes('تفتيت')) return fractureEntity('latest');
    if (low.includes('melt') || low.includes('ذوبان')) { setMode('melt'); return toggleMeltSphere(true); }
    if (low.includes('softbody') || low.includes('soft')) { setMode('softbody'); return convertToSoftbody('latest'); }
    if (low.includes('smoke') || low.includes('cloud')) { setMode('smoke'); return convertToSmokeHologram('latest'); }
    if (low.includes('merge') || low.includes('venom') || low.includes('دمج')) return triggerPairMerge('latest');
    showToolbar(!vrPanel.root.visible);
    return { ok: true, message: 'Toggled Model Manipulation Studio.' };
  });
})();
