        }

          vn[i * 3] /= l3; vn[i * 3 + 1] /= l3; vn[i * 3 + 2] /= l3;

        }
        out = { positions: P, normals: vn, uvs: uv, indices: T }; tr = [P.buffer, vn.buffer, uv.buffer, T.buffer];
      }
      out.bbox = { center: bc, size: bs };
      out.pca = { axes: axes, extents: pe };
      out.aspect = { e0e1: pe[0] / Math.max(pe[1], 1e-6), e1e2: pe[1] / Math.max(pe[2], 1e-6), e0e2: pe[0] / Math.max(pe[2], 1e-6) };
      out.fill = { aabb: vox / Math.max(bs[0] * bs[1] * bs[2], 1e-9), obb: vox / Math.max(pe[0] * pe[1] * pe[2], 1e-9) };
      out.symmetry = symm; out.sliceY = sliceY; out.sliceAxis = sliceAxis;
      out.ascii = { front: buildAscii('front'), side: buildAscii('side'), top: buildAscii('top') };
      out.yaw = { ang: ang, u: yu, v: yv, ratio: yu / Math.max(yv, 1e-6) };
      out.tris = nT;
      return { out: out, tr: tr };
    }

    const pos = [], idx = [], p = new V3();
    const rAt = (t) => { const f = t * (radii.length - 1), i = Math.floor(f), j = Math.min(i + 1, radii.length - 1); return radii[i] + (radii[j] - radii[i]) * (f - i); };

      for (let k = 0; k < radial; k++) { const a = k / radial * Math.PI * 2, c = Math.cos(a), s = Math.sin(a); pos.push(p.x + (N.x * c + B.x * s) * r, p.y + (N.y * c + B.y * s) * r, p.z + (N.z * c + B.z * s) * r); }
    }
    for (let i = 0; i < segs; i++) for (let k = 0; k < radial; k++) {

  function renderViews(out) {
    const PW = 240, cv = document.createElement('canvas'); cv.width = PW * 3; cv.height = PW;
    const c = cv.getContext('2d'), Pp = out.positions, I = out.indices, nT = I.length / 3, nV = Pp.length / 3, sz = out.bbox.size;
    const az = 0.6, el = 0.42;
    const views = [{ n: 'FRONT', f: new V3(0, 0, -1) }, { n: 'SIDE', f: new V3(-1, 0, 0) }, { n: '3/4', f: new V3(-Math.sin(az) * Math.cos(el), -Math.sin(el), -Math.cos(az) * Math.cos(el)) }];
    const sx = new Float32Array(nV), sy = new Float32Array(nV), sd = new Float32Array(nV);
    let scale = 1e9;
    views.forEach((v) => {
      v.r = new V3().crossVectors(v.f, UP).normalize(); v.u = new V3().crossVectors(v.r, v.f).normalize();
      let a0 = 1e9, a1 = -1e9, b0 = 1e9, b1 = -1e9;
      for (let i = 0; i < nV; i++) { const x = Pp[i * 3], y = Pp[i * 3 + 1], z = Pp[i * 3 + 2], u = x * v.r.x + y * v.r.y + z * v.r.z, w = x * v.u.x + y * v.u.y + z * v.u.z; if (u < a0) a0 = u; if (u > a1) a1 = u; if (w < b0) b0 = w; if (w > b1) b1 = w; }
      scale = Math.min(scale, PW * 0.80 / Math.max(a1 - a0, b1 - b0, 1e-4)); v.cx = (a0 + a1) / 2; v.cy = (b0 + b1) / 2;
    });
    c.fillStyle = '#0f172a'; c.fillRect(0, 0, PW * 3, PW);
    const stride = Math.max(1, Math.ceil(nT / 9000));
    views.forEach((v, vi) => {
      const ox = vi * PW; c.save(); c.beginPath(); c.rect(ox, 0, PW, PW); c.clip();
      for (let i = 0; i < nV; i++) { const x = Pp[i * 3], y = Pp[i * 3 + 1], z = Pp[i * 3 + 2]; sx[i] = ox + PW / 2 + ((x * v.r.x + y * v.r.y + z * v.r.z) - v.cx) * scale; sy[i] = PW / 2 - ((x * v.u.x + y * v.u.y + z * v.u.z) - v.cy) * scale; sd[i] = x * v.f.x + y * v.f.y + z * v.f.z; }
      const L = new V3().copy(v.f).multiplyScalar(-0.7).addScaledVector(v.r, -0.5).addScaledVector(v.u, 0.6).normalize();
      const list = [];
      for (let t = 0; t < nT; t += stride) {
        const a = I[t * 3], b = I[t * 3 + 1], d = I[t * 3 + 2];
        const e1x = Pp[b * 3] - Pp[a * 3], e1y = Pp[b * 3 + 1] - Pp[a * 3 + 1], e1z = Pp[b * 3 + 2] - Pp[a * 3 + 2], e2x = Pp[d * 3] - Pp[a * 3], e2y = Pp[d * 3 + 1] - Pp[a * 3 + 1], e2z = Pp[d * 3 + 2] - Pp[a * 3 + 2];
        let nx = e1y * e2z - e1z * e2y, ny = e1z * e2x - e1x * e2z, nz = e1x * e2y - e1y * e2x; const nl = Math.hypot(nx, ny, nz) || 1; nx /= nl; ny /= nl; nz /= nl;
        if (nx * v.f.x + ny * v.f.y + nz * v.f.z > 0) continue;
        list.push([(sd[a] + sd[b] + sd[d]) / 3, a, b, d, Math.max(0, nx * L.x + ny * L.y + nz * L.z)]);
      }
      list.sort((p, q) => q[0] - p[0]);
      c.lineWidth = 0.8; c.lineJoin = 'round';
      for (const t of list) {
        const g2 = 95 + t[4] * 160;
        const col = 'rgb(' + (g2 * 0.85 | 0) + ',' + (g2 * 0.96 | 0) + ',' + (Math.min(255, g2 + 20) | 0) + ')';
        c.fillStyle = col; c.strokeStyle = col; c.beginPath(); c.moveTo(sx[t[1]], sy[t[1]]); c.lineTo(sx[t[2]], sy[t[2]]); c.lineTo(sx[t[3]], sy[t[3]]); c.closePath(); c.fill(); c.stroke();
      }
      c.fillStyle = '#38bdf8'; c.font = 'bold 14px sans-serif'; c.fillText(v.n, ox + 8, 18);
      c.restore(); c.strokeStyle = '#334155'; c.strokeRect(ox + 0.5, 0.5, PW - 1, PW - 1);
    });
    c.fillStyle = '#cbd5e1'; c.font = '12px sans-serif'; c.fillText('Size: W=' + sz[0].toFixed(2) + ' H=' + sz[1].toFixed(2) + ' D=' + sz[2].toFixed(2) + 'm', 8, PW - 8);
    const url = cv.toDataURL('image/jpeg', 0.88); return IMG_WITH_PREFIX ? url : url.split(',')[1];
  }

  function extractJSON(t) { try { const m = String(t || '').match(/\{[\s\S]*\}/); return m ? JSON.parse(m[0]) : null; } catch (_) { return null; } }

  async function callAI(opts) {
    if (!APP.ai) throw new Error('APP.ai unavailable');
    const aiCaller = (APP.ai?.isLiveModelSelected?.() && APP.ai?.chatWithBuilder) ? APP.ai.chatWithBuilder : APP.ai.chat;
    let got = null;

    async function invoke(withImg) {
      got = null;
      const res = await withTimeout(aiCaller.call(APP.ai, {
        systemPrompt: opts.systemPrompt,
        prompt: opts.prompt,
        imageB64: withImg ? opts.imageB64 : undefined,
        tools: [opts.tool],
        maxSteps: opts.maxSteps || 1,
        onToolCall: async (name, args) => {
          if (name === opts.tool.name) {
            let a = args;
            if (typeof a === 'string') a = extractJSON(a);
            if (a && a.arguments) a = typeof a.arguments === 'string' ? extractJSON(a.arguments) : a.arguments;
            got = a;
          }
          return { ok: true };
        }
      }), 32000, 'AI timeout');
      if (!got && res?.text) got = extractJSON(res.text);
      if (!got && res && !res.ok && withImg && opts.imageB64) throw new Error(res.error || 'image failed');
      return { got, res };
    }

    try {
      const r1 = await invoke(!!opts.imageB64);
      if (r1.got) return r1.got;
      if (opts.imageB64) {
        // Retry without image in case selected free model is text-only
        const r2 = await invoke(false);
        return r2.got;
      }
      return null;
    } catch (err) {
      if (opts.imageB64) {
        try { const r2 = await invoke(false); return r2.got; } catch (_) {}
      }
      throw err;
    }
  }

  function cleanSpec(raw, base) {
    if (!raw || !raw.label || /^(sculpture|object|shape|blob)$/i.test(String(raw.label).trim())) {

    }
    const mats = ['wood', 'metal', 'stone', 'ceramic', 'organic', 'glass', 'fabric'], num = (x, d) => (isFinite(+x) ? +x : d);
    const sy = ['none', 'mirror_x', 'radial_y'].includes(raw.symmetry) ? raw.symmetry : (base.symmetry.mirrorX > 0.82 ? 'mirror_x' : 'none');
    const parts = (Array.isArray(raw.organizedSubParts) ? raw.organizedSubParts : []).slice(0, 12).map((p) => {

        op: p.op === 'subtract' ? 'sub' : 'add',

      label: String(raw.label || 'artifact').slice(0, 40),
      material: mats.includes(raw.material) ? raw.material : 'ceramic',

      facing: ['toward', 'away', 'left', 'right'].includes(raw.facing) ? raw.facing : 'none',
      lying: !!raw.lying
    };
  }

  function parseHint(text) {
    let s = String(text || '').toLowerCase().replace(/[\u2019]/g, "'").replace(/[^a-z0-9' ]+/g, ' ').replace(/\s+/g, ' ').trim();
    const pre = [
      /^i'?m going to (draw|sketch|make|create|sculpt)\s+/,
      /^i'?m (drawing|sketching|making|creating|sculpting)\s+/,
      /^i (am|will|want to|wanna|gonna|'ll)( be)?( going to | gonna )?(draw|sketch|make|create|sculpt|drawing|sketching)\s+/,
      /^(let's draw|lets draw|draw|sketch|sculpt)\s+/,
      /^(it is|it's|this is|that is|that's|a sculpture of|make my drawing)\s+/,
      /^(a|an|the|some)\s+/
    ];
    let changed = true;
    while (changed) { changed = false; for (const re of pre) { const n = s.replace(re, ''); if (n !== s) { s = n; changed = true; } } }
    s = s.replace(/\s+(please|now|thanks)$/, '').trim();
    const words = s.split(' ').filter(Boolean).slice(0, 3);
    if (!words.length) return null;
    const label = words.join(' '), head = words[words.length - 1];
    const sing = head.length > 3 && head.endsWith('s') && !head.endsWith('ss') ? head.slice(0, -1) : head;
    const keywords = [...new Set([label, head, sing].concat(SYN[head] || [], SYN[sing] || []))];
    return { label, keywords, text: String(text) };
  }

  function partsToOps(parts, base, avgR) {
    const c = base.bbox.center, z = base.bbox.size;
    return parts.map((p) => {
      const ctr = [c[0] + p.c[0] * z[0], c[1] + p.c[1] * z[1], c[2] + p.c[2] * z[2]], full = [p.s[0] * z[0], p.s[1] * z[1], p.s[2] * z[2]], k = avgR * (0.3 + 0.8 * p.round);
      const op = { t: p.op, c: ctr, k, r: 0, s: p.shape };
      if (p.shape === 'sphere') op.r = 0.25 * (full[0] + full[1] + full[2]) / 1.5;
      else if (p.shape === 'box') { op.s = 'box'; op.h = full.map((v) => v / 2); op.r = Math.min(...op.h) * 0.35 * p.round; }
      else if (p.shape === 'cylinder') { op.s = 'cyl'; op.ax = p.axis; const o = [0, 1, 2].filter((i) => i !== p.axis); op.r = 0.25 * (full[o[0]] + full[o[1]]); op.hh = full[p.axis] / 2; }
      else { op.s = 'cap'; let ai = 0; for (let i = 1; i < 3; i++) if (full[i] > full[ai]) ai = i; const o = [0, 1, 2].filter((i) => i !== ai); op.r = 0.25 * (full[o[0]] + full[o[1]]); const h = Math.max(full[ai] / 2 - op.r, 0); op.a = ctr.slice(); op.b = ctr.slice(); op.a[ai] -= h; op.b[ai] += h; }
      return op;
    });
  }

  const hash2 = (x, y) => { const h = Math.sin(x * 127.1 + y * 311.7) * 43758.5453; return h - Math.floor(h); };
  function vnoise(x, y) { const xi = Math.floor(x), yi = Math.floor(y), xf = x - xi, yf = y - yi, u = xf * xf * (3 - 2 * xf), v = yf * yf * (3 - 2 * yf); const a = hash2(xi, yi), b = hash2(xi + 1, yi), c = hash2(xi, yi + 1), d = hash2(xi + 1, yi + 1); return a + (b - a) * u + (c - a) * v + (a - b - c + d) * u * v; }
  function fbm(x, y, o) { let s = 0, a = 0.5; for (let i = 0; i < (o || 4); i++) { s += a * vnoise(x, y); x *= 2.03; y *= 2.03; a *= 0.5; } return s; }
  function pattern(kind, u, v, o) {
    switch (kind) {
      case 'wood': { const g = u * 12 + fbm(u * 3, v * 0.6, 4) * 5, ring = 0.5 + 0.5 * Math.sin(g * 6.2832); o[0] = ring * 0.45 + fbm(u * 36, v * 2, 3) * 0.3; o[1] = ring; o[2] = 0.62 + 0.22 * fbm(u * 28, v * 3, 2); break; }
      case 'metal': { const n = fbm(u * 2, v * 140, 3); o[0] = n * 0.35; o[1] = n; o[2] = 0.24 + 0.22 * n; break; }
      case 'stone': { const n = fbm(u * 8, v * 8, 4); o[0] = n; o[1] = n; o[2] = 0.80 + 0.14 * n; break; }
      case 'ceramic': { const sp = hash2(Math.floor(u * 220), Math.floor(v * 220)) > 0.985 ? 1 : 0; o[0] = 0.06 + 0.06 * fbm(u * 18, v * 18, 3); o[1] = fbm(u * 3, v * 3, 3) * 0.32 + sp; o[2] = 0.18 + 0.12 * fbm(u * 12, v * 12, 3); break; }
      case 'organic': { const n = fbm(u * 7, v * 7, 4), ve = Math.abs(Math.sin(n * 11)); o[0] = n * 0.55 + ve * 0.25; o[1] = n; o[2] = 0.65 + 0.14 * n; break; }
      case 'glass': o[0] = 0.5; o[1] = 0; o[2] = 0.06; break;
      default: { const w = Math.sin(u * 320) * Math.sin(v * 320); o[0] = 0.5 + 0.25 * w + 0.1 * fbm(u * 24, v * 24, 2); o[1] = fbm(u * 5, v * 5, 3); o[2] = 0.88; }
    }
  }
  const pbrCache = new Map();
  function bakePBR(kind, c1, c2) {
    const key = kind + ':' + c1.getHex() + ':' + c2.getHex();
    if (pbrCache.has(key)) return pbrCache.get(key);
    const out = bakePBRRaw(kind, c1, c2); Object.values(out).forEach((t) => { t.userData = t.userData || {}; t.userData.shared = true; }); pbrCache.set(key, out); return out;
  }
  function bakePBRRaw(kind, c1, c2) {
    const N = 256, mk = () => { const c = document.createElement('canvas'); c.width = c.height = N; return c; };
    const ca = mk(), cn = mk(), cr = mk(), ia = ca.getContext('2d').createImageData(N, N), inn = cn.getContext('2d').createImageData(N, N), ir = cr.getContext('2d').createImageData(N, N);
    const H = new Float32Array(N * N), o = [0, 0, 0], h1 = c1.getHex(), h2 = c2.getHex();
    const A = [(h1 >> 16) & 255, (h1 >> 8) & 255, h1 & 255], B = [(h2 >> 16) & 255, (h2 >> 8) & 255, h2 & 255];
    for (let y = 0; y < N; y++) for (let x = 0; x < N; x++) {
      pattern(kind, x / N, y / N, o); const i = y * N + x, m = clamp(o[1], 0, 1), sh = 0.76 + 0.24 * clamp(o[0], 0, 1);
      H[i] = o[0]; ia.data[i * 4] = (A[0] + (B[0] - A[0]) * m) * sh; ia.data[i * 4 + 1] = (A[1] + (B[1] - A[1]) * m) * sh; ia.data[i * 4 + 2] = (A[2] + (B[2] - A[2]) * m) * sh; ia.data[i * 4 + 3] = 255;
      const rr = clamp(o[2], 0.04, 1) * 255; ir.data[i * 4] = ir.data[i * 4 + 1] = ir.data[i * 4 + 2] = rr; ir.data[i * 4 + 3] = 255;
    }
    const str = { stone: 4.5, wood: 2.8, metal: 1.4, organic: 3.2, fabric: 4.2, ceramic: 1.5, glass: 0 }[kind] || 2;
    const hh = (x, y) => H[((y + N) % N) * N + ((x + N) % N)];
    for (let y = 0; y < N; y++) for (let x = 0; x < N; x++) {
      const gx = (hh(x + 1, y - 1) + 2 * hh(x + 1, y) + hh(x + 1, y + 1)) - (hh(x - 1, y - 1) + 2 * hh(x - 1, y) + hh(x - 1, y + 1));
      const gy = (hh(x - 1, y + 1) + 2 * hh(x, y + 1) + hh(x + 1, y + 1)) - (hh(x - 1, y - 1) + 2 * hh(x, y - 1) + hh(x + 1, y - 1));
      let nx = -gx * str * 0.25, ny = gy * str * 0.25, nz = 1; const l = Math.hypot(nx, ny, nz); const i = (y * N + x) * 4;
      inn.data[i] = (nx / l * 0.5 + 0.5) * 255; inn.data[i + 1] = (ny / l * 0.5 + 0.5) * 255; inn.data[i + 2] = (nz / l * 0.5 + 0.5) * 255; inn.data[i + 3] = 255;
    }
    ca.getContext('2d').putImageData(ia, 0, 0); cn.getContext('2d').putImageData(inn, 0, 0); cr.getContext('2d').putImageData(ir, 0, 0);
    const tex = (cv, srgb) => { const t = new THREE.CanvasTexture(cv); t.wrapS = t.wrapT = THREE.RepeatWrapping; t.repeat.set(2, 2); t.anisotropy = 4; if (srgb && THREE.SRGBColorSpace) t.colorSpace = THREE.SRGBColorSpace; return t; };
    return { map: tex(ca, true), normalMap: tex(cn, false), roughnessMap: tex(cr, false) };
  }

  function pbrMaterial(spec) {
    const t = bakePBR(spec.material, spec.c1, spec.c2);
    if (spec.material === 'glass') {
      return new THREE.MeshPhysicalMaterial({
        color: spec.c1, map: t.map, normalMap: t.normalMap, roughnessMap: t.roughnessMap,
        roughness: 0.12, metalness: 0, transmission: 0.92, ior: 1.5, thickness: 0.3, clearcoat: 1, side: THREE.DoubleSide
      });
    }
    if (spec.material === 'ceramic') {
      return new THREE.MeshPhysicalMaterial({
        map: t.map, normalMap: t.normalMap, roughnessMap: t.roughnessMap,
        roughness: 0.28, metalness: 0.02, clearcoat: 0.85, clearcoatRoughness: 0.14
      });
    }
    return new THREE.MeshStandardMaterial({
      map: t.map, normalMap: t.normalMap, roughnessMap: t.roughnessMap,
      roughness: 1.0, metalness: spec.material === 'metal' ? 0.88 : 0.04, side: THREE.FrontSide
    });
  }

  // Contextual geometric micro-accents added to Mode A meshes so recognized objects gain crisp details
  function addContextualDetails(group, spec, out) {
    const lbl = String(spec.label || '').toLowerCase();

    const minD = Math.max(0.025, Math.min(sx, sy, sz));
    const accentMat = new THREE.MeshStandardMaterial({
      color: spec.c2, roughness: spec.material === 'metal' ? 0.28 : 0.65, metalness: spec.material === 'metal' ? 0.88 : 0.08
    });

    if (/banana|apple|pear|fruit|lemon|orange|cherry/i.test(lbl)) {
      const stem = new THREE.Mesh(new THREE.CylinderGeometry(minD * 0.06, minD * 0.10, minD * 0.28, 10), new THREE.MeshStandardMaterial({ color: 0x4b3621, roughness: 0.85 }));
      stem.position.set(0, sy * 0.48, 0);

    } else if (/mug|cup|vase|bowl|pot|bottle/i.test(lbl)) {

      rim.rotation.x = Math.PI / 2;
      rim.position.set(0, sy * 0.46, 0);

    } else if (/sword|blade|axe|hammer|dagger/i.test(lbl)) {

      pommel.position.set(0, -sy * 0.48, 0);
      group.add(pommel);
    }
  }

    if (!glbIndex) glbIndex = collectGLBs();

    for (const kw of kws) { const e = map.get(kw); if (e) return e; }
    for (const kw of kws) for (const [k, v] of map) if (k.split(' ').includes(kw)) return v;

    for (const kw of kws) {
      const w = kw.replace(/\s+/g, '');
      for (const url of ['./assets/' + w + '.glb', './assets/' + w + '.gltf', './' + w + '.glb']) { try { const r = await fetch(url, { method: 'HEAD' }); if (r.ok) return { name: w, url }; } catch (_) {} }

  function fitGLB(model, spec, base) {
    const holder = new THREE.Group(), scaler = new THREE.Group(), pre = new THREE.Group();
    pre.add(model); scaler.add(pre); holder.add(scaler); holder.updateMatrixWorld(true);
    let box = new THREE.Box3().setFromObject(pre), sz = box.getSize(new V3()); const bs = base.bbox.size;
    if (spec.lying && sz.y > Math.max(sz.x, sz.z) * 1.4 && Math.max(bs[0], bs[2]) > bs[1] * 1.4) { pre.rotation.x = Math.PI / 2; holder.updateMatrixWorld(true); box.setFromObject(pre); sz = box.getSize(new V3()); }
    pre.position.copy(box.getCenter(new V3())).negate(); holder.updateMatrixWorld(true);
    let laZ = sz.z >= sz.x; const la = laZ ? new V3(0, 0, 1) : new V3(1, 0, 0);
    const faceDir = { toward: new V3(0, 0, 1), away: new V3(0, 0, -1), right: new V3(1, 0, 0), left: new V3(-1, 0, 0) }[spec.facing] || null, yw = base.yaw;
    const elong = yw.ratio >= 1.25; let D = null;
    if (elong) { D = new V3(Math.cos(yw.ang), 0, Math.sin(yw.ang)); if (faceDir && D.dot(faceDir) < 0) D.negate(); }
    else if (faceDir) { D = faceDir.clone(); la.set(0, 0, 1); laZ = true; }
    if (D) { if (la.dot(D) < -0.9999) holder.quaternion.setFromAxisAngle(UP, Math.PI); else holder.quaternion.setFromUnitVectors(la, D); }
    const gLa = laZ ? sz.z : sz.x, gOt = laZ ? sz.x : sz.z, tLa = elong ? yw.u : (yw.u + yw.v) / 2, tOt = elong ? yw.v : (yw.u + yw.v) / 2;
    const rLa = tLa / Math.max(gLa, 1e-4), rOt = tOt / Math.max(gOt, 1e-4), rY = bs[1] / Math.max(sz.y, 1e-4), su = Math.cbrt(rLa * rOt * rY), lo = 0.82, hi = 1.22;
    const fLa = su * clamp(rLa / su, lo, hi), fOt = su * clamp(rOt / su, lo, hi), fY = su * clamp(rY / su, lo, hi);
    scaler.scale.set(laZ ? fOt : fLa, fY, laZ ? fLa : fOt);
    return holder;
  }

  async function modeA(rec, spec, base, alive) {
    setStatus('Refining ' + spec.label + ' (' + spec.material + ')…', 'busy');
    const ops = partsToOps(spec.parts, base, rec.avgR);
    // Reuse `base` immediately if no extra SDF ops or symmetry changes were requested (2x faster!)
    const needsRemesh = ops.length > 0 || spec.symmetry !== 'none' || spec.sharpen > 0.45;
    const out = needsRemesh
      ? await runWorker({ strokes: rec.payload, ops, res: 48, smoothK: rec.avgR * (spec.sharpen > 0.55 ? 0.35 : 0.60), iter: spec.sharpen > 0.55 ? 2 : 5, sharpen: spec.sharpen, creaseDeg: 38, symmetry: spec.symmetry, center: base.bbox.center })
      : base;
    if (!alive()) return;
    setStatus('Applying ' + spec.material + ' PBR textures…', 'busy');
    const geo = geoFromOut(out);
    const mesh = new THREE.Mesh(geo, pbrMaterial(spec));
    mesh.frustumCulled = false; mesh.castShadow = true; mesh.receiveShadow = true;
    const grp = new THREE.Group();
    grp.add(mesh);
    addContextualDetails(grp, spec, out);
    finalize(rec, grp, spec.label, out.bbox.center, 1.15);
  }

  async function modeB(rec, spec, base, alive) {
    setStatus('Matching GLB for "' + spec.label + '"…', 'busy');
    const hit = await findGLB(spec.keywords.concat([spec.label]));
    if (!alive()) return;
    if (!hit) { toast('No GLB in /assets/ for "' + spec.label + '" — using Mode A PBR', 3400); return modeA(rec, spec, base, alive); }
    let model; try { model = await loadGLB(hit.url); } catch (e) { toast('GLB "' + hit.name + '" failed — Mode A PBR', 3000); return modeA(rec, spec, base, alive); }
    if (!alive()) { disposeDeep(model); return; }
    toast('Morphing into ' + hit.name, 2200);
    await morphInto(rec, fitGLB(model, spec, base), base, spec.label, alive);
  }

  function buildNN(src) {
    const n = src.length / 3, mn = [Infinity, Infinity, Infinity], mx = [-Infinity, -Infinity, -Infinity];
    for (let i = 0; i < n; i++) for (let a = 0; a < 3; a++) { const v = src[i * 3 + a]; if (v < mn[a]) mn[a] = v; if (v > mx[a]) mx[a] = v; }
    const span = Math.max(mx[0] - mn[0], mx[1] - mn[1], mx[2] - mn[2], 1e-4), G = Math.max(4, Math.min(20, Math.round(Math.cbrt(n) * 1.4))), cs = span / G * 1.0001;
    const cell = (x, a) => clamp(Math.floor((x - mn[a]) / cs), 0, G - 1), cells = new Map();
    for (let i = 0; i < n; i++) { const k = cell(src[i * 3], 0) + G * (cell(src[i * 3 + 1], 1) + G * cell(src[i * 3 + 2], 2)); let L = cells.get(k); if (!L) cells.set(k, L = []); L.push(i); }
    return { src, G, cells, cell, cs };
  }
  function nnQuery(nn, x, y, z) {
    const { src, G, cells, cell, cs } = nn, cx = cell(x, 0), cy = cell(y, 1), cz = cell(z, 2);
    let best = Infinity, bi = 0;
    for (let r = 0; r <= G; r++) {
      for (let dz = -r; dz <= r; dz++) for (let dy = -r; dy <= r; dy++) for (let dx = -r; dx <= r; dx++) {
        if (Math.max(Math.abs(dx), Math.abs(dy), Math.abs(dz)) !== r) continue;
        const ix = cx + dx, iy = cy + dy, iz = cz + dz;
        if (ix < 0 || iy < 0 || iz < 0 || ix >= G || iy >= G || iz >= G) continue;
        const L = cells.get(ix + G * (iy + G * iz)); if (!L) continue;
        for (const i of L) { const ex = src[i * 3] - x, ey = src[i * 3 + 1] - y, ez = src[i * 3 + 2] - z, d = ex * ex + ey * ey + ez * ez; if (d < best) { best = d; bi = i; } }
      }
      if (best < Infinity && best <= (r * cs) * (r * cs)) break;
    }
    return bi;
  }
  const easeOutBack = (k) => { const c1 = 1.2, c3 = c1 + 1; return 1 + c3 * Math.pow(k - 1, 3) + c1 * Math.pow(k - 1, 2); };

  function morphInto(rec, holder, base, label, alive) {
    return new Promise((resolve) => {
      holder.updateMatrixWorld(true);
      const P0 = base.positions, nP = P0.length / 3, stp = Math.max(1, Math.floor(nP / 1400)), src = [];
      for (let i = 0; i < nP; i += stp) src.push(P0[i * 3], P0[i * 3 + 1], P0[i * 3 + 2]);
      if (!src.length) { finalize(rec, holder, label, base.bbox.center, 0.9); return resolve(); }
      const nn = buildNN(src), parts = [];
      let yMin = Infinity, yMax = -Infinity;
      holder.traverse((m) => {
        if (!m.isMesh || !m.geometry || !m.geometry.attributes.position) return;
        const g = m.geometry.clone(); g.applyMatrix4(m.matrixWorld);
        const rest = Float32Array.from(g.attributes.position.array), n = rest.length / 3, start = new Float32Array(rest.length), delay = new Float32Array(n);
        for (let i = 0; i < n; i++) {
          const x = rest[i * 3], y = rest[i * 3 + 1], z = rest[i * 3 + 2], j = nnQuery(nn, x, y, z);
          start[i * 3] = src[j * 3]; start[i * 3 + 1] = src[j * 3 + 1]; start[i * 3 + 2] = src[j * 3 + 2];
          if (y < yMin) yMin = y; if (y > yMax) yMax = y;
        }
        parts.push({ g, rest, start, delay, n, orig: m.material });
      });
      if (!parts.length) { finalize(rec, holder, label, base.bbox.center, 0.9); return resolve(); }
      const H = Math.max(yMax - yMin, 1e-4);
      parts.forEach((p) => { for (let i = 0; i < p.n; i++) { const yn = (p.rest[i * 3 + 1] - yMin) / H; p.delay[i] = 0.45 * yn + Math.random() * 0.05; } });
      const morphRoot = new THREE.Group();
      parts.forEach((p) => {
        p.g.attributes.position.array.set(p.start); p.g.attributes.position.needsUpdate = true;
        const mesh = new THREE.Mesh(p.g, rec.holoMat), wire = new THREE.Mesh(p.g, rec.wireMat);
        mesh.frustumCulled = false; wire.frustumCulled = false; mesh.add(wire); p.mesh = mesh; p.wire = wire; morphRoot.add(mesh);
      });
      const bootTemps = rec.temps.slice(), D = 0.85, TOT = 0.45 + D;
      holoBoundsUpdate(rec, morphRoot);
      APP.scene.add(morphRoot);

      const SP = 240, spPos = new Float32Array(SP * 3).fill(-1000), spVel = new Float32Array(SP * 3), spLife = new Float32Array(SP);
      const spGeo = new THREE.BufferGeometry();
      spGeo.setAttribute('position', new THREE.BufferAttribute(spPos, 3));
      spGeo.setAttribute('aLife', new THREE.BufferAttribute(spLife, 1));
      const spMat = makeSparkDiscMat(0x7dd3fc);
      const sparks = new THREE.Points(spGeo, spMat); sparks.frustumCulled = false; APP.scene.add(sparks);

      let spHead = 0;
      const spawnAt = (x, y, z, speed) => {
        const s = spHead; spHead = (spHead + 1) % SP;
        spPos[s * 3] = x; spPos[s * 3 + 1] = y; spPos[s * 3 + 2] = z;
        const a = Math.random() * Math.PI * 2, b = (Math.random() - 0.3) * Math.PI * 0.9;
        spVel[s * 3] = Math.cos(a) * Math.cos(b) * speed; spVel[s * 3 + 1] = Math.sin(b) * speed + 0.08; spVel[s * 3 + 2] = Math.sin(a) * Math.cos(b) * speed;
        spLife[s] = 0.5 + Math.random() * 0.5;
      };
      const stepSparks = (dt) => {
        for (let s = 0; s < SP; s++) {
          if (spLife[s] <= 0) continue;
          spLife[s] -= dt;
          spPos[s * 3] += spVel[s * 3] * dt; spPos[s * 3 + 1] += spVel[s * 3 + 1] * dt; spPos[s * 3 + 2] += spVel[s * 3 + 2] * dt;
          const damp = Math.max(0, 1 - 2.4 * dt); spVel[s * 3] *= damp; spVel[s * 3 + 1] *= damp; spVel[s * 3 + 2] *= damp;
          if (spLife[s] <= 0) spPos[s * 3 + 1] = -1000;
        }
        spGeo.attributes.position.needsUpdate = true;
        spGeo.attributes.aLife.needsUpdate = true;
      };

      let t = 0, settled = false, unsub = null;
      const stop = () => { if (settled) return false; settled = true; try { unsub && unsub(); } catch (_) {} unsub = null; APP.scene.remove(morphRoot); APP.scene.remove(sparks); spGeo.dispose(); spMat.dispose(); return true; };
      const cancel = () => { if (stop()) parts.forEach((p) => p.g.dispose()); resolve(); };
      const done = () => {
        if (!stop()) return;
        parts.forEach((p) => {
          p.g.attributes.position.array.set(p.rest); p.g.attributes.position.needsUpdate = true; p.g.computeVertexNormals();
          p.mesh.remove(p.wire);
          const om = p.orig;
          p.mesh.material = Array.isArray(om) ? om.map((x) => x.clone()) : (om ? om.clone() : new THREE.MeshStandardMaterial({ color: 0xcfd8dc }));
        });
        bootTemps.forEach((o) => { o.visible = false; });
        if (alive()) finalize(rec, morphRoot, label, base.bbox.center, 0.85);
        resolve();
      };
      unsub = APP.onUpdate((dt) => {
        if (settled) return;
        if (!alive()) return cancel();
        const step = Math.min(dt, 0.05); t += step;
        for (const p of parts) {
          const a = p.g.attributes.position.array, s = p.start, r = p.rest, dl = p.delay;
          for (let i = 0; i < p.n; i++) {
            const e = easeOutBack(clamp((t - dl[i]) / D, 0, 1)), o = i * 3;
            a[o] = s[o] + (r[o] - s[o]) * e; a[o + 1] = s[o + 1] + (r[o + 1] - s[o + 1]) * e; a[o + 2] = s[o + 2] + (r[o + 2] - s[o + 2]) * e;
          }
          p.g.attributes.position.needsUpdate = true;
        }
        const k = clamp(t / TOT, 0, 1);
        rec.wireMat.opacity = 0.25 + 0.55 * (1 - k) * (0.5 + 0.5 * Math.sin(t * 16));
        if (rec.scanRing) rec.scanRing.position.y = yMin + k * H;
        bootTemps.forEach((o) => { o.visible = k < 0.45; });
        if (t < TOT) for (let q = 0; q < 8; q++) {
          const p = parts[(Math.random() * parts.length) | 0], i = (Math.random() * p.n) | 0, o = i * 3, a = p.g.attributes.position.array;
          spawnAt(a[o], a[o + 1], a[o + 2], 0.55);
        }
        stepSparks(step);
        if (t >= TOT) done();
      });
    });
  }

  }

  function undoLast() {
    endAllStrokes();
    const rec = S.undo.pop(); if (!rec) { toast('Nothing to undo'); return; }
    removeManifestedRecord(rec);

    const restored = rec.group, cur = S.strokes;
    cur.forEach((s) => { if (s.mesh) restored.add(s.mesh); });
    if (S.rawGroup && S.rawGroup !== restored) {
      if (S.rawEnt) { try { APP.removeEntity(S.rawEnt, false); } catch (_) { disposeDeep(S.rawGroup); } }
      else { if (S.rawGroup.parent) S.rawGroup.parent.remove(S.rawGroup); disposeDeep(S.rawGroup); }
    }
    restored.visible = true; restored.userData.locked = false;
    if (!restored.parent) APP.scene.add(restored);
    rec.strokes.forEach((s) => { if (s.mesh) s.mesh.material = penMat; });
    S.rawGroup = restored; S.rawEnt = rec.rawEnt; S.strokes = rec.strokes.concat(cur); S.viewFwd = rec.viewFwd; S.state = 0; S.dirty = true;
    setStatus('Raw strokes restored — hold Grip (+ Trigger) to edit', 'ok');
    toast('Undo: raw strokes restored');
  }

  function clearAll() {
    endAllStrokes();
    if (S.active) { S.active.cancelled = true; S.active = null; }
    S.busy = false;
    S.state = 0;

    // Remove ALL manifested sculptures, holos, sparks, and cached raw stroke groups in undo stack
    while (S.undo.length) {
      const rec = S.undo.pop();
      removeManifestedRecord(rec);
      if (rec.group) {
        if (rec.group.parent) rec.group.parent.remove(rec.group);
        disposeDeep(rec.group);
      }
    }
    for (const rv of S.reveals) {
      if (rv.sparks?.parent) rv.sparks.parent.remove(rv.sparks);
      rv.spGeo?.dispose(); rv.spMat?.dispose();
    }
    S.reveals = [];

    // Remove current un-manifested raw strokes and reset raw group
    discardStrokes();
    if (S.rawEnt) { try { APP.removeEntity(S.rawEnt, false); } catch (_) {} S.rawEnt = null; }
    if (S.rawGroup) {
      if (S.rawGroup.parent) S.rawGroup.parent.remove(S.rawGroup);
      disposeDeep(S.rawGroup);
    }
    newRawGroup();
    S.hint = null;
    S.dirty = true;
    setStatus('Cleared all drawings & sculpted models', 'ok');
    toast('Cleared all sculpt drawings & models');
  }
